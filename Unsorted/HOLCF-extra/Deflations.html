Theory Deflations

theory Deflations
  imports MUB_Closure
begin

(* ===================================================================== *)
(*  The lattice of finite deflations, and 'a defl by compact fixes       *)
(*                                                                       *)
(*  Built on MUB_Closure (the mub-closure machinery + the 'defl_of_set'  *)
(*  locale that turns a mub-closed set of compacts into a deflation).    *)
(*                                                                       *)
(*  - fin_defl order is fixed-point inclusion; pairwise joins exist      *)
(*    ('join_fin_defl' = defl_of_set of the mub-closure of the union of  *)
(*    fixed points), so finite deflations form a lattice.                *)
(*  - cast :: 'a defl → ('a → 'a) is injective (Algebraic.cast_eq_imp_eq)*)
(*    but NOT surjective onto {d. deflation d}: a deflation is the lub    *)
(*    of the finite deflations below it iff its image is bifinite, and    *)
(*    bifinite domains are not closed under retracts (RB ⊋ B, cf. Jung). *)
(*  - what IS true: an algebraic deflation 'cast A' is pinned down by     *)
(*    which COMPACT elements it fixes.  Writing 'dfixes d':              *)
(*      defl_below_iff_dfixes : A ⊑ B ⟷ dfixes (cast A) ⊆ dfixes (cast B)*)
(*      defl_eq_iff_dfixes    : A = B ⟷ dfixes (cast A) = dfixes (cast B)*)
(*    i.e. 'dfixes o cast' is an order embedding of 'a defl into sets of  *)
(*    compact basis elements.                                            *)
(* ===================================================================== *)


section ‹The lattice of finite deflations›

text ‹A deflation is below another exactly when its fixed-point set is contained
  in the other's.  This is the order-theoretic content behind @{const cast} being
  an order-embedding, phrased for raw deflations.›

lemma deflation_belowI_fixes:
  assumes d: "deflation d" and e: "deflation e"
    and fx: "⋀x. e⋅x = x ⟹ d⋅x = x"
  shows "e ⊑ d"
  using e fx by (rule deflation.belowI)

text ‹The compact fixed points of a deflation, as a set of basis elements.  This set
  is MUB-closed (@{thm [source] deflation_fixes_mub_closed}) and contains @{term ‹⊥›},
  so it satisfies the assumptions of the @{locale mub_closed_basis} locale.›

definition dfixes :: "('a::bifinite → 'a) ⇒ 'a compact_basis set"
  where "dfixes d = {a. d⋅(Rep_compact_basis a) = Rep_compact_basis a}"

lemma mub_closed_dfixes:
  assumes "deflation d"
  shows "mub_closed (dfixes d)"
proof (intro mub_closedI)
  let ?R = Rep_compact_basis
  fix S s assume fin: "finite S" and "S ⊆ dfixes d" "is_mub S s"
  from ‹finite S› ‹is_mub S s› have "is_mub (?R ` S) (?R s)"
    by (rule mub_compact_basis_fwd)
  moreover from ‹S ⊆ dfixes d› have "?R ` S ⊆ {x. d⋅x = x}"
    unfolding dfixes_def by auto
  ultimately have "?R s ∈ {x. d⋅x = x}"
    using deflation_fixes_mub_closed[OF assms, THEN mub_closedD]
      fin[THEN finite_imageI] by fast
  then show "s ∈ dfixes d" by (simp add: dfixes_def)
qed


lemma mub_closed_basis_dfixes:
  assumes "deflation d"
  shows "mub_closed_basis (dfixes d)"
  unfolding mub_closed_basis_def
  using mub_closed_dfixes[OF assms] by blast

lemma finite_dfixes:
  assumes "finite_deflation d"
  shows "finite (dfixes d)"
proof -
  have "inj_on Rep_compact_basis (dfixes d)"
    by (meson compact_basis.Rep_compact_basis_inject inj_onI)
  moreover have "Rep_compact_basis ` dfixes d ⊆ {x. d⋅x = x}"
    unfolding dfixes_def by auto
  moreover have "finite {x. d⋅x = x}" using assms by (rule finite_deflation.finite_fixes)
  ultimately show ?thesis by (meson finite_imageD finite_subset)
qed


text ‹Reconstruction: a finite deflation is recovered from its own compact fixed
  points by the @{locale mub_closed_basis} construction.  Thus ‹dfixes› and
  ‹defl_of_set› are mutually inverse between finite deflations and finite MUB-closed
  sets containing @{term ‹⊥›}.›

lemma defl_of_dfixes:
  assumes "finite_deflation d"
  shows "mub_closed_basis.d (dfixes d) = d"
proof -
  from assms have defld: "deflation d" by (rule finite_deflation_imp_deflation)
  interpret L: mub_closed_basis "dfixes d"
    by (rule mub_closed_basis_dfixes[OF defld])
  have fin: "finite (dfixes d)" using assms by (rule finite_dfixes)
  have defle: "deflation L.d" by (rule L.deflation_d)
  have fde: "finite_deflation L.d" by (rule L.finite_deflation_d[OF fin])
  have eq_fixes: "L.d⋅x = x ⟷ d⋅x = x" for x
  proof
    assume "d⋅x = x"
    moreover have "compact (d⋅x)" by (rule finite_deflation.compact[OF assms])
    ultimately have "compact x" by simp
    then obtain a where xa: "x = Rep_compact_basis a"
      using compact_basis.compact_imp_principal by blast
    from ‹d⋅x = x› xa have "a ∈ dfixes d" unfolding dfixes_def by simp
    thus "L.d⋅x = x" using xa L.d_fixes by simp
  next
    assume "L.d⋅x = x"
    moreover have "compact (L.d⋅x)" by (rule finite_deflation.compact[OF fde])
    ultimately have "compact x" by simp
    then obtain a where xa: "x = Rep_compact_basis a"
      using compact_basis.compact_imp_principal by blast
    from ‹L.d⋅x = x› xa have "a ∈ dfixes d" using L.d_fixes by simp
    thus "d⋅x = x" using xa unfolding dfixes_def by simp
  qed
  show "L.d = d"
    by (rule below_antisym; rule deflation_belowI_fixes) (simp_all add: eq_fixes defld defle)
qed

text ‹The MUB-closure of any set containing @{term ‹⊥›} satisfies the locale
  assumptions, so it determines a deflation.›

lemma mub_closed_basis_mub_closure:
  shows "mub_closed_basis (mub_closure T)"
  unfolding mub_closed_basis_def
  using mub_closed_mub_closure by blast

text ‹Pairwise join of finite deflations: the deflation determined by the MUB-closure
  of the union of their fixed-point sets.  By the lattice-embedding (fixed points
  ordered by inclusion) this is the least upper bound.›

definition join_fin_defl :: "'a::bifinite fin_defl ⇒ 'a fin_defl ⇒ 'a fin_defl"
  where "join_fin_defl A B =
    Abs_fin_defl (mub_closed_basis.d
      (mub_closure (dfixes (Rep_fin_defl A) ∪ dfixes (Rep_fin_defl B))))"

lemma join_fin_defl_aux:
  fixes A B :: "'a::bifinite fin_defl"
  defines "U ≡ mub_closure (dfixes (Rep_fin_defl A) ∪ dfixes (Rep_fin_defl B))"
  shows "mub_closed_basis U" (is ?P1)
    and "finite U" (is ?P2)
    and "finite_deflation (mub_closed_basis.d U)" (is ?P3)
    and "Rep_fin_defl (join_fin_defl A B) = mub_closed_basis.d U" (is ?P4)
proof -
  have finA: "finite (dfixes (Rep_fin_defl A))"
    by (rule finite_dfixes[OF finite_deflation_Rep_fin_defl])
  have finB: "finite (dfixes (Rep_fin_defl B))"
    by (rule finite_dfixes[OF finite_deflation_Rep_fin_defl])
  show P1: ?P1 unfolding U_def by (rule mub_closed_basis_mub_closure)
  show P2: ?P2 unfolding U_def
    using finite_mub_closure_cb[of "dfixes (Rep_fin_defl A) ∪ dfixes (Rep_fin_defl B)"]
      finA finB by simp
  show P3: ?P3 by (rule mub_closed_basis.finite_deflation_d[OF P1 P2])
  show ?P4 unfolding join_fin_defl_def U_def[symmetric]
    by (metis P3 Abs_fin_defl_inverse mem_Collect_eq)
qed

lemma dfixes_mub_closed_basis_d:
  assumes "mub_closed_basis S"
  shows "dfixes (mub_closed_basis.d S) = S"
  unfolding dfixes_def using mub_closed_basis.d_fixes[OF assms] by auto

text ‹The order on finite deflations is inclusion of their fixed-point sets — the
  promised lattice embedding into the MUB-closed sets.›

lemma fin_defl_below_dfixes:
  fixes A B :: "'a::bifinite fin_defl"
  shows "A ⊑ B ⟷ dfixes (Rep_fin_defl A) ⊆ dfixes (Rep_fin_defl B)"
proof
  assume "A ⊑ B"
  thus "dfixes (Rep_fin_defl A) ⊆ dfixes (Rep_fin_defl B)"
    unfolding dfixes_def using fin_defl_belowD by auto
next
  assume sub: "dfixes (Rep_fin_defl A) ⊆ dfixes (Rep_fin_defl B)"
  show "A ⊑ B"
  proof (rule fin_defl_belowI)
    fix x assume "Rep_fin_defl A⋅x = x"
    moreover have "compact (Rep_fin_defl A⋅x)"
      by (rule finite_deflation.compact[OF finite_deflation_Rep_fin_defl])
    ultimately have "compact x" by simp
    then obtain a where xa: "x = Rep_compact_basis a"
      using compact_basis.compact_imp_principal by blast
    from ‹Rep_fin_defl A⋅x = x› xa have "a ∈ dfixes (Rep_fin_defl A)"
      unfolding dfixes_def by simp
    with sub have "a ∈ dfixes (Rep_fin_defl B)" by blast
    thus "Rep_fin_defl B⋅x = x" using xa unfolding dfixes_def by simp
  qed
qed

lemma dfixes_join_fin_defl:
  fixes A B :: "'a::bifinite fin_defl"
  shows "dfixes (Rep_fin_defl (join_fin_defl A B)) =
         mub_closure (dfixes (Rep_fin_defl A) ∪ dfixes (Rep_fin_defl B))"
  using join_fin_defl_aux(4)[of A B]
    dfixes_mub_closed_basis_d[OF join_fin_defl_aux(1)[of A B]] by simp

lemma join_fin_defl_above1: "A ⊑ join_fin_defl A B"
  unfolding fin_defl_below_dfixes dfixes_join_fin_defl
  using mub_closure_extensive by blast

lemma join_fin_defl_above2: "B ⊑ join_fin_defl A B"
  unfolding fin_defl_below_dfixes dfixes_join_fin_defl
  using mub_closure_extensive by blast

lemma join_fin_defl_least:
  assumes "A ⊑ C" and "B ⊑ C"
  shows "join_fin_defl A B ⊑ C"
proof -
  have "dfixes (Rep_fin_defl A) ∪ dfixes (Rep_fin_defl B) ⊆ dfixes (Rep_fin_defl C)"
    using assms unfolding fin_defl_below_dfixes by blast
  hence "mub_closure (dfixes (Rep_fin_defl A) ∪ dfixes (Rep_fin_defl B))
           ⊆ dfixes (Rep_fin_defl C)"
    by (rule mub_closure_least[OF _ mub_closed_dfixes[OF deflation_Rep_fin_defl]])
  thus ?thesis unfolding fin_defl_below_dfixes dfixes_join_fin_defl .
qed

section ‹Algebraic deflations are determined by their compact fixes›

text ‹@{const cast} is injective but not surjective onto all deflations: only a
  deflation whose image is bifinite is the lub of the finite deflations below it.
  Still, an algebraic deflation @{term ‹cast⋅A›} is the directed lub of the finite
  deflations in its defining ideal, so it is pinned down by which COMPACT elements
  it fixes, i.e. by @{term ‹dfixes (cast⋅A)›}.›

text ‹Each finite deflation in the defining ideal of @{term A} lies below
  @{term ‹cast⋅A›}.›

lemma Rep_fin_defl_below_cast:
  assumes "a ∈ Rep_defl A"
  shows "Rep_fin_defl a ⊑ cast⋅A"
  using assms by (metis cast_defl_principal cast_below_cast defl.principal_below_iff_mem_rep)

text ‹Larger algebraic deflations fix more compacts (this direction is just
  monotonicity of @{term cast}).›

lemma dfixes_cast_mono:
  assumes "A ⊑ B"
  shows "dfixes (cast⋅A) ⊆ dfixes (cast⋅B)"
proof
  fix b assume "b ∈ dfixes (cast⋅A)"
  hence fx: "cast⋅A⋅(Rep_compact_basis b) = Rep_compact_basis b"
    by (simp add: dfixes_def)
  from assms have "cast⋅A ⊑ cast⋅B" by (rule monofun_cfun_arg)
  hence "cast⋅A⋅(Rep_compact_basis b) ⊑ cast⋅B⋅(Rep_compact_basis b)"
    by (rule monofun_cfun_fun)
  with fx have le: "Rep_compact_basis b ⊑ cast⋅B⋅(Rep_compact_basis b)" by simp
  have "cast⋅B⋅(Rep_compact_basis b) = Rep_compact_basis b"
  proof (rule below_antisym)
    show "cast⋅B⋅(Rep_compact_basis b) ⊑ Rep_compact_basis b" by (rule cast.below)
    show "Rep_compact_basis b ⊑ cast⋅B⋅(Rep_compact_basis b)" by (rule le)
  qed
  thus "b ∈ dfixes (cast⋅B)" by (simp add: dfixes_def)
qed

text ‹The converse is the substantive direction: an inclusion of compact fixes
  forces the deflation order.  We test each finite deflation @{term ‹Rep_fin_defl a›}
  in the ideal of @{term A} against @{term ‹cast⋅B›} using
  @{thm [source] finite_deflation.compact_belowI}, which only inspects compact
  fixed points.›

lemma defl_below_of_dfixes:
  assumes sub: "dfixes (cast⋅A) ⊆ dfixes (cast⋅B)"
  shows "A ⊑ B"
proof (unfold below_defl_def, rule subsetI)
  fix a assume aA: "a ∈ Rep_defl A"
  have ra: "Rep_fin_defl a ⊑ cast⋅A" using aA by (rule Rep_fin_defl_below_cast)
  have "Rep_fin_defl a ⊑ cast⋅B"
  proof (rule Rep_fin_defl.compact_belowI)
    fix x assume cx: "compact x" and fx: "Rep_fin_defl a⋅x = x"
    have caAx: "cast⋅A⋅x = x"
      using fx ra cast.belowD by fastforce
    let ?b = "Abs_compact_basis x"
    from cx have rb: "Rep_compact_basis ?b = x" by simp
    with caAx have "cast⋅A⋅(Rep_compact_basis ?b) = Rep_compact_basis ?b" by simp
    hence "?b ∈ dfixes (cast⋅A)" by (simp add: dfixes_def)
    with sub have "?b ∈ dfixes (cast⋅B)" ..
    hence "cast⋅B⋅(Rep_compact_basis ?b) = Rep_compact_basis ?b"
      by (simp add: dfixes_def)
    with rb show "cast⋅B⋅x = x" by simp
  qed
  hence "cast⋅(defl_principal a) ⊑ cast⋅B" by (simp add: cast_defl_principal)
  hence "defl_principal a ⊑ B" by (rule cast_below_imp_below)
  thus "a ∈ Rep_defl B" by (simp add: defl.principal_below_iff_mem_rep)
qed

text ‹Hence the compact-fixes map is an order embedding of @{typ ‹'a defl›} into
  sets of compact basis elements, and in particular injective: an algebraic
  deflation is completely determined by the compacts it fixes.›

theorem defl_below_iff_dfixes:
  "A ⊑ B ⟷ dfixes (cast⋅A) ⊆ dfixes (cast⋅B)"
  using dfixes_cast_mono defl_below_of_dfixes by blast

theorem defl_eq_iff_dfixes:
  "A = B ⟷ dfixes (cast⋅A) = dfixes (cast⋅B)"
  by (metis below_antisym defl_below_iff_dfixes set_eq_subset)
end
