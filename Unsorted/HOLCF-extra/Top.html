Theory Top

theory Top imports HOLCF begin

datatype 'a::type top = Of (top_down: 'a) | Top

instantiation top :: (below) below
begin
definition below_top_def:
"(⊑) a b ≡ case b of Top ⇒ True |
  Of b ⇒ (case a of Top ⇒ False | Of a ⇒ a ⊑ b)"

instance ..

theorem below_Top[simp]: "x ⊑ Top" by (simp add: below_top_def)

theorem Top_below[simp]: "Top ⊑ x ⟷ x = Top" by (cases x) (simp_all add: below_top_def)

theorem below_Of: "Of x ⊑ Of y ⟷ x ⊑ y" by (simp add: below_top_def)

end

instance top :: (po) po
  by (intro_classes)
   (auto simp add: below_top_def 
     intro: below_trans below_antisym
     split: top.splits)

lemma not_Top: assumes "x ≠ Top" obtains y where "x = Of y"
using assms by (cases x) simp_all

lemma ch2ch_top:
  assumes "chain S" 
    "Top ∉ range S"
  obtains T where "chain T" "S = (λi. Of (T i))"
proof -
  from ‹Top ∉ range S›[THEN notE, OF range_eqI] have "S i ≠ Top" for i by metis
  then obtain T where T: "S = (λi. Of (T i))" using not_Top by meson
  then have "chain T" using ‹chain S›[THEN chainE, unfolded below_top_def] 
    by (auto intro: chainI)
  from this T show ?thesis by (rule that)
qed

lemma top_chlub_Of: assumes "chain (T::nat⇒'a::cpo)"
  shows "range (λi. Of (T i)) <<| Of (⨆i. T i)"
proof (intro is_lubI is_ubI)
  show "x ∈ range (λi. Of (T i)) ⟹ x ⊑ Of (⨆i. T i)"
    for x by (auto simp add: is_ub_thelub[OF assms] below_Of)
  show "range (λi. Of (T i)) <| u ⟹ Of (⨆i. T i) ⊑ u" for u
    apply (induction u)
     apply (fastforce intro: lub_below[OF assms] dest: is_ubD simp add: below_Of)
    apply simp
    .
qed

lemma top_lub_Top:
"Top ∈ X ⟹ X <<| Top"
  by (intro is_lubI is_ubI) (auto dest: is_ubD)

lemma lub_Top_iff: "range (S::nat⇒'a::cpo top) <<| Top ⟷ Top ∈ range S" if "chain S"
proof (cases "Top ∈ range S")
  case True
  then show ?thesis by (simp add: top_lub_Top)
next
  case False
  from ‹chain S› this obtain T where "chain T" "S = (λi. Of (T i))"
    by (rule ch2ch_top)
  then have "range S <<| Of (⨆i. T i)" by (blast intro: top_chlub_Of)
  with ‹Top ∉ range S› show ?thesis by (blast dest: is_lub_unique)
qed

instantiation top :: (cpo) cpo
begin

instance
proof
  fix S :: "nat ⇒ 'a top" assume "chain S"
  show "∃x. range S <<| x"
  proof (cases "Top ∈ range S")
    case True
    then show ?thesis by (blast intro: top_lub_Top)
  next
    case False
    from ‹chain S› this show ?thesis by (blast elim: ch2ch_top intro: top_chlub_Of)
  qed
qed

theorem compact_Top: "compact (Top :: 'a top)"
  unfolding compact_def
proof (rule admI)
  fix Y :: "nat ⇒ 'a top" assume "chain Y"
  assume "∀i. ¬(Top ⊑ Y i)"
  then have "⋀i. Y i ≠ Top" by simp
  then have "Top ∉ range Y" by (metis rangeE)
  from ‹chain Y› this have "(⨆i. Y i) ≠ Top"
    using cpo_lubI lub_Top_iff by metis
  then show "¬(Top ⊑ (⨆i. Y i))" by simp
qed

end

instantiation top :: (pcpo) pcpo
begin

lemma of_bot_below: "Of ⊥ ⊑ y" by (cases y) (simp_all add: below_Of)

instance by (intro_classes, intro exI[of _ "Of ⊥"] allI of_bot_below)


theorem top_bot: "(⊥::'a top) = Of ⊥"
  by (rule bottomI[symmetric]) (rule of_bot_below)

end

definition "down ≡ (Λ x. Of x)"

theorem cont_Of[simp]: "cont Of"
  by (rule contI) (rule top_chlub_Of)

theorem down_beta: "down⋅x = Of x"
  unfolding down_def by simp
theorem top_cases[case_names down Top, cases type: top]:
  "(⋀p. x = down⋅p ⟹ P) ⟹ (x = Top ⟹ P) ⟹ P"
  by (cases x) (simp_all add: down_beta)

theorem down_inject: "down⋅x = down⋅y ⟹ x = y"
  by (simp add: down_beta)

lemma top_chain_cases [consumes 1]:
  fixes Y :: "nat ⇒ 'a::cpo top"
  assumes "chain Y"
  obtains (down) X where "chain X" "Y = (λi. down⋅(X i))" "(⨆i. Y i) = down⋅(⨆i. X i)"
  | (Top) "finite_chain Y" "∃i. Y i = Top" "(⨆i. Y i) = Top"
proof (cases "Top ∈ range Y")
  assume "Top ∈ range Y"
  then have 3: "(⨆i. Y i) = Top" using lub_Top_iff[OF assms] lub_eqI by blast
  with assms ‹Top ∈ range Y› have 1: "finite_chain Y"
    using finite_chain_def maxinch_is_thelub by fastforce
  from ‹Top ∈ range Y›[THEN rangeE] have 2: "∃i. Y i = Top" by metis
  from 1 2 3 show thesis by (rule that(2))
next
  assume "Top ∉ range Y"
  from ch2ch_top[OF assms this] obtain X where 1: "chain X" and 2: "Y = (λi. Of (X i))" .
  then have "range Y <<| Of (⨆i. X i)" using top_chlub_Of by blast
  then have 3: "(⨆i. Y i) = Of (⨆i. X i)" by (rule lub_eqI)
  from 1 2 3 show thesis by (rule that(1)[unfolded down_beta])
qed

definition "cmap_top ≡ (Λ f x. map_top (λx. f⋅x) x)"

lemma cmap_top_cont1[cont2cont]: "cont (map_top f)" if "cont f"
proof (rule contI)
  fix Y :: "nat ⇒ 'a top" assume "chain Y"
  then show "range (λi. map_top f (Y i)) <<| map_top f (⨆i. Y i)"
  proof (cases rule: top_chain_cases)
    case (down X)
    from that ‹chain X› have fX: "chain (λi. f (X i))" by (rule ch2ch_cont)
    from ‹Y = (λi. down⋅(X i))› have "(λi. map_top f (Y i)) = (λi. Of (f (X i)))"
      by (simp add: down_beta)
    then have "range (λi. map_top f (Y i)) <<| Of (⨆i. f (X i))" using top_chlub_Of[OF fX]
      by presburger
    also have "(⨆i. f (X i)) = f (⨆i. X i)" using that ‹chain X› by (rule cont2contlubE[symmetric])
    also from ‹(⨆i. Y i) = down⋅(⨆i. X i)› have "Of ... = map_top f (⨆i. Y i)"
      by (simp add: down_beta)
    finally show ?thesis .
  next
    case Top
    from ‹finite_chain Y› obtain i where "max_in_chain i Y" by (rule finite_chainE)
    with ‹(⨆i. Y i) = Top› have "Y i = Top" using maxinch_is_thelub[OF ‹chain Y›]
      by simp
    then have "map_top f (Y i) = Top" by simp
    then have "Top ∈ range (λi. map_top f (Y i))" using range_eqI by metis
    then have "... <<| Top" by (rule top_lub_Top)
    also have "Top = map_top f (⨆i. Y i)" using ‹(⨆i. Y i) = Top› by simp
    finally show ?thesis .
  qed
qed

lemma cmap_top_cont2[cont2cont]: "cont (λf. map_top (λx. f⋅x) x)"
  by (cases x) (auto simp add: down_beta intro: cont_compose)

lemma cmap_top_beta: "cmap_top⋅f⋅x = map_top (λx. f⋅x) x"
  unfolding cmap_top_def
  by (simp add: cmap_top_cont1 cmap_top_cont2 cont2cont_LAM)

lemma cmap_top_down[simp]: "cmap_top⋅f⋅(down⋅x) = down⋅(f⋅x)" by (simp add: cmap_top_beta down_beta)
lemma cmap_top_Top[simp]: "cmap_top⋅f⋅Top = Top" by (simp add: cmap_top_beta)
lemma cmap_top_ID[simp]: "cmap_top⋅ID = ID"
proof (rule cfun_eqI)
  show "cmap_top⋅ID⋅x = ID⋅x" for x
    by (cases x) simp_all
qed

lemma down_strict: "down⋅⊥ = ⊥" by (simp add: down_beta top_bot)

lemma down_below_of: "down⋅x ⊑ down⋅y ⟹ x ⊑ y" by (simp add: down_beta below_Of)
lemma down_below_iff[iff]: "down⋅x ⊑ down⋅y ⟷ x ⊑ y" by (rule iffI) (auto intro: monofun_cfun_arg down_below_of)

lemma down_ne_Top[simp]: "down⋅x ≠ Top" "Top ≠ down⋅x" by (simp_all add: down_beta)

lemma approx_chain_top:
  assumes "approx_chain a"
  shows "approx_chain (λi. cmap_top⋅(a i))"
proof -
  interpret approx_chain a by fact
  show ?thesis
  proof
    show "chain (λi. cmap_top⋅(a i))" by (simp add: ch2ch_Rep_cfunR)
    show "(⨆i. cmap_top⋅(a i)) = ID" by (simp add: contlub_cfun_arg[symmetric])
    show "cmap_top⋅(a i)⋅(cmap_top⋅(a i)⋅x) = cmap_top⋅(a i)⋅x" for i x
      by (cases x) (simp_all add: approx_idem)
    show "cmap_top⋅(a i)⋅x ⊑ x" for i x by (cases x) (simp_all add: approx_below)
    show "finite {x. cmap_top⋅(a i)⋅x = x}" for i
    proof -
      have "{x. cmap_top⋅(a i)⋅x = x} = insert Top {down⋅x | x. (a i)⋅x = x}"
        (is "?l = ?r")
      proof (rule set_eqI)
        fix x show "x ∈ ?l ⟷ x ∈ ?r"
          by (cases x) (auto dest: down_inject)
      qed
      also have "finite ..."
        using finite_deflation.finite_fixes[OF finite_deflation_approx]
        by simp
      finally show ?thesis.
    qed
  qed
qed

instance top :: (bifinite) bifinite
  by intro_classes (blast intro: bifinite[THEN exE, THEN approx_chain_top])

definition Iftop where "Iftop t f x ≡ case x of Top ⇒ t | Of x ⇒ f⋅x"

lemma Iftop_down: "Iftop t f (down⋅x) = f⋅x" by (simp add: Iftop_def down_beta)

lemma Iftop_top: "Iftop t f Top = t" by (simp add: Iftop_def)

lemma cont_Iftop: "cont (λx. Iftop t f x)" if t_is_top: "⋀x. x ⊑ t"
proof (rule contI)
  fix Y :: "nat ⇒ 'b top"  assume "chain Y"
  then show "range (λi. Iftop t f (Y i)) <<|
         Iftop t f (⨆i. Y i)"
  proof (cases Y rule: top_chain_cases)
    case (down X)
    from ‹Y = (λi. down⋅(X i))› have *:"(λi. Iftop t f (Y i)) = (λi. f⋅(X i))"
      by (simp add: Iftop_down)
    from ‹(⨆i. Y i) = down⋅(⨆i. X i)› have **: "Iftop t f (⨆i. Y i) = f⋅(⨆i. X i)"
      by (simp add: Iftop_down)
    from ‹chain X› show ?thesis  unfolding * **
      by (intro contE) auto
  next
    case Top
    from ‹finite_chain Y› obtain i where "max_in_chain i Y" by (rule finite_chainE)
    with ‹(⨆i. Y i) = Top› have "Top = Y i" using maxinch_is_thelub[OF ‹chain Y›] by simp
    then have "t = Iftop t f (Y i)" by (simp add: Iftop_top)
    then have "t ∈ range (λi. Iftop t f (Y i))" by (rule range_eqI)
    then have "range (λi. Iftop t f (Y i)) <<| t"
      unfolding is_lub_def is_ub_def using t_is_top by blast
    moreover have "t = Iftop t f (⨆i. Y i)" by (simp add: ‹(⨆i. Y i) = Top› Iftop_top)
    ultimately show ?thesis by presburger
  qed
qed


definition up_top_to_top_up :: "'a::cpo u top → 'a top u"
  where "up_top_to_top_up = (Λ x. Iftop (up⋅Top) (Λ (up⋅x). up⋅(down⋅x)) x)"

lemma utttu_beta: "up_top_to_top_up⋅x = Iftop (up⋅Top) (Λ (up⋅x). up⋅(down⋅x)) x"
  unfolding up_top_to_top_up_def
proof (rule beta_cfun, rule cont_Iftop)
  show "x ⊑ up⋅Top" for x by (cases x) simp_all
qed

definition top_up_to_up_top :: "'a::cpo top u → 'a u top"
  where "top_up_to_up_top = (Λ (up⋅x). Iftop Top (down oo up) x)"

lemma tutut_beta_minor: "(Λ x. Iftop Top (down oo up) x)⋅x = Iftop Top (down oo up) x"
  by (rule beta_cfun, rule cont_Iftop, rule below_Top)

theorem utttu_cases:
  "up_top_to_top_up⋅Top = up⋅Top"
  "up_top_to_top_up⋅(down⋅⊥) = ⊥"
  "up_top_to_top_up⋅(down⋅(up⋅x)) = up⋅(down⋅x)"
  by (simp_all add: utttu_beta Iftop_top Iftop_down)

theorem tutut_cases:
  "top_up_to_up_top⋅(up⋅Top) = Top"
  "top_up_to_up_top⋅(⊥) = down⋅⊥"
  "top_up_to_up_top⋅(up⋅(down⋅x)) = down⋅(up⋅x)"
  unfolding top_up_to_up_top_def
  by (simp_all add: tutut_beta_minor Iftop_top Iftop_down down_strict)

lemma top_up_iso1: "up_top_to_top_up⋅(top_up_to_up_top⋅x) = x"
proof -
  consider "x = up⋅Top" | y where "x = up⋅(down⋅y)" | "x = ⊥"
    by (metis top_cases upE)
  then show ?thesis by cases (simp_all add: utttu_cases tutut_cases)
qed

lemma top_up_iso2: "top_up_to_up_top⋅(up_top_to_top_up⋅x) = x"
proof -
  consider "x = Top" | y where "x = down⋅(up⋅y)" | "x = down⋅⊥"
    by (metis top_cases upE)
  then show ?thesis by cases (simp_all add: utttu_cases tutut_cases)
qed

instance top :: (profinite) profinite
proof
  from profinite obtain a :: "nat ⇒ 'a u → 'a u" where "approx_chain a" ..
  note this = approx_chain_top[OF this]
  from approx_chain_iso[OF this top_up_iso1 top_up_iso2]
  show "∃ (a :: nat ⇒ 'a top u → 'a top u). approx_chain a"
    by fast
qed
