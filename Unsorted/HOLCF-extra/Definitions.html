lemma obtain_compact_chain:
  fixes x :: "'a::profinite"
  obtains X where "chain X" "∀i. compact (X i)" "x = (⨆i. X i)"
proof -
  from compact_basis.obtain_principal_chain
  obtain B' where B': "∀i. B' i ⊑ B' (Suc i)" "up⋅x = (⨆i. Rep_compact_basis (B' i))"
    .
  define X' where "X' i = Rep_compact_basis (B' i)" for i
  have "chain X'" unfolding X'_def using B'(1) by (rule compact_basis.ch2ch_principal)
  have upx: "up⋅x = (⨆i. X' i)" unfolding X'_def by fact
  have "compact (X' i)" for i unfolding X'_def by simp
  obtain X n where upX: "∀i. up⋅(X i) = X' (i + n)"
    and "chain X" and x: "x = (⨆i. X i)"
    using ‹chain X'›
  proof (cases rule: up_chain_cases)
    case 1
    then have "(⨆i. X' i) = ⊥" by simp
    with upx show ?thesis by simp
  next
    case (2 X n)
    with upx show ?thesis using that by simp
  qed
  from upX ‹⋀i. compact (X' i)› compact_upD have "∀i. compact (X i)" 
    by metis
  from ‹chain X› this x show ?thesis by (rule that)
qed

lemma compact_induct':
  fixes x :: "'a::profinite"
  assumes adm: "adm P"
  assumes compact: "⋀xx. compact xx ⟹ xx ⊑ x ⟹ P xx"
  shows "P x"
proof -
  obtain X where X: "chain X" "∀i. compact (X i)" "x = (⨆i. X i)"
    by (rule obtain_compact_chain)
  from X(1, 3) have "⋀i. X i ⊑ x" using is_ub_thelub by blast
  from X(2) this have "⋀i. P (X i)" using compact by blast
  with adm[THEN admD, OF X(1)] X(3) show "P x" by presburger
qed

subsection ‹Step functions›

text ‹The step function ‹stepfn c y› is ‹y› above ‹c› and ‹⊥› elsewhere; it is continuous when
  ‹c› is compact.  It lies below @{term f} exactly when ‹y ⊑ f⋅c›.›

definition "stepfn c y ≡ (Λ x. if c ⊑ x then y else ⊥)"

lemma monofun_stepfn: "monofun (λx::'a::cpo. if c ⊑ x then y else ⊥)"
  by (rule monofunI) (auto dest: below_trans)

lemma cont_stepfn: "cont (λx::'a::cpo. if c ⊑ x then y else ⊥)" (is "cont ?f") if "compact c"
proof (rule contI2)
  show "monofun ?f" by (rule monofun_stepfn)
  fix Y :: "nat ⇒ 'a" assume "chain Y" and chain_F: "chain (λi. ?f (Y i))"
  show "?f (⨆i. Y i) ⊑ (⨆i. ?f (Y i))"
  proof (cases "c ⊑ (⨆i. Y i)")
    case True
    with ‹compact c› ‹chain Y› have "∃i. c ⊑ Y i" by (rule compactD2)
    then obtain i where i: "c ⊑ Y i" ..
    from i ‹chain Y›[THEN chain_mono] have Y_mono: "i ≤ j ⟹ c ⊑ Y j" for j
      by (rule below_trans)
    from i have 1: "?f (Y i) = y" by simp
    with Y_mono have "i ≤ j ⟹ ?f (Y i) = ?f (Y j)" for j by simp
    then have max: "max_in_chain i (λi. ?f (Y i))" by (rule max_in_chainI)
    from maxinch_is_thelub[OF chain_F] this 1 have 1: "(⨆i. ?f (Y i)) = y"
      by presburger
    from True have 2: "?f (⨆i. Y i) = y" by simp
    show ?thesis unfolding 1 2 by (rule below_refl)
  next
    case False
    then have 1: "?f (⨆i. Y i) = ⊥" by simp
    show ?thesis unfolding 1 by (rule minimal)
  qed
qed

lemma stepfn_beta: "stepfn c y⋅x = (if c ⊑ x then y else ⊥)" if "compact c"
  unfolding stepfn_def
  by (rule beta_cfun) (rule cont_stepfn[OF that])

lemma stepfn_below_iff: assumes "compact c" shows "stepfn c y ⊑ f ⟷ y ⊑ f⋅c"
  unfolding cfun_below_iff stepfn_beta[OF assms]
  apply (simp only: minimal split: if_splits)
  by simp (blast intro: monofun_cfun_arg[of c _ f] below_trans[of y "f⋅c"])
