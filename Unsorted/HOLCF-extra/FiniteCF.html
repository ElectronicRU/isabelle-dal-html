lemma (in po) finite_has_maximal:
  fixes A :: "'a set"
  shows "⟦finite A; A ≠ {}⟧ ⟹ ∃x∈A. ∀y∈A. x ⊑ y ⟶ x = y"
proof (induct rule: finite_ne_induct)
  case (singleton x)
    show ?case by simp
next
  case (insert a A)
  from ‹∃x∈A. ∀y∈A. x ⊑ y ⟶ x = y›
  obtain x where x: "x ∈ A"
           and x_eq: "⋀y. ⟦y ∈ A; x ⊑ y⟧ ⟹ x = y" by fast
  show ?case
  proof (intro bexI ballI impI)
    fix y
    assume "y ∈ insert a A" and "(if x ⊑ a then a else x) ⊑ y"
    thus "(if x ⊑ a then a else x) = y"
      by (auto split: if_splits intro: below_antisym dest: below_trans x_eq)
  next
    show "(if x ⊑ a then a else x) ∈ insert a A"
      by (simp add: x)
  qed
qed

lemma (in po) smyth_ub_mono:
  assumes "∀x ∈ A. ∃y ∈ B. x ⊑ y"
  "B <| b"
shows "A <| b"
  using assms by (blast intro: is_ubI below_trans dest: is_ubD)

lemma (in po) lub_mono_general:
  assumes "A <<| a" "B <<| b" and smyth_ord: "∀x ∈ A. ∃y ∈ B. x ⊑ y"
  shows "a ⊑ b"
proof -
  from ‹B <<| b› have "B <| b" by (rule is_lubD1)
  with smyth_ord have "A <| b" by (rule smyth_ub_mono)
  with ‹A <<| a› show "a ⊑ b" by (rule is_lubD2)
qed

lemma (in po) ch2ch_lub_gen:
  assumes lub: "⋀i. J i <<| Y i"
    and smyth_ord: "⋀i. ∀x ∈ J i. ∃y ∈ J (Suc i). x ⊑ y"
  shows "chain Y"
  by (rule chainI) (rule lub_mono_general[OF lub lub smyth_ord])


theorem ch2ch_lub:
  assumes chain: "⋀j. j ∈ J ⟹ chain (λi. Y i j)"
    and lub: "⋀i. Y i ` J <<| lub (Y i ` J)"
  shows "chain (λi. lub (Y i ` J))"
proof -
  have "Y (Suc i) ` J <| x ⟹ Y i ` J <| x" for i x
  proof (rule ub_imageI, drule ub_imageD)
    fix j assume "j ∈ J"
    then have "chain (λi. Y i j)" by (rule chain)
    then have "Y i j ⊑ Y (Suc i) j" by (rule chainE)
    also assume "Y (Suc i) j ⊑ x"
    finally show "Y i j ⊑ x" .
  qed
  from lub[THEN is_lubD2, OF this, OF lub[THEN is_lubD1]] show ?thesis
    by (rule chainI)
qed

theorem chain_lub_family:
  fixes Y :: "nat ⇒ 'a::type ⇒ 'b::cpo"
  assumes chain: "⋀j. j ∈ J ⟹ chain (λi. Y i j)"
    and lub: "⋀i. Y i ` J <<| lub (Y i ` J)"
  defines "L ≡ (⨆i. lub (Y i ` J))"
  shows "(λj. ⨆i. Y i j) ` J <<| L"
proof (rule is_lubI)
  have "chain (λi. lub (Y i ` J))" using chain lub by (rule ch2ch_lub)
  then have lub_L: "range (λi. lub (Y i ` J)) <<| L"
    unfolding L_def by (rule cpo_lubI)
  show "(λj. ⨆i. Y i j) ` J <| L"
  proof (rule ub_imageI)
    fix j assume j: "j ∈ J"
    {
      fix i
      have "Y i j ⊑ lub (Y i ` J)"
        using lub[of i, THEN is_lubD1, THEN ub_imageD, OF j] .
      also have "... ⊑ L"
        using lub_L[THEN is_lubD1, THEN ub_imageD] by blast
      finally have "Y i j ⊑ L" .
    }
    from chain[OF j] this show "(⨆i. Y i j) ⊑ L"
      by (rule lub_below)
  qed
  show "(λj. ⨆i. Y i j) ` J <| u ⟹ L ⊑ u" for u
  proof -
    assume ub: "(λj. ⨆i. Y i j) ` J <| u"
    {
      fix j assume j: "j ∈ J"
      from chain[OF j] have "Y i j ⊑ (⨆i. Y i j)" for i by (rule is_ub_thelub)
      also assume "(⨆i. Y i j) ⊑ u"
      finally have "Y i j ⊑ u" for i .
    }
    from this ub[THEN ub_imageD] have "Y i j ⊑ u" if "j ∈ J" for i j using that by blast
    from lub[THEN is_lubD2, OF ub_imageI, OF this] have "lub (Y i ` J) ⊑ u" for i by simp
    from lub_L[THEN is_lubD2, OF ub_imageI, OF this] show "L ⊑ u".
  qed
qed

corollary chain_lub_family_converse:
  fixes Y :: "nat ⇒ 'a::type ⇒ 'b::cpo"
  assumes chain: "⋀j. j ∈ J ⟹ chain (λi. Y i j)"
    and lub: "⋀i. Y i ` J <<| lub (Y i ` J)"
  shows "range (λi. lub (Y i ` J)) <<| (⨆j∈J. (⨆i. Y i j))"
proof -
  have "chain (λi. lub (Y i ` J))" using chain lub by (rule ch2ch_lub)
  then have *: "range (λi. lub (Y i ` J)) <<| (⨆i. lub (Y i ` J))"
    by (rule cpo_lubI)
  from chain lub have "((λj. ⨆i. Y i j) ` J) <<| (⨆i. lub (Y i ` J))"
    by (rule chain_lub_family)
  then have "lub ((λj. ⨆i. Y i j) ` J) = (⨆i. lub (Y i ` J))"
    by (rule lub_eqI)
  with * show ?thesis by simp
qed

lemma (in approx_chain) approx_ch2ch_lub:
  assumes "⋀i. (λx. approx i⋅x) ` S <<| (⨆x ∈ S. approx i⋅x)"
  shows "chain (λi. (⨆ x ∈ S. approx i⋅x))"
  by (rule ch2ch_lub) (simp_all add: assms)



lemma (in approx_chain) approx_lub_family:
  assumes "⋀i. (λx. approx i⋅x) ` S <<| (⨆x ∈ S. approx i⋅x)"
  shows "S <<| (⨆i. ⨆x ∈ S. approx i⋅x)"
proof -
  have "S = (λj. ⨆i. approx i⋅j) ` S"
  proof -
    have "(⨆i. approx i⋅j) = j" for j
      using lub_approx chain_approx[THEN contlub_cfun_fun] by simp
    then show ?thesis by simp
  qed
  also have "... <<| (⨆i. ⨆x ∈ S. approx i⋅x)"
    by (intro chain_lub_family assms) simp
  finally show ?thesis .
qed
