Theory Koenigs

theory Koenigs imports Main begin

context
  fixes rel :: "'a ⇒ 'a ⇒ bool" (infix ‹⊑› 50)
begin

primrec winnow :: "nat ⇒ (nat ⇒ 'a set) ⇒ (nat ⇒ 'a set)"
  where winnow_0: "winnow 0 S = S"
  | winnow_suc: "winnow (Suc i) S j = {x ∈ winnow i S j. ∃ y ∈ winnow i S (Suc j). x ⊑ y}"

abbreviation (input) upper_le_set (infix ‹≼♯› 50)
  where "upper_le_set X Y ≡ ∀y ∈ Y. ∃x ∈ X. x ⊑ y"

lemma winnow_chain: assumes "∀i. S i ≼♯ S (Suc i)"
  shows "∀i. winnow n S i ≼♯ winnow n S (Suc i)"
proof (induction n)
  case 0
  then show ?case using assms by simp
next
  case (Suc n)
  then show ?case by fastforce
qed

lemma winnow_ne_empty: assumes ne: "∀i. S i ≠ {}" and chain:  "∀i. S i ≼♯ S (Suc i)"
  shows "∀i. winnow n S i ≠ {}"
proof (induction n)
  case 0
  then show ?case using ne by simp
next
  case (Suc n)
  with winnow_chain[of S n, OF chain]
  show ?case by fastforce
qed

lemma winnow_descending: "winnow (Suc n) S i ⊆ winnow n S i" by simp

lemma winnow_antimono: "antimono (λn. winnow n S i)"
  unfolding antimono_iff_le_Suc by (simp add: winnow_descending)

definition winnow_fix where "winnow_fix S i = (⋂n. winnow n S i)"

theorem descending_finite_chain:
  fixes C :: "nat ⇒ 'a set"
  assumes "∀i. C (Suc i) ⊆ C i"
    and "∀i. C i ≠ {}"
    and "finite (C 0)"
  shows "(⋂i. C i) ≠ {}"
proof
  assume "(⋂n. C n) = {}"
  then have "∃n. x ∉ C n" if "x ∈ C 0" for x by blast
  note Least = this[THEN LeastI_ex]
  have "antimono C" unfolding antimono_iff_le_Suc by fact
  note mono = this[THEN antimonoD]
  define I where "I ≡ {LEAST n. x ∉ C n | x. x ∈ C 0}"
  with ‹finite (C 0)› have "finite I" by simp
  then have "n ≤ Max I" if "n ∈ I" for n by (simp add: that)
  with I_def mono Least have "x ∉ C (Max I)" if "x ∈ C 0" for x
    using that by blast
  with mono ‹∀i. C i ≠ {}› show False by blast
qed

lemma winnow_fix:
  assumes ne: "∀i. S i ≠ {}"
    and finite: "∀i. finite (S i)"
    and chain: "∀i. S i ≼♯ S (Suc i)"
  shows winnow_fix_ne: "winnow_fix S i ≠ {}"
    and winnow_fix_strong_chain: "∀x ∈ winnow_fix S i. ∃y ∈ winnow_fix S (Suc i). x ⊑ y"
proof -
  show fix_ne: "winnow_fix S i ≠ {}"
    unfolding winnow_fix_def
  proof (intro allI descending_finite_chain winnow_descending)
    fix i
    show "winnow n S i ≠ {}" for n using winnow_ne_empty[OF ne chain] by simp
    show "finite (winnow 0 S i)" using finite by simp
  qed
  show "∀x ∈ winnow_fix S i. ∃y ∈ winnow_fix S (Suc i). x ⊑ y"
  proof (intro strip)
    fix x assume "x ∈ winnow_fix S i"
    then have "x ∈ winnow n S i" for n by (simp add: winnow_fix_def)
    then have "x ∈ winnow (Suc n) S i" for n .
    then have 1: "∃y ∈ winnow n S (Suc i). x ⊑ y" for n by simp
    have "(⋂n. {y ∈ winnow n S (Suc i). x ⊑ y}) ≠ {}"
    proof (intro descending_finite_chain allI, clarsimp)
      show "{y ∈ winnow n S (Suc i). x ⊑ y} ≠ {}" for n
        using 1 by blast
      show "finite {y ∈ winnow 0 S (Suc i). x ⊑ y}" by (simp add: finite)
    qed
    then show "∃y ∈ winnow_fix S (Suc i). x ⊑ y"
      unfolding winnow_fix_def by blast
  qed
qed

lemma winnow_fix_subset: "winnow_fix S i ⊆ S i"
  unfolding winnow_fix_def
  by (intro Inter_lower range_eqI[where x = 0]) simp

lemma obtain_infinite_path:
  fixes S :: "nat ⇒ 'a set"
  assumes ne: "⋀i. S i ≠ {}" 
   and finite: "⋀i. finite (S i)" 
   and chain: "⋀i. S i ≼♯ S (Suc i)"
 obtains f where "∀i. f i ∈ S i" and "∀i. f i ⊑ f (Suc i)"
proof -
  from winnow_fix assms
  have "winnow_fix S i ≠ {}" and "∀x ∈ winnow_fix S i. ∃y ∈ winnow_fix S (Suc i). x ⊑ y"
    for i
    by auto
  then have "∃f. ∀i. f i ∈ winnow_fix S i ∧ f i ⊑ f (Suc i)"
    by (intro dependent_nat_choice) fast+
  with winnow_fix_subset show thesis using that by blast
qed

end
