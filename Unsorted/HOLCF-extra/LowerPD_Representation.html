Theory LowerPD_Representation

theory LowerPD_Representation
  imports HOLCF HOLCF.LowerPD Distributive Directed Top

begin

definition lower_mem :: "'a::bifinite ⇒ ('a)♭ ⇒ bool" (infix ‹∈♭› 50)
  where "x ∈♭ X ⟷ {x}♭ ∪♭ X = X"

definition lower_pd_set :: "('a::bifinite)♭ ⇒ 'a set"
where "lower_pd_set X ≡ {x. x ∈♭ X}"

lemma lower_mem_equiv: "x ∈♭ X ⟷ {x}♭ ⊑ X"
  unfolding lower_mem_def by (simp add: lower_join_eq lower_pd_lattice.sup.absorb_iff2)

lemma lower_mem_principal: "x ∈♭ {y}♭ ⟷ x ⊑ y"
  unfolding lower_mem_equiv by simp

corollary lower_principal_rep: "lower_pd_set {y}♭ = {x. x ⊑ y}"
  unfolding lower_mem_principal lower_pd_set_def ..

lemma lower_mem_union: "x ∈♭ X ∪♭ Y ⟷ x ∈♭ X ∨ x ∈♭ Y"
  unfolding lower_mem_equiv
  by (rule lower_unit_below_plus_iff)

corollary lowerset_union: "lower_pd_set (X ∪♭ Y) = lower_pd_set X ∪ lower_pd_set Y"
  unfolding lower_mem_union lower_pd_set_def by blast

definition set_approximants :: "'a::bifinite set ⇒ 'a compact_basis set"
  where "set_approximants X ≡ ⋃x ∈ X. approximants x"

lemma lower_pd_ideal_PDPlus:
  assumes "lower_le.ideal A"
    and "x ∈ A"
    and "y ∈ A"
  shows "PDPlus x y ∈ A"
proof -
  from assms obtain z where xy: "x ≤♭ z" "y ≤♭ z" and z: "z ∈ A"
    by (blast dest: lower_le.idealD2)
  from xy have "PDPlus x y ≤♭ z" by (simp add: lower_le_PDPlus_iff)
  from assms(1) this z show "PDPlus x y ∈ A" by (rule lower_le.idealD3)
qed

definition lower_pd_set_ideal :: "'a::bifinite set ⇒ 'a pd_basis set"
  where "lower_pd_set_ideal A ≡ {Abs_pd_basis B | B. B ⊆ set_approximants A ∧ finite B ∧ B ≠ {}}"

lemma lower_pd_ideal_is_set_ideal:
  assumes ideal: "lower_le.ideal A"
  defines "KA ≡ (⋃P∈A. Rep_pd_basis P)"
  shows "A = {B. Rep_pd_basis B ⊆ KA}"
proof (rule subset_antisym)
  show "A ⊆ {B. Rep_pd_basis B ⊆ KA}" unfolding KA_def by blast
  show "{B. Rep_pd_basis B ⊆ KA} ⊆ A"
  proof (safe)
    fix B assume B: "Rep_pd_basis B ⊆ KA"
    {
      fix k assume "k ∈ Rep_pd_basis B"
      with B obtain P where 1: "P ∈ A" and 2: "k ∈ Rep_pd_basis P"
        unfolding KA_def by auto
      from 2 have "PDUnit k ≤♭ P" unfolding lower_le_def
        by (auto simp add: Rep_PDUnit)
      from ideal this 1 have "PDUnit k ∈ A" by (rule lower_le.idealD3)
    }
    then have base: "k ∈ Rep_pd_basis B ⟹ PDUnit k ∈ A" for k .
    have ind: "Rep_pd_basis C ⊆ Rep_pd_basis B ⟹ C ∈ A" for C
    proof (induction C rule: pd_basis_induct)
      case (PDUnit a)
      then show ?case by (simp add: Rep_PDUnit base)
    next
      case (PDPlus t u)
      then have "t ∈ A" "u ∈ A" by (simp_all add: Rep_PDPlus)
      with ideal show "PDPlus t u ∈ A" by (rule lower_pd_ideal_PDPlus)
    qed
    then show "B ∈ A" by simp
  qed
qed

lemma exists_approx: "∃b. b ∈ approximants (a::'a::bifinite)"
  by (metis compact_basis.ideal_rep below.ideal_def)

lemma set_approximants_ne: "A ≠ {} ⟹ set_approximants A ≠ {}"
proof -
  assume "A ≠ {}"
  then obtain a where a: "a ∈ A" by blast
  obtain b where "b ∈ approximants a" by (rule exE[OF exists_approx])
  with a have "b ∈ set_approximants A" unfolding set_approximants_def by blast
  then show ?thesis by auto
qed

lemma lower_pd_set_ideal_mem: "x ∈ lower_pd_set_ideal A ⟷ Rep_pd_basis x ⊆ set_approximants A"
  unfolding lower_pd_set_ideal_def mem_Collect_eq
proof
  assume "∃B. x = Abs_pd_basis B ∧ B ⊆ set_approximants A ∧ finite B ∧ B ≠ {}"
  then obtain B where x: "x = Abs_pd_basis B" and B: "B ⊆ set_approximants A"
    and dom: "finite B" "B ≠ {}" by blast
  from dom have "B ∈ pd_basis" unfolding pd_basis_def by simp
  then have "Rep_pd_basis x = B" unfolding x by (rule Abs_pd_basis_inverse)

  with B show "Rep_pd_basis x ⊆ set_approximants A" by simp
next
  assume B: "Rep_pd_basis x ⊆ set_approximants A"
  show "∃B. x = Abs_pd_basis B ∧
        B ⊆ set_approximants A ∧ finite B ∧ B ≠ {}"
    by (rule exI[of _ "Rep_pd_basis x"])
      (simp add: B Rep_pd_basis_inverse[of x])
qed

lemma set_approximants_alt: "set_approximants A = {x. ∃y ∈ A. Rep_compact_basis x ⊑ y}"
  unfolding set_approximants_def approximants_def by blast

lemma set_approximants_downward_closed: "y ⊑ x ⟹ x ∈ set_approximants A ⟹ y ∈ set_approximants A"
  unfolding set_approximants_def approximants_def compact_le_def
  using below_trans by blast

lemma lower_pd_set_ideal: assumes "A ≠ {}" shows "lower_le.ideal (lower_pd_set_ideal A)"
proof (rule lower_le.idealI)
  obtain b where "b ∈ set_approximants A" using set_approximants_ne[OF assms] by blast
  then have "Abs_pd_basis {b} ∈ lower_pd_set_ideal A"
    unfolding lower_pd_set_ideal_def by blast
  then show "∃x. x ∈ lower_pd_set_ideal A" by auto
next
  fix x y assume "x ∈ lower_pd_set_ideal A" "y ∈ lower_pd_set_ideal A"
  then have xy: "Rep_pd_basis x ⊆ set_approximants A" "Rep_pd_basis y ⊆ set_approximants A"
    by (simp_all add: lower_pd_set_ideal_mem)
  then have "Rep_pd_basis (PDPlus x y) ⊆ set_approximants A"
    unfolding Rep_PDPlus by simp
  then have *: "PDPlus x y ∈ lower_pd_set_ideal A" by (simp add: lower_pd_set_ideal_mem)
  show "∃z ∈ lower_pd_set_ideal A. x ≤♭ z ∧ y ≤♭ z"
    using * PDPlus_lower_le PDPlus_commute[of x y] by metis
next
  show "x ≤♭ y ⟹ y ∈ lower_pd_set_ideal A ⟹ x ∈ lower_pd_set_ideal A"
    for x y
    unfolding lower_pd_set_ideal_mem lower_le_def
    using set_approximants_downward_closed by blast
qed

definition lower_pd_of_set :: "('a::bifinite) set ⇒ 'a lower_pd"
  where "lower_pd_of_set A = Abs_lower_pd (lower_pd_set_ideal A)"

lemma Rep_lower_pd_of_set: assumes "A ≠ {}" shows "Rep_lower_pd (lower_pd_of_set A) = lower_pd_set_ideal A"
  unfolding lower_pd_of_set_def
  by (rule Abs_lower_pd_inverse) (simp add: lower_pd_set_ideal[OF assms])

definition lowerset :: "'a::po set ⇒ 'a set"
  where "lowerset A = {y. ∃x ∈ A. y ⊑ x}"

lemma lowerset_idem: "lowerset (lowerset A) = lowerset A"
  unfolding lowerset_def using below_trans by blast

definition completion :: "'a::cpo set ⇒ 'a set"
  where "completion A = {(⨆i. S i) | S. chain S ∧ range S ⊆ A}"

definition closure :: "'a::cpo set ⇒ 'a set"
  where "closure A = completion (lowerset A)"

lemma approximant_lower:
  "(x::'a::bifinite compact_basis) ∈ set_approximants A ⟹ Rep_compact_basis x ∈ lowerset A"
  unfolding set_approximants_alt lowerset_def by blast

lemma lower_approximants:
  assumes "x ∈ lowerset A"
  shows "approximants x ⊆ set_approximants A"
proof -
  from assms[unfolded lowerset_def] obtain y where 1: "y ∈ A" and 2: "x ⊑ y"
    by blast
  from 2 have "approximants x ⊆ approximants y" by (rule compact_basis.rep_mono)
  also have "approximants y ⊆ set_approximants A"
    unfolding set_approximants_def using 1 by auto
  finally show ?thesis .
qed

lemma adm_set_approximants: "adm (λx::'a::bifinite. approximants x ⊆ set_approximants A)"
  by (rule admI) (blast dest: compact_basis.rep_lub)

lemma adm_not_set_approximants: "adm (λx::'a::bifinite. ¬(approximants x ⊆ set_approximants A))"
  apply (rule admI)
  unfolding compact_basis.rep_lub by fast

lemma adm_lower_mem: "adm (λx. x ∈♭ A)"
  unfolding lower_mem_equiv
  by auto

lemma adm_not_lower_mem: "adm (λx. ¬(x ∈♭ A))"
  unfolding lower_mem_equiv
  by auto

lemma closure_approximants: "approximants (x::'a::bifinite) ⊆ set_approximants A ⟷
  x ∈ closure A"
proof
  assume "x ∈ closure A"
  from this[unfolded closure_def completion_def]
  obtain S where x: "x = (⨆i. S i)" and chain: "chain S" and ss: "range S ⊆ lowerset A"
    by blast
  from ss lower_approximants have "⋀i. approximants (S i) ⊆ set_approximants A"
    by auto
  then have "approximants (⨆i. S i) ⊆ set_approximants A"
    unfolding compact_basis.rep_lub[OF chain] by auto
  with x show "approximants x ⊆ set_approximants A" by simp
next
  assume *: "approximants x ⊆ set_approximants A"
  obtain approx :: "nat ⇒ 'a → 'a" where "approx_chain approx" 
    by (rule exE[OF bifinite])
  then interpret approx_chain approx .
  define S where "S i = approx i⋅x" for i
  have chain: "chain S" unfolding S_def by auto
  have subset: "range S ⊆ lowerset A" unfolding S_def
  proof (rule subsetI, erule rangeE, hypsubst)
    fix i
    have "Abs_compact_basis (approx i⋅x) ∈ approximants x"
      by (simp add: approximants_def approx_below)
    then have "Abs_compact_basis (approx i⋅x) ∈ set_approximants A"
      by (rule subsetD[OF *])
    then have "Rep_compact_basis (Abs_compact_basis (approx i⋅x)) ∈ lowerset A"
      by (rule approximant_lower)
    then show "approx i⋅x ∈ lowerset A" by simp
  qed
  have lub: "x = (⨆i. S i)" unfolding S_def using lub_approx
    contlub_cfun_fun[OF chain_approx] by simp
  from lub chain subset show "x ∈ closure A"
    unfolding closure_def completion_def
    by auto
qed

lemma Abs_lower_pd_principal:
  "Rep_lower_pd (lower_principal x) = {u. u ≤♭ x}"
  unfolding lower_principal_def 
  by (intro Abs_lower_pd_inverse CollectI lower_le.ideal_principal)

lemma compact_pd_set_ideal:
  assumes "A ≠ {}"
  shows "{u. u ≤♭ PDUnit b} ⊆ lower_pd_set_ideal A ⟷ approximants (Rep_compact_basis b) ⊆ set_approximants A"
proof -
  have "{u. u ≤♭ PDUnit b} ⊆ lower_pd_set_ideal A ⟷ PDUnit b ∈ lower_pd_set_ideal A"
    (is "?ss ⟷ ?mem")
  proof
    show "?ss ⟹ ?mem" using lower_le_refl[of "PDUnit b"] by blast
    show "?mem ⟹ ?ss"
      using lower_le.idealD3[OF lower_pd_set_ideal[OF assms], of _ "PDUnit b"]
      by blast
  qed
  also have "PDUnit b ∈ lower_pd_set_ideal A ⟷ b ∈ set_approximants A"
    unfolding lower_pd_set_ideal_mem Rep_PDUnit by simp
  also have "b ∈ set_approximants A ⟷ approximants (Rep_compact_basis b) ⊆ set_approximants A"
    unfolding approximants_def
    using set_approximants_downward_closed
    by auto
  finally show ?thesis .
qed

lemma lower_pd_of_set_mem_iff:
  assumes "(A::'a::bifinite set) ≠ {}"
  shows "x ∈♭ lower_pd_of_set A ⟷ x ∈ closure A"
  unfolding closure_approximants[symmetric]
proof (induction x rule: compact_basis.principal_induct)
  case 1
  show "adm (λa. (a ∈♭ lower_pd_of_set A) =
             (approximants a ⊆ set_approximants A))"
    by (simp add: adm_set_approximants adm_not_set_approximants
        adm_lower_mem adm_not_lower_mem)
next
  case (2 b)
  have "Rep_compact_basis b ∈♭ lower_pd_of_set A ⟷
    lower_principal (PDUnit b) ⊑ lower_pd_of_set A"
    unfolding lower_mem_equiv by simp
  also have "... ⟷ {u. u ≤♭ PDUnit b} ⊆ lower_pd_set_ideal A"
    unfolding below_lower_pd_def
      Rep_lower_pd_of_set[OF assms]
      Abs_lower_pd_principal
    ..
  also note compact_pd_set_ideal[OF assms]
  finally show ?case .
qed

lemma lower_ext_subset:
  assumes "⋀p. {p}♭ ⊑ x ⟹ {p}♭ ⊑ y"
  shows "x ⊑ y"
proof -
  have "lower_principal h ⊑ x ⟹ lower_principal h ⊑ y" for h
  proof (induct rule: pd_basis_induct)
    case (PDUnit a)
    then show ?case using assms[of "Rep_compact_basis a"] by simp
  next
    case (PDPlus t u)
    from PDPlus.prems have "lower_principal t ⊑ x" "lower_principal u ⊑ x"
      by (metis lower_plus_below_iff lower_plus_principal)+
    with PDPlus.hyps show "lower_principal (PDPlus t u) ⊑ y"
      by (metis lower_plus_below_iff lower_plus_principal)
  qed

  then have "h ⊑ x ⟹ h ⊑ y" for h
    by (induct h rule: lower_pd.principal_induct, simp)
  then show "x ⊑ y" by simp
qed

lemma lower_pd_set_iff: "x ⊑ y ⟷ lower_pd_set x ⊆ lower_pd_set y"
  unfolding lower_pd_set_def lower_mem_equiv
  by (auto intro: lower_ext_subset below_trans)

corollary lower_pd_set_inj: "lower_pd_set x = lower_pd_set y ⟹ x = y"
  by (auto intro!: below_antisym simp add: lower_pd_set_iff)

lemma lower_pd_set_meet[simp]: "lower_pd_set (x ⊓ y) = lower_pd_set x ∩ lower_pd_set y"
  unfolding lower_pd_set_def lower_mem_equiv
  by auto

lemma lower_pd_set_join[simp]: "lower_pd_set (x ⊔ y) = lower_pd_set x ∪ lower_pd_set y"
  unfolding lower_join_eq lower_pd_set_def lower_mem_equiv lower_unit_below_plus_iff
  by auto

instance lower_pd :: (bifinite) distributive_cpo
proof
  fix a b c :: "'a lower_pd"
  show "c ⊓ (a ⊔ b) = (c ⊓ a) ⊔ (c ⊓ b)"
    by (auto intro!: lower_pd_set_inj)
qed

section ‹Lower powerdomain elements are generated by their maximal members›

text ‹The members of a lower powerdomain element form a Scott-closed set: closed downwards
  and under directed lubs.  By Zorn's lemma every member lies below a maximal one, so the set is
  the down-closure of its maximal members.›

lemma lower_mem_below: "x ∈♭ X ⟹ y ⊑ x ⟹ y ∈♭ X"
  unfolding lower_mem_equiv by (blast intro: below_trans monofun_cfun_arg)

lemma lower_mem_directed_lub:
  assumes D: "directed D" "D ≠ {}" and mem: "⋀x. x ∈ D ⟹ x ∈♭ X"
  shows "lub D ∈♭ X"
proof -
  obtain C where C: "chain C" "∀i. C i ∈ D" "(⨆i. C i) = lub D"
    by (rule obtain_directed_chain[OF D])
  have "(⨆i. C i) ∈♭ X" using admD[OF adm_lower_mem C(1)] C(2) mem by blast
  then show ?thesis by (simp only: C(3))
qed

lemma lower_mem_lub: "chain Y ⟹ x ∈♭ Y i ⟹ x ∈♭ (⨆i. Y i)"
  unfolding lower_mem_equiv by (blast intro: below_trans is_ub_thelub)

lemma lower_mem_chain: "chain Y ⟹ x ∈♭ Y i ⟹ i ≤ j ⟹ x ∈♭ Y j"
  unfolding lower_mem_equiv by (blast intro: below_trans chain_mono)

lemma lower_mem_compact_lub:
  assumes Y: "chain Y" and x: "compact x" and mem: "x ∈♭ (⨆i. Y i)"
  obtains i where "x ∈♭ Y i"
  using mem compact_below_lub_iff[OF compact_lower_unit[OF x] Y] by (auto simp add: lower_mem_equiv)

definition lower_maximal :: "'a::bifinite lower_pd ⇒ 'a set" where
  "lower_maximal X = {m. m ∈♭ X ∧ (∀y. y ∈♭ X ⟶ m ⊑ y ⟶ y = m)}"

theorem lower_mem_maximal_above:
  assumes x: "x ∈♭ X"
  obtains m where "m ∈ lower_maximal X" "x ⊑ m"
proof -
  have ub: "∃a ∈ lower_pd_set X. T <| a" if T: "T ⊆ lower_pd_set X" "directed T" for T
  proof (cases "T = {}")
    case True
    then show ?thesis using x by (auto simp add: lower_pd_set_def)
  next
    case False
    have "lub T ∈♭ X" using lower_mem_directed_lub[OF T(2) False] T(1) by (auto simp add: lower_pd_set_def)
    then show ?thesis using is_lubD1[OF directed_lub[OF T(2)]] by (auto simp add: lower_pd_set_def)
  qed
  obtain m where "m ∈ lower_pd_set X" "x ⊑ m" "∀z ∈ lower_pd_set X. m ⊑ z ⟶ m = z"
    using obtain_total_above[of x "lower_pd_set X"] x ub by (auto simp add: lower_pd_set_def)
  then show ?thesis using that by (auto simp add: lower_maximal_def lower_pd_set_def)
qed

corollary lower_pd_set_maximal: "lower_pd_set X = lowerset (lower_maximal X)"
proof (rule set_eqI, rule iffI)
  fix x assume "x ∈ lower_pd_set X"
  then obtain m where "m ∈ lower_maximal X" "x ⊑ m"
    by (auto elim: lower_mem_maximal_above simp add: lower_pd_set_def)
  then show "x ∈ lowerset (lower_maximal X)" unfolding lowerset_def by blast
next
  fix x assume "x ∈ lowerset (lower_maximal X)"
  then show "x ∈ lower_pd_set X"
    unfolding lowerset_def lower_maximal_def lower_pd_set_def by (blast intro: lower_mem_below)
qed

section ‹Binds and maps through members›

text ‹A bind lies below ‹Z› as soon as the image of every member does.›

lemma lower_bind_below:
  assumes g: "⋀y. y ∈♭ Y ⟹ g⋅y ⊑ Z"
  shows "lower_bind⋅Y⋅g ⊑ Z"
proof -
  have "W ⊑ Y ⟶ lower_bind⋅W⋅g ⊑ Z" for W
  proof (induct W rule: lower_pd_induct)
    case adm
    then show ?case by (rule adm_imp) simp_all
  next
    case (lower_unit a)
    then show ?case using g by (simp add: lower_mem_equiv)
  next
    case (lower_plus xs ys)
    then show ?case by simp
  qed
  then show ?thesis by simp
qed

lemma lower_bind_mem:
  "y ∈♭ Y ⟹ g⋅y ⊑ lower_bind⋅Y⋅g"
  using monofun_cfun_fun[of "lower_bind⋅{y}♭" "lower_bind⋅Y" g]
  by (simp add: lower_mem_equiv monofun_cfun_arg)

lemma lower_map_mem: "x ∈♭ X ⟹ f⋅x ∈♭ lower_map⋅f⋅X"
  using monofun_cfun_arg[of "{x}♭" X "lower_map⋅f"] by (simp add: lower_mem_equiv)

text ‹A compact member of an image lies below the image of a member.›

lemma lower_map_mem_compact:
  assumes v: "compact v" and mem: "v ∈♭ lower_map⋅f⋅Z"
  obtains w where "w ∈♭ Z" "v ⊑ f⋅w"
proof -
  have "∀v. compact v ⟶ {v}♭ ⊑ lower_map⋅f⋅W ⟶ (∃w. w ∈♭ W ∧ v ⊑ f⋅w)" for W
  proof (induct W rule: lower_pd_induct)
    case adm
    show ?case
    proof (rule admI, intro allI impI)
      fix Ws v assume Ws: "chain Ws"
        and all: "∀i. ∀v. compact v ⟶ {v}♭ ⊑ lower_map⋅f⋅(Ws i) ⟶ (∃w. w ∈♭ Ws i ∧ v ⊑ f⋅w)"
        and v: "compact v" and le: "{v}♭ ⊑ lower_map⋅f⋅(⨆i. Ws i)"
      have "{v}♭ ⊑ (⨆i. lower_map⋅f⋅(Ws i))" using le by (simp add: contlub_cfun_arg[OF Ws])
      then obtain i where "{v}♭ ⊑ lower_map⋅f⋅(Ws i)"
        using compact_below_lub_iff[OF compact_lower_unit[OF v]] Ws by auto
      then obtain w where "w ∈♭ Ws i" "v ⊑ f⋅w" using all v by blast
      then show "∃w. w ∈♭ (⨆i. Ws i) ∧ v ⊑ f⋅w" using lower_mem_lub[OF Ws] by blast
    qed
  next
    case (lower_unit a)
    then show ?case by (auto simp add: lower_mem_principal)
  next
    case (lower_plus xs ys)
    then show ?case by (auto simp add: lower_mem_union)
  qed
  then show ?thesis using v mem that by (auto simp add: lower_mem_equiv)
qed

section ‹Rows with a top adjoined›

text ‹Mapping ‹down› over a lower powerdomain element identifies it with a Top-free element of the
  lower powerdomain over rows with a top: its members are exactly the ‹down›s of the members.›

lemma lower_mem_map_down: "v ∈♭ lower_map⋅down⋅Z ⟷ (∃w. w ∈♭ Z ∧ v = down⋅w)"
proof
  assume mem: "v ∈♭ lower_map⋅down⋅Z"
  obtain ap :: "nat ⇒ 'a top → 'a top" where ap: "approx_chain ap" using bifinite by blast
  have "∃u. ap i⋅v = down⋅u ∧ u ∈♭ Z" for i
  proof -
    have "ap i⋅v ∈♭ lower_map⋅down⋅Z"
      using mem approx_chain.approx_below[OF ap] by (rule lower_mem_below)
    then obtain w where w: "w ∈♭ Z" and le: "ap i⋅v ⊑ down⋅w"
      using approx_chain.compact_approx[OF ap] by (blast elim: lower_map_mem_compact)
    show ?thesis
    proof (cases "ap i⋅v")
      case (down u)
      then show ?thesis using le w by (auto intro: lower_mem_below)
    next
      case Top
      then show ?thesis using le by simp
    qed
  qed
  then obtain us where us: "⋀i. ap i⋅v = down⋅(us i)" and mem_us: "⋀i. us i ∈♭ Z" by metis
  have ch: "chain us"
  proof (rule chainI)
    fix i
    have "ap i⋅v ⊑ ap (Suc i)⋅v"
      using approx_chain.chain_approx[OF ap] by (simp add: monofun_cfun_fun chainE)
    then show "us i ⊑ us (Suc i)" by (simp add: us)
  qed
  have "v = (⨆i. ap i⋅v)"
    using approx_chain.chain_approx[OF ap] approx_chain.lub_approx[OF ap] by (simp flip: contlub_cfun_fun)
  also have "… = down⋅(⨆i. us i)" by (simp add: us contlub_cfun_arg[OF ch])
  finally show "∃w. w ∈♭ Z ∧ v = down⋅w" using admD[OF adm_lower_mem ch] mem_us by blast
next
  assume "∃w. w ∈♭ Z ∧ v = down⋅w"
  then show "v ∈♭ lower_map⋅down⋅Z" by (auto intro: lower_map_mem)
qed

lemma Top_not_mem_map_down: "¬ Top ∈♭ lower_map⋅down⋅Z"
  by (simp add: lower_mem_map_down)

lemma lower_map_down_below_iff: "lower_map⋅down⋅X ⊑ lower_map⋅down⋅Z ⟷ X ⊑ Z"
proof
  assume le: "lower_map⋅down⋅X ⊑ lower_map⋅down⋅Z"
  have "x ∈♭ Z" if "x ∈♭ X" for x
  proof -
    have "down⋅x ∈♭ lower_map⋅down⋅Z"
      using lower_map_mem[OF that] le by (auto simp add: lower_mem_equiv intro: below_trans)
    then show ?thesis by (auto simp add: lower_mem_map_down dest: down_inject)
  qed
  then show "X ⊑ Z" by (simp add: lower_pd_set_iff lower_pd_set_def subset_iff)
qed (rule monofun_cfun_arg)

text ‹Conversely every Top-free element comes from the rows.›

theorem lower_map_down_onto:
  assumes nT: "¬ Top ∈♭ Y"
  obtains Z where "Y = lower_map⋅down⋅Z"
proof -
  define A where "A = {w. down⋅w ∈♭ Y}"
  have "⊥ ∈ A" unfolding A_def lower_mem_equiv by (simp add: down_strict)
  then have ne: "A ≠ {}" by blast
  have down_closed: "lowerset A = A"
    unfolding lowerset_def A_def by (auto intro: lower_mem_below monofun_cfun_arg)
  have closed: "closure A = A"
  proof (rule set_eqI, rule iffI)
    fix x assume "x ∈ closure A"
    then obtain S where x: "x = (⨆i. S i)" and S: "chain S" "range S ⊆ A"
      unfolding closure_def completion_def down_closed by blast
    have "down⋅(⨆i. S i) = (⨆i. down⋅(S i))" by (rule contlub_cfun_arg[OF S(1)])
    moreover have "(⨆i. down⋅(S i)) ∈♭ Y"
      using admD[OF adm_lower_mem ch2ch_Rep_cfunR[OF S(1)]] S(2) unfolding A_def by blast
    ultimately show "x ∈ A" by (simp add: A_def x)
  next
    fix x assume "x ∈ A"
    then show "x ∈ closure A"
      unfolding closure_def completion_def down_closed
      by (auto intro!: exI[of _ "λ_. x"])
  qed
  have "Y = lower_map⋅down⋅(lower_pd_of_set A)"
  proof (rule lower_pd_set_inj, rule set_eqI)
    fix v
    have "v ∈♭ Y ⟷ (∃w. w ∈ A ∧ v = down⋅w)"
      using nT by (cases v) (auto simp add: A_def dest: down_inject)
    then show "v ∈ lower_pd_set Y ⟷ v ∈ lower_pd_set (lower_map⋅down⋅(lower_pd_of_set A))"
      by (simp add: lower_pd_set_def lower_mem_map_down lower_pd_of_set_mem_iff[OF ne] closed)
  qed
  then show ?thesis by (rule that)
qed
