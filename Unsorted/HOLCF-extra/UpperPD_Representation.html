lemma koenig_construction_aux:
  fixes x Y f
  assumes chain: "∀i. Y i ≤♯ Y (Suc i)"
    and sup: "x = (⨆i. upper_principal (Y i))"
    and f_mem: "∀i. f i ∈ Rep_pd_basis (Y i)"
    and f_chain: "∀i. f i ⊑ f (Suc i)"
  shows "chain (λi. Rep_compact_basis (f i))"
    and "x ⊑ {⨆i. Rep_compact_basis (f i)}♯"
proof -
  from f_mem have le_unit: "⋀n. Y n ≤♯ PDUnit (f n)" unfolding upper_le_def Rep_PDUnit
    using below_refl by blast
  then have principal_below: "⋀n. upper_principal (Y n) ⊑ upper_principal (PDUnit (f n))"
    by simp
  from f_chain show fchain: "chain (λi. Rep_compact_basis (f i))" by simp
  then have "chain (λi. {Rep_compact_basis (f i)}♯)" by (rule chain_monofun) 
  with upper_pd.ch2ch_principal[OF chain] principal_below
  have "x ⊑ (⨆n. {Rep_compact_basis (f n)}♯)"
    unfolding sup
    by (auto intro: lub_mono)
  then show "x ⊑ {⨆i. Rep_compact_basis (f i)}♯"
    unfolding contlub_cfun_arg[OF fchain] by simp
qed

lemma upper_pd_has_point: "∃x. s ⊑ {x}♯"
proof -
  from upper_pd.obtain_principal_chain
  obtain Y where chain: "∀i. Y i ≤♯ Y (Suc i)"
    and sup: "s = (⨆i. upper_principal (Y i))".

  have 1: "Rep_pd_basis (Y i) ≠ {}" for i by simp
  have 2: "finite (Rep_pd_basis (Y i))" for i by simp
  have 3: "∀y ∈ Rep_pd_basis (Y (Suc i)). ∃x ∈ Rep_pd_basis (Y i). x ⊑ y" for i
    using chain[unfolded upper_le_def] ..
   
  obtain f where f: "∀i. f i ∈ Rep_pd_basis (Y i)" "∀i. f i ⊑ f (Suc i)"
    by (rule obtain_infinite_path[of "λi. Rep_pd_basis (Y i)" "(⊑)", OF 1 2 3 that])
  from koenig_construction_aux(2)[OF chain sup this]
  have "s ⊑ {⨆n. Rep_compact_basis (f n)}♯" .
  then show ?thesis ..
qed

theorem upper_pd_extensionality_lemma:
  fixes x k
  defines "y ≡ upper_principal k"
  assumes imp: "⋀p. x ⊑ {p}♯ ⟹ y ⊑ {p}♯"
  shows "y ⊑ x"
proof (rule classical)
  assume not_below: "¬(y ⊑ x)"

  from upper_pd.obtain_principal_chain
  obtain Y where chain: "∀i. Y i ≤♯ Y (Suc i)"
    and sup: "x = (⨆i. upper_principal (Y i))".
  from chain have chain': "chain (λi. upper_principal (Y i))"
    by (rule upper_pd.ch2ch_principal)
  from sup not_below chain' have "¬(y ⊑ upper_principal (Y i))" for i
    using below_lub[OF chain', of y] by iprover
  with y_def have not_below_compact: "¬(k ≤♯ (Y i))" for i by simp

  define Bad where "Bad i = {q ∈ Rep_pd_basis (Y i). (∀r ∈ Rep_pd_basis k. ¬(r ⊑ q))}"
    for i
  from not_below_compact have 1: "Bad i ≠ {}" for i
    unfolding upper_le_def Bad_def by fast
  have 2: "finite (Bad i)" for i unfolding Bad_def by simp
  have 3: "∀y ∈ Bad (Suc i). ∃x ∈ Bad i. x ⊑ y" for i
  proof
    fix y assume y: "y ∈ Bad (Suc i)"
    then have "y ∈ Rep_pd_basis (Y (Suc i))" unfolding Bad_def by auto
    with chain obtain x where "x ∈ Rep_pd_basis (Y i)" "x ⊑ y"
      unfolding upper_le_def by blast
    from y have "∀r ∈ Rep_pd_basis k. ¬(r ⊑ y)" unfolding Bad_def by simp
    with ‹x ⊑ y› below_trans
    have "∀r ∈ Rep_pd_basis k. ¬(r ⊑ x)"
      by blast
    with ‹x ∈ Rep_pd_basis (Y i)› have "x ∈ Bad i" unfolding Bad_def by auto
    with ‹x ⊑ y› show "∃x ∈ Bad i. x ⊑ y" ..
  qed

  from obtain_infinite_path[of Bad "(⊑)", OF 1 2 3]
  obtain f where f_mem: "∀i. f i ∈ Bad i" and f_chain: "∀i. f i ⊑ f (Suc i)" .
  from f_mem Bad_def have f_mem': "∀i. f i ∈ Rep_pd_basis (Y i)" by auto
  from koenig_construction_aux[OF chain sup f_mem' f_chain]
  have f_chain: "chain (λi. Rep_compact_basis (f i))"
    and x_below: "x ⊑ {⨆i. Rep_compact_basis (f i)}♯"
    by this+
  from x_below have "y ⊑ {⨆i. Rep_compact_basis (f i)}♯" by (rule imp)
  with f_chain have y_below: "y ⊑ (⨆i. {Rep_compact_basis (f i)}♯)"
    by (simp add: contlub_cfun_arg)
  from y_def have "compact y" by simp
  with y_below obtain i where "y ⊑ {Rep_compact_basis (f i)}♯"
    using chain_monofun[OF f_chain] compactD2 by blast
  with y_def have "k ≤♯ PDUnit (f i)" by simp
  then have "∃q ∈ Rep_pd_basis k. q ⊑ f i" unfolding upper_le_def
    by (simp add: Rep_PDUnit)
  with f_mem Bad_def have False by auto
  then show ?thesis ..
qed

theorem upper_pd_ext_subset:
  assumes "⋀p. x ⊑ {p}♯ ⟹ y ⊑ {p}♯"
  shows "y ⊑ x"
proof -
  have "h ⊑ y ⟹ h ⊑ x" for h
  proof (induct h rule: upper_pd.principal_induct, simp)
    fix k assume "upper_principal k ⊑ y"
    from below_trans[OF this assms]
    show "upper_principal k ⊑ x"
      by (rule upper_pd_extensionality_lemma)
  qed
  then show ?thesis by simp
qed

theorem upper_pd_below_iff_ext_subset:
"x ⊑ y ⟷ (∀p. y ⊑ {p}♯ ⟶ x ⊑ {p}♯)"
  by (auto intro: upper_pd_ext_subset below_trans)

definition upper_set where "upper_set x = {p. x ⊑ {p}♯}"

lemma upper_set_subset_iff: "x ⊑ y ⟷ upper_set x ⊇ upper_set y"
  unfolding upper_set_def
  by (subst upper_pd_below_iff_ext_subset) auto

lemma upper_set_join[simp]: "x ⇑ y ⟹ upper_set (x ⊔ y) = upper_set x ∩ upper_set y"
  unfolding upper_set_def using join_belowI joinableD[of x y] below_trans by blast

lemma upper_set_meet[simp]: "upper_set (x ⊓ y) = upper_set x ∪ upper_set y"
  unfolding upper_set_def upper_meet_eq upper_plus_below_unit_iff by auto

corollary upper_set_inj: "upper_set x = upper_set y ⟹ x = y"
  using upper_set_subset_iff by (auto intro!: below_antisym)

instance upper_pd :: (bifinite) distributive_cpo
proof
  fix a b c :: "'a upper_pd"
  assume "a ⇑ b"
  then show "c ⊓ (a ⊔ b) = (c ⊓ a) ⊔ (c ⊓ b)"
    by (auto intro!: upper_set_inj)
qed

lemma upper_principal_below_unit_iff:
  "upper_principal k ⊑ {x}♯ ⟷ (∃q ∈ Rep_pd_basis k. Rep_compact_basis q ⊑ x)"
proof (induct k rule: pd_basis_induct)
  case (PDUnit a)
  have "upper_principal (PDUnit a) ⊑ {x}♯ ⟷ {Rep_compact_basis a}♯ ⊑ {x}♯"
    by (simp only: upper_unit_Rep_compact_basis)
  also have "... ⟷ Rep_compact_basis a ⊑ x"
    by (rule upper_unit_below_iff)
  finally show ?case by (simp add: Rep_PDUnit)
next
  case (PDPlus s t)
  have "upper_principal (PDPlus s t) ⊑ {x}♯
        ⟷ upper_principal s ⊑ {x}♯ ∨ upper_principal t ⊑ {x}♯"
    by (simp only: upper_plus_principal[symmetric] upper_plus_below_unit_iff)
  also have "... ⟷ (∃q ∈ Rep_pd_basis (PDPlus s t). Rep_compact_basis q ⊑ x)"
    using PDPlus by (auto simp: Rep_PDPlus)
  finally show ?case .
qed

lemma chain_finite_nonempty_Inter:
  fixes ℱ :: "'a::type set set"
  assumes ne:    "ℱ ≠ {}"
      and chain: "⋀S T. ⟦S ∈ ℱ; T ∈ ℱ⟧ ⟹ S ⊆ T ∨ T ⊆ S"
      and fin:   "⋀S. S ∈ ℱ ⟹ finite S"
      and nemp:  "⋀S. S ∈ ℱ ⟹ S ≠ {}"
  shows "⋂ℱ ≠ {}"
proof -
  from ne obtain S0 where S0: "S0 ∈ ℱ" by blast
  define 𝒢 where "𝒢 = {S ∈ ℱ. S ⊆ S0}"
  have "𝒢 ⊆ Pow S0" by (auto simp: 𝒢_def)
  with fin[OF S0] have "finite 𝒢" by (blast intro: finite_subset)
  moreover have "𝒢 ≠ {}" using S0 by (auto simp: 𝒢_def)
  ultimately obtain m where mG: "m ∈ 𝒢"
      and m_min: "⋀T. ⟦T ∈ 𝒢; T ⊆ m⟧ ⟹ m = T"
    by (blast dest: finite_has_minimal)
  have least: "m ⊆ S" if "S ∈ ℱ" for S
  proof (cases "S ⊆ S0")
    case True
    with that have "S ∈ 𝒢" by (simp add: 𝒢_def)
    with chain mG have "S ⊆ m ∨ m ⊆ S" by (auto simp: 𝒢_def)
    with m_min ‹S ∈ 𝒢› show "m ⊆ S" by blast
  next
    case False
    have "m ⊆ S0" using mG by (simp add: 𝒢_def)
    also from chain[OF that S0] False have "S0 ⊆ S" by blast
    finally show "m ⊆ S" .
  qed
  from mG have "m ∈ ℱ" by (simp add: 𝒢_def)
  with nemp obtain z where "z ∈ m" by blast
  with least have "z ∈ ⋂ℱ" by blast
  thus ?thesis by blast
qed

text ‹Chains have greatest lower bounds in bifinite domains›
lemma bifinite_Chain_glb:
  fixes A :: "'a::bifinite set"
  assumes "Chain A" and "A ≠ {}"
  shows "∃a. is_glb A a"
proof -
  from bifinite obtain approx :: "nat ⇒ 'a → 'a" where "approx_chain approx" ..
  then interpret approx_chain approx .
  have lub_approx_app[simp]: "(⨆i. approx i⋅z) = z" for z
    by (simp add: contlub_cfun_fun[OF chain_approx, symmetric])
  define C where "C i = (λx. approx i⋅x) ` A" for i
  have C_Chain: "Chain (C i)" for i
    unfolding C_def
    using monofun_Rep_cfun2 assms(1)
    by (rule Chain_mono_image)
  have C_finite: "finite (C i)" for i unfolding C_def
    by (simp add: finite_deflation.finite_image)
  from assms(2) have C_ne: "C i ≠ {}" for i by (simp add: C_def)
  have C_glb: "is_glb (C i) (glb (C i))" and C_mem: "glb (C i) ∈ C i" for i
  proof -
    from order.finite_has_minimal[OF below_order C_finite C_ne]
    obtain m where m: "m ∈ C i" and m_min: "∀b ∈ C i. b ⊑ m ⟶ m = b" ..
    then have "∀b ∈ C i. m ⊑ b"
      using ChainE[OF C_Chain m] by metis
    with m have m_glb: "is_glb (C i) m"
      by (blast intro: is_glbI)
    then have glb_m: "glb (C i) = m"
      by (rule glb_eqI)
    from m_glb glb_m m show "is_glb (C i) (glb (C i))" and "glb (C i) ∈ C i"
      by simp_all
  qed
  
  define M where "M i = glb (C i)" for i
  have M_mem: "M i ∈ C i" for i unfolding M_def by (rule C_mem)
  have M_obtains: "∃a∈A. M i = approx i⋅a" for i
    using M_mem[of i] unfolding C_def by (auto simp: image_iff)
  from C_glb[THEN is_glb_below] have M_below_approx: "M i ⊑ approx i⋅x" if "x ∈ A" for i x
    unfolding M_def C_def using that by blast
  have M_chain: "chain M"
    by (rule chainI)
    (metis M_obtains M_below_approx below_trans
      monofun_cfun_fun[OF chainE[OF chain_approx]])
  have "is_glb A (⨆i. M i)"
  proof (rule is_glbI)
    fix x assume x: "x ∈ A"
    then have  "M i ⊑ approx i⋅x" for i
      by (rule M_below_approx)
    then have "(⨆i. M i) ⊑ (⨆i. approx i⋅x)"
      by (rule lub_mono[OF M_chain ch2ch_Rep_cfunL[OF chain_approx]])
    then show "(⨆i. M i) ⊑ x" by simp
  next
    fix b assume bnd: "∀x ∈ A. b ⊑ x"
    have "approx i⋅b ⊑ M i" for i
    proof -
      from M_obtains[of i] obtain a where a: "a ∈ A" and Ma: "M i = approx i⋅a" by blast
      from bnd a have "b ⊑ a" by blast
      then have "approx i⋅b ⊑ approx i⋅a"
        by (rule monofun_cfun_arg)
      with Ma show ?thesis by simp
    qed
    then have "(⨆i. approx i⋅b) ⊑ (⨆i. M i)"
      by (rule lub_mono[OF ch2ch_Rep_cfunL[OF chain_approx] M_chain]) 
    then show "b ⊑ (⨆i. M i)" by simp
  qed
  then show ?thesis ..
qed

lemma upper_principal_below_chain:
  assumes "Chain C" "C ≠ {}" "⋀x. x ∈ C ⟹ upper_principal k ⊑ {x}♯"
    "is_glb C g"
  shows "upper_principal k ⊑ {g}♯"
proof -
  define ℱ where
    "ℱ = (λc. {q ∈ Rep_pd_basis k. Rep_compact_basis q ⊑ c}) ` C"
  have "ℱ ≠ {}" using assms(2) by (simp add: ℱ_def)
  moreover have "S ⊆ T ∨ T ⊆ S" if "S ∈ ℱ" "T ∈ ℱ" for S T
  proof -
    from that obtain c d where cd: "c ∈ C" "d ∈ C"
      and S: "S = {q ∈ Rep_pd_basis k. Rep_compact_basis q ⊑ c}"
      and T: "T = {q ∈ Rep_pd_basis k. Rep_compact_basis q ⊑ d}"
      unfolding ℱ_def by blast
    from ChainE[OF assms(1) cd] show ?thesis
      unfolding S T using below_trans by blast
  qed
  moreover have "finite S" if "S ∈ ℱ" for S
    using that unfolding ℱ_def by auto
  moreover have "S ≠ {}" if "S ∈ ℱ" for S
  proof -
    from that obtain c where c: "c ∈ C"
      and S: "S = {q ∈ Rep_pd_basis k. Rep_compact_basis q ⊑ c}"
      unfolding ℱ_def by blast
    from c have "upper_principal k ⊑ {c}♯" by (rule assms(3))
    then show ?thesis unfolding S by (auto simp add: upper_principal_below_unit_iff)
  qed
  ultimately have "⋂ℱ ≠ {}" by (rule chain_finite_nonempty_Inter)
  then obtain q where q: "q ∈ ⋂ℱ" by blast
  from assms(2) obtain c0 where "c0 ∈ C" by blast
  with q have qmem: "q ∈ Rep_pd_basis k" unfolding ℱ_def by auto
  have q_lb: "Rep_compact_basis q ⊑ c" if "c ∈ C" for c
    using q that unfolding ℱ_def by auto
  from assms(4) q_lb have "Rep_compact_basis q ⊑ g"
    by (blast intro: is_glb_greatest)
  with qmem have "∃q∈Rep_pd_basis k. Rep_compact_basis q ⊑ g" by blast
  then show "upper_principal k ⊑ {g}♯"
    by (simp add: upper_principal_below_unit_iff)
qed

lemma upper_set_glb_closed:
  assumes "Chain C" "C ≠ {}" "C ⊆ upper_set x" "is_glb C g"
  shows "g ∈ upper_set x"
proof -
  from upper_pd.obtain_principal_chain
  obtain Y where chain: "∀i. Y i ≤♯ Y (Suc i)"
    and sup: "x = (⨆i. upper_principal (Y i))" .
  from chain have chain': "chain (λi. upper_principal (Y i))"
    by (rule upper_pd.ch2ch_principal)
  have prin_below_x: "upper_principal (Y i) ⊑ x" for i
    unfolding sup by (rule is_ub_thelub[OF chain'])
  with assms(3) have "c ∈ C ⟹ upper_principal (Y i) ⊑ {c}♯" for i c
    unfolding upper_set_def using below_trans by blast
  with assms(1,2,4) have "upper_principal (Y i) ⊑ {g}♯" for i
    by (intro upper_principal_below_chain[of C "Y i" g])
  with chain' have "x ⊑ {g}♯" unfolding sup
    by (rule lub_below)
  then show ?thesis by (simp add: upper_set_def)
qed

lemma po_Zorn_lemma_min:
  fixes A :: "'a::po set"
  assumes "⋀C. Chain C ⟹ C ⊆ A ⟹ ∃l ∈ A. ∀c ∈ C. l ⊑ c"
  shows "∃m ∈ A. ∀a ∈ A. a ⊑ m ⟶ a = m"
proof (rule predicate_Zorn)
  show "partial_order_on A (relation_of (λa b. b ⊑ a) A)"
    by (rule partial_order_on_relation_ofI) (auto intro: below_trans below_antisym)
next
  fix C assume "C ∈ Zorn.Chains (relation_of (λa b. b ⊑ a) A)"
  then have "Chain C" "C ⊆ A"
    by (auto simp: Zorn.Chains_def Chain_def relation_of_def)
  then have "∃l ∈ A. ∀c ∈ C. l ⊑ c" by (rule assms)
  then show "∃u ∈ A. ∀a ∈ C. (λa b. b ⊑ a) a u" by simp
qed

theorem upper_set_above_minimal:
  assumes "p ∈ upper_set x"
  shows "∃y. y ∈ upper_set x ∧ y ⊑ p ∧ (∀z ∈ upper_set x. z ⊑ y ⟶ z = y)"
proof -
  define P where "P = {q ∈ upper_set x. q ⊑ p}"
  have "∃m ∈ P. ∀a ∈ P. a ⊑ m ⟶ a = m"
  proof (rule po_Zorn_lemma_min)
    fix C assume C: "Chain C" "C ⊆ P"
    show "∃l ∈ P. ∀c ∈ C. l ⊑ c"
    proof (cases "C = {}")
      case True
      from assms have "p ∈ P" by (simp add: P_def)
      with True show ?thesis by blast
    next
      case False
      have C_sub: "C ⊆ upper_set x" using C(2) by (auto simp: P_def)
      from bifinite_Chain_glb[OF C(1) False] obtain g where g: "is_glb C g" ..
      from C(1) False C_sub g have gA: "g ∈ upper_set x" by (rule upper_set_glb_closed)
      from False obtain c0 where c0: "c0 ∈ C" by blast
      with C(2) have "c0 ⊑ p" by (auto simp: P_def)
      with g c0 have "g ⊑ p" using is_glb_below below_trans by blast
      with gA have "g ∈ P" by (simp add: P_def)
      moreover have "∀c ∈ C. g ⊑ c" using g by (blast intro: is_glb_below)
      ultimately show ?thesis by blast
    qed
  qed
  then obtain m where m: "m ∈ P" and m_min: "∀a ∈ P. a ⊑ m ⟶ a = m" by blast
  from m have mA: "m ∈ upper_set x" and mp: "m ⊑ p" by (auto simp: P_def)
  have "z = m" if z: "z ∈ upper_set x" "z ⊑ m" for z
  proof -
    from z(2) mp have "z ⊑ p" by (rule below_trans)
    with z(1) have "z ∈ P" by (simp add: P_def)
    with m_min z(2) show "z = m" by blast
  qed
  with mA mp show ?thesis by blast
qed
