Theory Cpo_Topology

theory Cpo_Topology imports HOLCF HOL.Topological_Spaces begin

definition "cpopen (A :: 'a::cpo set) ≡ (∀x ∈ A. ∀y. x ⊑ y ⟶ y ∈ A) ∧ (adm (λx. x ∉ A))"

lemma cpopen_upper: "cpopen A ⟹ x ∈ A ⟹ x ⊑ y ⟹ y ∈ A"
  by (simp add: cpopen_def)

lemma cpopen_nadm: "cpopen A ⟹ adm (λx. x ∉ A)"
  by (simp add: cpopen_def)

lemma cpopen_adm: "cpopen A ⟹ adm (λx. x ∈ A)"
  by (fast dest!: cpopen_upper intro!: adm_upward)

lemma adm_Inf: assumes "∀P ∈ U. adm P" shows "adm (Inf U)"
proof (rule admI)
  show "∀i. Inf U (Y i) ⟹ Inf U (⨆i. Y i)" if "chain Y" for Y
  proof
    fix P assume P: "P ∈ U"
    with assms have "adm P" by simp
    assume "∀i. Inf U (Y i)"
    with P have "∀i. P (Y i)" by auto
    with admD[OF ‹adm P› that] show "P (⨆i. Y i)" by simp
  qed
qed

theorem cpopen_Int: "cpopen A ⟹ cpopen B ⟹ cpopen (A ∩ B)"
  unfolding cpopen_def
  by auto

theorem cpopen_Union: "∀P ∈ U. cpopen P ⟹ cpopen (⋃U)"
  unfolding cpopen_def by fastforce


interpretation cpo_topology: topological_space cpopen
proof
  show "cpopen UNIV" by (simp add: cpopen_def)
qed (fact cpopen_Int cpopen_Union)+

hide_const Topological_Spaces.compact

lemma cpopen_chain_cases[consumes 2]:
  assumes cpopen: "cpopen P" and Y: "chain Y"
  obtains "(⨆i. Y i) ∉ P" "∀i. Y i ∉ P" |
    j where "(⨆i. Y i) ∈ P" "∀i. Y (i + j) ∈ P"
proof (cases "(⨆i. Y i) ∈ P")
  case 1: False
  then have 2: "∀i. Y i ∉ P"
    by (auto intro: is_ub_thelub[OF Y] cpopen_upper[OF cpopen])
  from 1 2 show ?thesis by (rule that(1))
next
  case 1: True
  have "¬¬(∃j. Y j ∈ P)"
  proof
    assume "∄j. Y j ∈ P"
    then have "∀j. Y j ∉ P"
      by simp
    with cpopen_nadm[OF cpopen, THEN admD, OF Y]
    have "(⨆i. Y i) ∉ P" by simp
    with 1 show False by contradiction
  qed
  then have 2: "∃j. Y j ∈ P" by simp
  then obtain j where "Y j ∈ P" ..
  with cpopen_upper[OF cpopen] chain_mono[OF Y]
  have 2: "Y (i + j) ∈ P" for i
    by auto
  from 1 2 show ?thesis by (fast intro: that(2))
qed

end
