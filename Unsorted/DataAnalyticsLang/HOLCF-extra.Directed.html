context po begin
definition directed :: "'a set ⇒ bool" where "directed S ≡ ∀x ∈ S. ∀y ∈ S. ∃z ∈ S. x ⊑ z ∧ y ⊑ z"

lemma directedI[intro]:
  assumes "⋀x y. x ∈ S ⟹ y ∈ S ⟹ ∃z ∈ S. x ⊑ z ∧ y ⊑ z"
  shows "directed S"
  using assms by (simp add: directed_def)

lemma directedE[elim]:
  assumes "directed S" "x ∈ S" "y ∈ S"
  obtains z where "z ∈ S" "x ⊑ z" "y ⊑ z"
  using assms by (auto simp add: directed_def)

lemma directed_finite_has_lub:
  assumes directed: "directed S"
    and finite_ne: "finite S" "S ≠ {}"
  shows "∃x ∈ S. S <<| x"
proof -
  from finite_ne
  obtain w where w: "w ∈ S"
    and w_max: "⋀x. x ∈ S ⟹ w ⊑ x ⟹ w = x"
    by (blast dest: finite_has_maximal)
  {
    fix x assume x: "x ∈ S"
    define T where "T ≡ {y ∈ S. x ⊑ y}"
    from finite_ne x
    have "finite T" "T ≠ {}" unfolding T_def
      by auto
    then obtain wx where "wx ∈ T"
      and wx_max': "⋀y. y ∈ T ⟹ wx ⊑ y ⟹ wx = y"
      by (blast dest: finite_has_maximal)
    from ‹wx ∈ T› T_def have "x ⊑ wx" "wx ∈ S" by auto
    have wx_max: "wx = y" if "y ∈ S" "wx ⊑ y" for y
    proof -
      from that below_trans[OF ‹x ⊑ wx›]
      have "y ∈ T" by (simp add: T_def)
      with that show ?thesis by (simp add: wx_max')
    qed
    have "wx = w" proof -
      from directed ‹wx ∈ S› ‹w ∈ S› obtain w'
        where "w' ∈ S" "wx ⊑ w'" "w ⊑ w'"
        by auto
      with w_max wx_max show ?thesis by blast
    qed
    with ‹x ⊑ wx› have "x ⊑ w" by simp
  }
  then have "S <| w" by (rule is_ubI)
  from this ‹w ∈ S› have "S <<| w" by (rule is_lub_maximal)
  with ‹w ∈ S› show ?thesis by rule
qed

corollary directed_finite_lub:
  "directed S ⟹ finite S ⟹ S ≠ {} ⟹ S <<| lub S"
  by (blast dest: directed_finite_has_lub intro: is_lub_lub)


corollary (in pcpo) directed_finite_lub_pcpo:
  fixes S :: "'a set"
  assumes "directed S" "finite S"
  shows "S <<| lub S"
proof (cases "S = {}")
  case True
  then have "S <<| ⊥" by (simp add: is_lub_def is_ub_def)
  then show ?thesis by (rule is_lub_lub)
next
  case False
  with assms show ?thesis by (rule directed_finite_lub)
qed

corollary finite_directed_lub_mem:
"lub S ∈ S" if "directed S" "finite S" "S ≠ {}"
proof -
  from that obtain w where "w ∈ S" "S <<| w"
    using directed_finite_has_lub by blast
  then show ?thesis by (subst lub_eqI)
qed

lemma directed_mono_image:
  fixes f :: "'a ⇒ 'b::po"
  assumes "⋀x y. x ⊑ y ⟹ f x ⊑ f y"
  "directed S"
shows "directed (f ` S)" 
  using assms(2) 
  unfolding directed_def po_class.directed_def
  by (blast intro: assms(1))

lemma chain_directed: "directed (range C)" if "chain C"
proof (rule, clarsimp)
  fix i j
  from chain_mono[OF that]
  show "∃k. C i ⊑ C k ∧ C j ⊑ C k"
    by (cases rule: le_cases[of i j]) blast+
qed

end

class dcpo = po +
  assumes dcpo: "⋀S. directed S ⟹ ∃x. S <<| x"
begin
theorem directed_lub: "directed S ⟹ S <<| lub S"
  using dcpo is_lub_lub by auto
end

subclass (in dcpo) cpo
proof
  fix S :: "nat ⇒ 'a"
  assume "chain S"
  from chain_directed[OF this] show "∃x. range S <<| x"
    using directed_lub by blast
qed

subclass (in dcpo) pcpo
proof
  have "directed {}" by auto
  then obtain x where "{} <<| x" using dcpo by blast
  then have "x ⊑ y" for y using is_lubD2 is_ubI by simp
  then show "∃x. ∀y. x ⊑ y" by auto
qed

instance bifinite ⊆ dcpo
proof
  fix S :: "'a set"
  assume "directed S"
  obtain a :: "nat ⇒ 'a → 'a" where "approx_chain a"
    using bifinite ..
  then interpret approx_chain a .
  define T where "T i = (λx. a i⋅x) ` S" for i
  {
    fix i
    have "directed (T i)"
      using monofun_cfun_arg ‹directed S› 
      unfolding T_def
      by (rule directed_mono_image)
    moreover have "finite (T i)"
      by (simp add: T_def finite_deflation.finite_image)
    ultimately have "T i <<| lub (T i)"
      by (rule directed_finite_lub_pcpo)
  }
  from approx_lub_family[OF this[unfolded T_def]]
  show "∃x. S <<| x" ..
qed

primrec chainify :: "'a::po set ⇒ (nat ⇒ 'a) ⇒ (nat ⇒ 'a)"
  where "chainify A f 0 = f 0"
  | "chainify A f (Suc n) = (SOME x. x ∈ A ∧ chainify A f n ⊑ x ∧ f (Suc n) ⊑ x)"

lemma chainifyPropI:
  assumes "directed S"
    "f (Suc i) ∈ S" "chainify S f i ∈ S"
  shows "chainify S f (Suc i) ∈ S
    ∧ chainify S f i ⊑ chainify S f (Suc i)
    ∧ f (Suc i) ⊑ chainify S f (Suc i)"
proof -
  from assms obtain x where "x ∈ S ∧ chainify S f i ⊑ x ∧ f (Suc i) ⊑ x"
    by auto
  from someI[of "λx. x ∈ S ∧ chainify S f i ⊑ x ∧ f (Suc i) ⊑ x", OF this]
  show ?thesis by simp
qed

lemma chainify_directed_mem:
  assumes "directed S" "⋀i. f i ∈ S"
  shows "chainify S f i ∈ S"
proof (induct i)
  case 0
  then show ?case by (simp add: assms(2))
next
  case (Suc i)
  from chainifyPropI[OF assms this] show ?case by blast
qed

lemma chainify_directed_chain:
  assumes "directed S" "⋀i. f i ∈ S"
  shows "chain (chainify S f)"
proof (rule chainI)
  fix i
  have "chainify S f i ∈ S"
    using assms by (rule chainify_directed_mem)
  from chainifyPropI[OF assms this]
  show "chainify S f i ⊑ chainify S f (Suc i)" by simp
qed

lemma chainify_directed_below:
  assumes "directed S" "⋀i. f i ∈ S"
  shows "f i ⊑ chainify S f i"
proof (cases i)
  case 0
  then show ?thesis by simp
next
  case (Suc i)
  have "chainify S f i ∈ S"
    using assms by (rule chainify_directed_mem)
  from chainifyPropI[OF assms this]
  show ?thesis unfolding Suc by simp
qed

lemma lub_subset_mono:
  assumes "A <<| a" "B <<| b" "A ⊆ B"
  shows "a ⊑ b"
  using assms(3) by (intro lub_mono_general[OF assms(1,2)]) blast

lemma obtain_directed_chain:
  fixes S :: "'a::bifinite set"
  assumes "directed S" "S ≠ {}"
  obtains C where "chain C" "∀i. C i ∈ S" "(⨆i. C i) = lub S"
proof -
  obtain a :: "nat ⇒ 'a → 'a" where "approx_chain a"
    using bifinite ..
  then interpret approx_chain a . 
  define T where "T i = (λx. a i⋅x) ` S" for i
  have "directed (T i)" for i
    using monofun_cfun_arg ‹directed S› 
    unfolding T_def
    by (rule directed_mono_image)
  moreover have "finite (T i)" for i
    by (simp add: T_def finite_deflation.finite_image)
  ultimately have T_lub: "T i <<| lub (T i)" for i
    by (rule directed_finite_lub_pcpo)
  then have SleT: "S <<| (⨆i. lub (T i))"
    unfolding T_def
    by (rule approx_lub_family)
  from T_lub have chainT: "chain (λi. lub (T i))"
    unfolding T_def
    by (rule approx_ch2ch_lub)
  have "T i ≠ {}" for i by (simp add: ‹S ≠ {}› T_def)
  with ‹⋀i. directed (T i)› ‹⋀i. finite (T i)›
  have "lub (T i) ∈ T i" for i
    by (rule finite_directed_lub_mem)
  then have "∃x ∈ S. a i⋅x = lub (T i)" for i
    unfolding T_def using imageE by metis
  with approx_below have "∃x ∈ S. lub (T i) ⊑ x" for i
    by metis
  then obtain f where f: "f i ∈ S" and f_above: "lub (T i) ⊑ f i" for i
    by metis
  define C where "C i ≡ chainify S f i" for i
  from ‹directed S› f have
    Cmem: "⋀i. C i ∈ S"
    and chainC: "chain C"
    and aboveC: "⋀i. f i ⊑ C i"
    unfolding C_def
    by (auto intro:
        chainify_directed_mem chainify_directed_chain
        chainify_directed_below)
  from f_above aboveC have "lub (T i) ⊑ C i" for i
    by (rule below_trans)
  with chainT chainC have TleC: "(⨆i. lub (T i)) ⊑ (⨆i. C i)"
    by (rule lub_mono)
  from SleT have "S <<| lub S" by (rule is_lub_lub)
  with cpo_lubI[OF chainC] have CleS: "(⨆i. C i) ⊑ lub S"
    by (blast intro: Cmem lub_subset_mono)
  from SleT have "lub S = (⨆i. lub (T i))" by (rule lub_eqI)
  with CleS TleC below_antisym have "(⨆i. C i) = lub S" by auto
  from chainC Cmem this show thesis using that by presburger
qed

text ‹In a bifinite (hence directed-complete) domain a continuous function preserves
  \emph{directed} lubs, not just lubs of chains: the image of a nonempty directed set has
  the image of the lub as its least upper bound.  The directed set is replaced by a cofinal
  chain via @{thm obtain_directed_chain}, on which ordinary continuity applies.

  Combined with distributivity this shows that binary meet preserves \emph{all} existing
  lubs — finite ones by the distributive law ‹meet_join_distrib›, directed ones by this lemma.›
lemma cont_directed_is_lub:
  fixes S :: "'a::bifinite set"
  assumes cont: "cont f" and dir: "directed S" and ne: "S ≠ {}"
  shows "f ` S <<| f (lub S)"
proof -
  have mono: "monofun f" using cont by (rule cont2mono)
  from dir ne obtain C where chainC: "chain C" and Cmem: "∀i. C i ∈ S"
    and Club: "(⨆i. C i) = lub S"
    by (rule obtain_directed_chain)
  have fC: "f (lub S) = (⨆i. f (C i))"
    using Club cont2contlubE[OF cont chainC] by simp
  show "f ` S <<| f (lub S)"
  proof (rule is_lubI)
    show "f ` S <| f (lub S)"
    proof (rule is_ubI)
      fix z assume "z ∈ f ` S"
      then obtain s where s: "s ∈ S" and z: "z = f s" by blast
      from dir have "S <<| lub S" by (rule directed_lub)
      with s have "s ⊑ lub S" by (auto dest: is_lubD1 is_ubD)
      then have "f s ⊑ f (lub S)" by (rule monofunE[OF mono])
      with z show "z ⊑ f (lub S)" by simp
    qed
  next
    fix u assume ub: "f ` S <| u"
    have chfC: "chain (λi. f (C i))" using mono chainC by (rule ch2ch_monofun)
    have "(⨆i. f (C i)) ⊑ u"
    proof (rule lub_below[OF chfC])
      fix i
      from Cmem have "f (C i) ∈ f ` S" by blast
      with ub show "f (C i) ⊑ u" by (rule is_ubD)
    qed
    with fC show "f (lub S) ⊑ u" by simp
  qed
qed

corollary cont_directed_lub:
  fixes S :: "'a::bifinite set"
  assumes "cont f" and "directed S" and "S ≠ {}"
  shows "lub (f ` S) = f (lub S)"
  using cont_directed_is_lub[OF assms] by (rule lub_eqI)

lemma directed_admD:
  fixes D :: "'a::bifinite set"
  assumes "adm P" and D: "directed D" "D ≠ {}"
    and forall: "∀x ∈ D. P x"
  shows "P (lub D)"
proof -
  from obtain_directed_chain[OF D]
  obtain C where "chain C" and mem: "∀i. C i ∈ D" and lub: "(⨆i. C i) = lub D" .
  from forall mem have "∀i. P (C i)" by fast
  with admD[OF ‹adm P› ‹chain C›] lub
  show "P (lub D)" by simp
qed

definition "Chain C ≡ ∀a ∈ C. ∀b ∈ C. a ⊑ b ∨ b ⊑ a"

lemma ChainE: assumes "Chain C" "a ∈ C" "b ∈ C"
  obtains "a ⊑ b" | "b ⊑ a"
  using assms unfolding Chain_def by auto

lemma Chain_directedI: "Chain C ⟹ directed C"
  unfolding Chain_def directed_def by metis

lemma Chain_mono_image: "monofun f ⟹ Chain C ⟹ Chain (f ` C)"
  unfolding Chain_def
  by (auto dest: monofunE)

lemma chain_ChainI: "Chain (range C)" if "chain C"
  unfolding Chain_def
proof (intro strip)
  fix a b assume "a ∈ range C" "b ∈ range C"
  then obtain i j where ab: "a = C i" "b = C j" by blast
  from chain_mono[OF that] have "C i ⊑ C j ∨ C j ⊑ C i"
    by (cases rule: le_cases[of i j]) auto
  with ab show "a ⊑ b ∨ b ⊑ a" by simp
qed

lemma po_Zorn_lemma:
  fixes A :: "'a::po set"
  assumes "⋀C. Chain C ⟹ C ⊆ A ⟹ ∃ u ∈ A. C <| u"
  shows "∃ m ∈ A. ∀ a ∈ A. m ⊑ a ⟶ a = m"
proof (rule predicate_Zorn)
  show "partial_order_on A (relation_of below A)"
    by (rule partial_order_on_relation_ofI)
      (auto intro: below_trans below_antisym)
next
  fix C assume "C ∈ Zorn.Chains (relation_of below A)"
  then have "Chain C" "C ⊆ A"
    by (auto simp add: Zorn.Chains_def Chain_def relation_of_def)
  then have "∃u ∈ A. C <| u" by (rule assms)
  then show "∃u ∈ A. ∀a ∈ C. a ⊑ u" by (simp add: is_ub_def)
qed

theorem obtain_total_above:
  fixes S :: "'a::po set" and x :: 'a
  assumes x: "x ∈ S" and directed: "⋀T. T ⊆ S ⟹ directed T ⟹ ∃a ∈ S. T <| a"
  obtains y where "y ∈ S" "x ⊑ y" "∀z ∈ S. y ⊑ z ⟶ y = z"
proof -
  define S_above where "S_above = {y ∈ S. x ⊑ y}"
  have S_above_iff: "y ∈ S_above ⟷ x ⊑ y ∧ y ∈ S" for y
    by (auto simp add: S_above_def)
  have "∃m ∈ S_above. ∀y ∈ S_above. m ⊑ y ⟶ y = m"
  proof (rule po_Zorn_lemma)
    fix C assume "Chain C" "C ⊆ S_above"
    then have "C ⊆ S" "directed C"
      by (auto intro: Chain_directedI simp: S_above_def)
    then obtain a where "a ∈ S" "C <| a" 
      by (blast dest: directed)
    show "∃u ∈ S_above. C <| u"
    proof (cases "C = {}")
      case True
      then show ?thesis
        by (intro bexI[where x=x]) (auto simp add: x S_above_def)
    next
      case False
      then obtain y where "y ∈ C" by blast
      with ‹C <| a› have "y ⊑ a" by (rule is_ubD)
      from ‹y ∈ C› ‹C ⊆ S_above› S_above_def
      have "x ⊑ y" by auto
      from below_trans[OF this ‹y ⊑ a›] ‹a ∈ S›
      have "a ∈ S_above" by (simp add: S_above_def)
      from this ‹C <| a› show ?thesis by blast
    qed
  qed
  then obtain m where "m ∈ S_above" and m_max: "∀y. y ∈ S_above ⟶ m ⊑ y ⟶ m = y"
    by blast
  from ‹m ∈ S_above› have "m ∈ S" "x ⊑ m" by (simp_all add: S_above_iff)
  have "m = y" if "y ∈ S" "m ⊑ y" for y
  proof -
    from ‹x ⊑ m› ‹m ⊑ y› have "x ⊑ y" by (rule below_trans)
    with ‹y ∈ S› have "y ∈ S_above" by (simp add: S_above_iff)
    with m_max ‹m ⊑ y› show "m = y" by blast
  qed
  from ‹m ∈ S› ‹x ⊑ m› this
  show thesis
    using that[of m]
    by blast
qed

lemma directed_join_of_finite_subsets:
  fixes A :: "'a::dcpo set"
  assumes lubs: "⋀B. B ∈ Fpow(A) ⟹ B <<| lub B"
  shows directed_lubs: "directed (lub ` Fpow(A))"
    and lub_lub: "A <<| lub (lub ` Fpow(A))"
proof -
  show "directed (lub ` Fpow(A))"
  proof (intro directedI, safe)
    fix x y assume xy: "x ∈ Fpow(A)" "y ∈ Fpow(A)"
    then have un: "x ∪ y ∈ Fpow(A)" by (auto simp add: Fpow_def)
    with xy lubs have "lub x ⊑ lub (x ∪ y) ∧ lub y ⊑ lub (x ∪ y)"
      using lub_subset_mono by fast
    with un show "∃z ∈ lub ` Fpow(A). lub x ⊑ z ∧ lub y ⊑ z" by fast
  qed
  then have pa_lub: "lub ` Fpow(A) <<| lub (lub ` Fpow(A))" by (rule directed_lub)
  show "A <<| lub (lub ` Fpow(A))"
  proof (intro is_lubI is_ubI)
    fix x assume "x ∈ A"
    then have "{x} ∈ Fpow(A)" by (simp add: Fpow_def)
    then have "x ∈ lub ` Fpow(A)" by force
    with pa_lub[THEN is_lubD1, THEN is_ubD] show "x ⊑ lub (lub ` Fpow(A))" .
  next
    fix u assume "A <| u"
    then have "B ∈ Fpow(A) ⟹ B <| u" for B by (auto simp add: Fpow_def is_ub_def)
    with lubs[THEN is_lubD2] have "B ∈ Fpow(A) ⟹ lub B ⊑ u" for B by simp
    then have "lub ` Fpow(A) <| u" by (rule ub_imageI)
    with pa_lub[THEN is_lubD2] show "lub (lub ` Fpow(A)) ⊑ u" .
  qed
qed

section ‹Lemmas concerning completion, complete sets and closed sets›

context fixes A :: "'a::bifinite set"
begin

definition "chain_complete ≡ ∀C. chain C ∧ range C ⊆ A ⟶ Lub C ∈ A"

definition "dcomplete ≡ ∀B. B ⊆ A ∧ directed B ∧ B ≠ {} ⟶ lub B ∈ A"

lemma chain_complete_iff_dcomplete:
  "chain_complete ⟷ dcomplete"
  unfolding chain_complete_def dcomplete_def
  apply (rule iffI)
  using obtain_directed_chain
   apply (metis image_subsetI subsetD)
  using chain_directed by blast

end


text ‹A compact element below the lub of a directed set is below a member; the approximants of
  the lub are the union of the approximants.›

lemma compact_directedD:
  fixes x :: "'a::bifinite"
  assumes D: "directed D" "D ≠ {}" and x: "compact x"
  shows "x ⊑ lub D ⟷ (∃y ∈ D. x ⊑ y)"
proof (safe)
  assume "x ⊑ lub D"
  from obtain_directed_chain[OF D]
  obtain C where "chain C" "∀i. C i ∈ D" "(⨆i. C i) = lub D" .
  with x ‹x ⊑ lub D› have "∃i. x ⊑ C i"
    by (auto intro: compactD2[of x C])
  with ‹∀i. C i ∈ D› show "∃y ∈ D. x ⊑ y" by fast
next
  fix y assume "y ∈ D"
  with directed_lub[OF D(1)] is_lubD1 is_ubD have "y ⊑ lub D" by blast
  moreover assume "x ⊑ y"
  ultimately show "x ⊑ lub D" by (rule below_trans[rotated])
qed

lemma directed_approximants:
  fixes D :: "('a::bifinite) set"
  assumes "directed D" "D ≠ {}"
  shows "approximants (lub D) = (⋃x ∈ D. approximants x)"
  unfolding approximants_def
  using compact_directedD[OF assms] by auto
