lemma tuple_bounded_complete:
  fixes A :: "('a, 'b::bounded_complete) tuple set" and b
  assumes "A <| b"
  shows "A <<| Abs_tuple (lub (tget ` A))"
proof -
  from assms have ub: "tget ` A <| tget b" by (auto simp: is_ub_def below_tuple_def)
  from ex_bound_lub[OF ub] have lubA: "tget ` A <<| lub (tget ` A)" .
  show "A <<| Abs_tuple (lub (tget ` A))"
  proof (rule is_lubI)
    show "A <| Abs_tuple (lub (tget ` A))"
    proof (rule is_ubI)
      fix t assume "t ∈ A"
      hence "tget t ⊑ lub (tget ` A)" using lubA by (meson imageI is_lubD1 is_ubD)
      thus "t ⊑ Abs_tuple (lub (tget ` A))" by (simp add: below_tuple_def Abs_tuple_inverse')
    qed
  next
    fix u assume "A <| u"
    hence "tget ` A <| tget u" by (auto simp: is_ub_def below_tuple_def)
    with lubA have "lub (tget ` A) ⊑ tget u" by (rule is_lubD2)
    thus "Abs_tuple (lub (tget ` A)) ⊑ u" by (simp add: below_tuple_def Abs_tuple_inverse')
  qed
qed

lemma tget_join:
  fixes a b :: "('a, 'b::bounded_complete) tuple"
  assumes "a ⇑ b"
  shows "tget (a ⊔ b) = (λp. tget a p ⊔ tget b p)"
proof -
  from assms obtain z where az: "a ⊑ z" and bz: "b ⊑ z" by (auto simp: joinable_def)
  hence abf: "tget a ⇑ tget b" by (auto simp: below_tuple_def intro: joinableI)
  from az bz have zub: "{a, b} <| z" by (auto simp: is_ub_def)
  have "tget (a ⊔ b) = lub (tget ` {a, b})"
    unfolding join_def using zub by (simp add: tuple_lub Abs_tuple_inverse')
  also have "... = tget a ⊔ tget b" by (simp add: join_def)
  also have "... = (λp. tget a p ⊔ tget b p)" using abf by (rule fun_join)
  finally show ?thesis .
qed

lemma tget_meet:
  fixes a b :: "('a, 'b::bounded_complete) tuple"
  shows "tget (a ⊓ b) = (λp. tget a p ⊓ tget b p)"
proof -
  let ?X = "Abs_tuple (λp. tget a p ⊓ tget b p)"
  have le: "tget (a ⊓ b) p ⊑ tget a p" "tget (a ⊓ b) p ⊑ tget b p" for p
    using meet_below1[of a b, unfolded below_tuple_def, THEN fun_belowD]
          meet_below2[of a b, unfolded below_tuple_def, THEN fun_belowD] by auto
  have "a ⊓ b = ?X"
  proof (rule below_antisym)
    show "a ⊓ b ⊑ ?X" unfolding below_tuple_def Abs_tuple_inverse'
      by (auto simp: fun_below_iff le intro: meet_greatest)
    show "?X ⊑ a ⊓ b"
    proof (rule meet_greatest)
      show "?X ⊑ a" unfolding below_tuple_def Abs_tuple_inverse'
        by (auto simp: fun_below_iff meet_below1)
      show "?X ⊑ b" unfolding below_tuple_def Abs_tuple_inverse'
        by (auto simp: fun_below_iff meet_below2)
    qed
  qed
  thus ?thesis by (simp add: Abs_tuple_inverse')
qed

instance tuple :: (type, distributive_cpo) distributive_cpo
proof
  fix a b c :: "('a, 'b) tuple"
  assume ab: "a ⇑ b"
  hence ptw: "tget a p ⇑ tget b p" for p
    by (auto simp: joinable_def below_tuple_def intro: joinableI dest: fun_belowD)
  have cab: "(c ⊓ a) ⇑ (c ⊓ b)" by (rule joinableI[OF meet_below1 meet_below1])
  have "tget (c ⊓ (a ⊔ b)) = (λp. tget c p ⊓ (tget a p ⊔ tget b p))"
    using ab by (simp add: tget_meet tget_join)
  also have "... = (λp. (tget c p ⊓ tget a p) ⊔ (tget c p ⊓ tget b p))"
    using ptw by (simp add: meet_join_distrib)
  also have "... = tget ((c ⊓ a) ⊔ (c ⊓ b))"
    using cab by (simp add: tget_join tget_meet)
  finally show "c ⊓ (a ⊔ b) = (c ⊓ a) ⊔ (c ⊓ b)" by (simp add: tget_inject)
qed

lemma deflation_le_cut: 
  assumes "deflation g"
    "deflation h"
    "g ⊑ h"
    and fg: "f = f oo g"
  shows
    "f = f oo h"
proof -
  from assms have "g⋅(h⋅x) = g⋅x" for x by (auto intro: deflation_below_comp1)
  then have 1: "g oo h = g" by (simp add: cfun_eq_iff)
  from fg have "f oo h = (f oo g) oo h" by simp
  also have "... = f oo (g oo h)" by (simp add: cfun_eq_iff)
  also note 1
  also note fg[symmetric]
  finally show ?thesis ..
qed

subsection ‹Componentwise reasoning on tuples›

lemma tuple_below_iff: "x ⊑ y ⟷ (∀d. x⟨d⟩ ⊑ y⟨d⟩)"
  by (simp add: below_tuple_def fun_below_iff)

lemma tuple_eq_iff: "x = y ⟷ (∀d. x⟨d⟩ = y⟨d⟩)"
  by (auto simp: po_eq_conv tuple_below_iff)

lemma tuple_joinableI:
  fixes x y :: "('d, 'v::bounded_complete) tuple"
  assumes "⋀d. x⟨d⟩ ⇑ y⟨d⟩"
  shows "x ⇑ y"
proof (rule joinableI)
  let ?z = "Abs_tuple (λd. x⟨d⟩ ⊔ y⟨d⟩) :: ('d, 'v) tuple"
  show "x ⊑ ?z" "y ⊑ ?z"
    using joinableD[OF assms] by (simp_all add: tuple_below_iff Abs_tuple_inverse')
qed

text ‹The single-key record ‹tupd d⋅⊥⋅x›.›

lemma tupd_bot_tget: "(tupd d⋅⊥⋅x)⟨e⟩ = (if e = d then x else ⊥)"
  by (simp add: tupd_fupd tget_strict)

text ‹The componentwise map of a family of functions.›

definition tmap :: "('d ⇒ 'v::pcpo → 'v) ⇒ ('d, 'v) tuple → ('d, 'v) tuple" where
  "tmap f = (Λ t. Abs_tuple (λd. f d⋅(t⟨d⟩)))"

lemma tmap_tget [simp]: "(tmap f⋅t)⟨d⟩ = f d⋅(t⟨d⟩)"
  unfolding tmap_def
  by (subst beta_cfun)
    (simp_all add: cont_Abs_tuple cont2cont_lambda cont2cont_fun cont_tget Abs_tuple_inverse')

text ‹The component at ‹d› of a function on tuples, read off the single-key records.›

definition tcomp :: "'d ⇒ (('d, 'v::pcpo) tuple → ('d, 'v) tuple) ⇒ 'v → 'v" where
  "tcomp d p = (Λ t. t⟨d⟩) oo p oo tupd d⋅⊥"

lemma tcomp_beta: "tcomp d p⋅x = (p⋅(tupd d⋅⊥⋅x))⟨d⟩"
  by (simp add: tcomp_def)

lemma tcomp_tmap [simp]: "tcomp d (tmap f) = f d"
  by (rule cfun_eqI) (simp add: tcomp_beta tupd_bot_tget)

subsection ‹Semi-factors of the tuple domain›

text ‹Records are ordered pointwise and their keys are independent, so a semi-factor of the
  tuples acts on each key separately: it is the componentwise map of its components, and these
  are semi-factors.  Hence the semi-factors of the tuples are the families of semi-factors.›

lemma semi_factor_defl_tmap:
  fixes f :: "'d::countable ⇒ 'v::{bifinite, distributive_cpo} → 'v"
  assumes f: "⋀d. semi_factor_defl (f d)"
  shows "semi_factor_defl (tmap f)"
proof (rule semi_factor_deflI)
  have sf: "stable_defl (f d)" for d by (rule semi_factor_defl.axioms(1)[OF f])
  note df = stable_defl.deflation[OF sf]
  show "stable_defl (tmap f)"
  proof standard
    fix t show "tmap f⋅(tmap f⋅t) = tmap f⋅t" by (simp add: tuple_eq_iff deflation.idem[OF df])
  next
    fix t show "tmap f⋅t ⊑ t" by (simp add: tuple_below_iff deflation.below[OF df])
  next
    fix x y :: "('d, 'v) tuple" assume "tmap f⋅y = y" "x ⊑ y"
    then show "tmap f⋅x = x"
      by (auto simp: tuple_eq_iff tuple_below_iff intro: stable_defl.downclosed[OF sf])
  qed
  fix x y :: "('d, 'v) tuple" assume y: "tmap f⋅y = y" and le: "tmap f⋅x ⊑ y"
  show "x ⇑ y"
  proof (rule tuple_joinableI)
    fix d
    from y have "(tmap f⋅y)⟨d⟩ = y⟨d⟩" by simp
    then have "f d⋅(y⟨d⟩) = y⟨d⟩" by simp
    moreover from le have "f d⋅(x⟨d⟩) ⊑ y⟨d⟩" by (simp add: tuple_below_iff)
    ultimately show "x⟨d⟩ ⇑ y⟨d⟩" by (rule semi_factor_defl.slide[OF f])
  qed
qed

lemma deflation_tupd:
  fixes p :: "('d, 'v::pcpo) tuple → ('d, 'v) tuple"
  assumes p: "deflation p"
  shows "p⋅(tupd d⋅⊥⋅x) = tupd d⋅⊥⋅(tcomp d p⋅x)"
proof (rule tuple_eqI)
  fix e
  have "(p⋅(tupd d⋅⊥⋅x))⟨e⟩ ⊑ (tupd d⋅⊥⋅x)⟨e⟩"
    using deflation.below[OF p] by (simp add: tuple_below_iff)
  then show "(p⋅(tupd d⋅⊥⋅x))⟨e⟩ = (tupd d⋅⊥⋅(tcomp d p⋅x))⟨e⟩"
    by (cases "e = d") (simp_all add: tupd_bot_tget tcomp_beta)
qed

lemma stable_defl_tget:
  fixes p :: "('d::countable, 'v::{bifinite, distributive_cpo}) tuple → ('d, 'v) tuple"
  assumes p: "stable_defl p"
  shows "(p⋅t)⟨d⟩ = tcomp d p⋅(t⟨d⟩)"
proof (rule below_antisym)
  interpret P: stable_defl p by (rule p)
  let ?s = "tupd d⋅⊥⋅(t⟨d⟩)" and ?u = "tupd d⋅⊥⋅((p⋅t)⟨d⟩)"
  have "?u ⊑ p⋅t" by (simp add: tuple_below_iff tupd_bot_tget)
  then have "p⋅?u = ?u" by (rule P.lower_fixed)
  moreover have "?u ⊑ ?s" using P.below[of t] by (simp add: tuple_below_iff tupd_bot_tget)
  ultimately have "?u ⊑ p⋅?s" by (rule P.below_image)
  then have "?u⟨d⟩ ⊑ (p⋅?s)⟨d⟩" by (simp add: tuple_below_iff)
  then show "(p⋅t)⟨d⟩ ⊑ tcomp d p⋅(t⟨d⟩)" by (simp add: tupd_bot_tget tcomp_beta)
  have "?s ⊑ t" by (simp add: tuple_below_iff tupd_bot_tget)
  then have "p⋅?s ⊑ p⋅t" by (rule monofun_cfun_arg)
  then have "(p⋅?s)⟨d⟩ ⊑ (p⋅t)⟨d⟩" by (simp add: tuple_below_iff)
  then show "tcomp d p⋅(t⟨d⟩) ⊑ (p⋅t)⟨d⟩" by (simp add: tcomp_beta)
qed

lemma tmap_tcomp:
  fixes p :: "('d::countable, 'v::{bifinite, distributive_cpo}) tuple → ('d, 'v) tuple"
  assumes "stable_defl p" shows "tmap (λd. tcomp d p) = p"
  by (rule cfun_eqI) (simp add: tuple_eq_iff stable_defl_tget[OF assms])

lemma semi_factor_defl_tcomp:
  fixes p :: "('d::countable, 'v::{bifinite, distributive_cpo}) tuple → ('d, 'v) tuple"
  assumes p: "semi_factor_defl p"
  shows "semi_factor_defl (tcomp d p)"
proof (rule semi_factor_deflI)
  interpret P: semi_factor_defl p by (rule p)
  note single = deflation_tupd[OF P.deflation]
  show "stable_defl (tcomp d p)"
  proof standard
    fix x
    have "tcomp d p⋅(tcomp d p⋅x) = (p⋅(tupd d⋅⊥⋅(tcomp d p⋅x)))⟨d⟩" by (rule tcomp_beta)
    also have "… = (p⋅(p⋅(tupd d⋅⊥⋅x)))⟨d⟩" by (simp only: single)
    also have "… = tcomp d p⋅x" by (simp add: P.idem tcomp_beta)
    finally show "tcomp d p⋅(tcomp d p⋅x) = tcomp d p⋅x" .
  next
    fix x
    have "(p⋅(tupd d⋅⊥⋅x))⟨d⟩ ⊑ (tupd d⋅⊥⋅x)⟨d⟩" using P.below by (simp add: tuple_below_iff)
    then show "tcomp d p⋅x ⊑ x" by (simp add: tcomp_beta tupd_bot_tget)
  next
    fix x y assume y: "tcomp d p⋅y = y" and xy: "x ⊑ y"
    have "p⋅(tupd d⋅⊥⋅y) = tupd d⋅⊥⋅y" by (simp add: single y)
    moreover have "tupd d⋅⊥⋅x ⊑ tupd d⋅⊥⋅y" using xy by (rule monofun_cfun_arg)
    ultimately have "p⋅(tupd d⋅⊥⋅x) = tupd d⋅⊥⋅x" by (rule P.downclosed)
    then show "tcomp d p⋅x = x" by (simp add: tcomp_beta tupd_bot_tget)
  qed
  fix x y assume y: "tcomp d p⋅y = y" and le: "tcomp d p⋅x ⊑ y"
  have fy: "p⋅(tupd d⋅⊥⋅y) = tupd d⋅⊥⋅y" by (simp add: single y)
  have "p⋅(tupd d⋅⊥⋅x) ⊑ tupd d⋅⊥⋅y" unfolding single using le by (rule monofun_cfun_arg)
  with fy have j: "tupd d⋅⊥⋅x ⇑ tupd d⋅⊥⋅y" by (rule P.slide)
  have "(tupd d⋅⊥⋅x)⟨d⟩ ⊑ (tupd d⋅⊥⋅x ⊔ tupd d⋅⊥⋅y)⟨d⟩"
    "(tupd d⋅⊥⋅y)⟨d⟩ ⊑ (tupd d⋅⊥⋅x ⊔ tupd d⋅⊥⋅y)⟨d⟩"
    using joinableD[OF j] by (simp_all add: tuple_below_iff)
  then show "x ⇑ y" by (simp add: tupd_bot_tget) (rule joinableI)
qed

text ‹The isomorphism between families of semi-factors and semi-factors of the tuples.›

definition semi_factor_of_family ::
  "('d::countable ⇒ 'v::{bifinite, distributive_cpo} semi_factor) ⇒ ('d, 'v) tuple semi_factor"
  where "semi_factor_of_family F = semi_factor_of_defl (tmap (λd. semi_factor_p (F d)))"

definition family_of_semi_factor ::
  "('d::countable, 'v::{bifinite, distributive_cpo}) tuple semi_factor ⇒ 'd ⇒ 'v semi_factor"
  where "family_of_semi_factor G d = semi_factor_of_defl (tcomp d (semi_factor_p G))"

lemma semi_factor_p_of_family:
  "semi_factor_p (semi_factor_of_family F) = tmap (λd. semi_factor_p (F d))"
  unfolding semi_factor_of_family_def
  by (rule semi_factor_p_of_defl, rule semi_factor_defl_tmap, rule semi_factor_defl_semi_factor_p)

lemma semi_factor_p_family_of:
  "semi_factor_p (family_of_semi_factor G d) = tcomp d (semi_factor_p G)"
  unfolding family_of_semi_factor_def
  by (rule semi_factor_p_of_defl, rule semi_factor_defl_tcomp, rule semi_factor_defl_semi_factor_p)

lemma family_of_semi_factor_of_family [simp]: "family_of_semi_factor (semi_factor_of_family F) = F"
  by (rule ext) (simp add: semi_factor_p_inject[symmetric] semi_factor_p_family_of
      semi_factor_p_of_family)

lemma semi_factor_of_family_of_semi_factor [simp]:
  "semi_factor_of_family (family_of_semi_factor G) = G"
  using semi_factor_defl.axioms(1)[OF semi_factor_defl_semi_factor_p[of G]]
  by (simp add: semi_factor_p_inject[symmetric] semi_factor_p_of_family semi_factor_p_family_of
      tmap_tcomp)

lemma tmap_below_iff: "tmap f ⊑ tmap g ⟷ (∀d. f d ⊑ g d)"
proof
  assume fg: "tmap f ⊑ tmap g"
  show "∀d. f d ⊑ g d"
  proof (intro allI cfun_belowI)
    fix d x
    have "(tmap f⋅(tupd d⋅⊥⋅x))⟨d⟩ ⊑ (tmap g⋅(tupd d⋅⊥⋅x))⟨d⟩"
      using monofun_cfun_fun[OF fg] by (simp add: tuple_below_iff)
    then show "f d⋅x ⊑ g d⋅x" by (simp add: tupd_bot_tget)
  qed
qed (auto intro!: cfun_belowI simp: tuple_below_iff monofun_cfun_fun)

lemma semi_factor_of_family_below_iff:
  "semi_factor_of_family F ⊑ semi_factor_of_family F' ⟷ F ⊑ F'"
  by (simp add: semi_factor_below_iff_p semi_factor_p_of_family tmap_below_iff fun_below_iff)

lemma family_of_semi_factor_below_iff:
  "family_of_semi_factor G ⊑ F ⟷ G ⊑ semi_factor_of_family F"
  by (metis semi_factor_of_family_below_iff semi_factor_of_family_of_semi_factor)

lemma below_family_of_semi_factor_iff:
  "F ⊑ family_of_semi_factor G ⟷ semi_factor_of_family F ⊑ G"
  by (metis semi_factor_of_family_below_iff semi_factor_of_family_of_semi_factor)

text ‹Being an order isomorphism, it preserves all glbs and lubs.›

lemma semi_factor_of_family_glb:
  "semi_factor_of_family (glb A) = glb (semi_factor_of_family ` A)"
proof (rule below_antisym)
  show "semi_factor_of_family (glb A) ⊑ glb (semi_factor_of_family ` A)"
    by (rule lattice_glb_greatest) (auto simp: semi_factor_of_family_below_iff intro: glb_below)
  show "glb (semi_factor_of_family ` A) ⊑ semi_factor_of_family (glb A)"
    unfolding family_of_semi_factor_below_iff[symmetric]
    by (rule lattice_glb_greatest) (simp add: family_of_semi_factor_below_iff glb_below)
qed

lemma semi_factor_of_family_lub:
  "semi_factor_of_family (lub A) = lub (semi_factor_of_family ` A)"
proof (rule below_antisym)
  show "semi_factor_of_family (lub A) ⊑ lub (semi_factor_of_family ` A)"
    unfolding below_family_of_semi_factor_iff[symmetric]
    by (rule lattice_lub_least) (simp add: is_ub_def below_family_of_semi_factor_iff lattice_lub_upper)
  show "lub (semi_factor_of_family ` A) ⊑ semi_factor_of_family (lub A)"
    by (rule lattice_lub_least)
      (auto simp: is_ub_def semi_factor_of_family_below_iff intro: lattice_lub_upper)
qed

lemma semi_factor_of_family_inf:
  "semi_factor_of_family (F ⊓ G) = semi_factor_of_family F ⊓ semi_factor_of_family G"
  unfolding meet_def using semi_factor_of_family_glb[of "{F, G}"] by simp

lemma semi_factor_of_family_sup:
  "semi_factor_of_family (F ⊔ G) = semi_factor_of_family F ⊔ semi_factor_of_family G"
  unfolding join_def using semi_factor_of_family_lub[of "{F, G}"] by simp

lemma semi_factor_of_family_top: "semi_factor_of_family lattice_top = lattice_top"
  by (rule below_antisym) (simp_all add: family_of_semi_factor_below_iff[symmetric])

lemma semi_factor_of_family_bot: "semi_factor_of_family ⊥ = ⊥"
  by (rule bottomI) (subst below_family_of_semi_factor_iff[symmetric], rule minimal)

lemma semi_factor_of_family_proj [simp]: "(t ⇂ semi_factor_of_family F)⟨d⟩ = t⟨d⟩ ⇂ F d"
  by (simp add: semi_factor_p_of_family)

subsection ‹Dimension sets as semi-factors of the tuple domain›

text ‹The coordinate cut is the family that is the top on the dimensions and ‹⊥› elsewhere.›

definition dcut :: "'a::countable set ⇒ ('a, 'b::scott_domain) tuple semi_factor" where
  "dcut A = semi_factor_of_family (λd. if d ∈ A then lattice_top else ⊥)"

lemma semi_factor_p_dcut:
  "semi_factor_p (dcut A) = (tcut A :: ('a::countable, 'b::scott_domain) tuple → ('a,'b) tuple)"
  by (rule cfun_eqI, rule tuple_eqI)
    (simp add: dcut_def semi_factor_p_of_family tcut_tget semi_factor_p_top semi_factor_p_bot)

text ‹Projecting a tuple onto the dimensions of an LOD is exactly the coordinate cut.›
lemma dcut_proj: "x ⇂ dcut A = tcut A⋅x"
  by (simp add: semi_factor_p_dcut)

text ‹A semi-factor is determined by its projection.›
lemma semi_factor_eqI:
  assumes "semi_factor_p F = semi_factor_p G" shows "F = G"
  using assms by (simp add: semi_factor_p_inject)

lemma tcut_compose: "tcut B⋅(tcut A⋅t) = tcut (A ∩ B)⋅t"
  by (rule tuple_eqI) (simp add: tcut_tget)

text ‹@{const dcut} is a lattice homomorphism, as the indicator families are.›
lemma dcut_inf: "dcut (A ∩ B) = dcut A ⊓ dcut B"
  unfolding dcut_def semi_factor_of_family_inf[symmetric]
  by (rule arg_cong[where f = semi_factor_of_family]) (simp add: fun_eq_iff fun_meet lattice_meet_idem lattice_meet_bot_left lattice_meet_bot_right)

lemma tcut_joinable: "tcut A⋅t ⇑ tcut B⋅t"
  by (rule joinableI[OF tcut_deflation.below tcut_deflation.below])

lemma tcut_join: "tcut A⋅t ⊔ tcut B⋅t = tcut (A ∪ B)⋅t"
  for t :: "('a, 'b::bounded_complete) tuple"
proof (rule tuple_eqI)
  fix a
  have "tget (tcut A⋅t ⊔ tcut B⋅t) a = tget (tcut A⋅t) a ⊔ tget (tcut B⋅t) a"
    by (subst tget_join[OF tcut_joinable]) simp
  thus "(tcut A⋅t ⊔ tcut B⋅t)⟨a⟩ = (tcut (A ∪ B)⋅t)⟨a⟩"
    by (cases "a ∈ A"; cases "a ∈ B") (simp_all add: tcut_tget)
qed

lemma dcut_sup: "dcut (A ∪ B) = dcut A ⊔ dcut B"
  unfolding dcut_def semi_factor_of_family_sup[symmetric]
  by (rule arg_cong[where f = semi_factor_of_family]) (simp add: fun_eq_iff fun_join lattice_join_bot_left lattice_join_bot_right)

lemma dcut_bot: "dcut {} = ⊥"
  by (simp add: dcut_def semi_factor_of_family_bot lambda_strict)

lemma dcut_top: "dcut UNIV = lattice_top"
  by (simp add: dcut_def semi_factor_of_family_top[unfolded lattice_top_fun])

lemma tcut_below: "tcut A⋅t ⊑ t"
  by (simp add: below_tuple_def fun_below_iff tcut_tget)

lemma tcut_mono_set: "A ⊆ B ⟹ tcut A⋅t ⊑ tcut B⋅t"
  by (auto simp: below_tuple_def fun_below_iff tcut_tget)

text ‹The support of a tuple: the coordinates it defines.  A key carries its own dimensions.›

definition tsupp :: "('d, 'v::pcpo) tuple ⇒ 'd set" where
  "tsupp K = {d. K⟨d⟩ ≠ ⊥}"

lemma tsupp_tcut: "tsupp (tcut A⋅K) = tsupp K ∩ A"
  by (auto simp: tsupp_def tcut_tget)

lemma tcut_tsupp: "tcut (tsupp K)⋅K = K"
  by (rule tuple_eqI) (simp add: tcut_tget tsupp_def)

lemma tcut_below_tcut: "tsupp K ∩ A ⊆ B ⟹ tcut A⋅K ⊑ tcut B⋅K"
  by (auto simp: below_tuple_def fun_below_iff tcut_tget tsupp_def)

lemma tsupp_mono: "x ⊑ y ⟹ tsupp x ⊆ tsupp y"
  by (auto simp: tsupp_def below_tuple_def fun_below_iff) (metis bottomI)

lemma adm_finite_bex: "finite A ⟹ (⋀a. a ∈ A ⟹ adm (P a)) ⟹ adm (λx. ∃a∈A. P a x)"
  by (induct rule: finite_induct) simp_all

text ‹Being defined on finitely many dimensions is an open condition.›

lemma adm_not_tsupp: "finite D ⟹ adm (λt. ¬ D ⊆ tsupp t)"
proof -
  assume "finite D"
  then have "adm (λt. ∃d∈D. t⟨d⟩ ⊑ ⊥)"
    by (rule adm_finite_bex) (intro adm_below cont_const, simp add: cont2cont_fun cont_tget)
  then show ?thesis by (simp add: tsupp_def subset_iff Bex_def)
qed

text ‹With chain-finite (e.g.\ flat) values, a cut to finitely many dimensions is compact, and
  so is a key with finite support.›

lemma compact_tcut:
  fixes t :: "('d, 'v::{pcpo, chfin}) tuple"
  assumes "finite A"
  shows "compact (tcut A⋅t)"
proof (rule compactI)
  have "(¬ tcut A⋅t ⊑ u) ⟷ (∃a∈A. ¬ t⟨a⟩ ⊑ u⟨a⟩)" for u
    by (auto simp: below_tuple_def fun_below_iff tcut_tget)
  moreover have "adm (λu. ∃a∈A. ¬ t⟨a⟩ ⊑ u⟨a⟩)"
    using assms by (rule adm_finite_bex) (simp add: cont2cont_fun cont_tget)
  ultimately show "adm (λu. ¬ tcut A⋅t ⊑ u)" by simp
qed

lemma compact_tsupp:
  fixes K :: "('d, 'v::{pcpo, chfin}) tuple"
  assumes "finite (tsupp K)"
  shows "compact K"
  using compact_tcut[OF assms, of K] by (simp add: tcut_tsupp)

lemma tget_lub: "chain Y ⟹ (⨆i. Y i)⟨d⟩ = (⨆i. (Y i)⟨d⟩)"
  by (simp add: lub_tuple Abs_tuple_inverse' lub_fun ch2ch_cont cont_tget)

text ‹Conversely, a compact key has finite support (whatever the values): otherwise cutting
  away ever smaller tails of an infinite sequence in its support approximates it by a chain of
  keys none of which is above it.  So compactness, the finitely observable keys, is what keys
  should be; for chain-finite values it is exactly finite support.›

lemma compact_finite_tsupp:
  fixes K :: "('d, 'v::pcpo) tuple"
  assumes "compact K"
  shows "finite (tsupp K)"
proof (rule ccontr)
  assume "infinite (tsupp K)"
  then obtain f :: "nat ⇒ 'd" where f: "inj f" "range f ⊆ tsupp K"
    using infinite_countable_subset by blast
  define Y where "Y i = tcut (- f ` {i..})⋅K" for i
  have Y: "chain Y"
    by (rule chainI) (auto simp: Y_def intro: tcut_mono_set)
  have "K ⊑ (⨆i. Y i)"
  proof -
    have "K⟨d⟩ ⊑ (⨆i. (Y i)⟨d⟩)" for d
    proof -
      have ch: "chain (λi. (Y i)⟨d⟩)"
        using Y by (simp add: chain_def below_tuple_def fun_below_iff)
      obtain j where "(Y j)⟨d⟩ = K⟨d⟩"
      proof (cases "d ∈ range f")
        case True
        then obtain j where d: "d = f j" by blast
        have "(Y (Suc j))⟨d⟩ = K⟨d⟩"
          unfolding Y_def by (simp add: tcut_tget d inj_image_mem_iff[OF f(1)])
        then show ?thesis by (rule that)
      next
        case False
        then have "(Y 0)⟨d⟩ = K⟨d⟩" by (auto simp: Y_def tcut_tget)
        then show ?thesis by (rule that)
      qed
      then show ?thesis using is_ub_thelub[OF ch, of j] by simp
    qed
    then show ?thesis by (simp add: below_tuple_def fun_below_iff tget_lub[OF Y])
  qed
  then obtain i where "K ⊑ Y i" using assms Y compact_below_lub_iff by blast
  then have "K⟨f i⟩ ⊑ (Y i)⟨f i⟩" by (simp add: below_tuple_def fun_below_iff)
  moreover have "(Y i)⟨f i⟩ = ⊥" by (simp add: Y_def tcut_tget)
  moreover have "K⟨f i⟩ ≠ ⊥" using f(2) by (auto simp: tsupp_def)
  ultimately show False by simp
qed

text ‹Cutting a compact key leaves it compact.›

lemma compact_tcut_compact:
  fixes K :: "('d, 'v::pcpo) tuple"
  assumes "compact K"
  shows "compact (tcut A⋅K)"
proof (rule compactI2)
  fix Y assume Y: "chain Y" and below: "tcut A⋅K ⊑ (⨆i. Y i)"
  define Z where "Z i = Abs_tuple (λd. if d ∈ A then (Y i)⟨d⟩ else K⟨d⟩)" for i
  have Zd: "(Z i)⟨d⟩ = (if d ∈ A then (Y i)⟨d⟩ else K⟨d⟩)" for i d
    by (simp add: Z_def Abs_tuple_inverse')
  have Z: "chain Z"
    using Y by (auto intro!: chainI simp: below_tuple_def fun_below_iff Zd chain_def)
  have "K ⊑ (⨆i. Z i)"
  proof -
    have "K⟨d⟩ ⊑ (⨆i. (Z i)⟨d⟩)" for d
    proof (cases "d ∈ A")
      case True
      from below have "(tcut A⋅K)⟨d⟩ ⊑ (⨆i. Y i)⟨d⟩" by (simp add: below_tuple_def fun_below_iff)
      with True show ?thesis by (simp add: Zd tget_lub[OF Y] tcut_tget)
    next
      case False then show ?thesis by (simp add: Zd)
    qed
    then show ?thesis by (simp add: below_tuple_def fun_below_iff tget_lub[OF Z])
  qed
  then obtain i where "K ⊑ Z i" using assms Z compact_below_lub_iff by blast
  then have "tcut A⋅K ⊑ Y i"
    by (auto simp: below_tuple_def fun_below_iff Zd tcut_tget split: if_splits)
  then show "∃i. tcut A⋅K ⊑ Y i" ..
qed

subsection ‹Compact tuples›


text ‹A tuple is compact exactly when it has finite support and compact components.  A component
  of a compact tuple is compact: a chain approximating the component, placed into the tuple,
  approximates the whole tuple.›

lemma compact_tget:
  fixes K :: "('d, 'v::pcpo) tuple"
  assumes "compact K" shows "compact (K⟨d⟩)"
proof (rule compactI2)
  fix Y :: "nat ⇒ 'v" assume Y: "chain Y" and le: "K⟨d⟩ ⊑ (⨆i. Y i)"
  define Z where "Z i = Abs_tuple ((tget K)(d := Y i))" for i
  have Zd: "(Z i)⟨e⟩ = (if e = d then Y i else K⟨e⟩)" for i e
    by (simp add: Z_def Abs_tuple_inverse')
  have Z: "chain Z"
    using Y by (simp add: chain_def tuple_below_iff Zd)
  have "K ⊑ (⨆i. Z i)"
    using le by (simp add: tuple_below_iff tget_lub[OF Z] Zd Y)
  then obtain i where "K ⊑ Z i" using compactD2[OF assms Z] by blast
  then have "K⟨d⟩ ⊑ (Z i)⟨d⟩" by (simp add: tuple_below_iff)
  then show "∃i. K⟨d⟩ ⊑ Y i" by (auto simp: Zd)
qed

lemma compact_tupleI:
  fixes K :: "('d, 'v::pcpo) tuple"
  assumes "finite (tsupp K)" and "⋀d. compact (K⟨d⟩)"
  shows "compact K"
proof (rule compactI)
  have "(¬ K ⊑ u) ⟷ (∃d∈tsupp K. ¬ K⟨d⟩ ⊑ u⟨d⟩)" for u
    by (auto simp: tuple_below_iff tsupp_def) (metis minimal)
  moreover have "adm (λu. ∃d∈tsupp K. ¬ K⟨d⟩ ⊑ u⟨d⟩)"
    using assms(1) proof (rule adm_finite_bex)
    show "adm (λu. ¬ K⟨d⟩ ⊑ u⟨d⟩)" for d
      by (rule adm_compact_not_below[OF assms(2)]) (simp add: cont2cont_fun cont_tget)
  qed
  ultimately show "adm (λu. ¬ K ⊑ u)" by simp
qed
