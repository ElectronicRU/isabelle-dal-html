Theory Interp

theory Interp imports Complex_Main begin

section ‹Language algebra. Initial algebra is defined as term‹etyp›, and a generic algebra is
defined as term‹interp›.›

datatype etyp =
  int | float | numeric | string | date | datetime | interval

type_synonym fid = string


datatype esyntax =
  Int_lit int
  | Bool_lit bool
  | Numeric_lit rat
  | Lit etyp string
  | Call fid "esyntax list"

record 'a interp =
  int_lit :: "int ⇒ 'a"
  bool_lit :: "bool ⇒ 'a"
  numeric_lit :: "rat ⇒ 'a"
  lit :: "etyp ⇒ string ⇒ 'a"
  call :: "fid ⇒ 'a list ⇒ 'a"

text ‹Catamorphism esyntax => interp.›
fun efold :: "'a interp ⇒ esyntax ⇒ 'a" where
  efold_int_lit: "efold R (Int_lit i) = int_lit R i"
| efold_bool_lit: "efold R (Bool_lit b) = bool_lit R b"
| efold_numeric_lit: "efold R (Numeric_lit r) = numeric_lit R r"
| efold_lit: "efold R (Lit t s) = lit R t s"
| efold_call: "efold R (Call f args) = call R f (map (efold R) args)"



definition Syntax_interp :: "esyntax interp" where
‹Syntax_interp = ⦇
  int_lit = Int_lit,
  bool_lit = Bool_lit,
  numeric_lit = Numeric_lit,
  lit = Lit,
  call = Call
⦈›


inductive_set generated:: "'a interp ⇒ 'a set" for i where
  int_lit: "int_lit i x ∈ generated i"
| bool_lit: "bool_lit i x ∈ generated i"
| numeric_lit: "numeric_lit i x ∈ generated i"
| lit: "lit i t x ∈ generated i"
| call: "(⋀ x::'a. x ∈ set args ⟹ x ∈ generated i) ⟹ call i fn args ∈ generated i"

abbreviation Generated :: "'a interp ⇒ 'a set" where "Generated i ≡ range (efold i)"

lemma efold_generated: "efold i e ∈ generated i"
  by (induction e, simp_all add: generated.intros, rule, auto)

lemma generated_efold: "x ∈ generated i ⟹ ∃ e. efold i e = x"
proof (induction x rule: generated.induct, auto intro!: efold.simps)
  case (call args fn)
  assume arg_exist: "⋀ arg. (arg ∈ set args ⟹ (∃ e. efold i e = arg))"
  have "∃ args0. map (efold i) args0 = args" by (metis arg_exist ex_map_conv)
  then obtain args0 where args0_map: "map (efold i) args0 = args" ..
  define e where "e = Call fn args0"
  then have "efold i e = call i fn args" by (simp add: args0_map)
  then show "∃ e. efold i e = call i fn args" ..
  qed

theorem generated_equiv: "generated i = Generated i"
  using generated_efold efold_generated by fast

definition is_hom :: "'a interp ⇒ 'b interp ⇒ ('a ⇒ 'b) ⇒ bool" where
‹is_hom A B f ≡ ∀ x. f (efold A x) = efold B x›

lemma Syntax_idem : "efold Syntax_interp x = x"
  by (induction x) (auto simp: Syntax_interp_def list.map_ident_strong)

lemma cata1 : "is_hom Syntax_interp i (efold i)"
  by (simp add: is_hom_def Syntax_idem)

lemma cata2 : "is_hom Syntax_interp i f ⟹ f x = efold i x"
  by (simp add: is_hom_def Syntax_idem)

theorem is_cata : "is_hom Syntax_interp i f = (f = efold i)"
  using cata1 cata2 by blast

type_synonym 'a esyntax' = "'a interp ⇒ 'a"

text ‹Functions which are "morally an AST"›
definition syntax_uniform where "syntax_uniform i f g ≡ f i = efold i g"

theorem tst: assumes "f i = efold i (f Syntax_interp)" shows True
  by simp

end
