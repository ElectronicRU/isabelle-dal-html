section ‹Finitary semi-factors and locally finitely detailed domains›

text ‹A semi-factor is ∗‹finitary› when all its elements are compact (equivalently, as a
  subdomain it has finite height): its projection takes every element, however large, to a
  compact one.  Finitary semi-factors are closed under meets (intersection), joins (pointwise
  joins of compacts) and contain the bottom semi-factor.›


default_sort "{bifinite, distributive_cpo}"

text ‹A domain is ∗‹locally finitely detailed› when every compact element lives at some finite
  level of detail, i.e.\ lies in a finitary semi-factor: every finite piece of information has a
  finite, self-contained level.  This is the semi-factor analogue of property I of dI-domains
  (for stable subdomains the least one containing a compact ‹k› is ‹↓k›, and a distributive lattice of
  finite height is finite), and strictly stronger.  (Compare also Libkin's requirement that the
  compact elements -- the storable objects -- form an ideal.)›

default_sort type

definition finitary :: "('a::pcpo → 'a) ⇒ bool" where
  "finitary d ⟷ (∀x. compact (d⋅x))"

definition faithful :: "('a::pcpo → 'a) ⇒ bool" where
  "faithful d ⟷ (∀x. d⋅x = ⊥ ⟶ x = ⊥)"

lemma finitary_bottom: "finitary ⊥"
  by (simp add: finitary_def)

text ‹Over a locally finitely detailed dI-domain the compact semi-factors, which are the levels of
  detail, are finitary (‹compact_semi_factor_proj› below).  The constructions act on
  the deflations; @{const semi_factor_of_defl} turns the result back into a semi-factor.›

definition finitary_semi_factor :: "'a::{bifinite, distributive_cpo} semi_factor ⇒ bool" where
  "finitary_semi_factor F ⟷ finitary (semi_factor_p F)"

lemma ex_finitary_semi_factor_iff:
  fixes k :: "'a::{bifinite, distributive_cpo}"
  shows "(∃F. finitary_semi_factor F ∧ k ⇂ F = k) ⟷ (∃e. semi_factor_defl e ∧ finitary e ∧ e⋅k = k)"
proof
  assume "∃F. finitary_semi_factor F ∧ k ⇂ F = k"
  then obtain F where "finitary_semi_factor F" "k ⇂ F = k" by blast
  then show "∃e. semi_factor_defl e ∧ finitary e ∧ e⋅k = k"
    by (intro exI[of _ "semi_factor_p F"])
      (simp add: semi_factor_defl_semi_factor_p finitary_semi_factor_def)
next
  assume "∃e. semi_factor_defl e ∧ finitary e ∧ e⋅k = k"
  then obtain e where "semi_factor_defl e" "finitary e" "e⋅k = k" by blast
  then show "∃F. finitary_semi_factor F ∧ k ⇂ F = k"
    by (intro exI[of _ "semi_factor_of_defl e"])
      (simp add: finitary_semi_factor_def semi_factor_p_of_defl)
qed

class locally_finitely_detailed = bifinite + distributive_cpo +
  assumes compact_level: "compact k ⟹ ∃F. finitary_semi_factor F ∧ k ⇂ F = k"

text ‹The domains of rows and keys: locally finitely detailed with algebraic semi-factors.  For
  now the latter is property I of dI-domains; it is meant to weaken to algebraicity of the
  semi-factors alone, which separated sums also give.›

class lod_domain = dI + locally_finitely_detailed

lemma semi_factor_defl_u_map:
  assumes f: "semi_factor_defl (f :: 'a::{bifinite, distributive_cpo} → 'a)"
  shows "semi_factor_defl (u_map⋅f)"
proof (rule semi_factor_deflI)
  show "stable_defl (u_map⋅f)" by (rule stable_defl_u_map[OF semi_factor_defl.axioms(1)[OF f]])
  fix x y :: "'a u" assume y: "u_map⋅f⋅y = y" and le: "u_map⋅f⋅x ⊑ y"
  show "x ⇑ y"
  proof (cases x)
    case bottom then show ?thesis by (simp add: bottom_joinable)
  next
    case (up a)
    with le obtain b where b: "y = up⋅b" "f⋅a ⊑ b" by (cases y) auto
    with y have "f⋅b = b" by simp
    from semi_factor_defl.slide[OF f this b(2)] up b(1) show ?thesis
      by (simp add: cfun_arg_joinable)
  qed
qed

lemma semi_factor_defl_iso:
  fixes abs :: "'r::{bifinite, distributive_cpo} → 't::{bifinite, distributive_cpo}" and rep
  assumes iso_abs: "⋀x. rep⋅(abs⋅x) = x" and iso_rep: "⋀y. abs⋅(rep⋅y) = y"
    and f: "semi_factor_defl f"
  shows "semi_factor_defl (abs oo f oo rep)"
proof (rule semi_factor_deflI)
  have iso: "iso abs rep" by (rule iso.intro[OF iso_abs iso_rep])
  interpret iso abs rep by (fact iso)
  show "stable_defl (abs oo f oo rep)"
    by (rule stable_defl_iso[OF iso_abs iso_rep semi_factor_defl.axioms(1)[OF f]])
  fix x y assume y: "(abs oo f oo rep)⋅y = y" and le: "(abs oo f oo rep)⋅x ⊑ y"
  have "rep⋅(abs⋅(f⋅(rep⋅y))) = rep⋅y" using y by simp
  then have fy: "f⋅(rep⋅y) = rep⋅y" by simp
  have "rep⋅((abs oo f oo rep)⋅x) ⊑ rep⋅y" using le by (rule monofun_cfun_arg)
  then have "f⋅(rep⋅x) ⊑ rep⋅y" by simp
  then have "rep⋅x ⇑ rep⋅y" by (rule semi_factor_defl.slide[OF f fy])
  then show "x ⇑ y" by (simp add: iso_rep_joinable_iff[OF iso])
qed

qed

subsection ‹Discriminator for separated sums›

text ‹In a separated sum every nonbottom element lies above a unique atom.  The discriminator
  sends it there: it is the lub of the step functions ‹stepfn a a› at the atoms, which are
  compact.  It is a faithful finitary semi-factor deflation, whose image is ‹⊥› and the atoms.›


definition discriminator :: "'a::{bifinite, bounded_complete, separated_sum} → 'a"
  where "discriminator = lub {stepfn a a |a. atom a}"

lemma discriminator_bound:
  "{stepfn a a |a. atom a} <| (ID :: 'a::{bifinite, bounded_complete, separated_sum} → 'a)"
  by (auto intro!: is_ubI simp: stepfn_below_iff atom_compact)

lemma discriminator_app:
  "discriminator⋅x = lub ((λa. if a ⊑ x then a else ⊥) ` {a. atom a})"
proof -
  have S: "{stepfn a a |a. atom a} = (λa. stepfn a a) ` {a. atom a}" by blast
  have "(λf. f⋅x) ` {stepfn a a |a. atom a} = (λa. stepfn a a⋅x) ` {a. atom a}"
    unfolding S image_image ..
  also have "… = (λa. if a ⊑ x then a else ⊥) ` {a. atom a}"
    by (rule image_cong[OF refl]) (simp add: stepfn_beta[OF atom_compact])
  finally have "(λf. f⋅x) ` {stepfn a a |a. atom a} = (λa. if a ⊑ x then a else ⊥) ` {a. atom a}" .
  then show ?thesis
    unfolding discriminator_def cfun_lub[OF discriminator_bound] by simp
qed

lemma discriminator_eq:
  assumes a: "atom a" and ax: "a ⊑ x"
  shows "discriminator⋅x = a"
  unfolding discriminator_app
proof (rule lub_eqI, rule is_lubI)
  show "(λa. if a ⊑ x then a else ⊥) ` {a. atom a} <| a"
    by (auto intro!: is_ubI dest: atom_unique[OF _ a _ ax])
next
  fix u assume "(λa. if a ⊑ x then a else ⊥) ` {a. atom a} <| u"
  moreover have "a ∈ (λa. if a ⊑ x then a else ⊥) ` {a. atom a}"
    using a ax by (auto intro!: image_eqI[of _ _ a])
  ultimately show "a ⊑ u" by (rule is_ubD)
qed

lemma discriminator_strict [simp]: "discriminator⋅⊥ = ⊥"
  unfolding discriminator_app
  by (rule lub_eqI) (auto intro!: is_lubI is_ubI dest: atom_nonbottom)

lemma discriminator_atom:
  assumes "x ≠ ⊥" shows "atom (discriminator⋅x)" and "discriminator⋅x ⊑ x"
proof -
  obtain a where "atom a" "a ⊑ x" using assms by (rule atom_below)
  then show "atom (discriminator⋅x)" "discriminator⋅x ⊑ x" by (simp_all add: discriminator_eq)
qed


lemma discriminator_below: "discriminator⋅x ⊑ x"
  by (cases "x = ⊥") (simp_all add: discriminator_atom)

lemma discriminator_fixes: "discriminator⋅x = x ⟷ x = ⊥ ∨ atom x"
  by (metis discriminator_atom(1) discriminator_eq below_refl discriminator_strict)

lemma discriminator_faithful: "faithful discriminator"
  unfolding faithful_def using discriminator_atom atom_nonbottom by blast

lemma discriminator_finitary: "finitary discriminator"
  unfolding finitary_def
  by (metis atom_compact compact_bottom discriminator_atom discriminator_strict)

lemma stable_discriminator: "stable_defl discriminator"
proof standard
  fix x show "discriminator⋅(discriminator⋅x) = discriminator⋅x"
    by (cases "x = ⊥") (simp_all add: discriminator_atom discriminator_eq)
next
  fix x show "discriminator⋅x ⊑ x" by (rule discriminator_below)
next
  fix x y assume "discriminator⋅y = y" and "x ⊑ y"
  then show "discriminator⋅x = x" by (auto simp: discriminator_fixes dest: atomD)
qed

lemma discriminator_semi_factor:
  "semi_factor_defl (discriminator :: 'a::{bifinite, distributive_cpo, separated_sum} → 'a)"
proof (rule semi_factor_deflI[OF stable_discriminator])
  fix x y :: 'a assume y: "discriminator⋅y = y" and le: "discriminator⋅x ⊑ y"
  show "x ⇑ y"
  proof (cases "x = ⊥")
    case True then show ?thesis by (simp add: bottom_joinable)
  next
    case False
    note a = discriminator_atom[OF False]
    have "y ≠ ⊥" using le atom_nonbottom[OF a(1)] by (metis below_bottom_iff)
    with y have "atom y" by (simp add: discriminator_fixes)
    with a(1) le have "discriminator⋅x = y" using atomD atom_nonbottom by blast
    with a(2) have "y ⊑ x" by simp
    then show ?thesis by (rule joinableI[OF below_refl])
  qed
qed

lemma discriminator_deflation: "deflation discriminator"
  by (rule stable_defl.deflation[OF stable_discriminator])

text ‹In a separated sum every nonbottom semi-factor is faithful: an element sent to ‹⊥› is
  joinable with every image element (slide rule), hence shares an atom with a nonbottom one; the
  atom is in the image (stability), and below the element, so it is sent to ‹⊥› as well.›

lemma semi_factor_defl_faithful:
  fixes d :: "'a::{bifinite, distributive_cpo, separated_sum} → 'a"
  assumes d: "semi_factor_defl d" and a: "d⋅a = a" "a ≠ ⊥"
  shows "faithful d"
  unfolding faithful_def
proof (intro allI impI, rule ccontr)
  interpret D: semi_factor_defl d by (rule d)
  fix x assume dx: "d⋅x = ⊥" and x: "x ≠ ⊥"
  have xa: "x ⇑ a" using a(1) by (rule D.slide) (simp add: dx)
  obtain l where l: "atom l" "l ⊑ x" "l ⊑ a"
    using separated_sum[OF x a(2) joinableD(1)[OF xa] joinableD(2)[OF xa]] by blast
  have "d⋅l = l" using a(1) l(3) by (rule D.downclosed)
  moreover have "d⋅l ⊑ d⋅x" using l(2) by (rule monofun_cfun_arg)
  ultimately have "l = ⊥" using dx by simp
  with atom_nonbottom[OF l(1)] show False by contradiction
qed


subsection ‹Finitary and faithful deflations under joins›

lemma finitary_join:
  fixes p q :: "'a::bounded_complete → 'a"
  assumes p: "deflation p" and q: "deflation q" and "finitary p" "finitary q"
  shows "finitary (p ⊔ q)"
  using assms(3,4)
  unfolding finitary_def cfun_join[OF deflation_joinable[OF p q]]
  by (blast intro: compact_join joinableI deflation.below[OF p] deflation.below[OF q])

text ‹Faithfulness is upward closed, so it passes to joins.›

lemma faithful_mono: "faithful p ⟹ p ⊑ q ⟹ faithful q"
  unfolding faithful_def by (metis monofun_cfun_fun below_bottom_iff)

lemma faithful_joinI1:
  fixes p q :: "'a::bounded_complete → 'a"
  shows "deflation p ⟹ deflation q ⟹ faithful p ⟹ faithful (p ⊔ q)"
  by (erule faithful_mono[OF _ joinableD(1)[OF deflation_joinable]])

lemma faithful_joinI2:
  fixes p q :: "'a::bounded_complete → 'a"
  shows "deflation p ⟹ deflation q ⟹ faithful q ⟹ faithful (p ⊔ q)"
  by (erule faithful_mono[OF _ joinableD(2)[OF deflation_joinable]])

text ‹In a separated sum, the faithful deflations are exactly those above the discriminator.›

lemma faithful_iff_discriminator_below:
  fixes d :: "'a::{bifinite, bounded_complete, separated_sum} → 'a"
  assumes d: "deflation d"
  shows "faithful d ⟷ discriminator ⊑ d"
proof
  assume f: "faithful d"
  show "discriminator ⊑ d"
  proof (rule cfun_belowI)
    fix x show "discriminator⋅x ⊑ d⋅x"
    proof (cases "x = ⊥")
      case False
      with f have "d⋅x ≠ ⊥" unfolding faithful_def by blast
      then obtain a where a: "atom a" "a ⊑ d⋅x" by (rule atom_below)
      have "a ⊑ x" using a(2) deflation.below[OF d] by (rule below_trans)
      with a show ?thesis by (simp add: discriminator_eq)
    qed simp
  qed
next
  assume "discriminator ⊑ d"
  with discriminator_faithful show "faithful d" by (rule faithful_mono)
qed

text ‹Hence the discriminator lies below every nonbottom semi-factor: it is the only atom of the
  lattice of semi-factors.›

lemma discriminator_below_semi_factor_defl:
  fixes d :: "'a::{bifinite, distributive_cpo, separated_sum} → 'a"
  assumes d: "semi_factor_defl d" and nz: "d ≠ ⊥"
  shows "discriminator ⊑ d"
proof -
  interpret D: semi_factor_defl d by (rule d)
  from nz obtain x where x: "d⋅x ≠ ⊥" using cfun_eqI[of d ⊥] by auto
  have "faithful d" by (rule semi_factor_defl_faithful[OF d D.idem x])
  then show ?thesis by (simp add: faithful_iff_discriminator_below[OF D.deflation])
qed



subsection ‹Local finite detail as a deflation predicate›

text ‹Every compact element of the image is fixed by a finitary level of the whole type.  The
  level is not relative to the image, which makes the predicate admissible for free: a compact
  element fixed by the limit of a chain is fixed at some finite stage.  At ‹ID› it is
  @{class locally_finitely_detailed}.  The strict combinators need faithful levels of their
  components; these come from the components being separated sums, as a sort constraint.
  Lifting needs nothing.›

locale locally_finitely_detailed_defl = deflation d for d :: "'a::{bifinite, distributive_cpo} → 'a" +
  assumes level_image: "compact k ⟹ d⋅k = k ⟹ ∃F. finitary_semi_factor F ∧ k ⇂ F = k"

lemma (in locally_finitely_detailed_defl) deflation: "deflation d" by (fact deflation_axioms)

lemmas (in locally_finitely_detailed_defl) level_image_defl =
  level_image[unfolded ex_finitary_semi_factor_iff]

text ‹The introduction rule asks for the levels as deflations, which is the form the
  constructions produce.›

lemma locally_finitely_detailed_deflI:
  assumes "deflation d"
    and "⋀k. compact k ⟹ d⋅k = k ⟹ ∃e. semi_factor_defl e ∧ finitary e ∧ e⋅k = k"
  shows "locally_finitely_detailed_defl d"
  using assms
  by (simp add: locally_finitely_detailed_defl_def locally_finitely_detailed_defl_axioms_def
      ex_finitary_semi_factor_iff)

lemma locally_finitely_detailed_defl_bottom:
  "locally_finitely_detailed_defl (⊥ :: 'a::{bifinite, distributive_cpo} → 'a)"
  by (rule locally_finitely_detailed_deflI[OF deflation_bottom])
    (use semi_factor_defl_bottom finitary_bottom in auto)

lemma locally_finitely_detailed_defl_ID:
  "locally_finitely_detailed_defl (ID :: 'a::locally_finitely_detailed → 'a)"
  by (rule locally_finitely_detailed_deflI[OF deflation_ID])
    (rule compact_level[unfolded ex_finitary_semi_factor_iff]; simp)

lemma locally_finitely_detailed_class_deflI:
  assumes d: "locally_finitely_detailed_defl (ID :: 'a::{bifinite, distributive_cpo} → 'a)"
  shows "OFCLASS('a, locally_finitely_detailed_class)"
  by intro_classes
    (rule locally_finitely_detailed_defl.level_image[OF d]; simp)

lemma adm_locally_finitely_detailed_defl:
  "adm (locally_finitely_detailed_defl :: ('a::{bifinite, distributive_cpo} → 'a) ⇒ bool)"
proof (rule admI)
  fix Y :: "nat ⇒ 'a → 'a" assume Y: "chain Y" and P: "∀i. locally_finitely_detailed_defl (Y i)"
  show "locally_finitely_detailed_defl (⨆i. Y i)"
  proof (rule locally_finitely_detailed_deflI)
    show "deflation (⨆i. Y i)"
      by (rule admD[OF adm_deflation Y]) (use P in ‹simp add: locally_finitely_detailed_defl_def›)
    fix k assume k: "compact k" and fk: "(⨆i. Y i)⋅k = k"
    have chk: "chain (λi. Y i⋅k)" by (rule ch2ch_Rep_cfunL[OF Y])
    have "k ⊑ (⨆i. Y i⋅k)" using fk by (simp add: contlub_cfun_fun[OF Y])
    then obtain n where "k ⊑ Y n⋅k" using compactD2[OF k chk] by blast
    then have "Y n⋅k = k"
      using deflation.below[OF locally_finitely_detailed_defl.deflation[OF P[rule_format]]]
      by (rule below_antisym[rotated])
    then show "∃e. semi_factor_defl e ∧ finitary e ∧ e⋅k = k"
      by (rule locally_finitely_detailed_defl.level_image_defl[OF P[rule_format] k])
  qed
qed

subsection ‹Tuples›

text ‹Under the isomorphism with families of semi-factors, the finitary semi-factors of the tuples
  are the finitely supported families of finitary semi-factors.›

lemma finitary_semi_factor_of_family:
  fixes F :: "'d::countable ⇒ 'v::{bifinite, distributive_cpo} semi_factor"
  shows "finitary_semi_factor (semi_factor_of_family F) ⟷
    finite {d. F d ≠ ⊥} ∧ (∀d. finitary_semi_factor (F d))"
proof safe
  assume fin: "finitary_semi_factor (semi_factor_of_family F)"
  then have cpt: "compact (t ⇂ semi_factor_of_family F)" for t
    by (simp add: finitary_semi_factor_def finitary_def)
  show "finitary_semi_factor (F d)" for d
    unfolding finitary_semi_factor_def finitary_def
  proof
    fix x
    have "compact ((tupd d⋅⊥⋅x ⇂ semi_factor_of_family F)⟨d⟩)" by (rule compact_tget[OF cpt])
    then show "compact (x ⇂ F d)" by (simp add: tupd_bot_tget)
  qed
  show "finite {d. F d ≠ ⊥}"
  proof (rule ccontr)
    assume inf: "infinite {d. F d ≠ ⊥}"
    have "∃x. x ⇂ F d ≠ ⊥" if "F d ≠ ⊥" for d
    proof (rule ccontr)
      assume "∄x. x ⇂ F d ≠ ⊥"
      then have "semi_factor_p (F d) = semi_factor_p ⊥" by (simp add: cfun_eq_iff semi_factor_p_bot)
      with that show False by (simp add: semi_factor_p_inject)
    qed
    then obtain x where x: "⋀d. F d ≠ ⊥ ⟹ x d ⇂ F d ≠ ⊥" by metis
    define t :: "('d, 'v) tuple" where "t = Abs_tuple x"
    have "{d. F d ≠ ⊥} ⊆ tsupp (t ⇂ semi_factor_of_family F)"
      using x by (auto simp: tsupp_def t_def Abs_tuple_inverse')
    with inf compact_finite_tsupp[OF cpt] show False by (metis finite_subset)
  qed
next
  assume fin: "finite {d. F d ≠ ⊥}" and F: "∀d. finitary_semi_factor (F d)"
  show "finitary_semi_factor (semi_factor_of_family F)"
    unfolding finitary_semi_factor_def finitary_def
  proof
    fix t show "compact (t ⇂ semi_factor_of_family F)"
    proof (rule compact_tupleI)
      have "tsupp (t ⇂ semi_factor_of_family F) ⊆ {d. F d ≠ ⊥}"
        by (auto simp: tsupp_def semi_factor_p_bot)
      then show "finite (tsupp (t ⇂ semi_factor_of_family F))" using fin by (rule finite_subset)
      show "compact ((t ⇂ semi_factor_of_family F)⟨d⟩)" for d
        using F by (simp add: finitary_semi_factor_def finitary_def)
    qed
  qed
qed

text ‹A compact tuple has finite support and compact components.  A finitary level containing
  it is the family of finitary levels containing the components on its support, and ‹⊥›
  elsewhere.›

instance tuple :: (countable, locally_finitely_detailed) locally_finitely_detailed
proof
  fix K :: "('a, 'b) tuple" assume K: "compact K"
  define F where "F d = (if d ∈ tsupp K
      then (SOME G. finitary_semi_factor G ∧ K⟨d⟩ ⇂ G = K⟨d⟩) else ⊥)" for d
  have F: "finitary_semi_factor (F d) ∧ K⟨d⟩ ⇂ F d = K⟨d⟩" for d
  proof (cases "d ∈ tsupp K")
    case True
    then show ?thesis using someI_ex[OF compact_level[OF compact_tget[OF K, of d]]]
      by (simp add: F_def)
  next
    case False
    then show ?thesis
      by (simp add: F_def tsupp_def finitary_semi_factor_def finitary_def semi_factor_p_bot)
  qed
  have "{d. F d ≠ ⊥} ⊆ tsupp K" by (auto simp: F_def)
  then have "finite {d. F d ≠ ⊥}" using compact_finite_tsupp[OF K] by (rule finite_subset)
  with F have "finitary_semi_factor (semi_factor_of_family F)"
    by (simp add: finitary_semi_factor_of_family)
  moreover have "K ⇂ semi_factor_of_family F = K" using F by (simp add: tuple_eq_iff)
  ultimately show "∃G. finitary_semi_factor G ∧ K ⇂ G = K" by blast
qed

subsection ‹Compact semi-factors project to compact keys›

text ‹Over a dI-domain every compact semi-factor is a finite join of principals of compacts
  (@{thm [source] compact_semi_factor_finite_join}).  Over a locally finitely detailed one the
  principal of a compact lies below a semi-factor all of whose elements are compact, and finite
  joins keep that property.  So compact semi-factors project every key to a compact one.
  (Without local finite detail this fails: in the separated sum of the powerset of the naturals
  and a point, the principal of the point is the whole domain.)  Projections of compact keys are
  compact over any dI-domain, everything below a compact element being compact.›

lemma compact_finite_lub:
  fixes u :: "'a::cpo"
  assumes fin: "finite C" and comp: "⋀c. c ∈ C ⟹ compact c" and lub: "C <<| u"
  shows "compact u"
proof (rule compactI2)
  fix Y :: "nat ⇒ 'a" assume Y: "chain Y" and uY: "u ⊑ (⨆i. Y i)"
  have "∃i. c ⊑ Y i" if "c ∈ C" for c
    by (rule compactD2[OF comp[OF that] Y])
      (rule below_trans[OF is_ubD[OF is_lubD1[OF lub] that] uY])
  then obtain f where f: "⋀c. c ∈ C ⟹ c ⊑ Y (f c)" by metis
  let ?n = "Max (insert 0 (f ` C))"
  have "c ⊑ Y ?n" if "c ∈ C" for c
    by (rule below_trans[OF f[OF that] chain_mono[OF Y]]) (simp add: fin that)
  then have "u ⊑ Y ?n" by (intro is_lubD2[OF lub] is_ubI)
  then show "∃i. u ⊑ Y i" by blast
qed

subsection ‹Keys maximal at an LOD›

definition maximal_at :: "'k::{bifinite, distributive_cpo} semi_factor ⇒ 'k ⇒ bool" where
  "maximal_at c k ⟷ proj c⋅k = k ∧ (∀t. proj c⋅t = t ⟶ k ⊑ t ⟶ t = k)"

text ‹Maximality passes down to coarser LODs: anything above the projection at the coarser LOD
  slides into a join with the key, which lives at the finer LOD and so cannot exceed it.›

lemma maximal_at_proj:
  assumes ab: "a ⊑ b" and k: "maximal_at b k"
  shows "maximal_at a (proj a⋅k)"
  unfolding maximal_at_def
proof (intro conjI allI impI)
  show "proj a⋅(proj a⋅k) = proj a⋅k" by simp
  fix t assume t: "proj a⋅t = t" and le: "proj a⋅k ⊑ t"
  have kb: "proj b⋅k = k" using k by (simp add: maximal_at_def)
  have tb: "proj b⋅t = t" using t lives_at_up[OF _ ab] by (simp add: lives_at_def)
  have j: "k ⇑ t" by (rule proj_slide[OF t le])
  have "proj b⋅(k ⊔ t) = k ⊔ t" by (simp add: proj_join[OF j] kb tb)
  then have "k ⊔ t = k" using k joinableD(1)[OF j] by (simp add: maximal_at_def)
  then have "t ⊑ k" using joinableD(2)[OF j] by simp
  then have "t ⊑ proj a⋅k" using monofun_cfun_arg[of t k "proj a"] by (simp add: t)
  with le show "t = proj a⋅k" by (rule below_antisym[rotated])
qed

lemma maximal_at_lives: "maximal_at c k ⟹ proj c⋅k = k"
  by (simp add: maximal_at_def)

lemma maximal_at_compact: "maximal_at c k ⟹ compact c ⟹ compact k"
  for c :: "'k::lod_domain semi_factor"
  by (metis maximal_at_lives proj_compact)

text ‹A key below one living at the same LOD is that key, when it is maximal there.›

lemma maximal_at_eq: "maximal_at c k ⟹ proj c⋅t = t ⟹ k ⊑ t ⟹ t = k"
  by (simp add: maximal_at_def)

text ‹Joins of compatible maximal keys are maximal at the join of their LODs.›

lemma maximal_at_join:
  assumes a: "maximal_at A a" and b: "maximal_at B b" and j: "a ⇑ b"
  shows "maximal_at (A ⊔ B) (a ⊔ b)"
  unfolding maximal_at_def
proof (intro conjI allI impI)
  have aA: "proj A⋅a = a" and bB: "proj B⋅b = b" using a b by (simp_all add: maximal_at_def)
  have pa: "proj A⋅(a ⊔ b) = a"
  proof -
    have "proj A⋅a ⊑ proj A⋅(a ⊔ b)" using joinableD(1)[OF j] by (rule monofun_cfun_arg)
    then have "a ⊑ proj A⋅(a ⊔ b)" by (simp only: aA)
    then show ?thesis using a by (simp add: maximal_at_def)
  qed
  have pb: "proj B⋅(a ⊔ b) = b"
  proof -
    have "proj B⋅b ⊑ proj B⋅(a ⊔ b)" using joinableD(2)[OF j] by (rule monofun_cfun_arg)
    then have "b ⊑ proj B⋅(a ⊔ b)" by (simp only: bB)
    then show ?thesis using b by (simp add: maximal_at_def)
  qed
  show "proj (A ⊔ B)⋅(a ⊔ b) = a ⊔ b" by (simp only: proj_sup pa pb)
  fix t assume t: "proj (A ⊔ B)⋅t = t" and le: "a ⊔ b ⊑ t"
  have at: "a ⊑ t" and bt: "b ⊑ t" using le joinableD[OF j] by (blast intro: below_trans)+
  have "proj A⋅t = a"
  proof -
    have "a ⊑ proj A⋅t" using monofun_cfun_arg[OF at, of "proj A"] by (simp add: aA)
    then show ?thesis using a by (simp add: maximal_at_def)
  qed
  moreover have "proj B⋅t = b"
  proof -
    have "b ⊑ proj B⋅t" using monofun_cfun_arg[OF bt, of "proj B"] by (simp add: bB)
    then show ?thesis using b by (simp add: maximal_at_def)
  qed
  ultimately show "t = a ⊔ b" using t by (simp add: proj_sup)
qed

text ‹A key maximal at ‹a› that has a common upper bound with ‹K› lies above the projection of
  ‹K›: the projection of their join is above the key, hence equal to it.  (Under a pinning slicer,
  conformity provides the common upper bound.)›

lemma proj_below_maximal:
  assumes K1: "maximal_at a K1" and KK2: "K ⊑ K2" and K1K2: "K1 ⊑ K2"
  shows "proj a⋅K ⊑ K1"
proof -
  have j: "K ⇑ K1" by (rule joinableI[OF KK2 K1K2])
  have "proj a⋅K1 ⊑ proj a⋅(K ⊔ K1)" using joinableD(2)[OF j] by (rule monofun_cfun_arg)
  then have "K1 ⊑ proj a⋅(K ⊔ K1)" using K1 by (simp add: maximal_at_def)
  then have eq: "proj a⋅(K ⊔ K1) = K1" using K1 by (simp add: maximal_at_def)
  have "proj a⋅K ⊑ proj a⋅(K ⊔ K1)" using joinableD(1)[OF j] by (rule monofun_cfun_arg)
  then show ?thesis by (simp only: eq)
qed

text ‹Round trip between two LODs that agree up to the LOD of a pinned key: from a key ‹K1› at
  ‹a› compatible with the pin, go to ‹a'› through the join with the pin, and back.›

lemma proj_round_trip:
  assumes aL: "a ⊔ L = a' ⊔ L" and K: "maximal_at L K"
    and K1: "proj a⋅K1 = K1" and le: "proj a⋅K ⊑ K1"
  shows "proj a⋅(proj a'⋅(K1 ⊔ K) ⊔ K) = K1"
proof -
  have j: "K1 ⇑ K" using proj_slide[OF K1 le] by (simp add: joinable_sym)
  let ?J = "K1 ⊔ K"
  have JK: "K ⊑ ?J" using j by (rule joinableD)
  have jJ: "proj a'⋅?J ⇑ K" by (rule joinableI[OF proj_below JK])
  have KL: "proj L⋅K = K" using K by (simp add: maximal_at_def)
  have LJ: "proj L⋅?J = K"
  proof -
    have "proj L⋅K ⊑ proj L⋅?J" using JK by (rule monofun_cfun_arg)
    then have "K ⊑ proj L⋅?J" by (simp only: KL)
    then show ?thesis using K by (simp add: maximal_at_def)
  qed
  have a_split: "a = (a ⊓ a') ⊔ (a ⊓ L)"
    by (metis aL lattice_meet_join_absorb lattice_meet_join_distrib1)
  have "proj a⋅?J = proj (a ⊓ a')⋅?J ⊔ proj a⋅K"
    by (subst a_split) (simp add: proj_sup proj_inf LJ)
  moreover have "proj a⋅?J = K1"
    by (simp add: proj_join[OF j] K1 join_absorb2[OF le])
  ultimately have "K1 = proj (a ⊓ a')⋅?J ⊔ proj a⋅K" by simp
  then show ?thesis by (simp add: proj_join[OF jJ] proj_inf)
qed

  shows "proj ((τ ⊓ c))⋅k = proj ((τ ⊓ c'))⋅k"
proof -

  also have "… = proj ((τ ⊓ c'))⋅k" using c by (simp add: lives_at_def)
  finally show ?thesis .
qed

subsection ‹Compact maps›

text ‹A map into a pcpo is compact in the pointwise order exactly when it has finite support and
  compact values.›

lemma compact_fun_iff:
  fixes f :: "'a ⇒ 'b::pcpo"
  shows "compact f ⟷ finite {x. f x ≠ ⊥} ∧ (∀x. compact (f x))"
proof
  assume c: "compact f"
  have "compact (f x)" for x
  proof (rule compactI2)
    fix C :: "nat ⇒ 'b" assume C: "chain C" and le: "f x ⊑ (⨆i. C i)"
    let ?Y = "λi. f(x := C i)"
    have Y: "chain ?Y" by (rule chainI, rule fun_belowI) (simp add: chainE[OF C])
    have "f ⊑ (⨆i. ?Y i)"
    proof (rule fun_belowI)
      fix y show "f y ⊑ (⨆i. ?Y i) y" by (cases "y = x") (simp_all add: lub_fun[OF Y] le)
    qed
    then obtain i where "f ⊑ ?Y i" using compactD2[OF c Y] by blast
    then show "∃i. f x ⊑ C i" by (metis fun_belowD fun_upd_same)
  qed
  moreover have "finite {x. f x ≠ ⊥}"
  proof (rule ccontr)
    assume "infinite {x. f x ≠ ⊥}"
    then obtain g :: "nat ⇒ 'a" where g: "inj g" "range g ⊆ {x. f x ≠ ⊥}"
      using infinite_countable_subset by blast
    let ?Y = "λn y. if y ∈ g ` {n..} then ⊥ else f y"
    have Y: "chain ?Y"
    proof (rule chainI, rule fun_belowI)
      fix n y show "?Y n y ⊑ ?Y (Suc n) y" by (cases "y ∈ g ` {n..}") auto
    qed
    have "f ⊑ (⨆n. ?Y n)"
    proof (rule fun_belowI)
      fix y
      obtain n where n: "y ∉ g ` {n..}"
      proof (cases "y ∈ range g")
        case True
        then obtain k where "y = g k" by blast
        then have "y ∉ g ` {Suc k..}" using g(1) by (auto simp: inj_eq)
        then show ?thesis by (rule that)
      next
        case False
        then show ?thesis by (intro that[of 0]) auto
      qed
      then have "f y = ?Y n y" by simp
      also have "… ⊑ (⨆n. ?Y n y)" by (rule is_ub_thelub[OF ch2ch_fun[OF Y]])
      finally show "f y ⊑ (⨆n. ?Y n) y" by (simp add: lub_fun[OF Y])
    qed
    then obtain n where "f ⊑ ?Y n" using compactD2[OF c Y] by blast
    then have "f (g n) ⊑ ⊥" using fun_belowD[of f "?Y n" "g n"] by simp
    then have "f (g n) = ⊥" by (rule bottomI)
    with g(2) show False by blast
  qed
  ultimately show "finite {x. f x ≠ ⊥} ∧ (∀x. compact (f x))" by blast
next
  assume "finite {x. f x ≠ ⊥} ∧ (∀x. compact (f x))"
  then have fin: "finite {x. f x ≠ ⊥}" and comp: "⋀x. compact (f x)" by blast+
  show "compact f"
  proof (rule compactI2)
    fix Y :: "nat ⇒ 'a ⇒ 'b" assume Y: "chain Y" and le: "f ⊑ (⨆i. Y i)"
    have "∃i. f x ⊑ Y i x" for x
    proof -
      have "f x ⊑ (⨆i. Y i x)" using fun_belowD[OF le, of x] by (simp add: lub_fun[OF Y])
      then show ?thesis by (rule compactD2[OF comp ch2ch_fun[OF Y]])
    qed
    then obtain h where h: "⋀x. f x ⊑ Y (h x) x" by metis
    let ?n = "Max (insert 0 (h ` {x. f x ≠ ⊥}))"
    have "f x ⊑ Y ?n x" for x
    proof (cases "f x = ⊥")
      case True then show ?thesis by simp
    next
      case False
      then have "h x ≤ ?n" by (intro Max_ge) (use fin in auto)
      then show ?thesis by (rule below_trans[OF h fun_belowD[OF chain_mono[OF Y]]])
    qed
    then show "∃i. f ⊑ Y i" by (blast intro: fun_belowI)
  qed
qed
