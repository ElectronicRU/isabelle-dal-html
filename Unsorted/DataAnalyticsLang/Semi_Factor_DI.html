section ‹Semi-factors over dI-domains›

text ‹Over a dI-domain the meet of any non-empty family of semi-factor deflations is again a
  semi-factor deflation.  Its image is the intersection of the images and it is stable
  (@{thm [source] stable_defl_glb}), so only the slide rule needs proof.  At a compact argument
  the family takes only finitely many values, and a finite subfamily, whose meet is a semi-factor
  by @{thm [source] semi_factor_defl_oo}, gives the slide rule there; algebraicity extends it to
  all arguments.›

text ‹Joinability with a fixed element is admissible: a chain bounded with ‹y› stays below the
  chain of joins with ‹y›.›

lemma adm_joinable: "adm (λx. x ⇑ (y :: 'a::bounded_complete))"
proof (rule admI)
  fix Y :: "nat ⇒ 'a" assume Y: "chain Y" and J: "∀i. Y i ⇑ y"
  have ch: "chain (λi. Y i ⊔ y)"
    by (rule chainI, rule join_mono1) (simp_all add: J chainE[OF Y])
  show "(⨆i. Y i) ⇑ y"
  proof (rule joinableI)
    show "(⨆i. Y i) ⊑ (⨆i. Y i ⊔ y)" by (rule lub_mono[OF Y ch]) (simp add: J joinableD(1))
    show "y ⊑ (⨆i. Y i ⊔ y)"
      by (rule below_trans[OF joinableD(2)[OF J[rule_format]] is_ub_thelub[OF ch]])
  qed
qed

lemma semi_factor_defl_finite_meet:
  fixes F :: "('a::{bifinite, distributive_cpo} → 'a) set"
  assumes "finite F" and "⋀d. d ∈ F ⟹ semi_factor_defl d"
  shows "∃e. semi_factor_defl e ∧ (∀z. e⋅z = z ⟷ (∀d∈F. d⋅z = z))"
  using assms
proof (induct F rule: finite_induct)
  case empty
  show ?case by (intro exI[of _ ID] conjI) (simp_all add: semi_factor_defl_ID)
next
  case (insert d F)
  then obtain e where e: "semi_factor_defl e" and efx: "∀z. e⋅z = z ⟷ (∀d∈F. d⋅z = z)"
    by blast
  have d: "semi_factor_defl d" using insert.prems by simp
  note fx = stable_defl_oo_fixed[OF semi_factor_defl.axioms(1)[OF d] semi_factor_defl.axioms(1)[OF e]]
  show ?case
  proof (intro exI[of _ "d oo e"] conjI allI)
    show "semi_factor_defl (d oo e)" by (rule semi_factor_defl_oo[OF d e])
    show "(d oo e)⋅z = z ⟷ (∀d'∈insert d F. d'⋅z = z)" for z using fx efx by simp
  qed
qed

lemma semi_factor_defl_glb_compact:
  fixes 𝒟 :: "('a::dI → 'a) set"
  assumes ne: "𝒟 ≠ {}" and sf: "⋀d. d ∈ 𝒟 ⟹ semi_factor_defl d"
    and k: "compact x" and y: "glb 𝒟⋅y = y" and xy: "glb 𝒟⋅x ⊑ y"
  shows "x ⇑ y"
proof -
  have st: "⋀d. d ∈ 𝒟 ⟹ stable_defl d" by (rule semi_factor_defl.axioms(1)[OF sf])
  note fx = stable_defl_glb_fixes[OF ne st]
  have "(λd. d⋅x) ` 𝒟 ⊆ {z. z ⊑ x}"
    using st by (auto intro: deflation.below stable_defl.deflation)
  then have "finite ((λd. d⋅x) ` 𝒟)" by (rule finite_subset) (rule finite_below_compact[OF k])
  then have "∃F⊆𝒟. finite F ∧ (λd. d⋅x) ` 𝒟 = (λd. d⋅x) ` F"
    by (rule finite_subset_image[OF _ subset_refl])
  then obtain F where F: "F ⊆ 𝒟" "finite F" and FD: "(λd. d⋅x) ` 𝒟 = (λd. d⋅x) ` F"
    by (elim exE conjE) (rule that)
  obtain e where e: "semi_factor_defl e" and efx: "⋀z. e⋅z = z ⟷ (∀d∈F. d⋅z = z)"
    using semi_factor_defl_finite_meet[OF F(2)] sf F(1) by blast
  interpret e: semi_factor_defl e by (rule e)
  have exF: "d⋅(e⋅x) = e⋅x" if "d ∈ F" for d using efx[of "e⋅x"] e.idem that by blast
  have "d⋅(e⋅x) = e⋅x" if d: "d ∈ 𝒟" for d
  proof -
    have "d⋅x ∈ (λd. d⋅x) ` F" using FD d by blast
    then obtain d' where d': "d' ∈ F" "d⋅x = d'⋅x" by blast
    have "e⋅x = d'⋅(e⋅x)" using exF[OF d'(1)] by simp
    also have "… ⊑ d'⋅x" by (rule monofun_cfun_arg[OF e.below])
    finally have "e⋅x ⊑ d⋅x" using d'(2) by simp
    then show ?thesis by (rule stable_defl.lower_fixed[OF st[OF d]])
  qed
  then have "glb 𝒟⋅(e⋅x) = e⋅x" using fx by blast
  then have "e⋅x ⊑ glb 𝒟⋅x" by (metis monofun_cfun_arg e.below)
  then have "e⋅x ⊑ y" using xy by (rule below_trans)
  moreover have "e⋅y = y" using efx y fx F(1) by blast
  ultimately show "x ⇑ y" using e.slide by blast
qed

lemma semi_factor_defl_glb:
  fixes 𝒟 :: "('a::dI → 'a) set"
  assumes ne: "𝒟 ≠ {}" and sf: "⋀d. d ∈ 𝒟 ⟹ semi_factor_defl d"
  shows "semi_factor_defl (glb 𝒟)"
proof (rule semi_factor_deflI)
  have st: "⋀d. d ∈ 𝒟 ⟹ stable_defl d" by (rule semi_factor_defl.axioms(1)[OF sf])
  show "stable_defl (glb 𝒟)" by (rule stable_defl_glb[OF ne st])
  fix x y assume y: "glb 𝒟⋅y = y" and xy: "glb 𝒟⋅x ⊑ y"
  obtain a :: "nat ⇒ 'a → 'a" where a: "approx_chain a" using bifinite by blast
  interpret approx_chain a by (fact a)
  have ch: "chain (λi. a i⋅x)" by (rule ch2ch_Rep_cfunL[OF chain_approx])
  have "a i⋅x ⇑ y" for i
    by (rule semi_factor_defl_glb_compact[OF ne sf compact_approx y])
      (blast, rule below_trans[OF monofun_cfun_arg[OF approx_below] xy])
  then have "(⨆i. a i⋅x) ⇑ y" by (rule admD[OF adm_joinable ch])
  then show "x ⇑ y" by (simp add: contlub_cfun_fun[OF chain_approx, symmetric])
qed


subsection ‹Meets of semi-factors›

text ‹Over a dI key the meet of a non-empty set of semi-factors is computed pointwise: its
  projection is the glb of the projections, so its image is the intersection of the images.›

lemma semi_factor_p_glb:
  fixes 𝒜 :: "'a::dI semi_factor set"
  assumes ne: "𝒜 ≠ {}"
  shows "semi_factor_p (glb 𝒜) = (⨅A∈𝒜. semi_factor_p A)"
proof -
  let ?P = "semi_factor_p ` 𝒜"
  have neP: "?P ≠ {}" using ne by blast
  have sfd: "semi_factor_defl (glb ?P)"
    by (rule semi_factor_defl_glb[OF neP]) (auto intro: semi_factor_defl_semi_factor_p)
  have "glb 𝒜 = semi_factor_of_defl (glb ?P)"
  proof (rule glb_eqI, rule is_glbI)
    show "semi_factor_of_defl (glb ?P) ⊑ A" if "A ∈ 𝒜" for A
      unfolding semi_factor_below_iff_p semi_factor_p_of_defl[OF sfd]
      by (rule glb_below) (use that in blast)
    show "B ⊑ semi_factor_of_defl (glb ?P)" if "∀A∈𝒜. B ⊑ A" for B
      unfolding semi_factor_below_iff_p semi_factor_p_of_defl[OF sfd]
      by (rule glb_greatest[OF neP]) (use that in ‹auto simp: semi_factor_below_iff_p›)
  qed
  then show ?thesis by (simp add: semi_factor_p_of_defl[OF sfd])
qed

subsection ‹Algebraicity›

lemma countable_compacts: "countable {x :: 'a::bifinite. compact x}"
proof -
  obtain f :: "'a compact_basis ⇒ nat" where "inj f" using compact_basis.countable ..
  then have "countable (range (Rep_compact_basis :: 'a compact_basis ⇒ 'a))"
    by (rule countable_image[OF countableI])
  moreover have "range (Rep_compact_basis :: 'a compact_basis ⇒ 'a) = {x. compact x}"
    using type_definition_compact_basis by (rule type_definition.Rep_range)
  ultimately show ?thesis by simp
qed

lemma stable_defl_semi_factor_p: "stable_defl (semi_factor_p F)"
  by (rule semi_factor_defl.axioms(1)[OF semi_factor_defl_semi_factor_p])

lemma deflation_semi_factor_p: "deflation (semi_factor_p F)"
  by (rule stable_defl.deflation[OF stable_defl_semi_factor_p])

lemma semi_factor_p_mono: "F ⊑ G ⟹ semi_factor_p F⋅x ⊑ semi_factor_p G⋅x"
  by (rule monofun_cfun_fun) (simp add: semi_factor_below_iff_p[symmetric])

text ‹Projections onto a lub of semi-factors are pointwise lubs (@{thm [source] proj_Sup}).›

lemma semi_factor_p_lub_is_lub: "(λF. semi_factor_p F⋅x) ` 𝒳 <<| semi_factor_p (lub 𝒳)⋅x"
proof -
  have "lub 𝒳 = Sup 𝒳" by (rule lub_eqI[OF semi_factor_is_lub])
  then have "semi_factor_p (lub 𝒳) = stable_subdom_p (Sup (subdom_of_semi_factor ` 𝒳))"
    by (simp add: semi_factor_p.rep_eq Sup_semi_factor.rep_eq)
  moreover have "(λF. semi_factor_p F⋅x) ` 𝒳 = (λA. stable_subdom_p A⋅x) ` (subdom_of_semi_factor ` 𝒳)"
    by (simp add: image_image semi_factor_p.rep_eq)
  ultimately show ?thesis using proj_Sup[where 𝒜 = "subdom_of_semi_factor ` 𝒳" and x = x] by simp
qed

lemma semi_factor_p_chain_lub: "semi_factor_p (⨆i. Y i)⋅x = (⨆i. semi_factor_p (Y i)⋅x)"
proof -
  have "range (λi. semi_factor_p (Y i)⋅x) <<| semi_factor_p (⨆i. Y i)⋅x"
    using semi_factor_p_lub_is_lub[where 𝒳 = "range Y" and x = x] by (simp add: image_image)
  then show ?thesis by (rule lub_eqI[symmetric])
qed

text ‹The principal semi-factor of ‹x›: the least one containing ‹x›.  Over a dI key its image is
  the intersection of the images containing ‹x› (@{thm [source] semi_factor_p_glb}).›

definition semi_factor_principal :: "'a::dI ⇒ 'a semi_factor" where
  "semi_factor_principal x = glb {F. semi_factor_p F⋅x = x}"

lemma semi_factor_principal_fixes:
  "semi_factor_p (semi_factor_principal x)⋅y = y ⟷
    (∀F. semi_factor_p F⋅x = x ⟶ semi_factor_p F⋅y = y)"
proof -
  let ?A = "{F :: 'a semi_factor. semi_factor_p F⋅x = x}"
  have "semi_factor_of_defl ID ∈ ?A" by (simp add: semi_factor_p_of_defl semi_factor_defl_ID)
  then have ne: "?A ≠ {}" by blast
  have "semi_factor_p (semi_factor_principal x) = glb (semi_factor_p ` ?A)"
    unfolding semi_factor_principal_def by (rule semi_factor_p_glb[OF ne])
  moreover have "glb (semi_factor_p ` ?A)⋅y = y ⟷ (∀d∈semi_factor_p ` ?A. d⋅y = y)"
    by (rule stable_defl_glb_fixes) (use ne in ‹auto intro: stable_defl_semi_factor_p›)
  ultimately show ?thesis by auto
qed

lemma semi_factor_principal_self: "semi_factor_p (semi_factor_principal x)⋅x = x"
  by (simp add: semi_factor_principal_fixes)

lemma semi_factor_principal_below_iff: "semi_factor_principal x ⊑ F ⟷ semi_factor_p F⋅x = x"
proof
  assume "semi_factor_principal x ⊑ F"
  then have "semi_factor_p (semi_factor_principal x)⋅x ⊑ semi_factor_p F⋅x"
    by (rule semi_factor_p_mono)
  then have "x ⊑ semi_factor_p F⋅x" by (simp add: semi_factor_principal_self)
  then show "semi_factor_p F⋅x = x" by (rule below_antisym[OF deflation.below[OF deflation_semi_factor_p]])
next
  assume "semi_factor_p F⋅x = x"
  then show "semi_factor_principal x ⊑ F" unfolding semi_factor_principal_def by (intro glb_below) simp
qed

lemma compact_semi_factor_principal:
  assumes k: "compact k"
  shows "compact (semi_factor_principal k)"
proof (rule compactI2)
  fix Y :: "nat ⇒ 'a semi_factor" assume Y: "chain Y" and le: "semi_factor_principal k ⊑ (⨆i. Y i)"
  have ch: "chain (λi. semi_factor_p (Y i)⋅k)" by (rule chainI, rule semi_factor_p_mono, rule chainE[OF Y])
  have "k = semi_factor_p (⨆i. Y i)⋅k" using le by (simp add: semi_factor_principal_below_iff)
  also have "… = (⨆i. semi_factor_p (Y i)⋅k)" by (rule semi_factor_p_chain_lub)
  finally have "k ⊑ (⨆i. semi_factor_p (Y i)⋅k)" by simp
  then obtain i where "k ⊑ semi_factor_p (Y i)⋅k" using compactD2[OF k ch] by blast
  then have "semi_factor_p (Y i)⋅k = k" by (rule below_antisym[OF deflation.below[OF deflation_semi_factor_p]])
  then have "semi_factor_principal k ⊑ Y i" by (simp add: semi_factor_principal_below_iff)
  then show "∃i. semi_factor_principal k ⊑ Y i" by blast
qed

text ‹Every semi-factor is the lub of the principals of its compact elements: its image is
  downward closed and the projection onto the lub fixes every compact of it, hence every element.›

lemma semi_factor_lub_principals:
  "F = lub {semi_factor_principal k | k. compact k ∧ semi_factor_p F⋅k = k}" (is "F = lub ?P")
proof (rule below_antisym)
  show "lub ?P ⊑ F" by (rule lattice_lub_least) (auto simp: is_ub_def semi_factor_principal_below_iff)
  show "F ⊑ lub ?P"
  proof (simp only: semi_factor_below_iff_p, rule cfun_belowI)
    fix x
    let ?Z = "lub ?P" and ?y = "semi_factor_p F⋅x"
    have fix_Z: "semi_factor_p ?Z⋅k = k" if "compact k" "semi_factor_p F⋅k = k" for k
    proof -
      have "semi_factor_principal k ⊑ ?Z" by (rule lattice_lub_upper) (use that in blast)
      then show ?thesis by (simp add: semi_factor_principal_below_iff)
    qed
    obtain C where C: "chain C" "∀i. compact (C i)" "?y = (⨆i. C i)" by (rule obtain_compact_chain)
    have fixF: "semi_factor_p F⋅(C i) = C i" for i
    proof -
      have "C i ⊑ semi_factor_p F⋅x" using is_ub_thelub[OF C(1)] C(3) by simp
      then show ?thesis by (rule stable_defl.lower_fixed[OF stable_defl_semi_factor_p])
    qed
    have "semi_factor_p ?Z⋅?y = (⨆i. semi_factor_p ?Z⋅(C i))"
      unfolding C(3) by (rule contlub_cfun_arg[OF C(1)])
    also have "… = (⨆i. C i)" using C(2) fixF by (simp add: fix_Z)
    also have "… = ?y" by (simp add: C(3))
    finally have yZ: "semi_factor_p ?Z⋅?y = ?y" .
    have "?y ⊑ x" by (rule deflation.below[OF deflation_semi_factor_p])
    then have "semi_factor_p ?Z⋅?y ⊑ semi_factor_p ?Z⋅x" by (rule monofun_cfun_arg)
    with yZ show "semi_factor_p F⋅x ⊑ semi_factor_p ?Z⋅x" by simp
  qed
qed

lemma semi_factor_algebraic: "F = lub {K. compact K ∧ K ⊑ F}" for F :: "'a::dI semi_factor"
proof (rule below_antisym)
  have "lub {semi_factor_principal k | k. compact k ∧ semi_factor_p F⋅k = k} ⊑ lub {K. compact K ∧ K ⊑ F}"
    by (rule lattice_lub_mono)
      (auto intro: compact_semi_factor_principal simp: semi_factor_principal_below_iff)
  then show "F ⊑ lub {K. compact K ∧ K ⊑ F}"
    by (rule below_trans[OF eq_imp_below[OF semi_factor_lub_principals]])
  show "lub {K. compact K ∧ K ⊑ F} ⊑ F" by (rule lattice_lub_least) (auto simp: is_ub_def)
qed

text ‹A compact semi-factor is the join of finitely many compact principals: enumerate the
  compacts of its image, and the joins of initial segments form a chain with lub the semi-factor.›

lemma compact_semi_factor_finite_join:
  assumes K: "compact K"
  shows "∃A. finite A ∧ A ⊆ {k. compact k} ∧ K = lub (semi_factor_principal ` A)"
proof -
  let ?P = "{k. compact k ∧ semi_factor_p K⋅k = k}"
  have "semi_factor_p K⋅⊥ = ⊥" by (rule bottomI, rule deflation.below[OF deflation_semi_factor_p])
  then have neP: "?P ≠ {}" by (intro exI[of _ ⊥] notI) simp
  have cP: "countable ?P" by (rule countable_subset[OF _ countable_compacts]) auto
  define e where "e = from_nat_into ?P"
  have e: "range e = ?P" unfolding e_def by (rule range_from_nat_into[OF neP cP])
  have eP: "compact (e i) ∧ semi_factor_p K⋅(e i) = e i" for i using rangeI[of e i] e by blast
  define G where "G n = lub (semi_factor_principal ` e ` {..n})" for n
  have chG: "chain G" unfolding G_def by (rule chainI, rule lattice_lub_mono) auto
  have GK: "G n ⊑ K" for n
    unfolding G_def by (rule lattice_lub_least) (auto simp: is_ub_def semi_factor_principal_below_iff eP)
  have "K ⊑ (⨆n. G n)"
  proof -
    have "K = lub {semi_factor_principal k | k. compact k ∧ semi_factor_p K⋅k = k}"
      by (rule semi_factor_lub_principals)
    also have "… ⊑ (⨆n. G n)"
    proof (rule lattice_lub_least, rule is_ubI)
      fix X assume "X ∈ {semi_factor_principal k | k. compact k ∧ semi_factor_p K⋅k = k}"
      then obtain k where X: "X = semi_factor_principal k" and k: "k ∈ ?P" by blast
      then have "k ∈ range e" using e by simp
      then obtain n where n: "k = e n" by blast
      have "X ⊑ G n" unfolding X n G_def by (rule lattice_lub_upper) simp
      also have "G n ⊑ (⨆n. G n)" by (rule is_ub_thelub[OF chG])
      finally show "X ⊑ (⨆n. G n)" .
    qed
    finally show ?thesis .
  qed
  then obtain n where "K ⊑ G n" using compactD2[OF K chG] by blast
  then have "K = G n" using GK[of n] by (rule below_antisym)
  then show ?thesis unfolding G_def by (intro exI[of _ "e ` {..n}"] conjI) (auto simp: eP)
qed

lemma countable_compact_semi_factors: "countable {K :: 'a::dI semi_factor. compact K}"
proof (rule countable_subset)
  show "{K. compact K} ⊆ (λA. lub (semi_factor_principal ` A)) ` {A. finite A ∧ A ⊆ {k :: 'a. compact k}}"
    using compact_semi_factor_finite_join by blast
  show "countable ((λA. lub (semi_factor_principal ` A)) ` {A. finite A ∧ A ⊆ {k :: 'a. compact k}})"
    by (rule countable_image, rule countable_Collect_finite_subset, rule countable_compacts)
qed

lemma semi_factor_compacts_enum: "∃c :: nat ⇒ 'a::dI semi_factor. range c = {K. compact K}"
  using range_from_nat_into[OF _ countable_compact_semi_factors] compact_bottom by blast

instance semi_factor :: (dI) bifinite
  by (rule bifinite_algebraic_lattice[OF semi_factor_compacts_enum semi_factor_algebraic])
