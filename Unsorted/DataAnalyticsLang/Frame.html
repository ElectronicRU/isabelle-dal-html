Theory Frame

theory Frame
  imports Slicer Views
begin

default_sort type

section ‹Frames›

text ‹A frame has a level of detail and two levels of filters.  The ∗‹ambient› filter is an open
  predicate on rows: expressions may add to it (an MDX-style subselect), but never take it apart
  or discard it.  The ∗‹slice› is a syntactic filter on keys that expressions coarsen;
  dropping detail from the frame always coarsens the slice as well.  The frame's LOD partitions
  the selected rows into groups by their projected keys.›

record ('l::type, 'f::type, 'r::pcpo) frame =
  flod :: 'l
  amb :: "'r sel"
  slice :: 'f

text ‹The record package's own ‹truncate› operation would shadow the truncation action.›

hide_const (open) frame.truncate

section ‹Frame transformers›

text ‹The level-of-detail constructs act on the frame an inner expression is evaluated under.
  ‹finclude D› adds detail ‹D› (requested detail: ‹INCLUDE› is not a combinator of its own, it
  compiles into the grouping inside an aggregate); ‹frestrict φ› adds ‹φ› to the ambient filter;
  ‹ffor τ G› needs the slicer: it drops detail by a truncation ‹τ›, from the frame and from the
  slice, and conjoins ‹G›.  Pegging conjoins a key's pin to the slice.›

definition finclude :: "'l::po ⇒ ('l, 'f::type, 'r::pcpo) frame ⇒ ('l, 'f, 'r) frame" where
  "finclude D F = F⦇flod := flod F ⊔ D⦈"

lemma flod_finclude: "flod (finclude D F) = flod F ⊔ D"
  by (simp add: finclude_def)

lemma amb_finclude [simp]: "amb (finclude D F) = amb F"
  by (simp add: finclude_def)

lemma slice_finclude [simp]: "slice (finclude D F) = slice F"
  by (simp add: finclude_def)

definition frestrict :: "'r::pcpo sel ⇒ ('l::type, 'f::type, 'r) frame ⇒ ('l, 'f, 'r) frame" where
  "frestrict φ F = F⦇amb := inf (amb F) φ⦈"

lemma flod_frestrict [simp]: "flod (frestrict φ F) = flod F"
  by (simp add: frestrict_def)

lemma slice_frestrict [simp]: "slice (frestrict φ F) = slice F"
  by (simp add: frestrict_def)

lemma amb_frestrict [simp]: "amb (frestrict φ F) = inf (amb F) φ"
  by (simp add: frestrict_def)

section ‹Frame semantics›

text ‹A dataspace fixes a slicer and a schema with keys; datasets are elements of the lower
  powerdomain of rows, their data points the maximal members.  Rows have keys, via ‹rkey›; a frame
  selects the rows in its ambient filter whose key its slice holds at.›

locale dataspace = slicer slicing + keyed_schema rules keying
  for slicing :: "('f, 'k::{domain, lod_domain}) slicer_ops"
    and rules :: "'r::{domain, lod_domain} schema_rules" and keying :: "('r, 'k) keying"
begin

definition frows :: "('k semi_factor, 'f, 'r) frame ⇒ 'r ⇒ bool" where
  "frows F r ⟷ selects (amb F) r ∧ holds (slice F) (rkey⋅r)"

lemma frows_cong: "amb F = amb F' ⟹ slice F = slice F' ⟹ frows F = frows F'"
  by (simp add: frows_def fun_eq_iff)

lemma frows_finclude [simp]: "frows (finclude D F) = frows F"
  by (simp add: frows_def fun_eq_iff)

lemma frows_frestrict: "frows (frestrict φ F) r ⟷ frows F r ∧ selects φ r"
  by (auto simp: frows_def)

definition ffor :: "'k semi_factor ⇒ 'f ⇒ ('k semi_factor, 'f, 'r) frame ⇒ ('k semi_factor, 'f, 'r) frame" where
  "ffor τ G F = F⦇flod := (τ ⊓ flod F), slice := coarsen τ (slice F) ❙* G⦈"

lemma flod_ffor [simp]: "flod (ffor τ G F) = (τ ⊓ flod F)"
  and amb_ffor [simp]: "amb (ffor τ G F) = amb F"
  and slice_ffor [simp]: "slice (ffor τ G F) = coarsen τ (slice F) ❙* G"
  by (simp_all add: ffor_def)

definition peg :: "'k ⇒ ('k semi_factor, 'f, 'r) frame ⇒ ('k semi_factor, 'f, 'r) frame" where
  "peg K F = F⦇slice := slice F ❙* spin K⦈"

lemma flod_peg [simp]: "flod (peg K F) = flod F"
  by (simp add: peg_def)

lemma amb_peg [simp]: "amb (peg K F) = amb F"
  by (simp add: peg_def)

lemma slice_peg [simp]: "slice (peg K F) = slice F ❙* spin K"
  by (simp add: peg_def)

lemma frows_peg_sub: "frows (peg K F) r ⟹ frows F r"
  by (simp add: frows_def)

lemma frows_peg: "compact K ⟹ frows (peg K F) r ⟷ frows F r ∧ K ⊑ rkey⋅r"
  by (auto simp: frows_def holds_spin)

text ‹Pegging commutes with the transformers that keep the slice.›

lemma peg_finclude: "peg K (finclude D F) = finclude D (peg K F)"
  by (simp add: peg_def finclude_def)

lemma peg_frestrict: "peg K (frestrict φ F) = frestrict φ (peg K F)"
  by (simp add: peg_def frestrict_def)

end

section ‹Key images›

text ‹The keys a frame has at an LOD are the projections of the data points it selects.›

context dataspace
begin

definition keyimg :: "'r lower_pd ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k semi_factor ⇒ 'k ⇒ bool" where
  "keyimg data F c k ⟷ (∃x∈lower_maximal data. frows F x ∧ k = proj c⋅(rkey⋅x))"

lemma keyimg_cong: "frows F = frows F' ⟹ keyimg data F c = keyimg data F' c"
  by (simp add: keyimg_def fun_eq_iff)

lemma keyimg_mono: "(⋀r. frows F r ⟹ frows F' r) ⟹ keyimg data F c k ⟹ keyimg data F' c k"
  unfolding keyimg_def by blast

text ‹Keys conform to the slice.›

lemma keyimg_slicer: "keyimg data F c k ⟹ ∃k'. k ⊑ k' ∧ holds (slice F) k'"
  unfolding keyimg_def frows_def using proj_below by blast

lemma keyimg_peg_iff: "compact K ⟹ keyimg data (peg K F) c K ⟷ keyimg data F c K"
  unfolding keyimg_def by (auto simp: frows_peg proj_below)

end

context dataspace
begin

text ‹An LOD is read on rows through their keys; reading at a join of LODs is reading at both.›

lemma rproj_sup_eq: "rproj (a ⊔ b) y = rproj (a ⊔ b) z ⟷ rproj a y = rproj a z ∧ rproj b y = rproj b z"
proof
  assume e: "rproj (a ⊔ b) y = rproj (a ⊔ b) z"
  show "rproj a y = rproj a z ∧ rproj b y = rproj b z"
    using arg_cong[OF e, of "Rep_cfun (proj a)"] arg_cong[OF e, of "Rep_cfun (proj b)"]
    by (simp add: proj_le lattice_join_ge1 lattice_join_ge2)
qed (simp add: proj_sup)

end

end
