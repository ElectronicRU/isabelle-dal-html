Theory Views

theory Views
  imports Schema
begin

default_sort "{bifinite, distributive_cpo}"

section ‹Views and functional dependencies›

text ‹A ∗‹view› reads a section of the row and a part of its key: a pair of a column-side and a
  key-side semi-factor.  Functional dependencies between views cover all four combinations of
  sides (columns to columns, columns to keys, keys to columns, keys to keys) at once.  The key of a
  row is computed from a declared column support, so columns-to-keys dependencies hold on every
  set of rows; the others are properties of datasets.›

type_synonym ('r, 'k) view = "'r semi_factor × 'k semi_factor"

text ‹Projecting a pair of a row and a key onto a view's parts.›

definition pproj :: "('r, 'k) view ⇒ 'r × 'k → 'r × 'k" where
  "pproj v = (Λ p. (fst p ⇂ fst v, proj (snd v)⋅(snd p)))"

lemma pproj_apply [simp]: "pproj v⋅(r, k) = (r ⇂ fst v, proj (snd v)⋅k)"
  by (simp add: pproj_def)

text ‹A keying computes keys from rows and declares the column support of each key part.›

record ('r, 'k) keying =
  key_fun :: "'r → 'k"
  key_supp :: "'k semi_factor ⇒ 'r semi_factor"

hide_const (open) keying.truncate

locale keyed =
  fixes keying :: "('r::{domain, lod_domain}, 'k::lod_domain) keying"
  assumes csupp_reads: "proj a⋅(key_fun keying⋅(x ⇂ key_supp keying a)) = proj a⋅(key_fun keying⋅x)"
    and csupp_bot: "key_supp keying ⊥ = ⊥"
    and csupp_sup: "key_supp keying (a ⊔ b) = key_supp keying a ⊔ key_supp keying b"
    and csupp_compact: "compact a ⟹ compact (key_supp keying a)"
begin

abbreviation rkey :: "'r → 'k" where "rkey ≡ key_fun keying"
abbreviation csupp :: "'k semi_factor ⇒ 'r semi_factor" where "csupp ≡ key_supp keying"

abbreviation rproj :: "'k semi_factor ⇒ 'r ⇒ 'k" where
  "rproj a x ≡ proj a⋅(rkey⋅x)"

definition vpart :: "('r, 'k) view ⇒ 'r ⇒ 'r × 'k" where
  "vpart v x = (x ⇂ fst v, rproj (snd v) x)"

lemma vpart_pair [simp]: "vpart (c, a) x = (x ⇂ c, rproj a x)"
  by (simp add: vpart_def)

text ‹A view part projects onto the parts of smaller views.›

lemma vpart_pproj: "v ⊑ w ⟹ pproj v⋅(vpart w x) = vpart v x"
  by (simp add: vpart_def below_prod_def proj_le)

text ‹A key part is read off any section covering its support.›

lemma rproj_section: "csupp a ⊑ c ⟹ rproj a (y ⇂ c) = rproj a y"
  by (metis csupp_reads proj_le)

lemma csupp_mono: "a ⊑ b ⟹ csupp a ⊑ csupp b"
  by (metis csupp_sup join_absorb lattice_join_ge1)

subsection ‹Dependencies on a set of rows›

text ‹A view determines a function on a set of rows when the function is monotone in the view
  there: a row reading less on the view gives less.  On all rows this makes the function factor
  continuously through the view; in particular rows agreeing on the view agree on the function.  A
  dependency between views is the case where the function is the other view.›

definition determines :: "'r set ⇒ ('r, 'k) view ⇒ ('r → 'b::cpo) ⇒ bool" where
  "determines P v g ⟷ (∀x∈P. ∀y∈P. vpart v x ⊑ vpart v y ⟶ g⋅x ⊑ g⋅y)"

text ‹The view part as a continuous function, for dependencies between views.›

definition vproj :: "('r, 'k) view ⇒ 'r → 'r × 'k" where
  "vproj v = (Λ x. vpart v x)"

lemma vproj_apply [simp]: "vproj v⋅x = vpart v x"
  by (simp add: vproj_def vpart_def)

abbreviation vfd :: "'r set ⇒ ('r, 'k) view ⇒ ('r, 'k) view ⇒ bool" where
  "vfd P v w ≡ determines P v (vproj w)"

lemma determines_belowD:
  "determines P v g ⟹ x ∈ P ⟹ y ∈ P ⟹ vpart v x ⊑ vpart v y ⟹ g⋅x ⊑ g⋅y"
  unfolding determines_def vproj_apply by blast

lemma determinesD: "determines P v g ⟹ x ∈ P ⟹ y ∈ P ⟹ vpart v x = vpart v y ⟹ g⋅x = g⋅y"
  by (rule below_antisym) (simp_all add: determines_belowD)

lemma vfdD: "vfd P v w ⟹ x ∈ P ⟹ y ∈ P ⟹ vpart v x = vpart v y ⟹ vpart w x = vpart w y"
  using determinesD[of P v "vproj w" x y] by simp

lemma determines_subset: "determines P v g ⟹ Q ⊆ P ⟹ determines Q v g"
  unfolding determines_def vproj_apply by blast

lemma determines_trans: "vfd P u v ⟹ determines P v g ⟹ determines P u g"
  unfolding determines_def vproj_apply by blast

lemma vpart_below_iff: "vpart (c, a) x ⊑ vpart (c, a) y ⟷ x ⇂ c ⊑ y ⇂ c ∧ rproj a x ⊑ rproj a y"
  by (simp add: below_prod_def)

lemma determines_key: "determines P (c, a) (h oo proj a oo rkey)"
  unfolding determines_def vproj_apply by (simp add: vpart_below_iff monofun_cfun_arg)

lemma determines_comp:
  assumes g: "determines P v g"
  shows "determines P v (h oo g)"
  unfolding determines_def vproj_apply
  using determines_belowD[OF g] by (simp add: monofun_cfun_arg)

lemma determines_pair:
  assumes f: "determines P v f" and g: "determines P v g"
  shows "determines P v (Λ x. h⋅(f⋅x)⋅(g⋅x))"
  unfolding determines_def vproj_apply
  using determines_belowD[OF f] determines_belowD[OF g] by (simp add: monofun_cfun)

lemma determines_prod:
  assumes f: "determines P v f" and g: "determines P v g"
  shows "determines P v (Λ x. (f⋅x, g⋅x))"
  unfolding determines_def vproj_apply
  using determines_belowD[OF f] determines_belowD[OF g] by (simp add: below_prod_def)

text ‹Views are ordered structurally, as a product: a larger view reads more.›

lemma vpart_le:
  assumes le: "v ⊑ w" and below: "vpart w x ⊑ vpart w y"
  shows "vpart v x ⊑ vpart v y"
proof -
  have c: "x ⇂ fst w ⊑ y ⇂ fst w" and k: "rproj (snd w) x ⊑ rproj (snd w) y"
    using below by (simp_all add: vpart_def below_prod_def)
  have "x ⇂ fst v ⊑ y ⇂ fst v"
    using monofun_cfun_arg[OF c, of "proj (fst v)"] le by (simp add: below_prod_def proj_le)
  moreover have "rproj (snd v) x ⊑ rproj (snd v) y"
    using monofun_cfun_arg[OF k, of "proj (snd v)"] le by (simp add: below_prod_def proj_le)
  ultimately show ?thesis by (simp add: vpart_def below_prod_def)
qed

lemma vfd_refl: "w ⊑ v ⟹ vfd P v w"
  unfolding determines_def vproj_apply by (blast intro: vpart_le)

lemma determines_mono: "determines P v g ⟹ v ⊑ w ⟹ determines P w g"
  by (rule determines_trans[OF vfd_refl])

text ‹Projections to two LODs below a third join monotonically.›

lemma proj_sup_mono:
  fixes x y :: "'a::{bifinite, distributive_cpo}"
  assumes "proj a⋅x ⊑ proj a⋅y" and "proj b⋅x ⊑ proj b⋅y"
  shows "proj (a ⊔ b)⋅x ⊑ proj (a ⊔ b)⋅y"
proof -
  have ub: "proj a⋅y ⊑ proj a⋅y ⊔ proj b⋅y" "proj b⋅y ⊑ proj a⋅y ⊔ proj b⋅y"
    using join_is_lub[OF proj_below proj_below, of a y b, THEN is_lubD1, THEN is_ubD] by simp_all
  show ?thesis unfolding proj_sup
    by (rule join_belowI) (rule below_trans[OF assms(1) ub(1)], rule below_trans[OF assms(2) ub(2)])
qed

lemma vpart_join:
  assumes v: "vpart v x ⊑ vpart v y" and w: "vpart w x ⊑ vpart w y"
  shows "vpart (v ⊔ w) x ⊑ vpart (v ⊔ w) y"
proof -
  have "x ⇂ fst v ⊑ y ⇂ fst v" "x ⇂ fst w ⊑ y ⇂ fst w"
    "rproj (snd v) x ⊑ rproj (snd v) y" "rproj (snd w) x ⊑ rproj (snd w) y"
    using v w by (simp_all add: vpart_def below_prod_def)
  then show ?thesis unfolding vpart_def lattice_join_prod fst_conv snd_conv below_prod_def
    by (simp add: proj_sup_mono)
qed

lemma vfd_sup: "vfd P u v ⟹ vfd P u w ⟹ vfd P u (v ⊔ w)"
  unfolding determines_def vproj_apply by (blast intro: vpart_join)

text ‹Columns-to-keys dependencies are a matter of computation.›

lemma vfd_support: "csupp a ⊑ c ⟹ vfd P (c, ⊥) (⊥, a)"
  unfolding determines_def vproj_apply
proof (intro ballI impI)
  fix x y assume le: "csupp a ⊑ c" and below: "vpart (c, ⊥) x ⊑ vpart (c, ⊥) y"
  have "x ⇂ c ⊑ y ⇂ c" using below by (simp add: vpart_below_iff)
  then have "rproj a (x ⇂ c) ⊑ rproj a (y ⇂ c)" by (rule monofun_cfun_arg[OF monofun_cfun_arg])
  then have "rproj a x ⊑ rproj a y" by (simp only: rproj_section[OF le])
  then show "vpart (⊥, a) x ⊑ vpart (⊥, a) y" by (simp add: vpart_below_iff proj_bot)
qed

subsection ‹The key part, the section, and the determination order›

text ‹The key part of a view drops its column part: it is the meet with the fixed view ‹(⊥, ⊤)›,
  so it is below the view and preserves joins.  The section of a view is the pure-column view
  reading the columns and the declared support of the key part.  Views are also preordered
  semantically, by ∗‹determination› on all rows: the structural order implies it, the key part
  is below the view and the section above it.  On pure-column views the two orders agree.›

definition vkey :: "('r, 'k) view ⇒ ('r, 'k) view" where
  "vkey v = (⊥, snd v)"

definition vsec :: "('r, 'k) view ⇒ ('r, 'k) view" where
  "vsec v = (fst v ⊔ csupp (snd v), ⊥)"

definition vdet :: "('r, 'k) view ⇒ ('r, 'k) view ⇒ bool" (infix ‹≼› 50) where
  "v ≼ w ⟷ vfd UNIV w v"

lemma vkey_meet: "vkey v = v ⊓ (⊥, lattice_top)"
  by (simp add: vkey_def prod_meet lattice_meet_commute[of _ lattice_top] lattice_meet_top_left)

lemma vkey_below: "vkey v ⊑ v"
  by (simp add: vkey_def below_prod_def)

lemma vkey_idem [simp]: "vkey (vkey v) = vkey v"
  by (simp add: vkey_def)

lemma vkey_join: "vkey (v ⊔ w) = vkey v ⊔ vkey w"
  by (simp add: vkey_def lattice_join_prod)

lemma vsec_idem [simp]: "vsec (vsec v) = vsec v"
  by (simp add: vsec_def csupp_bot)

lemma vdet_refl [simp]: "v ≼ v"
  unfolding vdet_def by (rule vfd_refl) simp

lemma vdet_trans: "u ≼ v ⟹ v ≼ w ⟹ u ≼ w"
  unfolding vdet_def by (rule determines_trans)

lemma below_vdet: "v ⊑ w ⟹ v ≼ w"
  unfolding vdet_def by (rule vfd_refl)

lemma vdet_vfd: "v ≼ w ⟹ vfd P w v"
  unfolding vdet_def by (erule determines_subset) simp

lemma determines_vdet: "determines P v g ⟹ v ≼ w ⟹ determines P w g"
  by (rule determines_trans[OF vdet_vfd])

lemma vkey_vdet: "vkey v ≼ v"
  by (rule below_vdet[OF vkey_below])

lemma vdet_vsec: "v ≼ vsec v"
  unfolding vdet_def determines_def vproj_apply
proof (intro ballI impI)
  fix x y assume below: "vpart (vsec v) x ⊑ vpart (vsec v) y"
  let ?s = "fst v ⊔ csupp (snd v)"
  have s: "x ⇂ ?s ⊑ y ⇂ ?s" using below by (simp add: vsec_def vpart_def below_prod_def)
  have "x ⇂ fst v ⊑ y ⇂ fst v"
    using monofun_cfun_arg[OF s, of "proj (fst v)"] by (simp add: proj_le lattice_join_ge1)
  moreover have "rproj (snd v) x ⊑ rproj (snd v) y"
  proof -
    have le: "csupp (snd v) ⊑ ?s" by (rule lattice_join_ge2)
    have "rproj (snd v) (x ⇂ ?s) ⊑ rproj (snd v) (y ⇂ ?s)" by (rule monofun_cfun_arg[OF monofun_cfun_arg[OF s]])
    then show ?thesis by (simp only: rproj_section[OF le])
  qed
  ultimately show "vpart v x ⊑ vpart v y" by (simp add: vpart_def below_prod_def)
qed

text ‹Factoring: a continuous function determined by a view on all rows factors continuously
  through the view's section, which rows realise (a section of a row is a row).›

theorem determines_factor:
  assumes "determines UNIV v f"
  shows "f = f oo proj (fst v ⊔ csupp (snd v))"
proof (rule cfun_eqI)
  fix x
  let ?s = "fst v ⊔ csupp (snd v)"
  have p: "vsec v = (?s, ⊥)" by (rule vsec_def)
  have s: "determines UNIV (?s, ⊥) f"
    using determines_vdet[OF assms vdet_vsec] unfolding p .
  have "vpart (?s, ⊥) (x ⇂ ?s) = vpart (?s, ⊥) x" by (simp add: proj_bot)
  then have "f⋅(x ⇂ ?s) = f⋅x" by (rule determinesD[OF s UNIV_I UNIV_I])
  then show "f⋅x = (f oo proj ?s)⋅x" by (simp only: cfcomp2)
qed

text ‹In general a function factors continuously through the view's parts, read as a pair, into
  the values with a top adjoined: the least upper bound of what rows reading at most a given pair
  give (a left Kan extension along the view part).  For a compact view it is continuous and reads
  only the view's parts of its argument; on view parts it is the function, if the view determines
  it.›

definition vfactor_set :: "('r, 'k) view ⇒ ('r → 'b::bounded_complete) ⇒ 'r × 'k ⇒ 'b top set" where
  "vfactor_set v f p = {down⋅(f⋅x) |x. vpart v x ⊑ p}"

definition vfactor :: "('r, 'k) view ⇒ ('r → 'b::bounded_complete) ⇒ 'r × 'k → 'b top" where
  "vfactor v f = (Λ p. lub (vfactor_set v f p))"

lemma compact_vpart: "compact v ⟹ compact x ⟹ compact (vpart v x)"
  by (simp add: vpart_def compact_proj_compact proj_compact[OF compact_snd])

lemma lattice_lub_max: "x ∈ S ⟹ (⋀y. y ∈ S ⟹ y ⊑ x) ⟹ lub S = (x :: 'a::lattice_complete)"
  by (rule below_antisym) (blast intro: lattice_lub_least is_ubI, rule lattice_lub_upper)

lemma vfactor_beta:
  fixes f :: "'r → 'b::bounded_complete"
  assumes cv: "compact v"
  shows "vfactor v f⋅p = lub (vfactor_set v f p)"
  unfolding vfactor_def
proof (rule beta_cfun)
  let ?G = "λp. lub (vfactor_set v f p)"
  have mono: "?G p ⊑ ?G q" if "p ⊑ q" for p q
    by (rule lattice_lub_least, rule is_ubI, rule lattice_lub_upper)
      (auto simp: vfactor_set_def intro: below_trans[OF _ that])
  obtain A :: "nat ⇒ 'r → 'r" where A: "approx_chain A" using bifinite by blast
  have app: "x = (⨆j. A j⋅x)" for x
    using approx_chain.lub_approx[OF A] approx_chain.chain_approx[OF A]
    by (metis ID1 contlub_cfun_fun)
  have chain_bound: "?G (⨆i. Y i) ⊑ (⨆i. ?G (Y i))" if Y: "chain Y" for Y
  proof (rule lattice_lub_least, rule is_ubI)
    have GY: "chain (λi. ?G (Y i))" using Y by (auto intro: chainI mono chainE)
    fix u assume "u ∈ vfactor_set v f (⨆i. Y i)"
    then obtain x where u: "u = down⋅(f⋅x)" and x: "vpart v x ⊑ (⨆i. Y i)"
      by (auto simp: vfactor_set_def)
    have chA: "chain (λj. A j⋅x)" using approx_chain.chain_approx[OF A] by (simp add: ch2ch_Rep_cfunL)
    have each: "down⋅(f⋅(A j⋅x)) ⊑ (⨆i. ?G (Y i))" for j
    proof -
      have "vpart v (A j⋅x) ⊑ vpart v x"
        using monofun_cfun_arg[OF approx_chain.approx_below[OF A], of "vproj v"] by simp
      then have "vpart v (A j⋅x) ⊑ (⨆i. Y i)" using x by (rule below_trans)
      then obtain i where i: "vpart v (A j⋅x) ⊑ Y i"
        using compactD2[OF compact_vpart[OF cv approx_chain.compact_approx[OF A]] Y] by blast
      have "down⋅(f⋅(A j⋅x)) ⊑ ?G (Y i)"
        by (rule lattice_lub_upper) (auto simp: vfactor_set_def intro: i)
      also have "… ⊑ (⨆i. ?G (Y i))" by (rule is_ub_thelub[OF GY])
      finally show ?thesis .
    qed
    have "u = (⨆j. down⋅(f⋅(A j⋅x)))"
      unfolding u by (subst app[of x]) (simp add: contlub_cfun_arg chA)
    then show "u ⊑ (⨆i. ?G (Y i))" using each by (simp add: lub_below chA)
  qed
  show "cont ?G" by (rule contI2) (auto intro: monofunI mono chain_bound)
qed

lemma vfactor_reads:
  assumes cv: "compact v"
  shows "vfactor v f⋅(s, κ) = vfactor v f⋅(s ⇂ fst v, proj (snd v)⋅κ)"
proof -
  have "vpart v x ⊑ (s, κ) ⟷ vpart v x ⊑ (s ⇂ fst v, proj (snd v)⋅κ)" for x
  proof
    assume "vpart v x ⊑ (s, κ)"
    then have "x ⇂ fst v ⊑ s" "rproj (snd v) x ⊑ κ" by (simp_all add: vpart_def below_prod_def)
    then show "vpart v x ⊑ (s ⇂ fst v, proj (snd v)⋅κ)"
      using monofun_cfun_arg[of _ _ "proj (fst v)"] monofun_cfun_arg[of _ _ "proj (snd v)"]
      by (fastforce simp: vpart_def below_prod_def)
  next
    assume "vpart v x ⊑ (s ⇂ fst v, proj (snd v)⋅κ)"
    then show "vpart v x ⊑ (s, κ)" by (rule below_trans) (simp add: below_prod_def proj_below)
  qed
  then have "vfactor_set v f (s, κ) = vfactor_set v f (s ⇂ fst v, proj (snd v)⋅κ)"
    by (simp add: vfactor_set_def)
  then show ?thesis by (simp add: vfactor_beta[OF cv])
qed

theorem vfactor_vpart:
  assumes cv: "compact v" and d: "determines UNIV v f"
  shows "vfactor v f⋅(vpart v x) = down⋅(f⋅x)"
  unfolding vfactor_beta[OF cv]
proof (rule lattice_lub_max)
  show "down⋅(f⋅x) ∈ vfactor_set v f (vpart v x)" by (auto simp: vfactor_set_def)
  show "u ⊑ down⋅(f⋅x)" if "u ∈ vfactor_set v f (vpart v x)" for u
    using that determines_belowD[OF d] by (auto simp: vfactor_set_def)
qed

corollary vfactor_vproj:
  "compact v ⟹ determines UNIV v f ⟹ vfactor v f oo vproj v = down oo f"
  by (rule cfun_eqI) (simp add: vfactor_vpart)

corollary determines_factor_ex:
  "determines UNIV v f ⟹ ∃h. f = h oo proj (fst v ⊔ csupp (snd v))"
  by (blast dest: determines_factor)

text ‹Joins: a view determined by two views' join is bounded by both; a join is the least
  upper bound up to determination.›

lemma vdet_join: "u ≼ w ⟹ v ≼ w ⟹ u ⊔ v ≼ w"
  unfolding vdet_def by (rule vfd_sup)

lemma vdet_join_mono: "u ≼ u' ⟹ v ≼ v' ⟹ u ⊔ v ≼ u' ⊔ v'"
  by (rule vdet_join) (blast intro: vdet_trans below_vdet lattice_join_ge1 lattice_join_ge2)+

lemma view_split: "v = (fst v, ⊥) ⊔ (⊥, snd v)" for v :: "('r, 'k) view"
  by (simp add: lattice_join_prod lattice_join_bot_left lattice_join_bot_right)

text ‹A key part is ∗‹exact› when it determines its declared column support on all rows: the key
  is one-to-one with those columns.  For exact key parts the section is equivalent to the view.›

definition exact_key :: "'k semi_factor ⇒ bool" where
  "exact_key a ⟷ (csupp a, ⊥) ≼ (⊥, a)"

theorem vsec_vdet_exact:
  assumes "exact_key (snd v)"
  shows "vsec v ≼ v"
proof -
  have "vsec v = (fst v, ⊥) ⊔ (csupp (snd v), ⊥)"
    by (simp add: vsec_def lattice_join_prod lattice_join_bot_left)
  moreover have "(fst v, ⊥) ≼ v" by (rule below_vdet) (simp add: below_prod_def)
  moreover have "(csupp (snd v), ⊥) ≼ v"
    using assms unfolding exact_key_def
    by (rule vdet_trans) (rule below_vdet, simp add: below_prod_def)
  ultimately show ?thesis by (simp add: vdet_join)
qed

lemma vdet_sections: "(c, ⊥) ≼ (c', ⊥) ⟷ c ⊑ c'"
proof
  assume d: "(c, ⊥) ≼ (c', ⊥)"
  have "proj (c ⊓ c')⋅x = proj c⋅x" for x
  proof -
    have "vpart (c', ⊥) (x ⇂ c') = vpart (c', ⊥) x" by (simp add: proj_bot)
    then have "vpart (c, ⊥) (x ⇂ c') = vpart (c, ⊥) x"
      using d unfolding vdet_def by (blast dest: vfdD)
    then show ?thesis by (simp add: proj_inf proj_bot)
  qed
  then have "c ⊓ c' = c" by (simp add: cfun_eqI semi_factor_p_inject[symmetric])
  then show "c ⊑ c'" by (metis lattice_meet_le2)
qed (simp add: below_vdet below_prod_def)

subsection ‹Weak dependencies›

text ‹A weak dependency only holds between rows where both sides are present, as for the two
  sides of a join.  Weak dependencies do not compose with each other (the middle view may be absent
  in one row), but they compose with dependencies on either side, given that presence passes from
  the outer view to the middle one.›

definition vpresent :: "('r, 'k) view ⇒ 'r ⇒ bool" where
  "vpresent v x ⟷ vpart v x ≠ ⊥"

definition weak_fd :: "'r set ⇒ ('r, 'k) view ⇒ ('r, 'k) view ⇒ bool" where
  "weak_fd P v w ⟷ (∀x∈P. ∀y∈P. vpart v x = vpart v y ⟶ vpresent v x ⟶
    vpresent w x ⟶ vpresent w y ⟶ vpart w x = vpart w y)"

lemma vfd_weak_fd: "vfd P v w ⟹ weak_fd P v w"
  unfolding weak_fd_def by (blast dest: vfdD)

lemma vfd_weak_fd_trans:
  "vfd P u v ⟹ weak_fd P v w ⟹ (∀x∈P. vpresent u x ⟶ vpresent v x) ⟹ weak_fd P u w"
  unfolding weak_fd_def by (blast dest: vfdD)

lemma weak_fd_vfd_trans:
  "weak_fd P u v ⟹ vfd P v w ⟹ (∀x∈P. vpresent w x ⟶ vpresent v x) ⟹ weak_fd P u w"
  unfolding weak_fd_def by (blast dest: vfdD)

text ‹The chase only guarantees the guarded form: equal ∗‹maximal› determinants.  It becomes a
  dependency between views when the determinant is maximal where present and the dependent part
  is absent where the determinant is.›

lemma fd_vfd:
  assumes f: "fd a b P" and m: "∀x∈P. x ⇂ a ≠ ⊥ ⟶ maximal_at a (x ⇂ a)"
    and abs: "∀x∈P. x ⇂ a = ⊥ ⟶ x ⇂ b = ⊥"
  shows "vfd P (a, ⊥) (b, ⊥)"
  unfolding determines_def vproj_apply
proof (intro ballI impI)
  fix x y assume x: "x ∈ P" and y: "y ∈ P" and below: "vpart (a, ⊥) x ⊑ vpart (a, ⊥) y"
  have le: "x ⇂ a ⊑ y ⇂ a" using below by (simp add: vpart_below_iff)
  have "x ⇂ b ⊑ y ⇂ b"
  proof (cases "x ⇂ a = ⊥")
    case True
    then show ?thesis using abs x by simp
  next
    case False
    then have mx: "maximal_at a (x ⇂ a)" using m x by blast
    have eq: "x ⇂ a = y ⇂ a" by (rule maximal_at_eq[OF mx _ le, symmetric]) simp
    show ?thesis using fdD[OF f x y mx eq] by simp
  qed
  then show "vpart (b, ⊥) x ⊑ vpart (b, ⊥) y" by (simp add: vpart_below_iff proj_bot)
qed

end

section ‹Dependencies over all conforming datasets›

locale keyed_schema = schema rules + keyed keying
  for rules :: "'r::{domain, lod_domain} schema_rules" and keying :: "('r, 'k::lod_domain) keying"
begin

definition universally :: "('r set ⇒ bool) ⇒ bool" where
  "universally Q ⟷ (∀X. conforms X ⟶ Q (lower_maximal X))"

lemma universallyD: "universally Q ⟹ conforms X ⟹ Q (lower_maximal X)"
  unfolding universally_def by blast

lemma universally_mono: "universally Q ⟹ (⋀P. Q P ⟹ T P) ⟹ universally T"
  unfolding universally_def by blast

lemma universally_mono2:
  "universally Q ⟹ universally Q' ⟹ (⋀P. Q P ⟹ Q' P ⟹ T P) ⟹ universally T"
  unfolding universally_def by blast

text ‹A functional dependency of the schema holds universally between views, given maximal
  determinants and absent dependents where the determinant is absent.›

lemma universally_fd_rule:
  assumes r: "fd_rule a b ∈ chase_rules"
    and m: "universally (λP. ∀x∈P. x ⇂ a ≠ ⊥ ⟶ maximal_at a (x ⇂ a))"
    and abs: "universally (λP. ∀x∈P. x ⇂ a = ⊥ ⟶ x ⇂ b = ⊥)"
  shows "universally (λP. vfd P (a, ⊥) (b, ⊥))"
  unfolding universally_def
  using fd_vfd conforms_fd[OF _ r] universallyD[OF m] universallyD[OF abs] by blast

subsection ‹Strong keys›

text ‹A key is ∗‹strong› when it is functionally equivalent to its column support in every
  conforming dataset; one direction holds by computation.  Strong keys transfer column dependencies
  to keys: recover the columns from the key, then compute the other key from them.›

definition strong_key :: "'k semi_factor ⇒ bool" where
  "strong_key a ⟷ universally (λP. vfd P (⊥, a) (csupp a, ⊥))"

lemma strong_key_equiv:
  "strong_key a ⟹ conforms X ⟹
    vfd (lower_maximal X) (⊥, a) (csupp a, ⊥) ∧ vfd (lower_maximal X) (csupp a, ⊥) (⊥, a)"
  unfolding strong_key_def by (auto dest: universallyD intro: vfd_support)

theorem strong_key_transfer:
  assumes a: "strong_key a" and c: "universally (λP. vfd P (csupp a, ⊥) (c, ⊥))"
    and b: "csupp b ⊑ c"
  shows "universally (λP. vfd P (⊥, a) (⊥, b))"
  using a c unfolding strong_key_def
  by (rule universally_mono2) (blast intro: determines_trans vfd_support[OF b])

subsection ‹Supports of content functions›

text ‹A content function reads a section of the row together with a part of its key: on all rows,
  these two (its column and key supports) determine it, which is what it depends on across
  datasets.  Within every conforming dataset the key part alone determines it, as when it reads a
  value column whose primary key is in the key.›

definition fun_support :: "('r, 'k) view ⇒ ('r → 'b::cpo) ⇒ bool" where
  "fun_support v f ⟷ determines UNIV v f ∧ universally (λP. determines P (vkey v) f)"

text ‹A function whose view is its own key part is ∗‹key-pure›: determined by its key part on all
  rows.  A function reading only a section is supported by any key part determining that section
  in data.›

lemma fun_support_key: "fun_support (vkey v) f ⟷ determines UNIV (vkey v) f"
  unfolding fun_support_def universally_def vkey_idem by (blast intro: determines_subset)

text ‹A view is ∗‹keyed› when within every conforming dataset its key part determines it; then
  every function of its parts is supported by it.›

definition keyed_view :: "('r, 'k) view ⇒ bool" where
  "keyed_view v ⟷ universally (λP. vfd P (vkey v) v)"

lemma keyed_view_key: "keyed_view (⊥, a)"
  unfolding keyed_view_def universally_def vkey_def by (simp add: vfd_refl)

lemma keyed_view_sup: "keyed_view v ⟹ keyed_view w ⟹ keyed_view (v ⊔ w)"
  unfolding keyed_view_def
proof (erule universally_mono2, assumption)
  fix P assume v: "vfd P (vkey v) v" and w: "vfd P (vkey w) w"
  have "vfd P (vkey (v ⊔ w)) v" "vfd P (vkey (v ⊔ w)) w"
    unfolding vkey_join by (rule determines_mono[OF v lattice_join_ge1], rule determines_mono[OF w lattice_join_ge2])
  then show "vfd P (vkey (v ⊔ w)) (v ⊔ w)" by (rule vfd_sup)
qed

lemma keyed_view_columns:
  assumes fd: "universally (λP. vfd P (⊥, a) (c, ⊥))"
  shows "keyed_view (c, a)"
proof -
  have j: "(c, a) = (c, ⊥) ⊔ (⊥, a)" using view_split[of "(c, a)"] by simp
  have "universally (λP. vfd P (⊥, a) ((c, ⊥) ⊔ (⊥, a)))"
    using fd by (rule universally_mono) (rule vfd_sup[OF _ vfd_refl[OF below_refl]])
  then show ?thesis unfolding keyed_view_def vkey_def by (simp only: j[symmetric] snd_conv)
qed

lemma keyed_view_determines:
  "keyed_view v ⟹ conforms X ⟹ determines (lower_maximal X) (vkey v) (h oo vproj v)"
  unfolding keyed_view_def by (blast dest: universallyD intro: determines_comp)

lemma fun_support_columns:
  assumes reads: "⋀x. f⋅(x ⇂ c) = f⋅x" and ka: "universally (λP. vfd P (⊥, a) (c, ⊥))"
  shows "fun_support (c, a) f"
proof -
  have c: "determines P (c, ⊥) f" for P
    unfolding determines_def vproj_apply vpart_below_iff by (metis reads monofun_cfun_arg)
  have "determines UNIV (c, a) f"
    by (rule determines_mono[OF c]) (simp add: below_prod_def)
  moreover have "universally (λP. determines P (⊥, a) f)"
    using ka by (rule universally_mono) (rule determines_trans[OF _ c])
  ultimately show ?thesis unfolding fun_support_def vkey_def by simp
qed

end
