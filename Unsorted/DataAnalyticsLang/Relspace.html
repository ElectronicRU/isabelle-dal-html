Theory Relspace

theory Relspace
  imports Main
begin

section ‹Relational vocabulary over a set of rows›

text ‹A ∗‹relspace› is just a set of rows ‹D›. Key spaces are given by arbitrary functions
  (projections) on rows; nothing here depends on their nature. All relations are stated through
  ∗‹co-occurrence› of keys, so that ‹D› is mentioned only in its definition.›

locale relspace =
  fixes D :: "'d set"
begin

definition cooc :: "'a ⇒ ('d ⇒ 'a) ⇒ ('d ⇒ 'b) ⇒ 'b ⇒ bool"
    ("_ ⟨_⨝_⟩ _" [51, 0, 0, 51] 50) where
  "k ⟨p⨝q⟩ κ ⟷ (∃d∈D. p d = k ∧ q d = κ)"

lemma cooc_fibres: "k ⟨p⨝q⟩ κ ⟷ D ∩ p -` {k} ∩ q -` {κ} ≠ {}"
  by (auto simp: cooc_def)

subsection ‹Functional dependence›

definition determines :: "('d ⇒ 'a) ⇒ ('d ⇒ 'b) ⇒ bool" (infix "⇛" 50) where
  "p ⇛ q ⟷ (∀k κ κ'. k ⟨p⨝q⟩ κ ⟶ k ⟨p⨝q⟩ κ' ⟶ κ = κ')"

lemma determines_rows: "p ⇛ q ⟷ (∀d∈D. ∀d'∈D. p d = p d' ⟶ q d = q d')"
  unfolding determines_def cooc_def by metis

lemma determines_comp: "p ⇛ f ∘ p"
  by (simp add: determines_rows)

lemma determines_refl: "p ⇛ p"
  by (simp add: determines_rows)

lemma determines_trans: "p ⇛ q ⟹ q ⇛ r ⟹ p ⇛ r"
  unfolding determines_rows by blast

lemma determines_pair: "p ⇛ q ⟹ p ⇛ r ⟹ p ⇛ (λd. (q d, r d))"
  unfolding determines_rows by blast

definition fequiv :: "('d ⇒ 'a) ⇒ ('d ⇒ 'b) ⇒ bool" (infix "⇔" 50) where
  "p ⇔ q ⟷ p ⇛ q ∧ q ⇛ p"

subsection ‹Splitting (unconditional independence)›

definition splits :: "('d ⇒ 'a) ⇒ ('d ⇒ 'b) ⇒ bool" where
  "splits p q ⟷ (∀k κ k' κ'. k ⟨p⨝q⟩ κ ⟶ k' ⟨p⨝q⟩ κ' ⟶ k ⟨p⨝q⟩ κ')"

lemma splits_rows:
  "splits p q ⟷ (∀d⇩1∈D. ∀d⇩2∈D. ∃d∈D. p d = p d⇩1 ∧ q d = q d⇩2)"
  unfolding splits_def cooc_def by blast

lemma splits_sym: "splits p q ⟹ splits q p"
  unfolding splits_rows by metis

lemma splits_antimono: "splits p q ⟹ p ⇛ p' ⟹ q ⇛ q' ⟹ splits p' q'"
  unfolding splits_rows determines_rows by metis

subsection ‹Divisibility at a grain›

text ‹‹t› divides ‹a› at grain ‹n›: up to functional equivalence, ‹a› is ‹t› joined with a
  cofactor that is independent of ‹t› and ‹n› together. Cofactors are labelled by rows.›

definition divides :: "('d ⇒ 'c) ⇒ ('d ⇒ 'b) ⇒ ('d ⇒ 'a) ⇒ bool" where
  "divides n t a ⟷ (∃i :: 'd ⇒ 'd. splits (λd. (t d, n d)) i ∧ a ⇔ (λd. (t d, i d)))"

subsection ‹Equivalence up to a grain›

text ‹‹a› and ‹t› are equivalent at grain ‹n› when co-occurring keys have the same
  co-occurring ‹n›-keys.›

definition equiv_upto :: "('d ⇒ 'c) ⇒ ('d ⇒ 'a) ⇒ ('d ⇒ 'b) ⇒ bool" where
  "equiv_upto n a t ⟷
     (∀k κ. k ⟨a⨝t⟩ κ ⟶ {ν. k ⟨a⨝n⟩ ν} = {ν. κ ⟨t⨝n⟩ ν})"

lemma equiv_upto_refl: "equiv_upto n a a"
  unfolding equiv_upto_def cooc_def by auto

lemma equiv_upto_sym:
  assumes "equiv_upto n a t" shows "equiv_upto n t a"
  unfolding equiv_upto_def
proof (intro allI impI)
  fix k κ assume "k ⟨t⨝a⟩ κ"
  then have "κ ⟨a⨝t⟩ k" by (auto simp: cooc_def)
  with assms show "{ν. k ⟨t⨝n⟩ ν} = {ν. κ ⟨a⨝n⟩ ν}"
    unfolding equiv_upto_def by simp
qed

lemma equiv_upto_trans:
  assumes at: "equiv_upto n a t" and ts: "equiv_upto n t s" shows "equiv_upto n a s"
  unfolding equiv_upto_def
proof (intro allI impI)
  fix k σ assume "k ⟨a⨝s⟩ σ"
  then obtain d where "d ∈ D" "a d = k" "s d = σ" by (auto simp: cooc_def)
  then have "k ⟨a⨝t⟩ t d" "t d ⟨t⨝s⟩ σ" by (auto simp: cooc_def)
  with at ts show "{ν. k ⟨a⨝n⟩ ν} = {ν. σ ⟨s⨝n⟩ ν}"
    unfolding equiv_upto_def by simp
qed

lemma fequiv_equiv_upto:
  assumes "a ⇔ t" shows "equiv_upto n a t"
  unfolding equiv_upto_def
proof (intro allI impI)
  fix k κ assume "k ⟨a⨝t⟩ κ"
  then obtain d⇩0 where d⇩0: "d⇩0 ∈ D" "a d⇩0 = k" "t d⇩0 = κ" by (auto simp: cooc_def)
  from assms have "a ⇛ t" "t ⇛ a" by (auto simp: fequiv_def)
  with d⇩0 show "{ν. k ⟨a⨝n⟩ ν} = {ν. κ ⟨t⨝n⟩ ν}"
    unfolding cooc_def determines_rows by metis
qed

lemma divides_equiv_upto:
  assumes "divides n t a" shows "equiv_upto n a t"
proof -
  from assms obtain i :: "'d ⇒ 'd"
    where sp: "splits (λd. (t d, n d)) i" and eq: "a ⇔ (λd. (t d, i d))"
    by (auto simp: divides_def)
  from eq have a_ti: "a ⇛ (λd. (t d, i d))" and ti_a: "(λd. (t d, i d)) ⇛ a"
    by (auto simp: fequiv_def)
  show ?thesis unfolding equiv_upto_def
  proof (intro allI impI)
    fix k κ assume "k ⟨a⨝t⟩ κ"
    then obtain d⇩0 where d⇩0: "d⇩0 ∈ D" "a d⇩0 = k" "t d⇩0 = κ" by (auto simp: cooc_def)
    have "κ ⟨t⨝n⟩ ν" if "k ⟨a⨝n⟩ ν" for ν
    proof -
      from that obtain d where d: "d ∈ D" "a d = k" "n d = ν" by (auto simp: cooc_def)
      from a_ti d d⇩0 have "(t d, i d) = (t d⇩0, i d⇩0)" unfolding determines_rows by metis
      with d d⇩0 show ?thesis by (auto simp: cooc_def)
    qed
    moreover have "k ⟨a⨝n⟩ ν" if "κ ⟨t⨝n⟩ ν" for ν
    proof -
      from that obtain d where d: "d ∈ D" "t d = κ" "n d = ν" by (auto simp: cooc_def)
      from sp d(1) d⇩0(1) obtain e where e: "e ∈ D" "(t e, n e) = (t d, n d)" "i e = i d⇩0"
        unfolding splits_rows by blast
      from e d d⇩0 have "(t e, i e) = (t d⇩0, i d⇩0)" by simp
      with ti_a e(1) d⇩0(1) have "a e = a d⇩0" unfolding determines_rows by metis
      with e d d⇩0 show ?thesis by (auto simp: cooc_def)
    qed
    ultimately show "{ν. k ⟨a⨝n⟩ ν} = {ν. κ ⟨t⨝n⟩ ν}" by blast
  qed
qed

subsection ‹Equivalence modulo a grain›

text ‹‹a› and ‹t› are equivalent modulo ‹n› when rows with distinct ‹n›-keys agree on ‹a›
  exactly when they agree on ‹t›.  This is a condition on pairs of rows, so it is inherited by
  every subset of rows; it is the largest such relation within @{const equiv_upto}.  At row grain
  it is functional equivalence; it is vacuous when both sides determine ‹n›.›

definition equiv_mod :: "('d ⇒ 'c) ⇒ ('d ⇒ 'a) ⇒ ('d ⇒ 'b) ⇒ bool" where
  "equiv_mod n a t ⟷ (∀d∈D. ∀d'∈D. n d ≠ n d' ⟶ (a d = a d' ⟷ t d = t d'))"

lemma equiv_mod_refl: "equiv_mod n a a"
  by (simp add: equiv_mod_def)

lemma equiv_mod_sym: "equiv_mod n a t ⟹ equiv_mod n t a"
  by (simp add: equiv_mod_def)

lemma equiv_mod_trans: "equiv_mod n a t ⟹ equiv_mod n t s ⟹ equiv_mod n a s"
  by (simp add: equiv_mod_def)

text ‹Adding the same component to both sides, and replacing either side by a functionally
  equivalent one, preserve the equivalence.›

lemma equiv_mod_pair: "equiv_mod n a t ⟹ equiv_mod n (λd. (a d, m d)) (λd. (t d, m d))"
  by (auto simp: equiv_mod_def)

lemma equiv_mod_fequiv: "a ⇔ a' ⟹ t ⇔ t' ⟹ equiv_mod n a t ⟹ equiv_mod n a' t'"
  unfolding equiv_mod_def fequiv_def determines_rows by metis

lemma fequiv_equiv_mod: "a ⇔ t ⟹ equiv_mod n a t"
  unfolding equiv_mod_def fequiv_def determines_rows by metis

lemma equiv_mod_equiv_upto:
  assumes "equiv_mod n a t" shows "equiv_upto n a t"
  unfolding equiv_upto_def
proof (intro allI impI)
  fix k κ assume "k ⟨a⨝t⟩ κ"
  then obtain d⇩0 where d⇩0: "d⇩0 ∈ D" "a d⇩0 = k" "t d⇩0 = κ" by (auto simp: cooc_def)
  have "κ ⟨t⨝n⟩ ν" if "k ⟨a⨝n⟩ ν" for ν
  proof -
    from that obtain d where d: "d ∈ D" "a d = k" "n d = ν" by (auto simp: cooc_def)
    show ?thesis
    proof (cases "n d = n d⇩0")
      case True with d d⇩0 show ?thesis by (auto simp: cooc_def)
    next
      case False
      with assms d d⇩0 have "t d = t d⇩0" unfolding equiv_mod_def by metis
      with d d⇩0 show ?thesis by (auto simp: cooc_def)
    qed
  qed
  moreover have "k ⟨a⨝n⟩ ν" if "κ ⟨t⨝n⟩ ν" for ν
  proof -
    from that obtain d where d: "d ∈ D" "t d = κ" "n d = ν" by (auto simp: cooc_def)
    show ?thesis
    proof (cases "n d = n d⇩0")
      case True with d d⇩0 show ?thesis by (auto simp: cooc_def)
    next
      case False
      with assms d d⇩0 have "a d = a d⇩0" unfolding equiv_mod_def by metis
      with d d⇩0 show ?thesis by (auto simp: cooc_def)
    qed
  qed
  ultimately show "{ν. k ⟨a⨝n⟩ ν} = {ν. κ ⟨t⨝n⟩ ν}" by blast
qed

end

lemma equiv_mod_subset:
  "D' ⊆ D ⟹ relspace.equiv_mod D n a t ⟹ relspace.equiv_mod D' n a t"
  unfolding relspace.equiv_mod_def by blast

end
