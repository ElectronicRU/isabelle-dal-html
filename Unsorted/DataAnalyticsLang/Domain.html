lemma map_restrict_self_dom: "m |` dom m = m" 
  by (simp add: ext domIff restrict_map_def)

lemma map_restrict_dom_inter: "m |` A = m |` (A ∩ dom m)"
  by (simp add: ext domIff restrict_map_def)

lemma map_restrict_dom_subset:
  fixes m:: "'col ⇀ 'val"
  fixes s:: "'col set"
  assumes "dom m ⊆ s"
  shows "m |` s = m"
  using assms map_restrict_self_dom map_restrict_dom_inter[of m s]
  by (metis inf.absorb_iff2)

lemma map_restrict_le: "m |` A ⊆⇩m m" by (metis restrict_in domIff restrict_out map_le_def)

lemma map_concat_restrict_dom:
  fixes m:: "'col ⇀ 'val"
  and n:: "'col ⇀ 'val"
  and h:: "'col set"
  assumes "dom m ∩ h = {}"
  shows "(m ++ n) |` h = n |` h"
  using assms unfolding restrict_map_def map_add_def fun_eq_iff by (auto split: option.split)

lemma map_restrict_union:
  fixes m:: "'col ⇀ 'val"
  and s1:: "'col set"
  and s2:: "'col set"
shows "m |` (s1 ∪ s2) = m |` s1 ++ m |` s2"
proof
  let ?m1 = "m |` s1"
  let ?m2 = "m |` s2"
  fix x
  show "(m |` (s1 ∪ s2)) x = (?m1 ++ ?m2) x"
  proof (cases "x ∈ dom ?m2")
    case True
    then show ?thesis by (simp add: Map.map_add_dom_app_simps)
  next
    case not_in: False
    then have "(?m1 ++ ?m2) x = ?m1 x" by (simp add: Map.map_add_dom_app_simps)
    also have "... = (m |` (s1 ∪ s2)) x"  by (metis not_in restrict_out restrict_in Un_iff domIff)
    finally show ?thesis ..
  qed
qed

abbreviation "nil ≡ Map.empty"

lemma disjoint_family_on_subset: fixes A B S shows
  "disjoint_family_on A S ⟹ ∀ x ∈ S. B x ⊆ A x ⟹ disjoint_family_on B S"
  unfolding disjoint_family_on_def by blast

locale unrestricted_rows =
  fixes
  table_of :: "'col ⇒ 'tab"
  and table_columns :: "'tab ⇒ 'col set"
  and data :: "'tab ⇒ ('col ⇀ 'val) set"
defines
  table_columns_def: "table_columns t ≡ {c. table_of c = t}"
assumes 
  data_dom: "⋀t. ∀m ∈ data t. dom m ⊆ table_columns t"
begin

lemma disjoint_tables: "disjoint_family table_columns"
proof -
  have "table_columns t1 ∩ table_columns t2 = {}" if "t1 ≠ t2" for t1 t2
    unfolding table_columns_def
    using that by auto
  then show ?thesis unfolding disjoint_family_on_def by auto
qed

abbreviation data' where "data' x ≡ insert nil (data x)"
definition choice_space where "choice_space ≡ (Π x ∈ UNIV. data' x)"

theorem choice_space_def': "choice_space = {g. ∀ x. g x = nil ∨ g x ∈ data x}"
  unfolding choice_space_def
  by blast

lemma choice_space_nil: "(λ_.nil) ∈ choice_space"
  unfolding choice_space_def' by simp

abbreviation factorized :: "('tab ⇒ ('col ⇀ 'val)) ⇒ bool"
  where "factorized g ≡ ∀ t. dom (g t) ⊆ table_columns t"

lemma choice_space_factorized[dest]: "g ∈ choice_space ⟹ factorized g"
  unfolding choice_space_def' mem_Collect_eq by (metis data_dom dom_empty empty_subsetI)

definition product :: "('tab ⇒ ('col ⇀ 'val)) ⇒ ('col ⇀ 'val)"
  where "product g c ≡ g (table_of c) c"

definition factor :: "('col ⇀ 'val) ⇒ ('tab ⇒ ('col ⇀ 'val))"
  where "factor m t c ≡ if table_of c = t then m c else None"

lemma prod_inv_factor: "product (factor m) = m" for m
  unfolding product_def factor_def by auto

lemma factor_inv_prod: assumes "factorized g" shows "factor (product g) = g"
proof (rule, rule)
  fix t :: 'tab and c
  have "dom (g t) ⊆ table_columns t" using assms by simp
  also have "... = {c. table_of c = t}" using table_columns_def by simp
  finally have "table_of c ≠ t ⟹ g t c = None" using table_columns_def by blast
  then have "g t c = (if table_of c = t then g t c else None)" by simp
  also have "... = (if table_of c = t then g (table_of c) c else None)" by simp
  also have "... = (if table_of c = t then product g c else None)"
    using product_def by simp
  also have "... = factor (product g) t c" using factor_def by simp
  finally show "factor (product g) t c = g t c" ..
qed

theorem prod_inj: assumes "⋀x. x ∈ S ⟹ factorized x" shows "inj_on product S"
  by (rule inj_on_inverseI[of _ factor]) (simp add: assms factor_inv_prod)

corollary prod_inj_choice_space: "inj_on product choice_space"
  using prod_inj choice_space_factorized by blast

theorem factor_inj: "inj factor"
  by (rule inj_on_inverseI[of _ product])
    (simp add: pointfree_idE prod_inv_factor)

abbreviation lim_table where "lim_table t m ≡ m |` table_columns t"

lemma factor_lim_table: "lim_table t m = factor m t"
  using factor_def table_columns_def by auto

corollary factorized_factor: "factorized (factor m)"
  by (metis factor_lim_table dom_restrict IntE subset_iff)

corollary factorized_factor_set: "g ∈ factor ` S ⟹ factorized g" for g
  using factorized_factor by auto

corollary lim_table_prod: assumes "factorized g" shows "lim_table t (product g) = g t"
  using assms factor_lim_table factor_inv_prod by simp

definition support :: "('a ⇒ ('col ⇀ 'val)) ⇒ 'a set"
  where "support g ≡ {t. g t ≠ nil}"

definition limit :: "'a set ⇒ ('a ⇒ ('col ⇀ 'val)) ⇒ ('a ⇒ ('col ⇀ 'val))"
  where "limit T g t ≡ if t ∈ T then g t else nil"

lemma support_limit: "support (limit T g) = support g ∩ T" for T g
  unfolding support_def limit_def
  by (rule set_eqI) simp

lemma limit_support: "support g ⊆ T ⟹ limit T g = g" for T g
  unfolding support_def limit_def
  by force

lemma limit_choice_space: "g ∈ choice_space ⟹ limit T g ∈ choice_space" for T g
  unfolding choice_space_def' limit_def
  by simp

lemma limit_limit: "limit T ∘ limit S = limit (T ∩ S)" for T S
  unfolding limit_def
  by fastforce

lemma support_factor: "t ∈ support (factor m) ⟷ dom m ∩ table_columns t ≠ {}" for t
proof
  assume "t ∈ support (factor m)"
  then have "factor m t ≠ nil" using support_def by fast
  then have "lim_table t m ≠ nil" using factor_lim_table by simp
  then have "{} ≠ dom (lim_table t m)" by blast
  also have "dom (lim_table t m) = dom m ∩ table_columns t" using Map.dom_restrict by simp
  finally show "dom m ∩ table_columns t ≠ {}" ..
next
  assume "dom m ∩ table_columns t ≠ {}"
  then obtain c where "c ∈ dom m ∩ table_columns t" by blast
  then have "m c ≠ None" and "table_of c = t" using table_columns_def by auto
  then have "factor m t c ≠ None" using factor_def by simp
  then have "factor m t ≠ nil" by auto
  then show "t ∈ support (factor m)" using support_def by fast
qed

definition subspace where "subspace U ≡ {g ∈ choice_space. support g ⊆ U}"

definition restricted_universe where
"restricted_universe U = product ` subspace U"

lemma choice_space_subspace: "choice_space = subspace UNIV"
  unfolding subspace_def by simp

theorem subspace_eq_factor: "subspace U = factor ` restricted_universe U" for U
proof -
  have ss_f: "g ∈ subspace U ⟹ factorized g" for g
    using subspace_def choice_space_factorized by simp
  have ru_f: "g ∈ factor ` restricted_universe U ⟹ factorized g" for g
    using factorized_factor_set by simp
  from ss_f ru_f have inj_on_Un: "inj_on product (subspace U ∪ (factor ` restricted_universe U))"
    using prod_inj by fastforce
  thm inj_on_Un_image_eq_iff

  have "product ` subspace U = restricted_universe U"
    unfolding subspace_def restricted_universe_def by presburger
  also have "... = product ` factor ` restricted_universe U"
    by (subst image_comp) (simp add: prod_inv_factor)
  finally show "subspace U = factor ` restricted_universe U"
    using inj_on_Un_image_eq_iff inj_on_Un by metis
qed

lemma subspace_limit_def: "subspace S = limit S ` choice_space"
proof (rule; rule subsetI)
  fix x assume "x ∈ subspace S"
  then have cs: "x ∈ choice_space" and supp: "support x ⊆ S"
    using subspace_def by simp_all
  from supp have "limit S x = x" by (rule limit_support)
  with cs show "x ∈ limit S ` choice_space" by force
next
  fix x assume "x ∈ limit S ` choice_space"
  then obtain y where y_def:"y ∈ choice_space" "x = limit S y" by blast
  then have 1: "support x ⊆ S" using support_limit by fast
  from y_def have 2: "x ∈ choice_space" using limit_choice_space by simp
  from 1 2 show "x ∈ subspace S" unfolding subspace_def by simp
qed

lemma limit_subspace: "limit S ` subspace T = subspace (S ∩ T)"
  using subspace_limit_def image_comp limit_limit by metis

lemma restricted_universe_factor_support:
  assumes "m ∈ restricted_universe T"
  shows "support (factor m) ⊆ T"
proof -
  from assms obtain g where "m = product g" and "g ∈ choice_space"
    and sub_t: "support g ⊆ T"
    using restricted_universe_def subspace_def by auto
  then have "factor m = g" using factor_inv_prod by auto
  with sub_t show ?thesis by simp
qed

definition "limit_tables T m ≡ m |` (table_of -`  T)"

lemma limit_tables_factor: "limit_tables T = product ∘ (limit T) ∘ factor"
  unfolding limit_tables_def factor_def product_def limit_def
  by (intro ext) simp

lemma limit_tables_restricted:
  "limit_tables S ` restricted_universe T = restricted_universe (S ∩ T)"
proof -
  have "limit_tables S ` (restricted_universe T) =
        (product ∘ (limit S) ∘ factor) ` restricted_universe T"
    using limit_tables_factor by simp
  also have "... = product ` limit S ` factor ` restricted_universe T"
    using image_comp by metis
  also have "factor ` restricted_universe T = subspace T" by (fact subspace_eq_factor[symmetric])
  also have "limit S ` subspace T = subspace (S ∩ T)" by (fact limit_subspace)
  also have "product ` subspace (S ∩ T) = restricted_universe (S ∩ T)"
    using restricted_universe_def by presburger
  finally show ?thesis .
qed

lemma subspace_mono: "S ⊆ T ⟹ subspace S ⊆ subspace T" unfolding subspace_def by auto

lemma restricted_universe_mono: "S ⊆ T ⟹ restricted_universe S ⊆ restricted_universe T"
  unfolding restricted_universe_def using subspace_mono by blast

lemma factor_subspace_mem: "factor m ∈ subspace T ⟹ m ∈ restricted_universe T"
  unfolding subspace_eq_factor
  by (simp add: factor_inj inj_image_mem_iff)

definition universe where "universe ≡ product ` choice_space"

lemma universe_restricted: "universe = restricted_universe UNIV"
  unfolding universe_def restricted_universe_def
  using choice_space_subspace by simp

corollary restricted_universe_limit: "limit_tables S ` universe = restricted_universe S"
  using limit_tables_restricted universe_restricted by simp

corollary factor_universe: "factor ` universe = choice_space"
  using universe_restricted subspace_eq_factor choice_space_subspace by simp

corollary subset_universe: "restricted_universe S ⊆ universe" for S
  unfolding universe_restricted using restricted_universe_mono by simp

lemma universe_nil: "nil ∈ universe" proof -
  have "nil = product (λ_. nil)" using product_def by auto
  also have "... ∈ universe" using choice_space_nil universe_def by auto
  finally show ?thesis.
qed

definition tables_of :: "('col ⇀ 'val) ⇒ 'tab set"
  where "tables_of m = table_of ` (dom m)"

lemma tables_of_support: "tables_of m = support (factor m)"
proof -
  have "tables_of m = table_of ` (dom m)" using tables_of_def by simp
  also have "... = {table_of c | c. m c ≠ None}" by blast
  also have "... = {table_of c | c. factor m (table_of c) c ≠ None}" using factor_def by simp
  also have "... = {t. ∃c. factor m t c ≠ None}" using factor_def by auto
  also have "... = {t. factor m t ≠ nil}" by auto
  also have "... = support (factor m)" using support_def by metis
  finally show ?thesis .
qed

lemma limit_tables_subset: assumes "tables_of m ⊆ S" shows "limit_tables S m = m" proof -
  have "limit_tables S m = product (limit S (factor m))"
    using limit_tables_factor by auto
  also have "limit S (factor m) = factor m" using limit_support tables_of_support assms by auto
  also have "product ... = m" by (rule prod_inv_factor)
  finally show ?thesis .
qed

lemma support_restricted_universe_mem: "g ∈ universe ⟹ g ∈ restricted_universe (tables_of g)"
proof -
  define T where "T = tables_of g"
  then have T_eq: "T = support (factor g)" using tables_of_support by simp

  assume "g ∈ universe"
  then have "factor g ∈ choice_space" using factor_universe by auto
  with T_eq have "factor g ∈ subspace T" using subspace_def by simp
  then show "g ∈ restricted_universe T" by (rule factor_subspace_mem)
qed

lemma restricted_universe_tables_of: "m ∈ restricted_universe S ⟹ tables_of m ⊆ S" for m S
  using restricted_universe_factor_support tables_of_support by simp

lemma restricted_universe_nil: "nil ∈ restricted_universe T" for T
proof
  have "tables_of nil = {}" using tables_of_def by fast
  then show "nil ∈ restricted_universe {}"
    using support_restricted_universe_mem universe_nil by metis
  show "restricted_universe {} ⊆ restricted_universe T"
    using restricted_universe_mono by simp
qed

abbreviation "agrees A x y ≡ x |` A = y |` A"

lemma agrees_iff: "agrees A x y ⟷ (∀a ∈ A. x a = y a)"
proof -
  have "agrees A x y ⟷ (∀a. (x |` A) a = (y |` A) a)" by auto
  also have "... = (∀a ∈ A. (x |` A) a = (y |` A) a) ∧ (∀a. a ∉ A ⟶ (x |` A) a = (y |` A) a)"
    by fastforce
  also have "(∀a. a ∉ A ⟶ (x |` A) a = (y |` A) a) ⟷ True" by simp
  also have "(∀a ∈ A. (x |` A) a = (y |` A) a) ⟷ (∀a ∈ A. x a = y a)" by simp
  finally show ?thesis by simp
qed

lemma agrees_union_iff[iff]: "agrees A x y ∧ agrees B x y ⟷ agrees (A ∪ B) x y"
  unfolding agrees_iff by blast

lemma agrees_mono: "A ⊆ B ⟹ agrees B x y ⟹ agrees A x y"
  unfolding agrees_iff by blast

abbreviation is_key_universal :: "'col set ⇒ prop"
  where "is_key_universal cols ≡ (⋀x y. x ∈ universe ⟹ y ∈ universe ⟹
    agrees cols x y ⟹
      agrees (table_cols_of cols) x y)"

lemma key_limit_project_project: "project cols (key_limit cols m) = project cols m"
  using project_project table_cols_of_superset by (simp add: inf.absorb_iff2)


theorem is_key_universalD[dest?]: "is_key cols ⟹ PROP is_key_universal cols" proof -
  assume inj_1: "is_key cols"
  fix x y
  assume x: "x ∈ universe" and y: "y ∈ universe"
  assume proj_eq: "project cols x = project cols y"
  let ?x1 = "key_limit cols x" and ?y1 = "key_limit cols y"
  from x have x1: "?x1 ∈ universe_of cols" by (rule key_limit_universe_of)
  from y have y1: "?y1 ∈ universe_of cols" by (rule key_limit_universe_of)
  have "project cols ?x1 = project cols x" by (rule key_limit_project_project)
  also have "... = project cols y" by (fact proj_eq)
  also have "... = project cols ?y1" by (rule key_limit_project_project[symmetric])
  finally show "?x1 = ?y1" using x1 y1 inj_1 inj_onD by metis
qed

lemma is_key_universalI[intro?]: "PROP is_key_universal cols ⟹ is_key cols" proof (rule inj_onI)
  assume inj_2: "x ∈ universe ⟹ y ∈ universe ⟹ project cols x = project cols y ⟹
    key_limit cols x = key_limit cols y" for x y
  fix x y
  assume x: "x ∈ universe_of cols" and y: "y ∈ universe_of cols"
  have conv: "z ∈ universe ∧ key_limit cols z = z" (is "?prop z") if "z ∈ universe_of cols" for z
    using subset_universe limit_tables_subset restricted_universe_tables_of that
    key_limit_tables subsetD by metis

  from x have x1: "?prop x" by (rule conv)
  from y have y1: "?prop y" by (rule conv)

  assume "project cols x = project cols y"
  with inj_2 x1 y1 show "x = y" by metis
qed

thm agrees_union_iff

theorem is_key_union: "is_key A ⟹ is_key B ⟹ is_key (A ∪ B)"
proof (rule is_key_universalI)
  assume "is_key A" then have aprop: "PROP is_key_universal A" by (rule is_key_universalD)
  assume "is_key B" then have bprop: "PROP is_key_universal B" by (rule is_key_universalD)
  fix x y assume x: "x ∈ universe" and y: "y ∈ universe"
  assume congrU: "agrees (A ∪ B) x y"
  then have "agrees A x y ∧ agrees B x y" by simp
  then have congrA: "agrees A x y" and congrB: "agrees B x y"
    by (satx+)
  from x y congrA have klA: "agrees (table_cols_of A) x y" by (rule aprop)
  from x y congrB have klB: "agrees (table_cols_of B) x y" by (rule bprop)
  from klA klB have "agrees (table_cols_of A ∪ table_cols_of B) x y" using agrees_union_iff by fast
  then show "agrees (table_cols_of (A ∪ B)) x y" using table_cols_of_union by simp
qed
