Theory Content_Law

theory Content_Law
  imports Frame "HOLCF-Library.List_Cpo"
begin

default_sort type

section ‹The content law›

text ‹An expression reads its data through a finite list of ∗‹generalized tables›.  An entry's
  ∗‹selector› ‹(a, ρ)› picks the fiber: the rows of the frame whose key agrees with the key at hand
  at ‹a ⊔ ρ ⊓ L›, for the frame LOD ‹L›, among the rows passing the ambient filter and the slice
  coarsened to ‹ρ›.  Its guard cuts the fiber down, and its ∗‹view› ‹(c, κ)› reads the rest: the
  content maps each row's key at ‹κ› to its section at ‹c›, which the key determines.  The
  ∗‹content law› says that the semantics at a key is a function of these contents.›

subsection ‹Observations›

text ‹Observations are continuous maps from rows into observed values.  A selection reads as a
  continuous truth value; selections pull back along continuous maps.›

definition sel_chi :: "'r::pcpo sel ⇒ 'r → tr" where
  "sel_chi P = (Λ x. if selects P x then TT else ⊥)"

lemma cont_sel_chi: "cont (λx. if selects P x then TT else (⊥::tr))"
proof (rule contI2)
  show m: "monofun (λx. if selects P x then TT else (⊥::tr))"
    by (rule monofunI) (auto dest: selects_up)
  fix Y :: "nat ⇒ 'a" assume Y: "chain Y"
  have ch: "chain (λi. if selects P (Y i) then TT else (⊥::tr))"
    by (rule ch2ch_monofun[OF m Y])
  show "(if selects P (⨆i. Y i) then TT else ⊥) ⊑ (⨆i. if selects P (Y i) then TT else (⊥::tr))"
  proof (cases "∃i. selects P (Y i)")
    case True
    then obtain i where i: "selects P (Y i)" by blast
    have "(if selects P (Y i) then TT else (⊥::tr)) ⊑ (⨆i. if selects P (Y i) then TT else (⊥::tr))"
      by (rule is_ub_thelub[OF ch])
    then show ?thesis using i by simp
  next
    case False
    then have "¬ selects P (⨆i. Y i)" by (intro admD[OF selects_open Y]) blast
    then show ?thesis by simp
  qed
qed

lemma sel_chi_apply: "sel_chi P⋅x = (if selects P x then TT else ⊥)"
  unfolding sel_chi_def by (rule beta_cfun[OF cont_sel_chi])

lift_definition sel_vimage :: "('a::pcpo → 'b::pcpo) ⇒ 'b sel ⇒ 'a sel" is "λg P x. P (g⋅x)"
proof (intro conjI allI impI)
  fix g :: "'a → 'b" and P :: "'b ⇒ bool" and x y :: 'a
  assume P: "(∀x y. P x ⟶ x ⊑ y ⟶ P y) ∧ adm (λx. ¬ P x)"
  show "P (g⋅y)" if "P (g⋅x)" "x ⊑ y" using P that monofun_cfun_arg by blast
next
  fix g :: "'a → 'b" and P :: "'b ⇒ bool"
  assume P: "(∀x y. P x ⟶ x ⊑ y ⟶ P y) ∧ adm (λx. ¬ P x)"
  show "adm (λx. ¬ P (g⋅x))" by (rule adm_subst[OF cont_Rep_cfun2]) (use P in blast)
qed

lemma selects_sel_vimage [simp]: "selects (sel_vimage g P) x ⟷ selects P (g⋅x)"
  by (simp add: sel_vimage.rep_eq)

text ‹The truth value ‹TT› is maximal and compact, so being ‹TT› is a selection.›

lift_definition sel_TT :: "tr sel" is "λt. t = TT"
proof (intro conjI allI impI)
  show "y = TT" if "x = TT" "x ⊑ y" for x y :: tr using that by simp
  show "adm (λt. t ≠ TT)" by (simp add: compact_TT eq_commute[of _ TT])
qed

lemma selects_sel_TT [simp]: "selects sel_TT t ⟷ t = TT"
  by (simp add: sel_TT.rep_eq)

text ‹Guards are continuous truth-valued functions of the parts of a declared view: the section
  of the row and the part of its key that they read.  On rows they are read through their view, so
  they read nothing else.›

record ('r::"{domain, distributive_cpo}", 'k::"{domain, distributive_cpo}", 'b::cpo) vfn =
  vfun :: "'r × 'k → 'b"
  oview :: "('r, 'k) view"

hide_const (open) vfn.truncate

type_synonym ('r, 'k) guard = "('r, 'k, tr) vfn"

abbreviation ocols :: "('r::{domain, distributive_cpo}, 'k::{domain, distributive_cpo}, 'b::cpo) vfn ⇒ 'r semi_factor" where
  "ocols f ≡ fst (oview f)"

abbreviation okeys :: "('r::{domain, distributive_cpo}, 'k::{domain, distributive_cpo}, 'b::cpo) vfn ⇒ 'k semi_factor" where
  "okeys f ≡ snd (oview f)"

text ‹Reading the section at a view.›

definition vsection :: "('r::{domain, distributive_cpo}, 'k::{domain, distributive_cpo}) view ⇒ ('r, 'k, 'r) vfn" where
  "vsection v = ⦇vfun = cfst, oview = v⦈"

lemma oview_vsection [simp]: "oview (vsection v) = v"
  by (simp add: vsection_def)

text ‹The guard every row passes, and the conjunction of guards, on the joined view.›

definition gtrue :: "('r::{domain, distributive_cpo}, 'k::{domain, distributive_cpo}) guard" where
  "gtrue = ⦇vfun = Λ _. TT, oview = ⊥⦈"

lemma oview_gtrue [simp]: "oview gtrue = ⊥"
  by (simp add: gtrue_def)

definition gand :: "('r::{domain, distributive_cpo}, 'k::{domain, distributive_cpo}) guard ⇒ ('r, 'k) guard ⇒ ('r, 'k) guard" where
  "gand γ δ = ⦇vfun = Λ p. (vfun γ⋅(pproj (oview γ)⋅p)) andalso (vfun δ⋅(pproj (oview δ)⋅p)),
     oview = oview γ ⊔ oview δ⦈"

lemma gand_view [simp]: "oview (gand γ δ) = oview γ ⊔ oview δ"
  by (simp add: gand_def)

text ‹An entry: a selector, picking the fiber; a guard, cutting it down; and a view, whose key
  part keys the content and whose section is recorded at each key.›

record 'k selector =
  lower_part :: "'k semi_factor"
  upper_part :: "'k semi_factor"

hide_const (open) selector.truncate

abbreviation sel :: "'k semi_factor ⇒ 'k semi_factor ⇒ 'k selector" where
  "sel a ρ ≡ ⦇lower_part = a, upper_part = ρ⦈"

record ('k::"{domain, distributive_cpo}", 'r::"{domain, distributive_cpo}") entry =
  esel :: "'k selector"
  eguard :: "('r, 'k) guard"
  eview :: "('r, 'k) view"

hide_const (open) entry.truncate

abbreviation elower :: "('k::{domain, distributive_cpo}, 'r::{domain, distributive_cpo}) entry ⇒ 'k semi_factor" where
  "elower e ≡ lower_part (esel e)"

abbreviation eupper :: "('k::{domain, distributive_cpo}, 'r::{domain, distributive_cpo}) entry ⇒ 'k semi_factor" where
  "eupper e ≡ upper_part (esel e)"

abbreviation ekeys :: "('k::{domain, distributive_cpo}, 'r::{domain, distributive_cpo}) entry ⇒ 'k semi_factor" where
  "ekeys e ≡ snd (eview e)"

abbreviation ecols :: "('k::{domain, distributive_cpo}, 'r::{domain, distributive_cpo}) entry ⇒ 'r semi_factor" where
  "ecols e ≡ fst (eview e)"

abbreviation mk_entry ::
  "'k selector ⇒ ('r, 'k) guard ⇒ ('r, 'k) view ⇒ ('k::{domain, distributive_cpo}, 'r::{domain, distributive_cpo}) entry" where
  "mk_entry S γ v ≡ ⦇esel = S, eguard = γ, eview = v⦈"

record ('k::"domain", 'f::type, 'r::"{bifinite, distributive_cpo}", 'v::type) expr =
  emind :: "'k semi_factor"
  edims :: "('k, 'r) entry list"
  esem :: "'r lower_pd ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k ⇀ 'v"
  csem :: "('k ⇒ 'r option) list ⇒ 'v option"

hide_const (open) expr.truncate

text ‹Contents are maps from keys to sections.  A graph, a single-valued set of pairs, gives one;
  the value it records at a key is ‹cval›.›

definition cval :: "('k × 'b) set ⇒ 'k ⇒ 'b" where
  "cval X q = (THE v. (q, v) ∈ X)"

lemma cval_singleton [simp]: "cval {(q, v)} q = v"
  by (simp add: cval_def)

lemma cval_eq: "single_valued X ⟹ (q, v) ∈ X ⟹ cval X q = v"
  unfolding cval_def by (rule the_equality) (auto dest: single_valuedD)

definition gmap :: "('k × 'b) set ⇒ 'k ⇒ 'b option" where
  "gmap X q = (if q ∈ fst ` X then Some (cval X q) else None)"

definition graph :: "('k ⇒ 'b option) ⇒ ('k × 'b) set" where
  "graph m = {(q, s). m q = Some s}"

lemma dom_gmap [simp]: "dom (gmap X) = fst ` X"
  by (auto simp: gmap_def dom_def)

lemma gmap_empty_iff [simp]: "gmap X = Map.empty ⟷ X = {}"
  unfolding dom_eq_empty_conv[symmetric] dom_gmap by simp

lemma gmap_apply: "single_valued X ⟹ (q, s) ∈ X ⟹ gmap X q = Some s"
  unfolding gmap_def by (simp add: cval_eq rev_image_eqI)

lemma gmap_singleton [simp]: "gmap {(q, s)} = [q ↦ s]"
  by (auto simp: gmap_def fun_eq_iff)

lemma single_valued_graph: "single_valued (graph m)"
  by (auto simp: single_valued_def graph_def)

lemma graph_gmap:
  assumes X: "single_valued X"
  shows "graph (gmap X) = X"
proof (rule set_eqI, rule iffI)
  fix x assume "x ∈ graph (gmap X)"
  then obtain q s where x: "x = (q, s)" and q: "gmap X q = Some s" by (auto simp: graph_def)
  then obtain s' where s': "(q, s') ∈ X" by (force simp: gmap_def split: if_splits)
  then show "x ∈ X" using q gmap_apply[OF X s'] by (simp add: x)
next
  fix x assume "x ∈ X"
  then show "x ∈ graph (gmap X)" using gmap_apply[OF X] by (cases x) (simp add: graph_def)
qed

lemma gmap_graph [simp]: "gmap (graph m) = m"
proof
  fix q show "gmap (graph m) q = m q"
  proof (cases "m q")
    case None
    then have "q ∉ fst ` graph m" by (force simp: graph_def)
    then show ?thesis using None by (simp add: gmap_def)
  next
    case (Some s)
    then have "(q, s) ∈ graph m" by (simp add: graph_def)
    from gmap_apply[OF single_valued_graph this] Some show ?thesis by simp
  qed
qed

context dataspace

text ‹Reading a guard or a section on rows.›

definition ofun :: "('r, 'k, 'b::cpo) vfn ⇒ 'r → 'b" where
  "ofun f = vfun f oo vproj (oview f)"

lemma ofun_apply: "ofun f⋅y = vfun f⋅(vpart (oview f) y)"
  by (simp add: ofun_def)

lemma ofun_vsection [simp]: "ofun (vsection v)⋅y = y ⇂ fst v"
  by (simp add: ofun_apply vsection_def vpart_def)

lemma ofun_gtrue [simp]: "ofun gtrue⋅y = TT"
  by (simp add: ofun_apply gtrue_def)

lemma gand_apply: "ofun (gand γ δ)⋅y = (ofun γ⋅y andalso ofun δ⋅y)"
  by (simp add: ofun_apply gand_def vpart_pproj lattice_join_ge1 lattice_join_ge2)

lemma gand_TT [simp]: "ofun (gand γ δ)⋅y = TT ⟷ ofun γ⋅y = TT ∧ ofun δ⋅y = TT"
  unfolding gand_apply by (cases "ofun γ⋅y" rule: trE) simp_all

text ‹The rows a test passes are a selection.›

definition gsel :: "('r, 'k) guard ⇒ 'r sel" where
  "gsel γ = sel_vimage (ofun γ) sel_TT"

lemma selects_gsel [simp]: "selects (gsel γ) y ⟷ ofun γ⋅y = TT"
  by (simp add: gsel_def)

text ‹Whether a row passes a slice, as a guard.›

definition guard_holds :: "'f ⇒ ('r, 'k) guard" where
  "guard_holds X = ⦇vfun = sel_chi (sopen X) oo csnd, oview = (⊥, slod X)⦈"

lemma guard_holds_apply: "ofun (guard_holds X)⋅y = (if holds X (rkey⋅y) then TT else ⊥)"
  by (simp add: ofun_apply guard_holds_def sel_chi_apply holds_def sopen_lod)

lemma guard_holds_view [simp]: "oview (guard_holds X) = (⊥, slod X)"
  by (simp add: guard_holds_def)

lemma guard_holds_TT [simp]: "ofun (guard_holds X)⋅y = TT ⟷ holds X (rkey⋅y)"
  by (simp add: guard_holds_apply)

subsection ‹Supports of guards›

text ‹A guard or section is ∗‹supported› when its view is keyed: within every
  conforming dataset its key part determines it on rows.›

definition supported :: "('r, 'k, 'b::cpo) vfn ⇒ bool" where
  "supported f ⟷ keyed_view (oview f)"

lemma supported_fun_support: "supported f ⟹ fun_support (oview f) (ofun f)"
proof -
  assume f: "supported f"
  have "determines UNIV (oview f) (vfun f oo vproj (oview f))"
    by (rule determines_comp) (rule vfd_refl[OF below_refl])
  moreover have "universally (λP. determines P (vkey (oview f)) (vfun f oo vproj (oview f)))"
    unfolding universally_def using keyed_view_determines f[unfolded supported_def] by blast
  ultimately show ?thesis unfolding fun_support_def ofun_def by blast
qed

lemma supported_determines:
  "supported f ⟹ conforms X ⟹ determines (lower_maximal X) (vkey (oview f)) (ofun f)"
  unfolding supported_def ofun_def by (rule keyed_view_determines)

text ‹Rows agreeing on a key part covering a supported function's key part agree on it.›

lemma supported_agree:
  assumes h: "supported h" and d: "conforms data" and x: "x ∈ lower_maximal data" and y: "y ∈ lower_maximal data"
    and le: "okeys h ⊑ G" and e: "rproj G y = rproj G x"
  shows "ofun h⋅y = ofun h⋅x"
proof (rule determinesD[OF supported_determines[OF h d] y x])
  have "rproj (okeys h) y = rproj (okeys h) x"
    using arg_cong[OF e, of "Rep_cfun (proj (okeys h))"] by (simp add: proj_le[OF le])
  then show "vpart (vkey (oview h)) y = vpart (vkey (oview h)) x" by (simp add: vkey_def proj_bot)
qed

text ‹Rows agreeing on a key part covering a keyed view's key part agree on its section.›

lemma view_agree:
  assumes v: "keyed_view v" and d: "conforms data" and x: "x ∈ lower_maximal data" and y: "y ∈ lower_maximal data"
    and le: "snd v ⊑ G" and e: "rproj G y = rproj G x"
  shows "y ⇂ fst v = x ⇂ fst v"
proof -
  have "supported (vsection v)" using v by (simp add: supported_def)
  from supported_agree[OF this d x y _ e] le show ?thesis by simp
qed

lemma keyed_view_bot_view: "keyed_view (⊥ :: ('r, 'k) view)"
  unfolding inst_prod_pcpo by (rule keyed_view_key)

text ‹A guard is ∗‹key-pure› when its view is its key part: then it is determined by its key
  part on all rows, independently of any dataset.  Slice tests are key-pure.›

definition key_pure :: "('r, 'k, 'b::cpo) vfn ⇒ bool" where
  "key_pure f ⟷ vkey (oview f) = oview f"

lemma key_pure_determines: "key_pure f ⟹ determines UNIV (oview f) (ofun f)"
  unfolding ofun_def by (rule determines_comp) (rule vfd_refl[OF below_refl])

lemma key_pure_guard_holds: "key_pure (guard_holds X)"
  by (simp add: key_pure_def vkey_def)

lemma supported_guard_holds: "supported (guard_holds X)"
  by (simp add: supported_def keyed_view_key)

lemma supported_gtrue: "supported gtrue"
  unfolding supported_def oview_gtrue inst_prod_pcpo by (rule keyed_view_key)

lemma supported_gand: "supported γ ⟹ supported δ ⟹ supported (gand γ δ)"
  by (simp add: supported_def keyed_view_sup)

text ‹An entry is ∗‹well-formed› when its view is keyed and its guard supported.  Then the
  view's key part determines its section on the rows of a conforming dataset: contents are graphs.
  This is a property of the expression, independent of the frame.›

definition entry_wf :: "('k, 'r) entry ⇒ bool" where
  "entry_wf e ⟷ keyed_view (eview e) ∧ supported (eguard e)"

lemma okeys_gtrue [simp]: "okeys gtrue = ⊥"
  unfolding oview_gtrue inst_prod_pcpo by (rule snd_conv)

lemma entry_wf_view: "keyed_view v ⟹ supported γ ⟹ entry_wf (mk_entry S γ v)"
  by (simp add: entry_wf_def)

lemma entry_wf_guard: "entry_wf e ⟹ supported γ ⟹ entry_wf (e⦇eguard := gand γ (eguard e)⦈)"
  by (simp add: entry_wf_def supported_gand)

definition group :: "'r lower_pd ⇒ 'k semi_factor ⇒ 'k semi_factor ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k ⇒ 'r set" where
  "group data a ρ F k = {y ∈ lower_maximal data. selects (amb F) y ∧ holds (coarsen ρ (slice F)) (rkey⋅y) ∧
     rproj (a ⊔ (ρ ⊓ flod F)) y = proj (a ⊔ (ρ ⊓ flod F))⋅k}"

text ‹The fiber of an entry: the rows of its group passing its guard.  Its content graph pairs
  each row's key at the view's key part with its section at the view's columns; the content is
  that graph as a map.›

definition efiber :: "'r lower_pd ⇒ ('k, 'r) entry ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k ⇒ 'r set" where
  "efiber data e F k = {y ∈ group data (elower e) (eupper e) F k. ofun (eguard e)⋅y = TT}"

definition cpair :: "('k, 'r) entry ⇒ 'r ⇒ 'k × 'r" where
  "cpair e y = (rproj (ekeys e) y, y ⇂ ecols e)"

lemma cpair_vpart: "cpair e y = (snd (vpart (eview e) y), fst (vpart (eview e) y))"
  by (simp add: cpair_def vpart_def)

definition cgraph :: "'r lower_pd ⇒ ('k, 'r) entry ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k ⇒ ('k × 'r) set" where
  "cgraph data e F k = cpair e ` efiber data e F k"

definition content :: "'r lower_pd ⇒ ('k, 'r) entry ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k ⇒ 'k ⇒ 'r option" where
  "content data e F k = gmap (cgraph data e F k)"

abbreviation contents :: "'r lower_pd ⇒ ('k, 'r) entry list ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k ⇒ ('k ⇒ 'r option) list" where
  "contents data es F k ≡ map (λe. content data e F k) es"

text ‹A ∗‹row entry› keys its view and guard within its lower part: rows of a fiber, which agree
  there, agree on the whole content pair, so its content has at most one key.›

definition row_entry :: "('k, 'r) entry ⇒ bool" where
  "row_entry e ⟷ ekeys e ⊑ elower e ∧ okeys (eguard e) ⊑ elower e"

lemma rproj_agree: "s ⊑ G ⟹ rproj G y = rproj G x ⟹ rproj s y = rproj s x"
  by (metis proj_le)

lemma row_entry_image:
  assumes e: "entry_wf e" and r: "row_entry e" and d: "conforms data" and Y: "Y ⊆ lower_maximal data"
    and x: "x ∈ Y" and aG: "elower e ⊑ G" and agree: "⋀y. y ∈ Y ⟹ rproj G y = rproj G x"
  shows "cpair e ` {y ∈ Y. ofun (eguard e)⋅y = TT} = (if ofun (eguard e)⋅x = TT then {cpair e x} else {})"
proof -
  have sG: "ekeys e ⊑ G" and gG: "okeys (eguard e) ⊑ G"
    using r aG unfolding row_entry_def by (blast intro: below_trans)+
  have xm: "x ∈ lower_maximal data" using x Y by blast
  have kv: "keyed_view (eview e)" and sup: "supported (eguard e)" using e by (simp_all add: entry_wf_def)
  let ?g = "λy. ofun (eguard e)⋅y"
  have g: "?g y = ?g x" if y: "y ∈ Y" for y
    using supported_agree[OF sup d xm _ gG agree[OF y]] y Y by blast
  have h: "cpair e y = cpair e x" if y: "y ∈ Y" for y
    using rproj_agree[OF sG agree[OF y]] view_agree[OF kv d xm _ sG agree[OF y]] y Y
    unfolding cpair_def by blast
  show ?thesis
  proof (cases "?g x = TT")
    case True
    have "{y ∈ Y. ?g y = TT} = Y"
    proof (rule set_eqI, rule iffI)
      fix y assume "y ∈ Y"
      then show "y ∈ {y ∈ Y. ?g y = TT}" using True by (simp add: g)
    qed simp
    moreover have "cpair e ` Y = (λ_. cpair e x) ` Y" by (rule image_cong[OF refl h])
    moreover have "Y ≠ {}" using x by blast
    ultimately show ?thesis using True by (simp only: image_constant_conv if_False) simp
  next
    case False
    have "{y ∈ Y. ?g y = TT} = {}"
    proof (rule equals0I)
      fix y assume "y ∈ {y ∈ Y. ?g y = TT}"
      then have y: "y ∈ Y" "?g y = TT" by simp_all
      then have "?g x = TT" by (simp only: g[OF y(1)])
      then show False using False by simp
    qed
    then show ?thesis using False by simp
  qed
qed

text ‹Content graphs of well-formed entries are single-valued.›

lemma cgraph_single_valued:
  assumes e: "entry_wf e" and d: "conforms data"
  shows "single_valued (cgraph data e F k)"
proof (rule single_valuedI)
  fix q u u' assume u: "(q, u) ∈ cgraph data e F k" and u': "(q, u') ∈ cgraph data e F k"
  obtain y where y: "y ∈ efiber data e F k" "q = rproj (ekeys e) y" "u = y ⇂ ecols e"
    using u unfolding cgraph_def cpair_def by blast
  obtain y' where y': "y' ∈ efiber data e F k" "q = rproj (ekeys e) y'" "u' = y' ⇂ ecols e"
    using u' unfolding cgraph_def cpair_def by blast
  have m: "y ∈ lower_maximal data" "y' ∈ lower_maximal data"
    using y(1) y'(1) by (auto simp: efiber_def group_def)
  have kv: "keyed_view (eview e)" using e by (simp add: entry_wf_def)
  have "y' ⇂ ecols e = y ⇂ ecols e"
    by (rule view_agree[OF kv d m(1) m(2) below_refl]) (use y(2) y'(2) in simp)
  then show "u = u'" using y(3) y'(3) by simp
qed

lemma graph_content: "entry_wf e ⟹ conforms data ⟹ graph (content data e F k) = cgraph data e F k"
  unfolding content_def by (rule graph_gmap[OF cgraph_single_valued])

subsection ‹Column supports of frames and sections of datasets›

text ‹A selection ∗‹reads› a section of the row when it is decided there.  A frame reads a
  section covering its ambient filter and the column support of its slice's LOD.  The
  ∗‹point section› of a dataset at a section is the set of its data points cut down to it; it is
  a plain set, not down-closed: contents are sets of values over data points.›

definition sel_reads :: "'r semi_factor ⇒ 'r sel ⇒ bool" where
  "sel_reads c P ⟷ (∀y. selects P (y ⇂ c) ⟷ selects P y)"

definition frame_reads :: "'r semi_factor ⇒ ('k semi_factor, 'f, 'r) frame ⇒ bool" where
  "frame_reads c F ⟷ sel_reads c (amb F) ∧ csupp (slod (slice F)) ⊑ c"

definition pts_section :: "'r semi_factor ⇒ 'r lower_pd ⇒ 'r set" where
  "pts_section c data = (λy. y ⇂ c) ` lower_maximal data"

lemma image_filter_factor:
  assumes "⋀y. Q (g y) = Q y" and "⋀y. h (g y) = h y"
  shows "h ` {y ∈ M. Q y} = h ` {z ∈ g ` M. Q z}"
proof -
  have "{z ∈ g ` M. Q z} = g ` {y ∈ M. Q y}" using assms(1) by auto
  then show ?thesis by (simp add: image_image assms(2))
qed

lemma holds_section: "csupp (slod X) ⊑ c ⟹ holds X (rkey⋅(y ⇂ c)) ⟷ holds X (rkey⋅y)"
  by (metis holds_lod rproj_section)

text ‹A view's parts are read off any section covering its columns and its key part's support.›

lemma vpart_section: "fst (vsec v) ⊑ c ⟹ vpart v (y ⇂ c) = vpart v y"
  unfolding vsec_def fst_conv
  by (simp add: vpart_def proj_le[OF below_trans[OF lattice_join_ge1]]
      rproj_section[OF below_trans[OF lattice_join_ge2]])

lemma ofun_section: "fst (vsec (oview f)) ⊑ c ⟹ ofun f⋅(y ⇂ c) = ofun f⋅y"
  by (simp add: ofun_apply vpart_section)

text ‹The content of an entry is computed from the point section at any section covering the
  group LOD's support, the frame, and the views of the guard and the entry.›

lemma content_section:
  assumes F: "frame_reads c F" and G: "csupp (elower e ⊔ (eupper e ⊓ flod F)) ⊑ c"
    and fc: "fst (vsec (eview e)) ⊑ c" and gc: "fst (vsec (oview (eguard e))) ⊑ c"
  shows "content data e F k = gmap (cpair e ` {z ∈ pts_section c data.
    selects (amb F) z ∧ holds (coarsen (eupper e) (slice F)) (rkey⋅z) ∧
    rproj (elower e ⊔ (eupper e ⊓ flod F)) z = proj (elower e ⊔ (eupper e ⊓ flod F))⋅k ∧
    ofun (eguard e)⋅z = TT})"
proof -
  let ?G = "elower e ⊔ (eupper e ⊓ flod F)"
  let ?P = "λz. selects (amb F) z ∧ holds (coarsen (eupper e) (slice F)) (rkey⋅z) ∧ rproj ?G z = proj ?G⋅k ∧
    ofun (eguard e)⋅z = TT"
  let ?h = "cpair e"
  have sl: "csupp (slod (coarsen (eupper e) (slice F))) ⊑ c"
    using F csupp_mono[OF below_trans[OF slod_coarsen lattice_meet_le1]] unfolding frame_reads_def
    by (blast intro: below_trans)
  have P: "?P (y ⇂ c) ⟷ ?P y" for y
    using F unfolding frame_reads_def sel_reads_def
    by (simp add: holds_section[OF sl] rproj_section[OF G] ofun_section[OF gc])
  have h: "?h (y ⇂ c) = ?h y" for y by (simp only: cpair_vpart vpart_section[OF fc])
  have "efiber data e F k = {y ∈ lower_maximal data. ?P y}" by (auto simp: efiber_def group_def)
  moreover have "?h ` {y ∈ lower_maximal data. ?P y} = ?h ` {z ∈ pts_section c data. ?P z}"
    unfolding pts_section_def by (rule image_filter_factor) (fact P h)+
  ultimately show ?thesis unfolding content_def cgraph_def by simp
qed

text ‹Groups with the same parameters and rows coincide.›

lemma group_cong:
  assumes "amb F = amb F'" "slice F = slice F'"
    and "a ⊔ (ρ ⊓ flod F) = a' ⊔ (ρ ⊓ flod F')"
    and "proj (a ⊔ (ρ ⊓ flod F))⋅k = proj (a' ⊔ (ρ ⊓ flod F'))⋅k'"
  shows "group data a ρ F k = group data a' ρ F' k'"
  using assms unfolding group_def by simp

text ‹A lower part covering the masked part of the minimal LOD absorbs it.›

lemma sat_absorb:
  fixes a ρ m L :: "'k semi_factor"
  assumes "(ρ ⊓ m) ⊑ a"
  shows "a ⊔ (ρ ⊓ (m ⊔ L)) = a ⊔ (ρ ⊓ L)"
proof -
  have "a ⊔ (ρ ⊓ (m ⊔ L)) = (a ⊔ (ρ ⊓ m)) ⊔ (ρ ⊓ L)"
    by (simp only: truncate_sup lattice_join_assoc)
  also have "a ⊔ (ρ ⊓ m) = a" by (rule lattice_join_absorb1[OF assms])
  finally show ?thesis .
qed

end

text ‹Projecting to a finer LOD after a coarser one changes nothing.›

lemma proj_absorb: "a ⊑ b ⟹ proj b⋅(proj a⋅k) = proj a⋅k"
  by (metis proj_inf lattice_meet_absorb2)

text ‹Joins of lists in complete lattices.›

lemma fold_join_absorb: "(⋀x. x ∈ set xs ⟹ x ⊑ b) ⟹ fold (⊔) xs b = (b::'a::lattice_complete)"
  by (induct xs) (simp_all add: lattice_join_absorb2)

lemma fold_join_bots: "fold (⊔) (map (λ_. ⊥) xs) (⊥::'a::lattice_complete) = ⊥"
  by (induct xs) (simp_all add: lattice_join_bot_left)

lemma fold_join_least:
  "(⋀x. x ∈ set xs ⟹ x ⊑ y) ⟹ b ⊑ y ⟹ fold (⊔) xs b ⊑ (y::'a::lattice_complete)"
  by (induct xs arbitrary: b) (simp_all add: lattice_join_least)

lemma fold_join_base: "b ⊑ fold (⊔) xs (b::'a::lattice_complete)"
proof (induct xs arbitrary: b)
  case (Cons x xs)
  have "b ⊑ x ⊔ b" by (rule lattice_join_ge2)
  also have "… ⊑ fold (⊔) xs (x ⊔ b)" by (rule Cons)
  finally show ?case by simp
qed simp

subsection ‹The laws›

text ‹A ∗‹semilawful› expression has supported entries, its minimal LOD is the join of their lower
  parts, and its presence and values are its content semantics of its contents, at every key of
  its key LOD and every conforming dataset (the content law).  A ∗‹lawful› one is moreover
  ∗‹saturated› (each lower part covers the part of the minimal LOD its upper part keeps, so groups
  depend on the frame LOD only through the key LOD) and has entries only at the frame's data keys
  (the key law).›

context dataspace
begin

abbreviation kd :: "('k, 'f, 'r, 'v) expr ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k semi_factor" where
  "kd E F ≡ emind E ⊔ flod F"

definition semilawful :: "('k, 'f, 'r, 'v) expr ⇒ bool" where
  "semilawful E ⟷ emind E = fold (⊔) (map elower (edims E)) ⊥ ∧ (∀e∈set (edims E). entry_wf e) ∧
     (∀data F k. conforms data ⟶ proj (kd E F)⋅k = k ⟶ esem E data F k = csem E (contents data (edims E) F k))"

definition lawful :: "('k, 'f, 'r, 'v) expr ⇒ bool" where
  "lawful E ⟷ semilawful E ∧ (∀e∈set (edims E). (eupper e ⊓ emind E) ⊑ elower e) ∧
     (∀data F k. conforms data ⟶ esem E data F k ≠ None ⟶ keyimg data F (kd E F) k)"

lemma semilawfulI:
  assumes "emind E = fold (⊔) (map elower (edims E)) ⊥"
    and "⋀e. e ∈ set (edims E) ⟹ entry_wf e"
    and "⋀data F k. ⟦conforms data; proj (emind E ⊔ flod F)⋅k = k⟧ ⟹
           esem E data F k = csem E (contents data (edims E) F k)"
  shows "semilawful E"
  unfolding semilawful_def using assms by blast

lemma semilawful_mind_lub: "semilawful E ⟹ emind E = fold (⊔) (map elower (edims E)) ⊥"
  unfolding semilawful_def by blast

lemma semilawful_wf: "semilawful E ⟹ e ∈ set (edims E) ⟹ entry_wf e"
  unfolding semilawful_def by blast

lemma semilawful_keyed: "semilawful E ⟹ e ∈ set (edims E) ⟹ keyed_view (eview e)"
  and semilawful_guard_supported: "semilawful E ⟹ e ∈ set (edims E) ⟹ supported (eguard e)"
  using semilawful_wf unfolding entry_wf_def by blast+

lemma semilawful_csem:
  "⟦semilawful E; conforms data; proj (kd E F)⋅k = k⟧ ⟹ esem E data F k = csem E (contents data (edims E) F k)"
  unfolding semilawful_def by blast

text ‹Equal contents give equal values, across datasets and frames.›

lemma semilawful_content_law:
  "⟦semilawful E; conforms data; conforms data'; proj (kd E F)⋅k = k; proj (kd E F')⋅k' = k';
    contents data (edims E) F k = contents data' (edims E) F' k'⟧ ⟹ esem E data F k = esem E data' F' k'"
  by (simp only: semilawful_csem)

lemma lawful_semiI:
  assumes "semilawful E"
    and "⋀e. e ∈ set (edims E) ⟹ (eupper e ⊓ emind E) ⊑ elower e"
    and "⋀data F k. conforms data ⟹ esem E data F k ≠ None ⟹ keyimg data F (emind E ⊔ flod F) k"
  shows "lawful E"
  unfolding lawful_def using assms by blast

lemma lawfulI:
  assumes "⋀e. e ∈ set (edims E) ⟹ (eupper e ⊓ emind E) ⊑ elower e"
    and "emind E = fold (⊔) (map elower (edims E)) ⊥"
    and "⋀e. e ∈ set (edims E) ⟹ entry_wf e"
    and "⋀data F k. ⟦conforms data; proj (emind E ⊔ flod F)⋅k = k⟧ ⟹
           esem E data F k = csem E (contents data (edims E) F k)"
    and "⋀data F k. conforms data ⟹ esem E data F k ≠ None ⟹ keyimg data F (emind E ⊔ flod F) k"
  shows "lawful E"
  by (rule lawful_semiI[OF semilawfulI[OF assms(2-4)] assms(1,5)])

lemma lawful_semilawful: "lawful E ⟹ semilawful E"
  unfolding lawful_def by blast

lemma lawful_saturated: "lawful E ⟹ e ∈ set (edims E) ⟹ (eupper e ⊓ emind E) ⊑ elower e"
  unfolding lawful_def by blast

lemma lawful_key_law: "lawful E ⟹ conforms data ⟹ esem E data F k ≠ None ⟹ keyimg data F (kd E F) k"
  unfolding lawful_def by blast

lemma lawful_keyed: "lawful E ⟹ e ∈ set (edims E) ⟹ keyed_view (eview e)"
  by (rule semilawful_keyed[OF lawful_semilawful])

lemma lower_below_mind:
  assumes E: "semilawful E" and e: "e ∈ set (edims E)"
  shows "elower e ⊑ emind E"
proof -
  have "elower e ∈ set (map elower (edims E))" using e by force
  then show ?thesis unfolding semilawful_mind_lub[OF E] by (rule fold_join_upper)
qed

lemma group_below_kd:
  "semilawful E ⟹ e ∈ set (edims E) ⟹ elower e ⊔ (eupper e ⊓ flod F) ⊑ kd E F"
  by (rule join_belowI[OF below_trans[OF lower_below_mind lattice_join_ge1]
        below_trans[OF truncate_below lattice_join_ge2]])

text ‹Contents agree when the groups do.›

lemma content_group_cong:
  "group data (elower e) (eupper e) F k = group data' (elower e) (eupper e) F' k' ⟹
    content data e F k = content data' e F' k'"
  by (simp add: content_def cgraph_def efiber_def)

text ‹Contents only depend on the point section at a section covering the frame, the key LOD and
  the sections of the entries' views.›

definition entry_reads :: "'r semi_factor ⇒ ('k, 'r) entry ⇒ bool" where
  "entry_reads c e ⟷ fst (vsec (eview e)) ⊑ c ∧ fst (vsec (oview (eguard e))) ⊑ c"

lemma contents_section:
  assumes E: "semilawful E" and F: "frame_reads c F" and K: "csupp (kd E F) ⊑ c"
    and cols: "⋀e. e ∈ set (edims E) ⟹ entry_reads c e"
    and sec: "pts_section c data = pts_section c data'"
  shows "contents data (edims E) F k = contents data' (edims E) F k"
proof (rule map_cong[OF refl])
  fix e assume e: "e ∈ set (edims E)"
  have G: "csupp (elower e ⊔ (eupper e ⊓ flod F)) ⊑ c"
    by (rule below_trans[OF csupp_mono[OF group_below_kd[OF E e]] K])
  have r: "fst (vsec (eview e)) ⊑ c" "fst (vsec (oview (eguard e))) ⊑ c"
    using cols[OF e] by (simp_all add: entry_reads_def)
  show "content data e F k = content data' e F k"
    unfolding content_section[OF F G r] sec ..
qed

text ‹The ∗‹grain› of an expression at a frame joins the group LODs of its entries.  Contents, and
  so values, depend on the key only through its part at the grain: an expression emits at its
  grain, and its values at finer keys are broadcast.›

definition grain :: "('k, 'f, 'r, 'v) expr ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k semi_factor" where
  "grain E F = fold (⊔) (map (λe. elower e ⊔ (eupper e ⊓ flod F)) (edims E)) ⊥"

lemma group_below_grain: "e ∈ set (edims E) ⟹ elower e ⊔ (eupper e ⊓ flod F) ⊑ grain E F"
  unfolding grain_def by (rule fold_join_upper) force

lemma grain_below_kd: "semilawful E ⟹ grain E F ⊑ kd E F"
  unfolding grain_def by (rule fold_join_least) (auto intro: group_below_kd)

theorem sem_grain:
  assumes E: "semilawful E" and d: "conforms data" and k: "proj (kd E F)⋅k = k"
  shows "esem E data F k = esem E data F (proj (grain E F)⋅k)"
proof (rule semilawful_content_law[OF E d d k])
  show "proj (kd E F)⋅(proj (grain E F)⋅k) = proj (grain E F)⋅k"
    by (rule proj_absorb[OF grain_below_kd[OF E]])
  show "contents data (edims E) F k = contents data (edims E) F (proj (grain E F)⋅k)"
  proof (rule map_cong[OF refl])
    fix e assume e: "e ∈ set (edims E)"
    have "proj (elower e ⊔ (eupper e ⊓ flod F))⋅(proj (grain E F)⋅k) = proj (elower e ⊔ (eupper e ⊓ flod F))⋅k"
      by (rule proj_le[OF group_below_grain[OF e]])
    then show "content data e F k = content data e F (proj (grain E F)⋅k)"
      by (intro content_group_cong) (simp add: group_def)
  qed
qed

text ‹For lawful expressions the frame LOD inside the minimal LOD does not matter: groups only see
  the key LOD.›

lemma group_kd:
  "lawful E ⟹ e ∈ set (edims E) ⟹ elower e ⊔ (eupper e ⊓ flod F) = elower e ⊔ (eupper e ⊓ kd E F)"
  by (rule sat_absorb[symmetric, OF lawful_saturated])

lemma contents_kd:
  assumes E: "lawful E" and F: "amb F = amb F'" "slice F = slice F'" "kd E F = kd E F'"
  shows "contents data (edims E) F k = contents data (edims E) F' k"
proof (rule map_cong[OF refl])
  fix e assume e: "e ∈ set (edims E)"
  have "group data (elower e) (eupper e) F k = group data (elower e) (eupper e) F' k"
    by (rule group_cong[OF F(1,2)]) (simp_all only: group_kd[OF E e] F(3))
  then show "content data e F k = content data e F' k" by (rule content_group_cong)
qed

text ‹Pegging the frame to a key does not change that key's contents: the pin's demand within
  a mask is part of the group already, by saturation.›

lemma peg_group:
  assumes E: "lawful E" and e: "e ∈ set (edims E)" and cK: "compact K" and K: "proj (kd E F)⋅K = K"
  shows "group data (elower e) (eupper e) (peg K F) K = group data (elower e) (eupper e) F K"
proof -
  let ?G = "elower e ⊔ (eupper e ⊓ flod F)" and ?T = "(eupper e ⊓ kd E F)"
  have le: "?T ⊑ ?G" unfolding group_kd[OF E e] by (rule lattice_join_ge2)
  have pin: "holds (coarsen (eupper e) (spin K)) (rkey⋅y)" if y: "rproj ?G y = proj ?G⋅K" for y
  proof -
    have "proj (eupper e)⋅K = proj ?T⋅K" by (simp only: proj_inf K)
    also have "… = proj ?T⋅(proj ?G⋅K)" by (rule proj_le[OF le, symmetric])
    also have "… = proj ?T⋅(rproj ?G y)" by (simp only: y)
    also have "… = rproj ?T y" by (rule proj_le[OF le])
    finally have "proj (eupper e)⋅K ⊑ rkey⋅y" using proj_below[of ?T "rkey⋅y"] by simp
    then show ?thesis by (rule holds_coarsen_spin[OF cK])
  qed
  show ?thesis unfolding group_def using pin by (auto simp: coarsen_conj)
qed

lemma peg_contents:
  assumes E: "lawful E" and cK: "compact K" and K: "proj (kd E F)⋅K = K"
  shows "contents data (edims E) (peg K F) K = contents data (edims E) F K"
proof (rule map_cong[OF refl])
  fix e assume e: "e ∈ set (edims E)"
  show "content data e (peg K F) K = content data e F K"
    by (rule content_group_cong[OF peg_group[OF E e cK K]])
qed

lemma key_lives: "lawful E ⟹ conforms data ⟹ esem E data F k ≠ None ⟹ proj (kd E F)⋅k = k"
  by (auto dest!: lawful_key_law simp: keyimg_def)

text ‹The effective-LOD law: the frame LOD matters only through the key LOD.›

theorem eff_law:
  assumes E: "lawful E" and d: "conforms data" and F: "amb F = amb F'" "slice F = slice F'" "kd E F = kd E F'"
  shows "esem E data F = esem E data F'"
proof
  fix k
  show "esem E data F k = esem E data F' k"
  proof (cases "proj (kd E F)⋅k = k")
    case True
    then have "proj (kd E F')⋅k = k" by (simp add: F(3))
    then show ?thesis
      by (rule semilawful_content_law[OF lawful_semilawful[OF E] d d True _ contents_kd[OF E F]])
  next
    case False
    then have "proj (kd E F')⋅k ≠ k" by (simp add: F(3))
    then have "esem E data F' k = None" using key_lives[OF E d, of F' k] by blast
    moreover have "esem E data F k = None" using False key_lives[OF E d, of F k] by blast
    ultimately show ?thesis by simp
  qed
qed

text ‹The peg law: pegging the frame to a key does not change the value there.›

theorem peg_law:
  assumes E: "lawful E" and d: "conforms data" and cK: "compact K"
  shows "esem E data (peg K F) K = esem E data F K"
proof (cases "proj (kd E F)⋅K = K")
  case True
  have "proj (kd E (peg K F))⋅K = K" using True by simp
  then show ?thesis
    by (rule semilawful_content_law[OF lawful_semilawful[OF E] d d _ True peg_contents[OF E cK True]])
next
  case False
  then have "proj (kd E (peg K F))⋅K ≠ K" by simp
  then have "esem E data (peg K F) K = None" using key_lives[OF E d, of "peg K F" K] by blast
  moreover have "esem E data F K = None" using False key_lives[OF E d, of F K] by blast
  ultimately show ?thesis by simp
qed

text ‹Locality: the semantics at a frame depends on the dataset only through its point section at
  a section covering the frame, the key LOD and the entries' columns.›

theorem sem_section:
  assumes E: "lawful E" and d: "conforms data" and d': "conforms data'"
    and F: "frame_reads c F" and K: "csupp (kd E F) ⊑ c"
    and cols: "⋀e. e ∈ set (edims E) ⟹ entry_reads c e"
    and sec: "pts_section c data = pts_section c data'"
  shows "esem E data F k = esem E data' F k"
proof (cases "proj (kd E F)⋅k = k")
  case True
  show ?thesis
    by (rule semilawful_content_law[OF lawful_semilawful[OF E] d d' True True
          contents_section[OF lawful_semilawful[OF E] F K cols sec]])
next
  case False
  then have "esem E data F k = None" "esem E data' F k = None"
    using key_lives[OF E d] key_lives[OF E d'] by blast+
  then show ?thesis by simp
qed

end

subsection ‹Lemmas for the combinators›

lemma concat_map_blocks:
  assumes "concat (map f xs) = concat (map g xs)" and "⋀x. x ∈ set xs ⟹ length (f x) = length (g x)"
    and "x ∈ set xs"
  shows "f x = g x"
  using assms
proof (induct xs)
  case (Cons y ys)
  have l: "length (f y) = length (g y)" by (rule Cons.prems(2)) simp
  have "f y = g y ∧ concat (map f ys) = concat (map g ys)"
    using Cons.prems(1) by (simp add: append_eq_append_conv[OF disjI1[OF l]])
  then show ?case using Cons by auto
qed simp

lemma proj_sup_eq:
  "proj (a ⊔ b)⋅x = proj (a ⊔ b)⋅z ⟷ proj a⋅x = proj a⋅z ∧ proj b⋅x = proj b⋅z"
proof
  assume e: "proj (a ⊔ b)⋅x = proj (a ⊔ b)⋅z"
  show "proj a⋅x = proj a⋅z ∧ proj b⋅x = proj b⋅z"
    using arg_cong[OF e, of "Rep_cfun (proj a)"] arg_cong[OF e, of "Rep_cfun (proj b)"]
    by (simp add: proj_le lattice_join_ge1 lattice_join_ge2)
qed (simp add: proj_sup)
