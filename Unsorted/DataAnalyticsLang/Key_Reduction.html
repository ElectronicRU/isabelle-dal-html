theory Key_Reduction

section ‹Evaluating components at their own keys›

text ‹Collecting evaluates every component under the frame raised by all the components' minimal
  LODs, at the full key.  A component may instead be evaluated under the plain frame at its own key
  when raising the frame does not refine its groups.  Raising by ‹M› refines the group of an entry
  with selector ‹(a, ρ)› from ‹a ⊔ ρ ⊓ L› to ‹a ⊔ ρ ⊓ (L ⊔ M)›.  Two cases leave the content alone, as
  long as the full key is a data key: a row entry, whose coarse content has at most one element; and
  an entry whose coarse group LOD determines the masked part of ‹M› in every conforming dataset.›

context dataspace
begin

definition reducible :: "'k semi_factor ⇒ ('k, 'f, 'r, 'v) expr ⇒ ('k semi_factor, 'f, 'r) frame ⇒ bool" where
  "reducible M E F ⟷ (∀e∈set (edims E).
     row_entry e ∨ universally (λP. vfd P (⊥, elower e ⊔ (eupper e ⊓ flod F)) (⊥, (eupper e ⊓ M))))"

end

context dataspace
begin

lemma content_reduce:
  assumes ES: "semilawful E" and d: "conforms data" and M: "emind E ⊑ M" and e: "e ∈ set (edims E)"
    and red: "row_entry e ∨ universally (λP. vfd P (⊥, elower e ⊔ (eupper e ⊓ flod F)) (⊥, (eupper e ⊓ M)))"
    and ki: "keyimg data F (M ⊔ flod F) k"
  shows "content data e (finclude M F) k = content data e F (proj (kd E F)⋅k)"
proof -
  let ?a = "elower e" and ?ρ = "eupper e"
  let ?Gc = "?a ⊔ (?ρ ⊓ flod F)" and ?T = "(?ρ ⊓ M)"
  let ?Gf = "?a ⊔ (?ρ ⊓ (flod F ⊔ M))" and ?W = "M ⊔ flod F"
  have Gf: "?Gf = ?Gc ⊔ ?T" by (simp add: truncate_sup lattice_join_assoc)
  have aM: "?a ⊑ M" by (rule below_trans[OF lower_below_mind[OF ES e] M])
  have Gc_kd: "?Gc ⊑ kd E F" by (rule group_below_kd[OF ES e])
  have "(?ρ ⊓ (flod F ⊔ M)) ⊑ ?W"
    using truncate_below[of ?ρ "flod F ⊔ M"] by (simp only: lattice_join_commute)
  then have Gf_W: "?Gf ⊑ ?W" by (rule join_belowI[OF below_trans[OF aM lattice_join_ge1]])
  have Gc_W: "?Gc ⊑ ?W"
    by (rule below_trans[OF _ Gf_W]) (simp add: Gf lattice_join_ge1)
  obtain x where x: "x ∈ lower_maximal data" "frows F x" and kx: "k = rproj ?W x"
    using ki by (auto simp: keyimg_def)
  have pk: "proj G⋅k = rproj G x" if "G ⊑ ?W" for G
    unfolding kx by (rule proj_le[OF that])
  have pkd: "proj ?Gc⋅(proj (kd E F)⋅k) = proj ?Gc⋅k" by (rule proj_le[OF Gc_kd])
  let ?P = "λy. y ∈ lower_maximal data ∧ selects (amb F) y ∧ holds (coarsen ?ρ (slice F)) (rkey⋅y)"
  have fine: "group data ?a ?ρ (finclude M F) k = {y. ?P y ∧ rproj ?Gf y = rproj ?Gf x}"
    by (auto simp: group_def flod_finclude pk[OF Gf_W])
  have coarse: "group data ?a ?ρ F (proj (kd E F)⋅k) = {y. ?P y ∧ rproj ?Gc y = rproj ?Gc x}"
    by (auto simp: group_def pkd pk[OF Gc_W])
  have fine_eq: "rproj ?Gf y = rproj ?Gf x ⟷ rproj ?Gc y = rproj ?Gc x ∧ rproj ?T y = rproj ?T x" for y
    unfolding Gf by (rule rproj_sup_eq)
  have Px: "?P x" using x by (auto simp: frows_def intro: holds_coarsen)
  have det: "determines (lower_maximal data) v g" if "universally (λP. determines P v g)" for v and g :: "'r → 'b::cpo"
    using universallyD[OF that] d .
  show ?thesis
    using red
  proof
    assume row: "row_entry e"
    let ?g = "λy. ofun (eguard e)⋅y"
    have img: "cpair e ` {y ∈ Y. ?g y = TT} = (if ?g x = TT then {cpair e x} else {})"
      if Y: "Y ⊆ {y. ?P y ∧ rproj ?Gc y = rproj ?Gc x}" and xY: "x ∈ Y" for Y
      by (rule row_entry_image[OF semilawful_wf[OF ES e] row d _ xY lattice_join_ge1]) (use Y in blast)+
    have "cgraph data e (finclude M F) k = cpair e ` {y ∈ {y. ?P y ∧ rproj ?Gf y = rproj ?Gf x}. ?g y = TT}"
      unfolding cgraph_def efiber_def fine ..
    also have "… = (if ?g x = TT then {cpair e x} else {})"
      by (rule img) (use Px in ‹auto simp: fine_eq›)
    also have "… = cpair e ` {y ∈ {y. ?P y ∧ rproj ?Gc y = rproj ?Gc x}. ?g y = TT}"
      by (rule img[symmetric]) (use Px in auto)
    also have "… = cgraph data e F (proj (kd E F)⋅k)"
      unfolding cgraph_def efiber_def coarse ..
    finally show ?thesis unfolding content_def by simp
  next
    assume fd: "universally (λP. vfd P (⊥, ?Gc) (⊥, ?T))"
    have "rproj ?T y = rproj ?T x" if y: "?P y" "rproj ?Gc y = rproj ?Gc x" for y
    proof -
      have "vpart (⊥, ?Gc) y = vpart (⊥, ?Gc) x" using y(2) by (simp add: proj_bot)
      then have "vpart (⊥, ?T) y = vpart (⊥, ?T) x"
        using vfdD[OF det[OF fd]] y(1) x(1) by blast
      then show ?thesis by simp
    qed
    then have "{y. ?P y ∧ rproj ?Gf y = rproj ?Gf x} = {y. ?P y ∧ rproj ?Gc y = rproj ?Gc x}"
      unfolding fine_eq by blast
    then show ?thesis by (intro content_group_cong) (simp only: fine coarse)
  qed
qed

theorem sem_reduce:
  assumes ES: "semilawful E" and d: "conforms data" and M: "emind E ⊑ M" and red: "reducible M E F"
    and ki: "keyimg data F (M ⊔ flod F) k"
  shows "esem E data (finclude M F) k = esem E data F (proj (kd E F)⋅k)"
proof (rule semilawful_content_law[OF ES d d])
  have W: "kd E (finclude M F) = M ⊔ flod F"
  proof -
    have "emind E ⊔ (flod F ⊔ M) = (emind E ⊔ M) ⊔ flod F" by (simp only: lattice_join_ac)
    then show ?thesis by (simp only: flod_finclude lattice_join_absorb2[OF M])
  qed
  obtain x where "k = rproj (M ⊔ flod F) x" using ki by (auto simp: keyimg_def)
  then show "proj (kd E (finclude M F))⋅k = k" unfolding W by simp
  show "proj (kd E F)⋅(proj (kd E F)⋅k) = proj (kd E F)⋅k" by simp
  show "contents data (edims E) (finclude M F) k = contents data (edims E) F (proj (kd E F)⋅k)"
  proof (rule map_cong[OF refl])
    fix e assume e: "e ∈ set (edims E)"
    have r: "row_entry e ∨ universally (λP. vfd P (⊥, elower e ⊔ (eupper e ⊓ flod F)) (⊥, (eupper e ⊓ M)))"
      using red e by (auto simp: reducible_def)
    show "content data e (finclude M F) k = content data e F (proj (kd E F)⋅k)"
      by (rule content_reduce[OF ES d M e r ki])
  qed
qed

end

text ‹The gate without widening: under reducibility by its own minimal LOD, the gate evaluates
  the expression under the plain frame, at its own key.›

context dataspace
begin

theorem gate_reduce:
  assumes E: "semilawful E" and d: "conforms data" and red: "reducible (emind E) E F"
  shows "esem (gate E) data F k = (if keyimg data F (emind E ⊔ flod F) k
      then esem E data F (proj (emind E ⊔ flod F)⋅k) else None)"
proof -
  have ES: "semilawful E" by (fact E)
  show ?thesis using sem_reduce[OF ES d below_refl red] by simp
qed

end
