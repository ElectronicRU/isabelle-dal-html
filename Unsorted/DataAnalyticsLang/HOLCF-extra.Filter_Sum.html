Theory HOLCF-extra.Filter_Sum

theory Filter_Sum imports Distributive begin


section ‹Filter sums›

text ‹A pcpo is a ∗‹filter sum› when any two nonbottom elements with a common upper bound also
  have a common nonbottom lower bound.  In a bounded-complete pcpo this says that joinable
  nonbottom elements have a nonbottom meet (@{text filter_sum_meet} below).  Equivalently, every
  connected component of the punctured space ‹D - {⊥}› is a filter, i.e. ‹D ≅ (Σ⇩i F⇩i)⇩⊥›.  This
  is what keeps the smash product distributive.  (When every component has a least element, ‹D›
  is a ∗‹separated sum›, the lifted disjoint sum of pointed domains.)›

class filter_sum = pcpo +
  assumes filter_sum: "x ≠ ⊥ ⟹ y ≠ ⊥ ⟹ x ⊑ u ⟹ y ⊑ u ⟹ ∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y"

text ‹Bounded-above nonbottom elements have nonbottom lower bounds with any third element
  together: if ‹z› has one with ‹y›, it has one with ‹x›.  No lattice operations needed.›

lemma filter_sum_lower_cong:
  fixes x y z :: "'a::filter_sum"
  assumes x: "x ≠ ⊥" and xu: "x ⊑ u" and yu: "y ⊑ u"
    and w: "w ≠ ⊥" "w ⊑ z" "w ⊑ y"
  shows "∃v. v ≠ ⊥ ∧ v ⊑ z ∧ v ⊑ x"
proof -
  have "w ⊑ u" using ‹w ⊑ y› yu by (rule below_trans)
  from filter_sum[OF ‹w ≠ ⊥› x this xu] obtain v where "v ≠ ⊥" "v ⊑ w" "v ⊑ x" by fast
  with ‹w ⊑ z› show ?thesis by (fast dest: below_trans)
qed

text ‹Separation through any chain of maps with lub ‹ID› (e.g. the ‹take› functions of a domain):
  it suffices to separate approximants ‹t n⋅x› from arbitrary ‹y›.  A nonbottom ‹x› has a
  nonbottom approximant, and a witness below it lies below ‹x›.›

lemma filter_sum_chainI:
  fixes t :: "nat ⇒ 'a::pcpo → 'a"
  assumes t: "chain t" "(⨆i. t i) = ID"
    and sep: "⋀n x y u. t n⋅x ≠ ⊥ ⟹ y ≠ ⊥ ⟹ t n⋅x ⊑ u ⟹ y ⊑ u ⟹
      ∃l. l ≠ ⊥ ∧ l ⊑ t n⋅x ∧ l ⊑ y"
  shows "OFCLASS('a, filter_sum_class)"
proof
  fix x y u :: 'a assume "x ≠ ⊥" "y ≠ ⊥" "x ⊑ u" "y ⊑ u"
  have chx: "chain (λn. t n⋅x)" by (rule ch2ch_Rep_cfunL[OF t(1)])
  have reach: "(⨆n. t n⋅x) = x" by (simp add: contlub_cfun_fun[OF t(1), symmetric] t(2))
  have below: "t n⋅x ⊑ x" for n using is_ub_thelub[OF chx, of n] by (simp add: reach)
  have "∃n. t n⋅x ≠ ⊥" using ‹x ≠ ⊥› lub_eq_bottom_iff[OF chx] by (simp add: reach)
  then obtain n :: nat where n: "t n⋅x ≠ ⊥" by (elim exE)
  have "t n⋅x ⊑ u" using below ‹x ⊑ u› by (rule below_trans)
  from sep[OF n ‹y ≠ ⊥› this ‹y ⊑ u›] obtain l :: 'a where l: "l ≠ ⊥" "l ⊑ t n⋅x" "l ⊑ y"
    by (elim exE conjE)
  from l(2) below have "l ⊑ x" by (rule below_trans)
  with l(1,3) show "∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y" by (intro exI[of _ l]) simp
qed

subsection ‹The meet form, in bounded-complete pcpos›

text ‹A nonbottom meet is the same as a nonbottom common lower bound; through this, the
  lattice-free statements above become statements about meets.›

lemma nonbot_meet_iff:
  fixes x y :: "'a::bounded_complete"
  shows "x ⊓ y ≠ ⊥ ⟷ (∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y)"
proof
  assume "x ⊓ y ≠ ⊥"
  then show "∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y"
    by (intro exI[of _ "x ⊓ y"] conjI meet_below1 meet_below2)
next
  assume "∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y"
  then obtain l :: 'a where "l ≠ ⊥" "l ⊑ x" "l ⊑ y" by (elim exE conjE)
  then show "x ⊓ y ≠ ⊥" by (metis bottomI meet_greatest)
qed

lemma filter_sum_meet:
  fixes x y :: "'a::{filter_sum, bounded_complete}"
  assumes "x ⇑ y" "x ⊓ y = ⊥" shows "x = ⊥ ∨ y = ⊥"
proof -
  from ‹x ⇑ y› obtain u :: 'a where "x ⊑ u" "y ⊑ u" unfolding joinable_def by (elim exE conjE)
  with filter_sum[of x y u] ‹x ⊓ y = ⊥› show ?thesis
    by (auto simp: nonbot_meet_iff[symmetric])
qed

text ‹Joinable nonbottom elements meet any third element to ‹⊥› together.›

lemma filter_sum_meet_bottom_cong:
  fixes x y z :: "'a::{filter_sum, bounded_complete}"
  assumes "x ⇑ y" "x ≠ ⊥" "y ≠ ⊥" "z ⊓ x = ⊥" shows "z ⊓ y = ⊥"
proof (rule ccontr)
  assume "z ⊓ y ≠ ⊥"
  then obtain w :: 'a where w: "w ≠ ⊥" "w ⊑ z" "w ⊑ y"
    unfolding nonbot_meet_iff by (elim exE conjE)
  from ‹x ⇑ y› obtain u :: 'a where u: "x ⊑ u" "y ⊑ u" unfolding joinable_def by (elim exE conjE)
  have "∃v. v ≠ ⊥ ∧ v ⊑ z ∧ v ⊑ x"
    using filter_sum_lower_cong[of x u y w z] ‹x ≠ ⊥› u w by simp
  then have "z ⊓ x ≠ ⊥" by (simp add: nonbot_meet_iff)
  with ‹z ⊓ x = ⊥› show False by simp
qed

text ‹Lifting needs no assumption on the argument beyond pointedness: ‹up⋅⊥› lies below every
  nonbottom element.›

instance u :: (pcpo) filter_sum
proof
  fix x y u :: "'a u" assume "x ≠ ⊥" "y ≠ ⊥"
  then show "∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y"

qed

text ‹A common upper bound forces both elements into the same summand; separate there.›

instance ssum :: ("{pcpo, filter_sum}", "{pcpo, filter_sum}") filter_sum
proof
  fix x y u :: "'a ⊕ 'b" assume nb: "x ≠ ⊥" "y ≠ ⊥" and le: "x ⊑ u" "y ⊑ u"
  have L: "∃l. l ≠ ⊥ ∧ l ⊑ (sinl⋅a :: 'a ⊕ 'b) ∧ l ⊑ sinl⋅b"
    if "a ≠ ⊥" "b ≠ ⊥" "a ⊑ c" "b ⊑ c" for a b c :: 'a
  proof -
    from filter_sum[OF that] obtain l :: 'a where "l ≠ ⊥" "l ⊑ a" "l ⊑ b"
      by (elim exE conjE)
    then show ?thesis by (intro exI[of _ "sinl⋅l"]) simp
  qed
  have R: "∃l. l ≠ ⊥ ∧ l ⊑ (sinr⋅a :: 'a ⊕ 'b) ∧ l ⊑ sinr⋅b"
    if "a ≠ ⊥" "b ≠ ⊥" "a ⊑ c" "b ⊑ c" for a b c :: 'b
  proof -
    from filter_sum[OF that] obtain l :: 'b where "l ≠ ⊥" "l ⊑ a" "l ⊑ b"
      by (elim exE conjE)
    then show ?thesis by (intro exI[of _ "sinr⋅l"]) simp
  qed
  show "∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y"
  proof (cases x)
    case bottom with nb show ?thesis by simp
  next
    case (sinl a)
    from le(1) sinl obtain c where u: "u = sinl⋅c" and ac: "a ⊑ c" by (cases u) auto
    from le(2) nb(2) u obtain b where y: "y = sinl⋅b" "b ≠ ⊥" and bc: "b ⊑ c" by (cases y) auto
    from L[OF sinl(2) y(2) ac bc] show ?thesis by (simp only: sinl(1) y(1))
  next
    case (sinr a)
    from le(1) sinr obtain c where u: "u = sinr⋅c" and ac: "a ⊑ c" by (cases u) auto
    from le(2) nb(2) u obtain b where y: "y = sinr⋅b" "b ≠ ⊥" and bc: "b ⊑ c" by (cases y) auto
    from R[OF sinr(2) y(2) ac bc] show ?thesis by (simp only: sinr(1) y(1))
  qed
qed

text ‹Componentwise: the pair of the two components' witnesses.›

instance sprod :: ("{pcpo, filter_sum}", "{pcpo, filter_sum}") filter_sum
proof
  fix x y u :: "'a ⊗ 'b" assume nb: "x ≠ ⊥" "y ≠ ⊥" and le: "x ⊑ u" "y ⊑ u"
  have S: "∃l. l ≠ ⊥ ∧ l ⊑ (:a, b:) ∧ l ⊑ (:a', b':)"
    if "a ≠ ⊥" "b ≠ ⊥" "a' ≠ ⊥" "b' ≠ ⊥" "a ⊑ c" "a' ⊑ c" "b ⊑ d" "b' ⊑ d"
    for a a' c :: 'a and b b' d :: 'b
  proof -
    from filter_sum[OF ‹a ≠ ⊥› ‹a' ≠ ⊥› ‹a ⊑ c› ‹a' ⊑ c›]
    obtain la :: 'a where "la ≠ ⊥" "la ⊑ a" "la ⊑ a'" by (elim exE conjE)
    moreover from filter_sum[OF ‹b ≠ ⊥› ‹b' ≠ ⊥› ‹b ⊑ d› ‹b' ⊑ d›]
    obtain lb :: 'b where "lb ≠ ⊥" "lb ⊑ b" "lb ⊑ b'" by (elim exE conjE)
    ultimately show ?thesis using that
      by (intro exI[of _ "(:la, lb:)"] conjI) (simp_all only: spair_below spair_bottom_iff simp_thms)
  qed
  obtain x1 x2 where x: "x = (:x1, x2:)" "x1 ≠ ⊥" "x2 ≠ ⊥" using nb(1) by (cases x) auto
  obtain y1 y2 where y: "y = (:y1, y2:)" "y1 ≠ ⊥" "y2 ≠ ⊥" using nb(2) by (cases y) auto
  obtain u1 u2 where u: "u = (:u1, u2:)" using le(1) x by (cases u) auto
  have "x1 ⊑ u1" "x2 ⊑ u2" "y1 ⊑ u1" "y2 ⊑ u2"
    using le(1)[unfolded x(1) u] le(2)[unfolded y(1) u]
    by (simp_all only: spair_below x(2,3) y(2,3) simp_thms)
  from S[OF x(2,3) y(2,3) this(1,3,2,4)] show "∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y"
    by (simp only: x(1) y(1))
qed

section ‹Separated sums›

text ‹An ∗‹atom› is a minimal nonbottom element.  A pcpo is a ∗‹separated sum› when any two
  nonbottom elements with a common upper bound lie above a common atom: atomicity and
  separation in one statement.  Equivalently every component of the punctured space
  ‹D - {⊥}› has a least element, i.e.\ ‹D ≅ (Σ⇩i C⇩i)⇩⊥› for pointed ‹C⇩i›, the separated sum of
  the ‹C⇩i›; that least element is the component's unique atom.›

definition atom :: "'a::pcpo ⇒ bool" where
  "atom l ⟷ l ≠ ⊥ ∧ (∀z. z ⊑ l ⟶ z = ⊥ ∨ z = l)"

lemma atomI: "l ≠ ⊥ ⟹ (⋀z. z ⊑ l ⟹ z = ⊥ ∨ z = l) ⟹ atom l"
  by (simp add: atom_def)

lemma atom_nonbottom: "atom l ⟹ l ≠ ⊥"
  by (simp add: atom_def)

lemma atomD: "atom l ⟹ z ⊑ l ⟹ z = ⊥ ∨ z = l"
  by (simp add: atom_def)

class separated_sum = pcpo +
  assumes separated_sum: "x ≠ ⊥ ⟹ y ≠ ⊥ ⟹ x ⊑ u ⟹ y ⊑ u ⟹ ∃l. atom l ∧ l ⊑ x ∧ l ⊑ y"

instance separated_sum ⊆ filter_sum
proof
  fix x y u :: 'a assume "x ≠ ⊥" "y ≠ ⊥" "x ⊑ u" "y ⊑ u"
  from separated_sum[OF this] show "∃l. l ≠ ⊥ ∧ l ⊑ x ∧ l ⊑ y"
    by (blast dest: atom_nonbottom)
qed

lemma atom_below:
  fixes x :: "'a::separated_sum"
  assumes "x ≠ ⊥" obtains l where "atom l" "l ⊑ x"
  using separated_sum[OF assms assms below_refl below_refl] by blast

text ‹Two atoms with a common upper bound are equal.›

lemma atom_unique:
  fixes a b :: "'a::separated_sum"
  assumes a: "atom a" and b: "atom b" and "a ⊑ u" "b ⊑ u" shows "a = b"
proof -
  from separated_sum[OF atom_nonbottom[OF a] atom_nonbottom[OF b] assms(3,4)]
  obtain l where l: "atom l" "l ⊑ a" "l ⊑ b" by blast
  have "l = a" using atomD[OF a l(2)] atom_nonbottom[OF l(1)] by blast
  moreover have "l = b" using atomD[OF b l(3)] atom_nonbottom[OF l(1)] by blast
  ultimately show ?thesis by simp
qed

text ‹In an algebraic domain atoms are compact: some compact approximant is nonbottom, and
  lies below the atom.›

lemma atom_compact:
  fixes a :: "'a::bifinite" assumes a: "atom a" shows "compact a"
proof -
  obtain C where C: "chain C" "∀i. compact (C i)" "a = (⨆i. C i)" by (rule obtain_compact_chain)
  have "∃i. C i ≠ ⊥" using C(1,3) atom_nonbottom[OF a] lub_eq_bottom_iff by blast
  then obtain i where i: "C i ≠ ⊥" by blast
  have "C i ⊑ a" using is_ub_thelub[OF C(1)] C(3) by simp
  with atomD[OF a] i have "C i = a" by blast
  with C(2) show ?thesis by metis
qed

subsection ‹Atoms of the domain constructions›

lemma atom_up_bottom: "atom (up⋅(⊥::'a::pcpo))"
proof (rule atomI)
  fix z :: "'a u" assume "z ⊑ up⋅⊥"
  then show "z = ⊥ ∨ z = up⋅⊥" by (cases z) simp_all
qed simp

lemma atom_sinl_iff: "atom (sinl⋅x :: 'a::pcpo ⊕ 'b::pcpo) ⟷ atom x"
proof
  assume a: "atom (sinl⋅x :: 'a ⊕ 'b)"
  show "atom x"
  proof (rule atomI)
    show "x ≠ ⊥" using atom_nonbottom[OF a] by auto
    fix z assume "z ⊑ x"
    then have "(sinl⋅z :: 'a ⊕ 'b) ⊑ sinl⋅x" by simp
    from atomD[OF a this] show "z = ⊥ ∨ z = x" by simp
  qed
next
  assume a: "atom x"
  show "atom (sinl⋅x :: 'a ⊕ 'b)"
  proof (rule atomI)
    show "(sinl⋅x :: 'a ⊕ 'b) ≠ ⊥" using atom_nonbottom[OF a] by simp
    fix z :: "'a ⊕ 'b" assume z: "z ⊑ sinl⋅x"
    show "z = ⊥ ∨ z = sinl⋅x"
    proof (cases z)
      case (sinl z') with z atomD[OF a, of z'] show ?thesis by auto
    qed (use z in auto)
  qed
qed

lemma atom_sinr_iff: "atom (sinr⋅x :: 'a::pcpo ⊕ 'b::pcpo) ⟷ atom x"
proof
  assume a: "atom (sinr⋅x :: 'a ⊕ 'b)"
  show "atom x"
  proof (rule atomI)
    show "x ≠ ⊥" using atom_nonbottom[OF a] by auto
    fix z assume "z ⊑ x"
    then have "(sinr⋅z :: 'a ⊕ 'b) ⊑ sinr⋅x" by simp
    from atomD[OF a this] show "z = ⊥ ∨ z = x" by simp
  qed
next
  assume a: "atom x"
  show "atom (sinr⋅x :: 'a ⊕ 'b)"
  proof (rule atomI)
    show "(sinr⋅x :: 'a ⊕ 'b) ≠ ⊥" using atom_nonbottom[OF a] by simp
    fix z :: "'a ⊕ 'b" assume z: "z ⊑ sinr⋅x"
    show "z = ⊥ ∨ z = sinr⋅x"
    proof (cases z)
      case (sinr z') with z atomD[OF a, of z'] show ?thesis by auto
    qed (use z in auto)
  qed
qed

lemma atom_spair:
  assumes a: "atom (a::'a::pcpo)" and b: "atom (b::'b::pcpo)" shows "atom (:a, b:)"
proof (rule atomI)
  show "(:a, b:) ≠ ⊥" using atom_nonbottom[OF a] atom_nonbottom[OF b] by simp
  fix z assume z: "z ⊑ (:a, b:)"
  show "z = ⊥ ∨ z = (:a, b:)"
  proof (cases z)
    case (spair z1 z2)
    with z atom_nonbottom[OF a] atom_nonbottom[OF b] have "z1 ⊑ a" "z2 ⊑ b"
      by (simp_all add: spair_below)
    with atomD[OF a] atomD[OF b] spair show ?thesis by auto
  qed simp
qed

    from separated_sum[OF sinr(2) y(2) ac bc] obtain l where "atom l" "l ⊑ a" "l ⊑ b" by blast
    with sinr y show ?thesis by (intro exI[of _ "sinr⋅l"]) (simp add: atom_sinr_iff)

instance sprod :: (separated_sum, separated_sum) separated_sum
proof
  fix x y u :: "'a ⊗ 'b" assume nb: "x ≠ ⊥" "y ≠ ⊥" and le: "x ⊑ u" "y ⊑ u"
  obtain x1 x2 where x: "x = (:x1, x2:)" "x1 ≠ ⊥" "x2 ≠ ⊥" using nb(1) by (cases x) auto
  obtain y1 y2 where y: "y = (:y1, y2:)" "y1 ≠ ⊥" "y2 ≠ ⊥" using nb(2) by (cases y) auto
  obtain u1 u2 where u: "u = (:u1, u2:)" using le(1) x by (cases u) auto
  have "x1 ⊑ u1" "x2 ⊑ u2" "y1 ⊑ u1" "y2 ⊑ u2"
    using le(1)[unfolded x(1) u] le(2)[unfolded y(1) u]
    by (simp_all only: spair_below x(2,3) y(2,3) simp_thms)
  then obtain l1 l2 where "atom l1" "l1 ⊑ x1" "l1 ⊑ y1" "atom l2" "l2 ⊑ x2" "l2 ⊑ y2"
    using separated_sum[OF x(2) y(2)] separated_sum[OF x(3) y(3)] by meson
  then show "∃l. atom l ∧ l ⊑ x ∧ l ⊑ y"
    using x y by (intro exI[of _ "(:l1, l2:)"])
      (simp add: atom_spair spair_below atom_nonbottom)
qed

section ‹Distributivity of smash products›

text ‹Separation of both components keeps ‹⊗› distributive: joinable nonbottom pairs meet a third
  pair to ‹⊥› together, so either both sides collapse or everything is componentwise.›

instance sprod :: ("{filter_sum, distributive_cpo}", "{filter_sum, distributive_cpo}")
  distributive_cpo
proof
  fix a b c :: "'a ⊗ 'b" assume "a ⇑ b"
  show "c ⊓ (a ⊔ b) = c ⊓ a ⊔ c ⊓ b"
  proof (cases "a = ⊥ ∨ b = ⊥ ∨ c = ⊥")
    case True then show ?thesis by (elim disjE) simp_all
  next
    case False
    then obtain a1 a2 b1 b2 c1 c2 where
      ab: "a = (:a1, a2:)" "b = (:b1, b2:)" "c = (:c1, c2:)" and
      nb: "a1 ≠ ⊥" "a2 ≠ ⊥" "b1 ≠ ⊥" "b2 ≠ ⊥" "c1 ≠ ⊥" "c2 ≠ ⊥"
      by (cases a; cases b; cases c) auto
    from ‹a ⇑ b› have j: "a1 ⇑ b1" "a2 ⇑ b2" by (simp_all add: ab nb spair_joinable_iff)
    have cong: "c1 ⊓ a1 = ⊥ ⟷ c1 ⊓ b1 = ⊥" "c2 ⊓ a2 = ⊥ ⟷ c2 ⊓ b2 = ⊥"
      using filter_sum_meet_bottom_cong[OF j(1) nb(1,3)] filter_sum_meet_bottom_cong[OF j(2) nb(2,4)]
        filter_sum_meet_bottom_cong[OF joinable_sym[THEN iffD1, OF j(1)] nb(3,1)]
        filter_sum_meet_bottom_cong[OF joinable_sym[THEN iffD1, OF j(2)] nb(4,2)]
      by blast+
    have mj: "x ⊓ y ⇑ x ⊓ z" for x y z :: "'c::bounded_complete"
      by (rule joinableI[OF meet_below1 meet_below1])
    have lhs: "c ⊓ (a ⊔ b) = (:c1 ⊓ a1 ⊔ c1 ⊓ b1, c2 ⊓ a2 ⊔ c2 ⊓ b2:)"
      by (simp only: ab spair_join[OF nb(1,2,3,4) j] spair_meet meet_join_distrib[OF j(1)]
        meet_join_distrib[OF j(2)])
    have rhs: "c ⊓ a ⊔ c ⊓ b = (:c1 ⊓ a1, c2 ⊓ a2:) ⊔ (:c1 ⊓ b1, c2 ⊓ b2:)"
      by (simp only: ab spair_meet)
    show ?thesis
    proof (cases "c1 ⊓ a1 = ⊥ ∨ c2 ⊓ a2 = ⊥")
      case True
      with cong show ?thesis unfolding lhs rhs by auto
    next
      case False
      with cong have "c1 ⊓ a1 ≠ ⊥" "c2 ⊓ a2 ≠ ⊥" "c1 ⊓ b1 ≠ ⊥" "c2 ⊓ b2 ≠ ⊥" by simp_all
      then show ?thesis unfolding lhs rhs by (simp add: spair_join mj)
    qed
  qed
qed

text ‹If one component is flat, the other needs no separation: joinable nonbottom elements of a
  flat type are equal, so that component contributes no proper joins.  These cannot be
  instances (they would conflict with the arity above), so they are stated as the class law.›

lemma sprod_flat_left_distrib:
  fixes a b c :: "'a::flat ⊗ 'b::distributive_cpo"
  assumes "a ⇑ b" shows "c ⊓ (a ⊔ b) = c ⊓ a ⊔ c ⊓ b"
proof (cases "a = ⊥ ∨ b = ⊥ ∨ c = ⊥")
  case True then show ?thesis by (elim disjE) simp_all
next
  case False
  then obtain a1 a2 b1 b2 c1 c2 where
    ab: "a = (:a1, a2:)" "b = (:b1, b2:)" "c = (:c1, c2:)" and
    nb: "a1 ≠ ⊥" "a2 ≠ ⊥" "b1 ≠ ⊥" "b2 ≠ ⊥" "c1 ≠ ⊥" "c2 ≠ ⊥"
    by (cases a; cases b; cases c) auto
  from ‹a ⇑ b› have j: "a1 ⇑ b1" "a2 ⇑ b2" by (simp_all add: ab nb spair_joinable_iff)
  from j(1) nb(1,3) have "a1 = b1" by (blast elim: flat_joinableE)
  with ab(2) have b: "b = (:a1, b2:)" by simp
  have mj: "x ⊓ y ⇑ x ⊓ z" for x y z :: "'c::bounded_complete"
    by (rule joinableI[OF meet_below1 meet_below1])
  have lhs: "c ⊓ (a ⊔ b) = (:c1 ⊓ a1, c2 ⊓ a2 ⊔ c2 ⊓ b2:)"
    by (simp only: ab(1,3) b spair_join[OF nb(1,2,1,4) joinable_refl j(2)] spair_meet join_idem
      meet_join_distrib[OF j(2)])
  have rhs: "c ⊓ a ⊔ c ⊓ b = (:c1 ⊓ a1, c2 ⊓ a2:) ⊔ (:c1 ⊓ a1, c2 ⊓ b2:)"
    by (simp only: ab(1,3) b spair_meet)
  show ?thesis
  proof (cases "c1 ⊓ a1 = ⊥ ∨ c2 ⊓ a2 = ⊥ ∨ c2 ⊓ b2 = ⊥")
    case True then show ?thesis unfolding lhs rhs by (elim disjE) simp_all
  next    case False then show ?thesis unfolding lhs rhs by (simp add: spair_join mj)
  qed
qed

global_interpretation sprod_flat_left: distributive_cpo
  "below :: ('a::flat ⊗ 'b::distributive_cpo) ⇒ _ ⇒ bool"
  rewrites "po.is_ub (⊑) = ((<|) :: ('a ⊗ 'b) set ⇒ _)"
    and "po.is_lub (⊑) = ((<<|) :: ('a ⊗ 'b) set ⇒ _)"
    and "po.lub (⊑) = (lub :: ('a ⊗ 'b) set ⇒ _)"
    and "po.is_glb (⊑) = (is_glb :: ('a ⊗ 'b) set ⇒ _)"
    and "po.glb (⊑) = (glb :: ('a ⊗ 'b) set ⇒ _)"
    and "po.meet (⊑) = ((⊓) :: 'a ⊗ 'b ⇒ _)"
    and "po.join (⊑) = ((⊔) :: 'a ⊗ 'b ⇒ _)"
    and "bounded_complete.joinable (⊑) = ((⇑) :: 'a ⊗ 'b ⇒ _)"
  by unfold_locales (simp_all only: class_locale_eqs sprod_flat_left_distrib)

lemma sprod_flat_right_distrib:
  fixes a b c :: "'a::distributive_cpo ⊗ 'b::flat"
  assumes "a ⇑ b" shows "c ⊓ (a ⊔ b) = c ⊓ a ⊔ c ⊓ b"
proof (cases "a = ⊥ ∨ b = ⊥ ∨ c = ⊥")
  case True then show ?thesis by (elim disjE) simp_all
next
  case False
  then obtain a1 a2 b1 b2 c1 c2 where
    ab: "a = (:a1, a2:)" "b = (:b1, b2:)" "c = (:c1, c2:)" and
    nb: "a1 ≠ ⊥" "a2 ≠ ⊥" "b1 ≠ ⊥" "b2 ≠ ⊥" "c1 ≠ ⊥" "c2 ≠ ⊥"
    by (cases a; cases b; cases c) auto
  from ‹a ⇑ b› have j: "a1 ⇑ b1" "a2 ⇑ b2" by (simp_all add: ab nb spair_joinable_iff)
  from j(2) nb(2,4) have "a2 = b2" by (blast elim: flat_joinableE)
  with ab(2) have b: "b = (:b1, a2:)" by simp
  have mj: "x ⊓ y ⇑ x ⊓ z" for x y z :: "'c::bounded_complete"
    by (rule joinableI[OF meet_below1 meet_below1])
  have lhs: "c ⊓ (a ⊔ b) = (:c1 ⊓ a1 ⊔ c1 ⊓ b1, c2 ⊓ a2:)"
    by (simp only: ab(1,3) b spair_join[OF nb(1,2,3,2) j(1) joinable_refl] spair_meet join_idem
      meet_join_distrib[OF j(1)])
  have rhs: "c ⊓ a ⊔ c ⊓ b = (:c1 ⊓ a1, c2 ⊓ a2:) ⊔ (:c1 ⊓ b1, c2 ⊓ a2:)"
    by (simp only: ab(1,3) b spair_meet)
  show ?thesis
  proof (cases "c2 ⊓ a2 = ⊥ ∨ c1 ⊓ a1 = ⊥ ∨ c1 ⊓ b1 = ⊥")
    case True then show ?thesis unfolding lhs rhs by (elim disjE) simp_all
  next
    case False then show ?thesis unfolding lhs rhs by (simp add: spair_join mj)
  qed
qed

  by unfold_locales (simp_all only: class_locale_eqs sprod_flat_right_distrib)
