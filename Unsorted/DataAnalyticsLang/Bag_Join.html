Theory Bag_Join

theory Bag_Join
  imports "HOLCF-extra.BagPD" LOD
begin

default_sort bifinite

section ‹Outer joins of bags›

text ‹A binary outer join of two bags of rows. The sides keep their own row types; a pair of
  rows is combined by a monotone @{text c} (the pair, or the join of two disjoint blocks of a
  universal row), and an orphan is combined with bottom. Rows meet through key maps into a
  common key order: a pair is a \emph{decided match} when the keys are equal and live, which
  makes them maximal. Other keys never match: partial keys, and maximal keys that are not live
  such as a decided NULL, which pass through as orphans.

  A decided match of @{text "n⋅x"} and @{text "m⋅y"} gives @{text "n m"} joined rows. Besides,
  each row @{text x} of multiplicity @{text n} keeps @{text "n N(x)"} orphan copies, where
  @{text "N(x)"} bounds the partners that refinements can still bring:
  ▪ @{text ∞} on an @{text Open} side (the 1-side of an N:1 join: more referencing rows may join);
  ▪ on a @{text Closed} side with a decided match, the undecided partner mass compatible with
    its key;
  ▪ on a @{text Closed} side without a decided match, the largest partner mass that a live key
    above its key can collect, and at least one copy.
  Refinement only merges rows, so partner masses only shrink and the join is monotone in both
  arguments. No deduplication is involved: duplicate rows simply multiply.›

subsection ‹Sides and indices›

datatype join_side = Open | Closed

text ‹The rows of a join on the basis are indexed by joined pairs and by the orphans of either
  side.›

datatype ('a, 'b) join_index =
    Joined "'a compact_basis" "'b compact_basis"
  | Left_Orphan "'a compact_basis"
  | Right_Orphan "'b compact_basis"

definition join_indices :: "'a bag_basis ⇒ 'b bag_basis ⇒ ('a, 'b) join_index set"
  where "join_indices t u =
    case_prod Joined ` (bsupp t × bsupp u) ∪ (Left_Orphan ` bsupp t ∪ Right_Orphan ` bsupp u)"

lemma join_indices_iff [simp]:
  "Joined x y ∈ join_indices t u ⟷ x ∈ bsupp t ∧ y ∈ bsupp u"
  "Left_Orphan x ∈ join_indices t u ⟷ x ∈ bsupp t"
  "Right_Orphan y ∈ join_indices t u ⟷ y ∈ bsupp u"
  by (auto simp: join_indices_def)

lemma finite_join_indices [simp]: "finite (join_indices t u)"
  by (simp add: join_indices_def)

lemma sum_join_indices:
  "sum g (join_indices t u) = (∑x∈bsupp t. ∑y∈bsupp u. g (Joined x y)) +
    ((∑x∈bsupp t. g (Left_Orphan x)) + (∑y∈bsupp u. g (Right_Orphan y)))"
proof -
  have "sum g (join_indices t u) = sum g (case_prod Joined ` (bsupp t × bsupp u)) +
      (sum g (Left_Orphan ` bsupp t) + sum g (Right_Orphan ` bsupp u))"
    unfolding join_indices_def by (subst sum.union_disjoint; auto)+
  then show ?thesis
    by (simp add: sum.reindex inj_on_def sum.cartesian_product prod.case_distrib)
qed

primrec join_label ::
    "('a compact_basis ⇒ 'b compact_basis ⇒ 'c compact_basis) ⇒ ('a, 'b) join_index ⇒ 'c compact_basis"
  where
    "join_label c (Joined x y) = c x y"
  | "join_label c (Left_Orphan x) = c x compact_bot"
  | "join_label c (Right_Orphan y) = c compact_bot y"

primrec join_swap :: "('a, 'b) join_index ⇒ ('b, 'a) join_index"
  where
    "join_swap (Joined x y) = Joined y x"
  | "join_swap (Left_Orphan x) = Right_Orphan x"
  | "join_swap (Right_Orphan y) = Left_Orphan y"

lemma join_swap_swap [simp]: "join_swap (join_swap i) = i"
  by (cases i) simp_all

lemma join_swap_indices: "i ∈ join_indices t u ⟹ join_swap i ∈ join_indices u t"
  by (cases i) simp_all

text ‹Joined rows as pairs, and keys of basis elements.›

definition pair_basis :: "'a compact_basis ⇒ 'b compact_basis ⇒ ('a × 'b) compact_basis"
  where "pair_basis x y = Abs_compact_basis (Rep_compact_basis x, Rep_compact_basis y)"

lemma Rep_pair_basis [simp]:
  "Rep_compact_basis (pair_basis x y) = (Rep_compact_basis x, Rep_compact_basis y)"
  by (simp add: pair_basis_def)

lemma pair_basis_mono:
  assumes "x ⊑ x'" and "y ⊑ y'"
  shows "pair_basis x y ⊑ pair_basis x' y'"
proof (rule compact_below_iff [THEN iffD2])
  show "Rep_compact_basis (pair_basis x y) ⊑ Rep_compact_basis (pair_basis x' y')"
    using assms [THEN compact_below_iff [THEN iffD1]] by (simp only: Rep_pair_basis Pair_below_iff)
qed

abbreviation on_basis :: "('a → 'k::cpo) ⇒ 'a compact_basis ⇒ 'k"
  where "on_basis f x ≡ f⋅(Rep_compact_basis x)"

lemma monofun_on_basis: "monofun (on_basis f)"
  by (rule monofunI, rule monofun_cfun_arg, erule compact_below_iff [THEN iffD1])

subsection ‹Partner masses›

text ‹Keys range over @{text Keys}; the live keys are maximal there.›

locale bag_join_keys =
  fixes Live Keys :: "'k::cpo set"
  assumes live_maximal: "k ∈ Live ⟹ k ⊑ k' ⟹ k' ∈ Keys ⟹ k' = k"
begin

definition decided_match :: "'k ⇒ 'k ⇒ bool"
  where "decided_match k k' ⟷ k ∈ Live ∧ k' = k"

lemma decided_match_sym: "decided_match k k' ⟷ decided_match k' k"
  by (auto simp: decided_match_def)

text ‹For a row with key @{text k} against the bag @{text u} on the other side, whose rows have
  keys @{text key}: the decided partner mass, the undecided partner mass below @{text k}, and
  the largest mass that a live key above @{text k} can collect.›

definition decided_mass :: "('b compact_basis ⇒ 'k) ⇒ 'b bag_basis ⇒ 'k ⇒ enat"
  where "decided_mass key u k = bag_mass u (λy. decided_match k (key y))"

definition undecided_mass :: "('b compact_basis ⇒ 'k) ⇒ 'b bag_basis ⇒ 'k ⇒ enat"
  where "undecided_mass key u k = bag_mass u (λy. key y ⊑ k ∧ key y ≠ k)"

definition reachable_mass :: "('b compact_basis ⇒ 'k) ⇒ 'b bag_basis ⇒ 'k ⇒ enat"
  where "reachable_mass key u k = (SUP k'∈{k'∈Live. k ⊑ k'}. bag_mass u (λy. key y ⊑ k'))"

fun orphan_count :: "join_side ⇒ ('b compact_basis ⇒ 'k) ⇒ 'b bag_basis ⇒ 'k ⇒ enat"
  where
    "orphan_count Open key u k = ∞"
  | "orphan_count Closed key u k =
      (if decided_mass key u k = 0 then max 1 (reachable_mass key u k) else undecided_mass key u k)"

lemma decided_mass_live: "decided_mass key u k ≠ 0 ⟹ k ∈ Live"
  by (auto simp: decided_mass_def bag_mass_nonzero_iff decided_match_def)

lemma orphan_count_0: "orphan_count w key u k = 0 ⟹ w = Closed ∧ decided_mass key u k ≠ 0"
  by (cases w) (auto simp: max_def split: if_split_asm)

lemma decided_undecided_mass:
  "k ∈ Live ⟹ decided_mass key u k + undecided_mass key u k = bag_mass u (λy. key y ⊑ k)"
  unfolding decided_mass_def undecided_mass_def
  by (subst bag_mass_disj[symmetric]) (auto simp: decided_match_def intro: bag_mass_cong)

lemma reachable_mass_upper:
  "k' ∈ Live ⟹ k ⊑ k' ⟹ bag_mass u (λy. key y ⊑ k') ≤ reachable_mass key u k"
  unfolding reachable_mass_def by (rule SUP_upper) simp

lemma reachable_mass_key_anti: "k ⊑ k1 ⟹ reachable_mass key u k1 ≤ reachable_mass key u k"
  unfolding reachable_mass_def by (rule SUP_subset_mono) (auto intro: below_trans)

text ‹A row without a decided match holds enough copies for whatever a row above it emits.›

lemma orphan_count_step:
  assumes "k ⊑ k1" and "decided_mass key u k = 0"
  shows "decided_mass key u k1 + orphan_count w key u k1 ≤ orphan_count w key u k"
proof (cases w)
  case Open
  then show ?thesis
    by simp
next
  case Closed
  show ?thesis
  proof (cases "decided_mass key u k1 = 0")
    case True
    have "max 1 (reachable_mass key u k1) ≤ max 1 (reachable_mass key u k)"
      using reachable_mass_key_anti[OF assms(1)] by (rule max.mono[OF order_refl])
    then show ?thesis
      using Closed True assms(2) by simp
  next
    case nz: False
    then have live: "k1 ∈ Live"
      by (rule decided_mass_live)
    have "decided_mass key u k1 + orphan_count w key u k1 = bag_mass u (λy. key y ⊑ k1)"
      using Closed nz decided_undecided_mass[OF live] by simp
    also have "… ≤ reachable_mass key u k"
      by (rule reachable_mass_upper[OF live assms(1)])
    also have "… ≤ orphan_count w key u k"
      using Closed assms(2) by simp
    finally show ?thesis .
  qed
qed

text ‹Refining the other side only lowers the counts.›

lemma decided_mass_up:
  assumes "t ≤⇩b t'" and "monofun key" and keys: "⋀x. key x ∈ Keys"
    and "decided_mass key t k ≠ 0"
  shows "decided_mass key t' k ≠ 0"
proof -
  obtain x where x: "x ∈ bsupp t" "decided_match k (key x)"
    using assms(4) by (auto simp: decided_mass_def bag_mass_nonzero_iff)
  obtain x' where "x' ∈ bsupp t'" "x ⊑ x'"
    using bag_le_lower[OF assms(1) x(1)] by blast
  moreover have "decided_match k (key x')"
    using x(2) monofunE[OF assms(2) ‹x ⊑ x'›] live_maximal keys by (auto simp: decided_match_def)
  ultimately show ?thesis
    by (auto simp: decided_mass_def bag_mass_nonzero_iff)
qed

lemma undecided_mass_anti:
  assumes "t ≤⇩b t'" and "monofun key" and keys: "⋀x. key x ∈ Keys" and "k ∈ Live"
  shows "undecided_mass key t' k ≤ undecided_mass key t k"
  unfolding undecided_mass_def
proof (rule bag_le_mass[OF assms(1)])
  fix x x'
  assume "x ⊑ x'" and x': "key x' ⊑ k ∧ key x' ≠ k"
  have "key x ⊑ key x'"
    using assms(2) ‹x ⊑ x'› by (rule monofunE)
  show "key x ⊑ k ∧ key x ≠ k"
  proof
    show "key x ⊑ k"
      using ‹key x ⊑ key x'› x' by (blast intro: below_trans)
    show "key x ≠ k"
      using ‹key x ⊑ key x'› x' live_maximal[OF assms(4)] keys by blast
  qed
qed

lemma reachable_mass_anti:
  assumes "t ≤⇩b t'" and "monofun key"
  shows "reachable_mass key t' k ≤ reachable_mass key t k"
  unfolding reachable_mass_def
  by (rule SUP_subset_mono[OF order_refl], rule bag_le_mass[OF assms(1)])
    (metis assms(2) monofunE below_trans)

lemma orphan_count_anti:
  assumes "t ≤⇩b t'" and "monofun key" and "⋀x. key x ∈ Keys"
  shows "orphan_count w key t' k ≤ orphan_count w key t k"
proof (cases w)
  case Open
  then show ?thesis
    by simp
next
  case Closed
  show ?thesis
  proof (cases "decided_mass key t k = 0")
    case nz: False
    then show ?thesis
      using Closed undecided_mass_anti[OF assms decided_mass_live[OF nz]] decided_mass_up[OF assms nz]
      by simp
  next
    case True
    show ?thesis
    proof (cases "decided_mass key t' k = 0")
      case True
      have "max 1 (reachable_mass key t' k) ≤ max 1 (reachable_mass key t k)"
        using reachable_mass_anti[OF assms(1,2)] by (rule max.mono[OF order_refl])
      then show ?thesis
        using Closed True ‹decided_mass key t k = 0› by simp
    next
      case nz: False
      then have live: "k ∈ Live"
        by (rule decided_mass_live)
      have "undecided_mass key t' k ≤ undecided_mass key t k"
        by (rule undecided_mass_anti[OF assms live])
      also have "… ≤ bag_mass t (λx. key x ⊑ k)"
        using decided_undecided_mass[OF live, of key t] by (metis le_iff_add add.commute)
      also have "… ≤ reachable_mass key t k"
        by (rule reachable_mass_upper[OF live below_refl])
      also have "… ≤ max 1 (reachable_mass key t k)"
        by simp
      finally show ?thesis
        using Closed nz True by simp
    qed
  qed
qed

subsection ‹The join on the basis›

primrec join_mult ::
    "join_side ⇒ join_side ⇒ ('a compact_basis ⇒ 'k) ⇒ ('b compact_basis ⇒ 'k) ⇒
      'a bag_basis ⇒ 'b bag_basis ⇒ ('a, 'b) join_index ⇒ enat"
  where
    "join_mult wl wr lkey rkey t u (Joined x y) =
      (if decided_match (lkey x) (rkey y) then bmult t x * bmult u y else 0)"
  | "join_mult wl wr lkey rkey t u (Left_Orphan x) = bmult t x * orphan_count wl rkey u (lkey x)"
  | "join_mult wl wr lkey rkey t u (Right_Orphan y) = bmult u y * orphan_count wr lkey t (rkey y)"

lemma join_mult_nonzero: "∃i∈join_indices t u. join_mult wl wr lkey rkey t u i ≠ 0"
proof -
  obtain x where x: "x ∈ bsupp t"
    using bsupp_nonempty by blast
  show ?thesis
  proof (cases "orphan_count wl rkey u (lkey x) = 0")
    case True
    then obtain y where "y ∈ bsupp u" "decided_match (lkey x) (rkey y)"
      by (auto dest!: orphan_count_0 simp: decided_mass_def bag_mass_nonzero_iff)
    then show ?thesis
      using x by (intro bexI[of _ "Joined x y"]) (simp_all add: bsupp_iff)
  next
    case False
    then show ?thesis
      using x by (intro bexI[of _ "Left_Orphan x"]) (simp_all add: bsupp_iff)
  qed
qed

definition BJoin ::
    "('a compact_basis ⇒ 'b compact_basis ⇒ 'c compact_basis) ⇒
      join_side ⇒ join_side ⇒ ('a compact_basis ⇒ 'k) ⇒ ('b compact_basis ⇒ 'k) ⇒
      'a bag_basis ⇒ 'b bag_basis ⇒ 'c bag_basis"
  where "BJoin c wl wr lkey rkey t u =
    BFamily (join_indices t u) (join_label c) (join_mult wl wr lkey rkey t u)"

text ‹The join is symmetric up to swapping the sides.›

lemma BJoin_swap: "BJoin c wl wr lkey rkey t u = BJoin (λy x. c x y) wr wl rkey lkey u t"
proof -
  have bij: "bij_betw join_swap (join_indices t u) (join_indices u t)"
    by (rule bij_betw_byWitness[where f' = join_swap]) (auto intro: join_swap_indices)
  have "join_label (λy x. c x y) (join_swap i) = join_label c i" for i
    by (cases i) simp_all
  moreover have "join_mult wr wl rkey lkey u t (join_swap i) = join_mult wl wr lkey rkey t u i" for i
    by (cases i) (simp_all add: decided_match_sym mult.commute)
  ultimately show ?thesis
    unfolding BJoin_def
    using BFamily_reindex[OF bij, of "join_label (λy x. c x y)" "join_mult wr wl rkey lkey u t"]
    by simp
qed

subsection ‹Monotonicity›

text ‹A plan @{text P} from @{text t} to @{text t'} lifts to the joins with @{text u}. A pair
  follows its left row. A left orphan without a decided match sends copies to the pairs that its
  refined row now matches, and the rest to the refined orphan. Right orphans stay where they
  are.›

context
  fixes P :: "'a compact_basis ⇒ 'a compact_basis ⇒ enat"
    and wl wr :: join_side
    and lkey :: "'a compact_basis ⇒ 'k" and rkey :: "'b compact_basis ⇒ 'k"
    and t' :: "'a bag_basis" and u :: "'b bag_basis"
begin

fun lift_plan :: "('a, 'b) join_index ⇒ ('a, 'b) join_index ⇒ enat"
  where
    "lift_plan (Joined x y) (Joined x' y') =
      (if y' = y then if decided_match (lkey x) (rkey y) then P x x' * bmult u y else 0 else 0)"
  | "lift_plan (Left_Orphan x) (Joined x' y) =
      (if decided_mass rkey u (lkey x) = 0 ∧ decided_match (lkey x') (rkey y)
       then P x x' * bmult u y else 0)"
  | "lift_plan (Left_Orphan x) (Left_Orphan x') =
      P x x' * orphan_count wl rkey u (lkey x')"
  | "lift_plan (Right_Orphan y) (Right_Orphan y') =
      (if y' = y then bmult u y * orphan_count wr lkey t' (rkey y) else 0)"
  | "lift_plan i j = 0"

end

lemma BJoin_mono_left:
  assumes tt: "t ≤⇩b t'" and lkey: "monofun lkey" and keys: "⋀x. lkey x ∈ Keys"
    and c: "⋀x x' y y'. x ⊑ x' ⟹ y ⊑ y' ⟹ c x y ⊑ c x' y'"
  shows "BJoin c wl wr lkey rkey t u ≤⇩b BJoin c wl wr lkey rkey t' u"
proof -
  obtain P where P: "bag_plan t t' P"
    using tt unfolding bag_le_def by blast
  have Pkey: "lkey x ⊑ lkey x'" if "P x x' ≠ 0" for x x'
    using lkey bag_plan_below[OF P that] by (rule monofunE)
  ― ‹a decided key does not move›
  have Pfix: "lkey x' = lkey x" if "P x x' ≠ 0" "decided_mass rkey u (lkey x) ≠ 0" for x x'
    using live_maximal decided_mass_live[OF that(2)] Pkey[OF that(1)] keys by blast
  let ?m = "join_mult wl wr lkey rkey t u" and ?n = "join_mult wl wr lkey rkey t' u"
  let ?p = "lift_plan P wl wr lkey rkey t' u"
  have sum_decided_mass:
    "(∑y∈bsupp u. if decided_match k (rkey y) then e * bmult u y else 0) = e * decided_mass rkey u k"
    for k e
    by (simp add: sum.inter_filter[symmetric] decided_mass_def bag_mass_def sum_distrib_left if_distrib)
  show ?thesis
    unfolding BJoin_def
  proof (rule BFamily_le[where p = ?p])
    show "∃i∈join_indices t u. ?m i ≠ 0"
      by (rule join_mult_nonzero)
  next
    fix i j
    assume "i ∈ join_indices t u" "j ∈ join_indices t' u" "?p i j ≠ 0"
    then show "join_label c i ⊑ join_label c j"
      by (cases i; cases j) (auto split: if_split_asm intro!: c dest: bag_plan_below[OF P])
  next
    fix i
    assume i: "i ∈ join_indices t u"
    show "sum (?p i) (join_indices t' u) ≤ ?m i"
    proof (cases i)
      case (Joined x y)
      then have y: "y ∈ bsupp u"
        using i by simp
      have "sum (?p i) (join_indices t' u) =
          (if decided_match (lkey x) (rkey y) then sum (P x) (bsupp t') * bmult u y else 0)"
        using y by (simp add: Joined sum_join_indices sum_distrib_right)
      then show ?thesis
        by (simp add: Joined bag_plan_row[OF P])
    next
      case (Left_Orphan x)
      have bound: "(∑y∈bsupp u. ?p i (Joined x' y)) + ?p i (Left_Orphan x')
          ≤ P x x' * orphan_count wl rkey u (lkey x)" for x'
      proof (cases "P x x' = 0")
        case False
        show ?thesis
        proof (cases "decided_mass rkey u (lkey x) = 0")
          case True
          have "P x x' * decided_mass rkey u (lkey x') + P x x' * orphan_count wl rkey u (lkey x')
              ≤ P x x' * orphan_count wl rkey u (lkey x)"
            using orphan_count_step[OF Pkey[OF False] True]
            by (simp add: distrib_left[symmetric] mult_left_mono)
          then show ?thesis
            using True by (simp add: Left_Orphan sum_decided_mass mult.commute)
        next
          case nz: False
          then show ?thesis
            using Pfix[OF False nz] by (simp add: Left_Orphan)
        qed
      qed (simp add: Left_Orphan)
      have "sum (?p i) (join_indices t' u) =
          (∑x'∈bsupp t'. (∑y∈bsupp u. ?p i (Joined x' y)) + ?p i (Left_Orphan x'))"
        by (simp add: Left_Orphan sum_join_indices sum.distrib)
      also have "… ≤ (∑x'∈bsupp t'. P x x' * orphan_count wl rkey u (lkey x))"
        by (intro sum_mono bound)
      also have "… = ?m i"
        by (simp add: Left_Orphan sum_distrib_right[symmetric] bag_plan_row[OF P])
      finally show ?thesis .
    next
      case (Right_Orphan y)
      then have y: "y ∈ bsupp u"
        using i by simp
      have "sum (?p i) (join_indices t' u) = bmult u y * orphan_count wr lkey t' (rkey y)"
        using y by (simp add: Right_Orphan sum_join_indices)
      also have "… ≤ ?m i"
        using orphan_count_anti[OF tt lkey keys] by (simp add: Right_Orphan mult_left_mono)
      finally show ?thesis .
    qed
  next
    fix j
    assume j: "j ∈ join_indices t' u"
    show "?n j ≤ (∑i∈join_indices t u. ?p i j)"
    proof (cases j)
      case (Joined x' y)
      then have y: "y ∈ bsupp u"
        using j by simp
      show ?thesis
      proof (cases "decided_match (lkey x') (rkey y)")
        case True
        have cover: "P x x' * bmult u y ≤ (∑y'∈bsupp u. ?p (Joined x y') j) + ?p (Left_Orphan x) j"
          for x
        proof (cases "P x x' = 0 ∨ decided_mass rkey u (lkey x) = 0")
          case False
          then have "decided_match (lkey x) (rkey y)"
            using True Pfix by auto
          then show ?thesis
            using y by (simp add: Joined)
        qed (auto simp: Joined True)
        have "?n j ≤ (∑x∈bsupp t. P x x') * bmult u y"
          using True bag_plan_col[OF P, of x'] by (simp add: Joined mult_right_mono)
        also have "… ≤ (∑x∈bsupp t. (∑y'∈bsupp u. ?p (Joined x y') j) + ?p (Left_Orphan x) j)"
          unfolding sum_distrib_right by (intro sum_mono cover)
        also have "… = (∑i∈join_indices t u. ?p i j)"
          by (simp add: Joined sum_join_indices sum.distrib)
        finally show ?thesis .
      qed (simp add: Joined)
    next
      case (Left_Orphan x')
      have "?n j ≤ (∑x∈bsupp t. P x x') * orphan_count wl rkey u (lkey x')"
        using bag_plan_col[OF P, of x'] by (simp add: Left_Orphan mult_right_mono)
      also have "… = (∑i∈join_indices t u. ?p i j)"
        by (simp add: Left_Orphan sum_join_indices sum_distrib_right)
      finally show ?thesis .
    next
      case (Right_Orphan y)
      then have "y ∈ bsupp u"
        using j by simp
      then show ?thesis
        by (simp add: Right_Orphan sum_join_indices)
    qed
  next
    fix i
    assume i: "i ∈ join_indices t u" and "?m i ≠ 0"
    show "∃j∈join_indices t' u. ?n j ≠ 0 ∧ join_label c i ⊑ join_label c j"
    proof (cases i)
      case (Joined x y)
      with i ‹?m i ≠ 0› have x: "x ∈ bsupp t" and y: "y ∈ bsupp u"
        and d: "decided_match (lkey x) (rkey y)"
        by (simp_all split: if_split_asm)
      obtain x' where x': "x' ∈ bsupp t'" "P x x' ≠ 0"
        using bag_plan_row_ex[OF P x] .
      have "decided_mass rkey u (lkey x) ≠ 0"
        using y d by (auto simp: decided_mass_def bag_mass_nonzero_iff)
      then have "decided_match (lkey x') (rkey y)"
        using d Pfix[OF x'(2)] by simp
      then show ?thesis
        using x' y bag_plan_below[OF P x'(2)]
        by (intro bexI[of _ "Joined x' y"]) (auto simp: Joined bsupp_iff intro: c)
    next
      case (Left_Orphan x)
      with i have x: "x ∈ bsupp t"
        by simp
      obtain x' where x': "x' ∈ bsupp t'" "P x x' ≠ 0"
        using bag_plan_row_ex[OF P x] .
      note xx' = bag_plan_below[OF P x'(2)]
      show ?thesis
      proof (cases "orphan_count wl rkey u (lkey x') = 0")
        case True
        then obtain y where "y ∈ bsupp u" "decided_match (lkey x') (rkey y)"
          by (auto dest!: orphan_count_0 simp: decided_mass_def bag_mass_nonzero_iff)
        then show ?thesis
          using x' xx' by (intro bexI[of _ "Joined x' y"]) (auto simp: Left_Orphan bsupp_iff intro: c)
      next
        case False
        then show ?thesis
          using x' xx' by (intro bexI[of _ "Left_Orphan x'"]) (auto simp: Left_Orphan bsupp_iff intro: c)
      qed
    next
      case (Right_Orphan y)
      with i have y: "y ∈ bsupp u"
        by simp
      show ?thesis
      proof (cases "orphan_count wr lkey t' (rkey y) = 0")
        case True
        then obtain x' where "x' ∈ bsupp t'" "decided_match (rkey y) (lkey x')"
          by (auto dest!: orphan_count_0 simp: decided_mass_def bag_mass_nonzero_iff)
        then show ?thesis
          using y
          by (intro bexI[of _ "Joined x' y"]) (auto simp: Right_Orphan bsupp_iff decided_match_sym intro: c)
      next
        case False
        then show ?thesis
          using y by (intro bexI[of _ "Right_Orphan y"]) (auto simp: Right_Orphan bsupp_iff)
      qed
    qed
  qed simp_all
qed

lemma BJoin_mono_right:
  assumes "u ≤⇩b u'" and "monofun rkey" and "⋀y. rkey y ∈ Keys"
    and c: "⋀x x' y y'. x ⊑ x' ⟹ y ⊑ y' ⟹ c x y ⊑ c x' y'"
  shows "BJoin c wl wr lkey rkey t u ≤⇩b BJoin c wl wr lkey rkey t u'"
  unfolding BJoin_swap[of c wl wr lkey rkey t] by (rule BJoin_mono_left[OF assms(1-3)]) (rule c)

subsection ‹The join of bags into pairs›

definition bag_pair_join ::
    "join_side ⇒ join_side ⇒ ('a → 'k) ⇒ ('b → 'k) ⇒ 'a bag_pd → 'b bag_pd → ('a × 'b) bag_pd"
  where "bag_pair_join wl wr lkey rkey = bag_pd.extension (λt. bag_pd.extension (λu.
    bag_principal (BJoin pair_basis wl wr (on_basis lkey) (on_basis rkey) t u)))"

lemma bag_pair_join_principal:
  assumes "⋀x. lkey⋅x ∈ Keys" and "⋀y. rkey⋅y ∈ Keys"
  shows "bag_pair_join wl wr lkey rkey⋅(bag_principal t)⋅(bag_principal u) =
    bag_principal (BJoin pair_basis wl wr (on_basis lkey) (on_basis rkey) t u)"
  unfolding bag_pair_join_def
  by (simp add: bag_pd.extension_principal bag_pd.extension_mono BJoin_mono_left BJoin_mono_right
      monofun_on_basis pair_basis_mono assms)

text ‹Sanity check, an N:1 join of two rows with the same decided key: the referencing row is
  consumed by the match, while the referenced row keeps an open-ended number of orphan copies.›

lemma bag_pair_join_unit_match:
  fixes lkey :: "'a → 'k" and rkey :: "'b → 'k"
  assumes a: "Cpo.compact a" and b: "Cpo.compact b" and live: "lkey⋅a ∈ Live"
    and k: "rkey⋅b = lkey⋅a" and keys: "⋀x. lkey⋅x ∈ Keys" "⋀y. rkey⋅y ∈ Keys"
  shows "bag_pair_join Closed Open lkey rkey⋅(bag_unit⋅a)⋅(bag_unit⋅b) =
    bag_unit⋅(a, b) ∪⇩b bag_omega⋅(bag_unit⋅(⊥, b))"
proof -
  define a' b' where "a' = Abs_compact_basis a" and "b' = Abs_compact_basis b"
  have ra: "Rep_compact_basis a' = a" and rb: "Rep_compact_basis b' = b"
    using a b by (simp_all add: a'_def b'_def)
  have d: "decided_match (on_basis lkey a') (on_basis rkey b')"
    using live k by (simp add: ra rb decided_match_def)
  have l: "orphan_count Closed (on_basis rkey) (BUnit b') (on_basis lkey a') = 0"
    using d by (simp add: decided_mass_def undecided_mass_def bag_mass_BUnit ra rb k)
  have "BJoin pair_basis Closed Open (on_basis lkey) (on_basis rkey) (BUnit a') (BUnit b') =
      BPlus (BUnit (pair_basis a' b')) (BOmega (BUnit (pair_basis compact_bot b')))"
    by (rule bag_basis_eqI)
      (simp add: BJoin_def bmult_BFamily_sum join_mult_nonzero sum_join_indices d l bmult_BUnit
        bmult_BOmega eq_commute[of _ "pair_basis _ _"] del: orphan_count.simps(2))
  moreover have "bag_unit⋅a = bag_principal (BUnit a')" "bag_unit⋅b = bag_principal (BUnit b')"
    "bag_unit⋅(a, b) = bag_principal (BUnit (pair_basis a' b'))"
    "bag_unit⋅(⊥::'a, b) = bag_principal (BUnit (pair_basis compact_bot b'))"
    using bag_unit_Rep_compact_basis [of a'] bag_unit_Rep_compact_basis [of b']
      bag_unit_Rep_compact_basis [of "pair_basis a' b'"]
      bag_unit_Rep_compact_basis [of "pair_basis (compact_bot::'a compact_basis) b'"]
    by (simp_all add: ra rb)
  ultimately show ?thesis
    by (simp only: bag_pair_join_principal[OF keys] bag_omega_principal bag_plus_principal)
qed

end

subsection ‹Joins on one row domain›

text ‹Both sides live in one row domain @{typ 'r}, in disjoint blocks @{text A} and @{text B}. The
  left key columns @{text S} and the right key columns @{text T} are different columns, related by
  the transports @{text φ} and @{text ψ} (the column correspondence of the join); the live keys are
  the maximal points of @{text T}. A joined pair is the join of its two blocks, which exists because
  the blocks are disjoint, so no compatibility condition on pairs is needed.›

text ‹The join of two continuous functions whose values are always joinable is continuous.›

lemma cont_join_joinable:
  assumes f: "cont f" and g: "cont g" and j: "⋀x. f x ⇑ g x"
  shows "cont (λx. f x ⊔ g x)"
proof (rule contI2)
  show "monofun (λx. f x ⊔ g x)"
  proof (rule monofunI)
    fix x y :: 'a
    assume "x ⊑ y"
    then have "f x ⊑ f y" "g x ⊑ g y"
      using f g by (simp_all add: cont2monofunE)
    then show "f x ⊔ g x ⊑ f y ⊔ g y"
      using joinableD[OF j[of y]] by (blast intro: join_belowI below_trans)
  qed
  fix Y :: "nat ⇒ 'a"
  assume Y: "chain Y" and ch: "chain (λi. f (Y i) ⊔ g (Y i))"
  have "f (⨆i. Y i) ⊑ (⨆i. f (Y i) ⊔ g (Y i))"
    unfolding cont2contlubE[OF f Y] by (rule lub_mono[OF ch2ch_cont[OF f Y] ch joinableD(1)[OF j]])
  moreover have "g (⨆i. Y i) ⊑ (⨆i. f (Y i) ⊔ g (Y i))"
    unfolding cont2contlubE[OF g Y] by (rule lub_mono[OF ch2ch_cont[OF g Y] ch joinableD(2)[OF j]])
  ultimately show "f (⨆i. Y i) ⊔ g (⨆i. Y i) ⊑ (⨆i. f (Y i) ⊔ g (Y i))"
    by (rule join_belowI)
qed

locale bag_join_columns =
  fixes A B S T :: "'r::{bifinite, distributive_cpo} semi_factor" and φ ψ :: "'r → 'r"
  assumes disjoint: "A ⊓ B = ⊥"
    and ψφ: "ψ oo φ = proj S" and φψ: "φ oo ψ = proj T"
    and φ_T: "proj T oo φ = φ" and ψ_S: "proj S oo ψ = ψ"
begin

lemma ψφ_app [simp]: "ψ⋅(φ⋅x) = proj S⋅x"
  using ψφ by (simp add: cfun_eq_iff)

lemma φψ_app [simp]: "φ⋅(ψ⋅x) = proj T⋅x"
  using φψ by (simp add: cfun_eq_iff)

lemma φ_T_app [simp]: "proj T⋅(φ⋅x) = φ⋅x"
  using φ_T by (simp add: cfun_eq_iff)

lemma ψ_S_app [simp]: "proj S⋅(ψ⋅x) = ψ⋅x"
  using ψ_S by (simp add: cfun_eq_iff)

lemma φ_S_app [simp]: "φ⋅(proj S⋅x) = φ⋅x"
  by (metis φ_T_app φψ_app ψφ_app)

lemma ψ_T_app [simp]: "ψ⋅(proj T⋅x) = ψ⋅x"
  by (metis ψ_S_app φψ_app ψφ_app)

text ‹A decided match reads the same from either side.›

lemma match_transport:
  "(maximal_at T (φ⋅x) ∧ proj T⋅y = φ⋅x) ⟷ (maximal_at S (proj S⋅x) ∧ proj S⋅x = ψ⋅y)"
proof
  assume m: "maximal_at T (φ⋅x) ∧ proj T⋅y = φ⋅x"
  have "proj S⋅x = ψ⋅y"
    by (metis m ψφ_app ψ_T_app)
  moreover have "maximal_at S (proj S⋅x)"
    unfolding maximal_at_def
  proof (intro conjI allI impI)
    fix t
    assume t: "proj S⋅t = t" and le: "proj S⋅x ⊑ t"
    have "φ⋅t = φ⋅x"
      using maximal_at_eq[OF m[THEN conjunct1] φ_T_app] monofun_cfun_arg[OF le, of φ] by simp
    then show "t = proj S⋅x"
      by (metis t ψφ_app)
  qed simp
  ultimately show "maximal_at S (proj S⋅x) ∧ proj S⋅x = ψ⋅y"
    by blast
next
  assume m: "maximal_at S (proj S⋅x) ∧ proj S⋅x = ψ⋅y"
  have "proj T⋅y = φ⋅x"
    by (metis m φψ_app φ_S_app)
  moreover have "maximal_at T (φ⋅x)"
    unfolding maximal_at_def
  proof (intro conjI allI impI)
    fix t
    assume t: "proj T⋅t = t" and le: "φ⋅x ⊑ t"
    have "ψ⋅t = proj S⋅x"
      using maximal_at_eq[OF m[THEN conjunct1] ψ_S_app] monofun_cfun_arg[OF le, of ψ] by simp
    then show "t = φ⋅x"
      by (metis t φψ_app φ_S_app)
  qed simp
  ultimately show "maximal_at T (φ⋅x) ∧ proj T⋅y = φ⋅x"
    by blast
qed

lemma blocks_joinable: "proj A⋅x ⇑ proj B⋅y"
proof (rule proj_slide)
  show "proj B⋅(proj B⋅y) = proj B⋅y"
    by simp
  have "proj B⋅(proj A⋅x) = ⊥"
    using proj_inf[of B A x] disjoint by (simp add: lattice_meet_commute proj_bot)
  then show "proj B⋅(proj A⋅x) ⊑ proj B⋅y"
    by simp
qed

definition block_join :: "'r × 'r → 'r"
  where "block_join = (Λ(x, y). proj A⋅x ⊔ proj B⋅y)"

lemma block_join_beta [simp]: "block_join⋅(x, y) = proj A⋅x ⊔ proj B⋅y"
  unfolding block_join_def by (simp add: split_beta' cont_join_joinable blocks_joinable)

sublocale keys: bag_join_keys "{k. maximal_at T k}" "{k. proj T⋅k = k}"
  by unfold_locales (simp add: maximal_at_eq)

definition bag_join :: "join_side ⇒ join_side ⇒ 'r bag_pd → 'r bag_pd → 'r bag_pd"
  where "bag_join wl wr = (Λ X Y. bag_map⋅block_join⋅(keys.bag_pair_join wl wr φ (proj T)⋅X⋅Y))"

lemma bag_join_eq: "bag_join wl wr⋅X⋅Y = bag_map⋅block_join⋅(keys.bag_pair_join wl wr φ (proj T)⋅X⋅Y)"
  by (simp add: bag_join_def)

text ‹Sanity check: an N:1 join of an @{text A}-row and a @{text B}-row with matching decided keys
  gives the joined row once and the referenced row open-endedly often.›

lemma bag_join_unit_match:
  assumes a: "Cpo.compact a" "proj A⋅a = a" and b: "Cpo.compact b" "proj B⋅b = b"
    and m: "maximal_at T (φ⋅a)" "proj T⋅b = φ⋅a"
  shows "bag_join Closed Open⋅(bag_unit⋅a)⋅(bag_unit⋅b) =
    bag_unit⋅(a ⊔ b) ∪⇩b bag_omega⋅(bag_unit⋅b)"
proof -
  have "keys.bag_pair_join Closed Open φ (proj T)⋅(bag_unit⋅a)⋅(bag_unit⋅b) =
      bag_unit⋅(a, b) ∪⇩b bag_omega⋅(bag_unit⋅(⊥, b))"
    using a b m by (intro keys.bag_pair_join_unit_match) simp_all
  moreover have "block_join⋅(a, b) = a ⊔ b" "block_join⋅(⊥, b) = b"
    using a b deflation.below[OF proj_defl, of A ⊥] by (simp_all add: join_absorbI)
  ultimately show ?thesis
    by (simp add: bag_join_eq)
qed
