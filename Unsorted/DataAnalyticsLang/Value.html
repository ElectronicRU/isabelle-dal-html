Theory Value

section ‹Values›

text ‹Bottom is a data-quality lapse or divergence: a dangling foreign key, a failed cast, a
  short row.  Missing values that are not lapses live at three levels.
  ▪ ∗‹Data› (records and record-level expressions), @{text "'v datum"}: ‹NA› (not
    applicable; joins through it give ‹NA›), ‹Val⋅Null› (present but unknown; joins
    through it give ‹⊥›), or a known value.  Source nulls map to one of the two per column.
  ▪ ∗‹Keys› (dimensions), @{text "'v with_na"}: nulls become ‹NA›, which can be pinned
    (@{text to_key}).
  ▪ ∗‹Measures›, @{text "'v with_null"}: ‹NA› becomes ‹⊥›, i.e.\ no entry
    (@{text from_na}); aggregations treat ‹⊥› and ‹NA› alike.
  A record-level function is ∗‹NA-blind› when it commutes with @{text squash}, the
  projection forgetting ‹NA›; lifted operators are, coalescing is not.›

text ‹A datum is ‹NA›, ‹Val⋅Null›, or a known value.›

section ‹Coercions between the levels›

text ‹Binding through ‹NA›: ‹NA› propagates, a value is passed on.  A join through a
  relation that may be inapplicable is a bind.›

fixrec na_bind :: "('a::domain → 'b::domain with_na) → 'a with_na → 'b with_na" where
  "na_bind⋅f⋅NA = NA"
| "na_bind⋅f⋅(Val⋅x) = f⋅x"

lemma na_bind_strict [simp]: "na_bind⋅f⋅⊥ = ⊥"
  by fixrec_simp

text ‹Measures: ‹NA› becomes ‹⊥› (no entry).›

fixrec from_na :: "'a::domain with_na → 'a" where
  "from_na⋅NA = ⊥"
| "from_na⋅(Val⋅x) = x"

lemma from_na_strict [simp]: "from_na⋅⊥ = ⊥"
  by fixrec_simp

text ‹Measures embed back strictly: @{text na_emb} and @{const from_na} form an
  embedding-projection pair.  Their composition @{text squash} is the projection that forgets
  ‹NA› (and the not-null root, which data never takes).›

definition na_emb :: "'a::domain → 'a with_na" where
  "na_emb = strictify⋅Val"

lemma na_emb_strict [simp]: "na_emb⋅⊥ = ⊥"
  by (simp add: na_emb_def)

lemma na_emb_defined [simp]: "x ≠ ⊥ ⟹ na_emb⋅x = Val⋅x"
  by (simp add: na_emb_def strictify_conv_if)

lemma ep_pair_na: "ep_pair na_emb from_na"
proof
  fix x :: 'a show "from_na⋅(na_emb⋅x) = x" by (cases "x = ⊥") simp_all
next
  fix y :: "'a with_na" show "na_emb⋅(from_na⋅y) ⊑ y"
    by (cases y) (simp_all add: na_emb_def strictify_conv_if)
qed

lemma from_na_na_emb [simp]: "from_na⋅(na_emb⋅x) = x"
  by (rule ep_pair.e_inverse[OF ep_pair_na])

definition squash :: "'a::domain with_na → 'a with_na" where
  "squash = na_emb oo from_na"

lemma squash_simps [simp]: "squash⋅⊥ = ⊥" "squash⋅NA = ⊥" "squash⋅(Val⋅x) = na_emb⋅x"
  by (simp_all add: squash_def)

lemma squash_na_emb [simp]: "squash⋅(na_emb⋅x) = na_emb⋅x"
  unfolding squash_def by (simp add: ep_pair.e_inverse[OF ep_pair_na])

lemma deflation_squash: "deflation squash"
  unfolding squash_def by (rule ep_pair.deflation_e_p[OF ep_pair_na])

lemma from_na_squash [simp]: "from_na⋅(squash⋅x) = from_na⋅x"
  unfolding squash_def by (simp add: ep_pair.e_inverse[OF ep_pair_na])

  by fixrec_simp

definition to_key :: "'v::domain datum → 'v with_na" where
  "to_key = na_bind⋅null_na"

text ‹Measures project to plain values, ‹Null› becoming ‹⊥›.  @{const Known} is strict,
  so it is the embedding itself.›

fixrec from_null :: "'v::domain with_null → 'v" where
  "from_null⋅Null = ⊥"
| "v ≠ ⊥ ⟹ from_null⋅(Known⋅v) = v"

lemma from_null_strict [simp]: "from_null⋅⊥ = ⊥"
  by fixrec_simp

lemma ep_pair_null: "ep_pair (Known :: 'v::domain → 'v with_null) from_null"
proof
  fix x :: 'v show "from_null⋅(Known⋅x) = x" by (cases "x = ⊥") simp_all
next
  fix y :: "'v with_null" show "Known⋅(from_null⋅y) ⊑ y" by (cases y) simp_all
qed

lemma from_null_Known [simp]: "from_null⋅(Known⋅x) = x"
  by (rule ep_pair.e_inverse[OF ep_pair_null])

text ‹The plain value of a datum: every missing value becomes ‹⊥›.›

definition plain :: "'v::domain datum → 'v" where
  "plain = from_null oo from_na"

lemma plain_simps [simp]:
  "plain⋅⊥ = ⊥" "plain⋅NA = ⊥" "plain⋅(Val⋅x) = from_null⋅x"
  by (simp_all add: plain_def)

lemma plain_na_emb [simp]: "plain⋅(na_emb⋅x) = from_null⋅x"
  by (simp add: plain_def)

section ‹Record-level operators›

text ‹Operators are lifted layer by layer, each layer's missing value infectious.  The outer
  layer is checked first, so ‹NA› wins over ‹Null›; NA-blindness forces that choice.›

fixrec na_lift2 :: "('a::domain → 'b::domain → 'c::domain) → 'a with_na → 'b with_na → 'c with_na"
  where
  "na_lift2⋅f⋅NA⋅NA = NA"
| "na_lift2⋅f⋅NA⋅(Val⋅y) = NA"
| "na_lift2⋅f⋅(Val⋅x)⋅NA = NA"
| "na_lift2⋅f⋅(Val⋅x)⋅(Val⋅y) = na_emb⋅(f⋅x⋅y)"

fixrec null_lift2 ::
  "('a::domain → 'b::domain → 'c::domain) → 'a with_null → 'b with_null → 'c with_null"
  where
  "null_lift2⋅f⋅Null⋅Null = Null"
| "y ≠ ⊥ ⟹ null_lift2⋅f⋅Null⋅(Known⋅y) = Null"
| "x ≠ ⊥ ⟹ null_lift2⋅f⋅(Known⋅x)⋅Null = Null"
| "x ≠ ⊥ ⟹ y ≠ ⊥ ⟹ null_lift2⋅f⋅(Known⋅x)⋅(Known⋅y) = Known⋅(f⋅x⋅y)"

fixrec na_lift1 :: "('a::domain → 'b::domain) → 'a with_na → 'b with_na" where
  "na_lift1⋅f⋅NA = NA"
| "na_lift1⋅f⋅(Val⋅x) = na_emb⋅(f⋅x)"

fixrec null_lift1 :: "('a::domain → 'b::domain) → 'a with_null → 'b with_null" where
  "null_lift1⋅f⋅Null = Null"
| "x ≠ ⊥ ⟹ null_lift1⋅f⋅(Known⋅x) = Known⋅(f⋅x)"

abbreviation dlift1 :: "('a::domain → 'b::domain) → 'a datum → 'b datum" where
  "dlift1 ≡ na_lift1 oo null_lift1"

lemma na_lift1_strict [simp]: "na_lift1⋅f⋅⊥ = ⊥"
  by fixrec_simp

lemma null_lift1_strict [simp]: "null_lift1⋅f⋅⊥ = ⊥"
  by fixrec_simp

abbreviation dlift2 :: "('a::domain → 'b::domain → 'c::domain) → 'a datum → 'b datum → 'c datum" where
  "dlift2 ≡ na_lift2 oo null_lift2"

lemma na_lift2_strict [simp]: "na_lift2⋅f⋅⊥⋅y = ⊥" "na_lift2⋅f⋅x⋅⊥ = ⊥"
  by fixrec_simp (cases x; fixrec_simp)

lemma null_lift2_strict [simp]: "null_lift2⋅f⋅⊥⋅y = ⊥" "null_lift2⋅f⋅x⋅⊥ = ⊥"
  by fixrec_simp (cases x; fixrec_simp)

text ‹A record-level function is NA-blind when swapping ‹NA› for ‹⊥› in its inputs does
  the same to its output.›

definition na_blind1 :: "('a::domain with_na → 'b::domain with_na) ⇒ bool" where
  "na_blind1 g ⟷ (∀x. g⋅(squash⋅x) = squash⋅(g⋅x))"

definition na_blind2 ::
  "('a::domain with_na → 'b::domain with_na → 'c::domain with_na) ⇒ bool" where
  "na_blind2 g ⟷ (∀x y. g⋅(squash⋅x)⋅(squash⋅y) = squash⋅(g⋅x⋅y))"

lemma na_blind2_na_lift2:
  assumes l: "⋀y. f⋅⊥⋅y = ⊥" and r: "⋀x. f⋅x⋅⊥ = ⊥"
  shows "na_blind2 (na_lift2⋅f)"
  unfolding na_blind2_def
proof (intro allI)
  fix x y show "na_lift2⋅f⋅(squash⋅x)⋅(squash⋅y) = squash⋅(na_lift2⋅f⋅x⋅y)"
    by (cases x; cases y) (auto simp: l r strictify_conv_if na_emb_def)
qed

text ‹The inner layer is strict, so every lifted record-level operator is NA-blind.›

lemma na_blind2_dlift2: "na_blind2 (dlift2⋅f)"
  by (simp add: na_blind2_na_lift2)

text ‹On plain values, a lifted strict operator is the operator itself.›

lemma plain_dlift1:
  assumes "f⋅⊥ = ⊥" shows "plain⋅(na_lift1⋅(null_lift1⋅f)⋅x) = f⋅(plain⋅x)"
proof (cases x)
  case (Val a) then show ?thesis using assms by (cases a) simp_all
qed (simp_all add: assms)

lemma plain_dlift2:
  assumes "⋀y. f⋅⊥⋅y = ⊥" "⋀x. f⋅x⋅⊥ = ⊥"
  shows "plain⋅(na_lift2⋅(null_lift2⋅f)⋅x⋅y) = f⋅(plain⋅x)⋅(plain⋅y)"
proof (cases x)
  case x: (Val a) show ?thesis
  proof (cases y)
    case y: (Val b) show ?thesis unfolding x y using assms by (cases a; cases b) simp_all
  qed (simp_all add: x assms)
qed (cases y; simp add: assms)+

subsection ‹Equality›

text ‹Equality of flat values, strict in both arguments.  @{typ tr} is @{typ "bool lift"}, so
  this covers both fields of @{text scalar}.›

definition lift_eq :: "'a lift → 'a lift → tr" where
  "lift_eq = (FLIFT a. FLIFT b. Def (a = b))"

lemma lift_eq_Def [simp]: "lift_eq⋅(Def a)⋅(Def b) = Def (a = b)"
  by (simp add: lift_eq_def)

lemma lift_eq_strict [simp]: "lift_eq⋅⊥⋅y = ⊥" "lift_eq⋅x⋅⊥ = ⊥"
  by (simp_all add: lift_eq_def) (cases x; simp)

lemma lift_eq_defined: "x ≠ ⊥ ⟹ y ≠ ⊥ ⟹ lift_eq⋅x⋅y = Def (x = y)"
  by (cases x; cases y) simp_all

text ‹Scalars of different types are unequal; under static typing that case does not arise.›

fixrec scalar_eq :: "scalar → scalar → tr" where
  "a ≠ ⊥ ⟹ b ≠ ⊥ ⟹ scalar_eq⋅(SBool⋅a)⋅(SBool⋅b) = lift_eq⋅a⋅b"
| "m ≠ ⊥ ⟹ n ≠ ⊥ ⟹ scalar_eq⋅(SInt⋅m)⋅(SInt⋅n) = lift_eq⋅m⋅n"
| "a ≠ ⊥ ⟹ n ≠ ⊥ ⟹ scalar_eq⋅(SBool⋅a)⋅(SInt⋅n) = FF"
| "m ≠ ⊥ ⟹ b ≠ ⊥ ⟹ scalar_eq⋅(SInt⋅m)⋅(SBool⋅b) = FF"

lemma scalar_eq_strict [simp]: "scalar_eq⋅⊥⋅y = ⊥" "scalar_eq⋅x⋅⊥ = ⊥"
  by fixrec_simp (cases x; fixrec_simp)

lemma scalar_eq_defined: "x ≠ ⊥ ⟹ y ≠ ⊥ ⟹ scalar_eq⋅x⋅y = Def (x = y)"
  by (cases x; cases y) (simp_all add: lift_eq_defined FF_def)

text ‹Record-level equality is the lifted scalar equality, hence NA-blind.›

abbreviation datum_eq :: "scalar datum → scalar datum → tr datum" where
  "datum_eq ≡ dlift2⋅scalar_eq"

subsection ‹Truth›

text ‹The connectives are strict, so lifted to data they make ‹NA› and ‹Null›
  infectious (‹NOT NA = NA›).  @{text truth_d} projects a datum to @{typ tr}, sending every
  missing value and every non-boolean to ‹⊥›; it is a homomorphism for the connectives.  A
  slicer condition holds where @{text truth_d} is @{term TT}.›

definition tr_and :: "tr → tr → tr" where
  "tr_and = (FLIFT a. FLIFT b. Def (a ∧ b))"

definition tr_or :: "tr → tr → tr" where
  "tr_or = (FLIFT a. FLIFT b. Def (a ∨ b))"

lemma tr_and_simps [simp]:
  "tr_and⋅⊥⋅y = ⊥" "tr_and⋅x⋅⊥ = ⊥" "tr_and⋅(Def a)⋅(Def b) = Def (a ∧ b)"
  by (simp_all add: tr_and_def) (cases x rule: lift.exhaust; simp)

lemma tr_or_simps [simp]:
  "tr_or⋅⊥⋅y = ⊥" "tr_or⋅x⋅⊥ = ⊥" "tr_or⋅(Def a)⋅(Def b) = Def (a ∨ b)"
  by (simp_all add: tr_or_def) (cases x rule: lift.exhaust; simp)

fixrec sbool :: "scalar → tr" where
  "b ≠ ⊥ ⟹ sbool⋅(SBool⋅b) = b"

lemma sbool_strict [simp]: "sbool⋅⊥ = ⊥"
  by fixrec_simp

fixrec snot :: "scalar → scalar" where
  "a ≠ ⊥ ⟹ snot⋅(SBool⋅a) = SBool⋅(neg⋅a)"

fixrec sand :: "scalar → scalar → scalar" where
  "a ≠ ⊥ ⟹ b ≠ ⊥ ⟹ sand⋅(SBool⋅a)⋅(SBool⋅b) = SBool⋅(tr_and⋅a⋅b)"

fixrec sor :: "scalar → scalar → scalar" where
  "a ≠ ⊥ ⟹ b ≠ ⊥ ⟹ sor⋅(SBool⋅a)⋅(SBool⋅b) = SBool⋅(tr_or⋅a⋅b)"

text ‹Unconditional constructor equations; ill-typed applications give ‹⊥›.›

lemma sbool_simps [simp]: "sbool⋅(SBool⋅b) = b" "sbool⋅(SInt⋅n) = ⊥"
  by (cases "b = ⊥"; simp) (cases "n = ⊥"; fixrec_simp)

lemma snot_simps [simp]: "snot⋅(SBool⋅a) = SBool⋅(neg⋅a)" "snot⋅(SInt⋅n) = ⊥"
  by (cases "a = ⊥"; simp; fixrec_simp) (cases "n = ⊥"; fixrec_simp)

lemma sand_simps [simp]:
  "sand⋅(SBool⋅a)⋅(SBool⋅b) = SBool⋅(tr_and⋅a⋅b)"
  "sand⋅(SInt⋅m)⋅y = ⊥" "sand⋅x⋅(SInt⋅n) = ⊥"
    apply (cases "a = ⊥"; cases "b = ⊥"; simp; fixrec_simp)
   apply (cases "m = ⊥"; fixrec_simp)
  apply (cases x; cases "n = ⊥"; fixrec_simp)
  done

lemma sor_simps [simp]:
  "sor⋅(SBool⋅a)⋅(SBool⋅b) = SBool⋅(tr_or⋅a⋅b)"
  "sor⋅(SInt⋅m)⋅y = ⊥" "sor⋅x⋅(SInt⋅n) = ⊥"
    apply (cases "a = ⊥"; cases "b = ⊥"; simp; fixrec_simp)
   apply (cases "m = ⊥"; fixrec_simp)
  apply (cases x; cases "n = ⊥"; fixrec_simp)
  done

definition truth :: "scalar with_na → tr" where
  "truth = sbool oo from_na"

definition truth_d :: "scalar datum → tr" where
  "truth_d = sbool oo plain"

lemma scalar_connectives_strict [simp]:
  "snot⋅⊥ = ⊥" "sand⋅⊥⋅y = ⊥" "sand⋅x⋅⊥ = ⊥" "sor⋅⊥⋅y = ⊥" "sor⋅x⋅⊥ = ⊥"
      apply fixrec_simp
     apply fixrec_simp
    apply (cases x; fixrec_simp)
   apply fixrec_simp
  apply (cases x; fixrec_simp)
  done

lemma sbool_connectives:
  "sbool⋅(snot⋅s) = neg⋅(sbool⋅s)"
  "sbool⋅(sand⋅s⋅t) = tr_and⋅(sbool⋅s)⋅(sbool⋅t)"
  "sbool⋅(sor⋅s⋅t) = tr_or⋅(sbool⋅s)⋅(sbool⋅t)"
  by (cases s; simp; cases t; simp)+

lemma truth_d_not: "truth_d⋅(dlift1⋅snot⋅x) = neg⋅(truth_d⋅x)"
  by (simp add: truth_d_def plain_dlift1 sbool_connectives)

lemma truth_d_and: "truth_d⋅(dlift2⋅sand⋅x⋅y) = tr_and⋅(truth_d⋅x)⋅(truth_d⋅y)"
  by (simp add: truth_d_def plain_dlift2 sbool_connectives)

lemma truth_d_or: "truth_d⋅(dlift2⋅sor⋅x⋅y) = tr_or⋅(truth_d⋅x)⋅(truth_d⋅y)"
  by (simp add: truth_d_def plain_dlift2 sbool_connectives)

section ‹The not-null level of keys›

text ‹The atoms of @{typ "'a with_na"} are ‹NA› and ‹Val⋅⊥›, so its discriminator is the
  projection @{text not_null}, which observes only whether a key is ‹NA›.  It is a finitary
  semi-factor, i.e.\ a level of detail, and @{text is_na} factors through it.›

fixrec is_na :: "'a::domain with_na → tr" where
  "is_na⋅NA = TT"
| "is_na⋅(Val⋅x) = FF"

lemma is_na_strict [simp]: "is_na⋅⊥ = ⊥"
  by fixrec_simp

fixrec not_null :: "'a::domain with_na → 'a with_na" where
  "not_null⋅NA = NA"
| "not_null⋅(Val⋅x) = Val⋅⊥"

lemma not_null_strict [simp]: "not_null⋅⊥ = ⊥"
  by fixrec_simp

lemma atom_NA: "atom NA"
  by (rule atomI) (simp, case_tac z, simp_all)

lemma atom_Val_bottom: "atom (Val⋅⊥)"
  by (rule atomI) (simp, case_tac z, simp_all)

lemma discriminator_with_na:
  "discriminator⋅x = not_null⋅(x :: 'a::{domain, bounded_complete} with_na)"
proof (cases x)
  case bottom then show ?thesis by simp
next
  case NA then show ?thesis by (simp add: discriminator_eq[OF atom_NA below_refl])
next
  case (Val v)
  have "Val⋅⊥ ⊑ x" unfolding Val by (rule monofun_cfun_arg) simp
  then show ?thesis unfolding Val by (simp add: discriminator_eq[OF atom_Val_bottom])
qed

lemma not_null_level:
  "semi_factor_defl (not_null :: 'a::{domain, distributive_cpo} with_na → 'a with_na)"
  "finitary (not_null :: 'a::{domain, distributive_cpo} with_na → 'a with_na)"
  using discriminator_semi_factor discriminator_finitary
  by (simp_all add: cfun_eqI[OF discriminator_with_na, symmetric])

lemma is_na_not_null: "is_na⋅(not_null⋅x) = is_na⋅x"
  by (cases x) simp_all
