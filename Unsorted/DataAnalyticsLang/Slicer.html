default_sort type

section ‹Levels of detail as syntactic dimension sets›

text ‹
  Dimensions are drawn from a (countable, nominal) label set \<^typ>‹'d›; each label is
  evaluated on rows by a continuous row-level expression into a value domain.  Two labels may
  share an expression: they are then functionally dependent, but distinct.  The key of a row is
  the tuple of all its dimensions' values; a level of detail then cuts it.

  The image of a row under the cut key need ∗‹not› be a stable subdomain of the tuple domain
  (with an expression dimension ‹c ≈ a + b› the allowable sub-rows are not closed under existing
  joins), and we never require it to be: the domain stays the source rows, and functional
  dependence is stated relationally via kernel orders.
›

locale dim_eval =
  fixes deval :: "'d ⇒ ('r::pcpo → 'v::pcpo)" (‹⟦_⟧›)
begin

definition keyof :: "'r → ('d, 'v) tuple" where
  "keyof = (Λ r. Abs_tuple (λd. ⟦d⟧⋅r))"

lemma keyof_beta: "keyof⋅r = Abs_tuple (λd. ⟦d⟧⋅r)"
  unfolding keyof_def by (simp add: cont_Abs_tuple)

lemma keyof_tget [simp]: "(keyof⋅r)⟨d⟩ = ⟦d⟧⋅r"
  by (simp add: keyof_beta Abs_tuple_inverse')

subsection ‹Kernel orders and functional dependence›

text ‹The kernel order of a projection ‹p› on keys (in practice a deflation, such as an LOD's):
  rows compared by what ‹p› keeps of their keys.›

definition kern :: "(('d, 'v) tuple → ('d, 'v) tuple) ⇒ 'r ⇒ 'r ⇒ bool" where
  "kern p x y ⟷ p⋅(keyof⋅x) ⊑ p⋅(keyof⋅y)"

lemma is_cong_kern: "is_cong (kern p)"
  unfolding kern_def by unfold_locales (auto intro: below_trans monofun_cfun_arg)

lemma kern_bot: "kern ⊥ x y"
  by (simp add: kern_def)

lemma kern_below:
  assumes "deflation p" "deflation q" "p ⊑ q" and "kern q x y"
  shows "kern p x y"
proof -
  have "p⋅(q⋅(keyof⋅x)) ⊑ p⋅(q⋅(keyof⋅y))" using assms(4) unfolding kern_def by (rule monofun_cfun_arg)
  then show ?thesis unfolding kern_def by (simp add: deflation_below_comp1[OF assms(1-3)])
qed

text ‹Functional dependence of ‹p› on ‹q›, relative to a support predicate \<^term>‹S› on source
  rows: rows that ‹q› cannot tell apart (in the kernel order), ‹p› cannot either.  Taking
  \<^term>‹S› true everywhere gives schema-level dependence; taking \<^term>‹S› to be what a filter
  selects gives the data-level one.›

definition fdep :: "('r ⇒ bool) ⇒ (('d, 'v) tuple → ('d, 'v) tuple) ⇒ (('d, 'v) tuple → ('d, 'v) tuple) ⇒ bool" where
  "fdep S p q ⟷ (∀x y. S x ⟶ S y ⟶ kern q x y ⟶ kern p x y)"

lemma fdep_refl: "fdep S p p"
  by (simp add: fdep_def)

lemma fdep_trans: "fdep S p q ⟹ fdep S q r ⟹ fdep S p r"
  unfolding fdep_def by blast

lemma fdep_below: "deflation p ⟹ deflation q ⟹ p ⊑ q ⟹ fdep S p q"
  unfolding fdep_def by (blast intro: kern_below)

lemma fdep_support_mono: "(⋀x. S' x ⟹ S x) ⟹ fdep S p q ⟹ fdep S' p q"
  unfolding fdep_def by blast

end

section ‹Filters›

text ‹A filter is a continuous predicate into \<^typ>‹one› (Sierpinski space): a row ∗‹passes› it
  when it maps to \<^term>‹ONE›, a Scott-open predicate.  Several filters are combined by
  continuous conjunction.›

definition passes :: "('r::pcpo → one) ⇒ 'r ⇒ bool" where
  "passes φ r ⟷ φ⋅r = ONE"

lemma passes_up: "passes φ x ⟹ x ⊑ y ⟹ passes φ y"
  unfolding passes_def by (cases "φ⋅y") (auto dest: monofun_cfun_arg[of x y φ])

lemma passes_open: "adm (λr. ¬ passes φ r)"
proof -
  have "¬ passes φ r ⟷ φ⋅r ⊑ ⊥" for r
    unfolding passes_def by (cases "φ⋅r") simp_all
  then show ?thesis by simp
qed

definition fconj :: "('r::pcpo → one) ⇒ ('r → one) ⇒ ('r → one)" where
  "fconj φ ψ = (Λ r. seq⋅(φ⋅r)⋅(ψ⋅r))"

lemma passes_fconj [simp]: "passes (fconj φ ψ) r ⟷ passes φ r ∧ passes ψ r"
  unfolding passes_def fconj_def by (auto simp: seq_conv_if split: if_splits)

lemma passes_ftrue [simp]: "passes (Λ _. ONE) r"
  by (simp add: passes_def)

section ‹Datasets›

text ‹A dataset is a Scott-closed set \<^term>‹S› of rows whose ∗‹maximal› elements are the
  actual data points; the rest are their partial approximations (sub-rows).  These need not be
  maximal in the row ∗‹type›.  Restricting to a predicate keeps the data points satisfying it and
  everything below them.  The result is generally ∗‹not› a stable subdomain (it need not be
  closed under joins of compatible sub-rows), but it is again a dataset, with exactly those
  points as its maximal elements.  We assume finitely many data points, which also makes
  closedness automatic.›

definition pts :: "'r::pcpo set ⇒ 'r set" where
  "pts S = {m ∈ S. ∀y∈S. m ⊑ y ⟶ y = m}"

definition filtered :: "'r::pcpo set ⇒ ('r ⇒ bool) ⇒ 'r set" where
  "filtered S P = {x. ∃m ∈ pts S. P m ∧ x ⊑ m}"

lemma adm_below_finite: "finite P ⟹ adm (λx. ∃m∈P. x ⊑ m)"
  by (induct rule: finite_induct) simp_all

locale dataset =
  fixes S :: "'r::pcpo set"
  assumes downward: "y ∈ S ⟹ x ⊑ y ⟹ x ∈ S"
    and max_above: "x ∈ S ⟹ ∃m∈pts S. x ⊑ m"
    and finite_pts: "finite (pts S)"
begin

lemma S_eq: "S = {x. ∃m∈pts S. x ⊑ m}"
  using max_above downward unfolding pts_def by blast

lemma closed: "adm (λx. x ∈ S)"
  by (subst S_eq) (simp add: adm_below_finite finite_pts)

lemma filtered_subset: "filtered S P ⊆ S"
  unfolding filtered_def pts_def using downward by blast

lemma pts_filtered: "pts (filtered S P) = {m ∈ pts S. P m}"
  using filtered_subset unfolding filtered_def pts_def by (auto 6 3)

lemma dataset_filtered: "dataset (filtered S P)"
proof
  show "y ∈ filtered S P ⟹ x ⊑ y ⟹ x ∈ filtered S P" for x y
    unfolding filtered_def by (blast intro: below_trans)
  show "x ∈ filtered S P ⟹ ∃m∈pts (filtered S P). x ⊑ m" for x
    unfolding pts_filtered by (auto simp: filtered_def)
  show "finite (pts (filtered S P))"
    unfolding pts_filtered using finite_pts by simp
qed

lemma filtered_filtered: "filtered (filtered S P) Q = filtered S (λx. P x ∧ Q x)"
  unfolding filtered_def[of "filtered S P"] pts_filtered
  by (auto simp: filtered_def)

lemma filtered_top: "filtered S (λ_. True) = S"
  by (subst (2) S_eq) (simp add: filtered_def)

end

section ‹Slicers›

subsection ‹Selections›

text ‹What slices select: Scott-open predicates, i.e.\ upward closed with an admissible
  complement.  These correspond one-to-one to filters \<^typ>‹'r → one›, and form a distributive
  bounded lattice under the pointwise operations.›

typedef 'r sel = "{P :: 'r::pcpo ⇒ bool. (∀x y. P x ⟶ x ⊑ y ⟶ P y) ∧ adm (λx. ¬ P x)}"
  morphisms selects Abs_sel
  by (rule exI[of _ "λ_. True"]) simp

setup_lifting type_definition_sel

lemma selects_up: "selects s x ⟹ x ⊑ y ⟹ selects s y"
  using selects[of s] by blast

lemma selects_open: "adm (λx. ¬ selects s x)"
  using selects[of s] by blast

lemma sel_eqI: "(⋀x. selects a x ⟷ selects b x) ⟹ a = b"
  by (simp add: selects_inject[symmetric] fun_eq_iff)

lift_definition sel_of :: "('r::pcpo → one) ⇒ 'r sel" is passes
  using passes_up passes_open by blast

instantiation sel :: (pcpo) "{distrib_lattice, bounded_lattice}"
begin

lift_definition less_eq_sel :: "'a sel ⇒ 'a sel ⇒ bool" is "(≤)" .
lift_definition less_sel :: "'a sel ⇒ 'a sel ⇒ bool" is "(<)" .
lift_definition inf_sel :: "'a sel ⇒ 'a sel ⇒ 'a sel" is inf
  by (auto intro!: adm_disj)
lift_definition sup_sel :: "'a sel ⇒ 'a sel ⇒ 'a sel" is sup
  by (auto intro!: adm_conj)
lift_definition bot_sel :: "'a sel" is bot by simp
lift_definition top_sel :: "'a sel" is top by simp

instance proof
  fix x y z :: "'a sel"
  show "x < y ⟷ x ≤ y ∧ ¬ y ≤ x" by transfer (rule less_le_not_le)
  show "x ≤ x" by transfer (rule order_refl)
  show "x ≤ y ⟹ y ≤ z ⟹ x ≤ z" by transfer (erule (1) order_trans)
  show "x ≤ y ⟹ y ≤ x ⟹ x = y" by transfer (erule (1) antisym)
  show "inf x y ≤ x" by transfer (rule inf_le1)
  show "inf x y ≤ y" by transfer (rule inf_le2)
  show "x ≤ y ⟹ x ≤ z ⟹ x ≤ inf y z" by transfer (erule (1) le_infI)
  show "x ≤ sup x y" by transfer (rule sup_ge1)
  show "y ≤ sup x y" by transfer (rule sup_ge2)
  show "y ≤ x ⟹ z ≤ x ⟹ sup y z ≤ x" by transfer (erule (1) le_supI)
  show "sup x (inf y z) = inf (sup x y) (sup x z)" by transfer (rule sup_inf_distrib1)
  show "bot ≤ x" by transfer (rule bot_least)
  show "x ≤ top" by transfer (rule top_greatest)
qed

end

lemma selects_inf [simp]: "selects (inf a b) x ⟷ selects a x ∧ selects b x"
  by (simp add: inf_sel.rep_eq)

lemma selects_top [simp]: "selects top x"
  by (simp add: top_sel.rep_eq)

subsection ‹Semilattice homomorphisms›

text ‹Slices are an algebra given by laws (e.g.\ a quotient of syntax), not a type class:
  conjunction alone is assumed to form a semilattice with unit.  Their selections, their
  dimensions and their coarsening are all homomorphisms of such, into different targets.›

locale semilattice_hom = src: semilattice_neutr f z + tgt: semilattice_neutr g w
  for f :: "'a ⇒ 'a ⇒ 'a" and z :: 'a and g :: "'b ⇒ 'b ⇒ 'b" and w :: 'b +
  fixes h :: "'a ⇒ 'b"
  assumes hom_f: "h (f a b) = g (h a) (h b)"
    and hom_z: "h z = w"

subsection ‹Slice evaluation›

text ‹Slices are given by their conjunction ‹❙*› and its unit ‹❙1›, nothing more: the slicer
  constructs only conjoin.  The order is the one conjunction induces.  Slices act on keys, which
  carry levels of detail (semi-factors of the keys, compact ones for LODs).  Each slice comes with the LOD it is written
  at, ‹slod›: what it ∗‹looks at›, not what it selects (an upper bound, not the least one).  Its
  meaning ‹sopen› is a Scott-open predicate on keys, and a meet-semilattice homomorphism.  A slice
  reads a key only through its truncation to the slice's LOD, so it applies to keys finer than its
  LOD as well.  ‹holds› is the plain predicate.›

text ‹The slicer interface is a record of operations.  Besides evaluation, a slicer
  ∗‹coarsens› slices to a given LOD and ∗‹pins› keys; their laws are explained below.›

record ('f, 'k) slicer_ops =
  sl_conj :: "'f ⇒ 'f ⇒ 'f"
  sl_unit :: 'f
  sl_lod :: "'f ⇒ 'k semi_factor"
  sl_open :: "'f ⇒ 'k sel"
  sl_coarsen :: "'k semi_factor ⇒ 'f ⇒ 'f"
  sl_pin :: "'k ⇒ 'f"

hide_const (open) slicer_ops.truncate

locale slicer = conj: semilattice_neutr "sl_conj slicing" "sl_unit slicing" +
  slod: semilattice_hom "sl_conj slicing" "sl_unit slicing" "(⊔)" ⊥ "sl_lod slicing" +
  sopen: semilattice_hom "sl_conj slicing" "sl_unit slicing" inf top "sl_open slicing"
  for slicing :: "('f, 'k::lod_domain) slicer_ops" +
  assumes sopen_lod: "selects (sl_open slicing f) (proj (sl_lod slicing f)⋅k) ⟷ selects (sl_open slicing f) k"
    and slod_compact: "compact (sl_lod slicing f)"
    and coarsen_conj: "sl_coarsen slicing A (sl_conj slicing a b) =
      sl_conj slicing (sl_coarsen slicing A a) (sl_coarsen slicing A b)"
    and coarsen_inside: "sl_lod slicing f ⊑ A ⟹ sl_coarsen slicing A f = f"
    and slod_coarsen: "sl_lod slicing (sl_coarsen slicing A f) ⊑ sl_lod slicing f ⊓ A"
    and coarsen_cut: "selects (sl_open slicing f) k ⟹
      selects (sl_open slicing (sl_coarsen slicing A f)) (proj A⋅k)"
    and coarsen_slod: "sl_coarsen slicing A f = sl_coarsen slicing (A ⊓ sl_lod slicing f) f"
    and coarsen_coarsen: "sl_coarsen slicing A (sl_coarsen slicing B f) = sl_coarsen slicing (A ⊓ B) f"
    and sopen_spin: "compact K ⟹ selects (sl_open slicing (sl_pin slicing K)) k ⟷ K ⊑ k"
begin

abbreviation conj :: "'f ⇒ 'f ⇒ 'f" (infixl ‹❙*› 70) where "conj ≡ sl_conj slicing"
abbreviation sunit :: 'f (‹❙1›) where "sunit ≡ sl_unit slicing"
abbreviation slod :: "'f ⇒ 'k semi_factor" where "slod ≡ sl_lod slicing"
abbreviation sopen :: "'f ⇒ 'k sel" where "sopen ≡ sl_open slicing"
abbreviation coarsen :: "'k semi_factor ⇒ 'f ⇒ 'f" where "coarsen ≡ sl_coarsen slicing"
abbreviation spin :: "'k ⇒ 'f" where "spin ≡ sl_pin slicing"

lemmas slod_conj [simp] = slod.hom_f and slod_unit [simp] = slod.hom_z

definition sle :: "'f ⇒ 'f ⇒ bool" (infix ‹❙≤› 50) where
  "a ❙≤ b ⟷ a = a ❙* b"

definition slt :: "'f ⇒ 'f ⇒ bool" (infix ‹❙<› 50) where
  "a ❙< b ⟷ a ❙≤ b ∧ a ≠ b"

sublocale sord: semilattice_neutr_order conj sunit sle slt
  by unfold_locales (simp_all add: sle_def slt_def)

lemma sopen_mono: "a ❙≤ b ⟹ sopen a ≤ sopen b"
  unfolding sle_def by (metis sopen.hom_f inf.orderI)

definition holds :: "'f ⇒ 'k ⇒ bool" where
  "holds f = selects (sopen f)"

lemma holds_conj [simp]: "holds (a ❙* b) k ⟷ holds a k ∧ holds b k"
  by (simp add: holds_def sopen.hom_f)

lemma holds_unit [simp]: "holds ❙1 k"
  by (simp add: holds_def sopen.hom_z)

lemma holds_up: "holds f k ⟹ k ⊑ k' ⟹ holds f k'"
  unfolding holds_def by (rule selects_up)

lemma holds_open: "adm (λk. ¬ holds f k)"
  unfolding holds_def by (rule selects_open)

lemma holds_mono: "a ❙≤ b ⟹ holds a k ⟹ holds b k"
  unfolding sle_def by (metis holds_conj)

lemma holds_lod: "holds f (proj (slod f)⋅k) ⟷ holds f k"
  by (simp add: holds_def sopen_lod)

text ‹Truncating to any LOD the slice is written within does not change whether it holds.›

lemma holds_proj: "slod f ⊑ c ⟹ holds f (proj c⋅k) ⟷ holds f k"
  by (metis holds_lod proj_le)

lemma holds_bot_lod:
  assumes "slod f = ⊥" shows "holds f k ⟷ holds f ⊥"
  using holds_lod[of f k] by (simp add: assms proj_bot)

subsection ‹Coarsening›

text ‹A slicer adds ∗‹coarsening› of a slice to a given LOD.  ‹GROUP BY ONLY› coarsens to the new
  frame LOD; ‹FOR› coarsens to the slice's own LOD with dimensions dropped by a transformer.
  Coarsening distributes over conjunction, leaves slices written inside the target alone, keeps
  only what the target looks at, holds at the truncation of every key the original holds at, and
  coarsening twice is coarsening once, to the meet of the targets.
  It only depends on the part of the target the slice is written at: coarsening is about what is
  cut away.
  It is ∗‹not› extensional: coarsening ‹b = 2 ∧ b ≠ 2› away from ‹b› gives ‹True›, coarsening
  ‹False› gives ‹False›.›


lemma holds_coarsen_cut: "holds f k ⟹ holds (coarsen A f) (proj A⋅k)"
  unfolding holds_def by (rule coarsen_cut)

lemma coarsen_unit [simp]: "coarsen A ❙1 = ❙1"
  by (rule coarsen_inside) simp

lemma coarsen_hom: "semilattice_hom conj sunit conj sunit (coarsen A)"
  by unfold_locales (fact coarsen_conj coarsen_unit)+

lemma coarsen_mono: "a ❙≤ b ⟹ coarsen A a ❙≤ coarsen A b"
  unfolding sle_def by (metis coarsen_conj)

lemma slod_coarsen_target: "slod (coarsen A f) ⊑ A"
  using slod_coarsen[of A f] by (rule below_trans[OF _ lattice_meet_le2])

lemma holds_coarsen: "holds f k ⟹ holds (coarsen A f) k"
  using holds_coarsen_cut proj_below by (rule holds_up)

lemma holds_coarsen_reads: "holds (coarsen A f) (proj A⋅k) ⟷ holds (coarsen A f) k"
  by (rule holds_proj[OF slod_coarsen_target])

text ‹A slice written entirely outside the target coarsens to a constant, which is true as soon
  as the slice holds anywhere.  Syntactically this fails: ‹False ∧ a = 1› coarsens to ‹False›
  away from ‹a›.›

lemma coarsen_outside:
  assumes "slod f ⊓ A = ⊥" and "holds f t"
  shows "holds (coarsen A f) k"
proof -
  have lod: "slod (coarsen A f) = ⊥"
    using slod_coarsen[of A f] assms(1) by simp
  have "holds (coarsen A f) (proj A⋅t)" using assms(2) by (rule holds_coarsen_cut)
  then have "holds (coarsen A f) ⊥" using holds_bot_lod[OF lod, of "proj A⋅t"] by blast
  then show ?thesis by (rule holds_up) simp
qed

text ‹Coarsening to less is weaker.›

lemma holds_coarsen_le:
  assumes "A ⊑ B" and "holds (coarsen B f) k" shows "holds (coarsen A f) k"
proof -
  have "coarsen A f = coarsen A (coarsen B f)" by (simp add: coarsen_coarsen lattice_meet_absorb1[OF assms(1)])
  then show ?thesis using holds_coarsen[OF assms(2)] by simp
qed

lemma coarsen_top [simp]: "coarsen lattice_top f = f"
  by (rule coarsen_inside) simp

subsection ‹Pins›

text ‹‹spin K› is the slice pinning the key ‹K›: on keys, the principal upset of ‹K›.  It is only
  specified for compact keys, whose upsets are open.  The slicer picks the LOD a pin is written
  at; nothing is assumed about it, but it is always one the key lives at.  Nothing is assumed of
  coarsened pins either: by @{thm [source] coarsen_cut} they hold above the truncated key, and a
  syntactic slicer may lose more (coarsening ‹date = d› to the month drops it).›


lemma holds_spin: "compact K ⟹ holds (spin K) k ⟷ K ⊑ k"
  unfolding holds_def by (rule sopen_spin)

lemma slod_spin_lives:
  assumes "compact K" shows "proj (slod (spin K))⋅K = K"
proof -
  have "holds (spin K) K" by (simp add: holds_spin[OF assms])
  then have "holds (spin K) (proj (slod (spin K))⋅K)" by (simp only: holds_lod)
  then have "K ⊑ proj (slod (spin K))⋅K" by (simp only: holds_spin[OF assms])
  then show ?thesis using proj_below by (rule below_antisym[rotated])
qed

lemma holds_coarsen_spin: "compact K ⟹ proj A⋅K ⊑ k ⟹ holds (coarsen A (spin K)) k"
  by (rule holds_up[OF holds_coarsen_cut]) (simp_all add: holds_spin)

text ‹A slice below a pin only holds above the pinned key.›

lemma holds_pinned:
  assumes "compact K" and "f ❙≤ spin K" and "holds f k"
  shows "K ⊑ k"
proof -
  have "holds (spin K) k" using assms(2,3) by (rule holds_mono)
  then show ?thesis by (simp only: holds_spin[OF assms(1)])
qed

end

subsection ‹Rows versus keys›

text ‹A row predicate reading only what a projection ‹p› keeps of the key is the same thing as a
  key predicate: its up-image is open (when the projected keys are compact), and its preimage
  gives back the row predicate.
  Preimages of open key predicates are open for any ‹p›.›

context dim_eval
begin

definition keyup :: "(('d, 'v) tuple → ('d, 'v) tuple) ⇒ ('r ⇒ bool) ⇒ ('d, 'v) tuple ⇒ bool" where
  "keyup p V t ⟷ (∃r. V r ∧ p⋅(keyof⋅r) ⊑ t)"

definition rowpre :: "(('d, 'v) tuple → ('d, 'v) tuple) ⇒ (('d, 'v) tuple ⇒ bool) ⇒ 'r ⇒ bool" where
  "rowpre p U r ⟷ U (p⋅(keyof⋅r))"

lemma keyup_up: "keyup p V t ⟹ t ⊑ t' ⟹ keyup p V t'"
  unfolding keyup_def by (blast intro: below_trans)

lemma rowpre_keyup:
  assumes "⋀x y. kern p x y ⟹ V x ⟹ V y"
  shows "rowpre p (keyup p V) = V"
  using assms unfolding fun_eq_iff rowpre_def keyup_def kern_def by blast

lemma rowpre_up: "(⋀t t'. U t ⟹ t ⊑ t' ⟹ U t') ⟹ rowpre p U r ⟹ r ⊑ r' ⟹ rowpre p U r'"
  unfolding rowpre_def by (blast intro: monofun_cfun_arg)

lemma rowpre_open: "adm (λt. ¬ U t) ⟹ adm (λr. ¬ rowpre p U r)"
  unfolding rowpre_def using adm_subst[of "λr. p⋅(keyof⋅r)" "λt. ¬ U t"] by simp

lemma keyup_open:
  assumes "⋀r. compact (p⋅(keyof⋅r))"
  shows "adm (λt. ¬ keyup p V t)"
proof (rule admI)
  fix Y :: "nat ⇒ ('d, 'v) tuple"
  assume Y: "chain Y" "∀i. ¬ keyup p V (Y i)"
  show "¬ keyup p V (⨆i. Y i)"
  proof
    assume "keyup p V (⨆i. Y i)"
    then obtain r where "V r" "p⋅(keyof⋅r) ⊑ (⨆i. Y i)" by (auto simp: keyup_def)
    then obtain i where "p⋅(keyof⋅r) ⊑ Y i" using Y(1) assms compact_below_lub_iff by blast
    with ‹V r› Y(2) show False by (auto simp: keyup_def)
  qed
qed

end
