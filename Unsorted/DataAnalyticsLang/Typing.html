Theory Typing

theory Typing imports Main begin

unbundle lattice_syntax

section ‹Typing terms that obey lattice subtyping›
locale typing =
  fixes of_type :: "'val ⇒ 'typ::{complete_lattice} ⇒ bool"
  assumes of_type_mono: "t ≤ t' ⟹ of_type v t ⟹ of_type v t'"
      and of_type_Inf: "(⋀ t. t ∈ ts ⟹ of_type v t) ⟹ of_type v (⨅ ts)"

begin

definition principal_type :: "'val ⇒ 'typ" where "principal_type v = ⨅ {t. of_type v t}"

theorem principal_type_is_principal: "of_type v t ⟷ principal_type v ≤ t"
proof
  fix v t
  assume "of_type v t"
  then show "principal_type v ≤ t" by (simp add: Inf_lower principal_type_def)
next
  fix v t
  assume 0: "principal_type v ≤ t"
  then show "of_type v t" using of_type_Inf of_type_mono principal_type_def 0 by fastforce
qed

lemma principal_type_elim[elim]: "principal_type v ≤ t ⟹ of_type v t"
  using principal_type_is_principal by fast

theorem Sup_principal_type: "⨆(principal_type ` V) = ⨅{t. ∀v ∈ V. of_type v t}"
proof -
  have "⨆(principal_type ` V) = ⨅{t. ∀ v ∈ V. principal_type v ≤ t}" by (simp add: Sup_eq_Inf)
  also have "... = ⨅{t. ∀ v ∈ V. of_type v t}" using principal_type_is_principal by simp
  finally show ?thesis .
qed

interpretation typing_self: typing
  where of_type = ‹(≤)::('typ::{complete_lattice} ⇒ 'typ ⇒ bool)›
  by (auto intro!: typing.intro simp: Inf_greatest)
end

section ‹Typing judgments›

datatype 't judg = Of 't | Empty

instantiation judg :: (ord) ord begin

definition less_eq_judg where
"x ≤ y ≡ case x of Empty ⇒ True | Of x ⇒ (case y of Empty ⇒ False | Of y ⇒ x ≤ y)"

definition
  "(x::'a judg) < y ≡ x ≤ y ∧ x ≠ y"

lemma less_eq_empty: "Empty ≤ x" by (auto simp add: less_eq_judg_def)
lemma less_eq_judg_Of: "x ≤ y ⟷ Of x ≤ Of y" by (auto simp add: less_eq_judg_def)

instance ..

end

instantiation judg :: (order) order begin

instance
  by standard (auto simp add: less_eq_judg_def less_judg_def intro: order_trans split: judg.splits)

end

instantiation judg :: (bounded_lattice) bounded_lattice begin
definition sup_judg where
"a ⊔ b ≡ case a of Empty ⇒ b 
  | Of x ⇒ (case b of Empty ⇒ Of x | Of y ⇒ Of (x ⊔ y))"
definition inf_judg where
"a ⊓ b ≡ case a of Empty ⇒ Empty 
  | Of x ⇒ (case b of Empty ⇒ Empty | Of y ⇒ Of (x ⊓ y))"
definition top_judg where "⊤ = Of ⊤"
definition bot_judg where "⊥ = Empty"

instance
  by standard
    (auto simp add:
        sup_judg_def inf_judg_def top_judg_def bot_judg_def less_eq_judg_def
    split: judg.splits)

end

type_synonym 'v model_function = "'v list ⇒ 'v option"

type_synonym 't judgment_function = "'t list ⇒ 't judg"

context typing begin

primrec judges_type where
  "judges_type v (Of t) = of_type v t"
| "judges_type v Empty = False"

context fixes mf :: "'val model_function" begin
context fixes ts :: "'typ list" begin

abbreviation of_typeAll where "of_typeAll ≡ list_all2 of_type"

definition fn_value_set :: "'val set" where 
  "fn_value_set ≡ { v | vs v. of_typeAll vs ts ∧ mf vs = Some v }"

abbreviation sup_type where "sup_type ≡ ⨆(principal_type ` fn_value_set)"
abbreviation min_common_type where "min_common_type ≡  ⨅{t. ∀ v ∈ fn_value_set. of_type v t}"

lemma sup_type_eq_min_common_type: "sup_type = min_common_type"
  by (rule Sup_principal_type)

definition fn_type_judgment :: "'typ judg" where
  "fn_type_judgment ≡ if (fn_value_set = {}) then Empty else Of min_common_type"

lemma fn_type_judgment_Empty: "fn_type_judgment = Empty ⟷ fn_value_set = {}"
  using fn_type_judgment_def
  by (cases "fn_value_set = {}") auto

lemma fn_type_judgment_Of: "fn_value_set ≠ {} ⟹
  fn_type_judgment = Of min_common_type"
  using fn_type_judgment_def
  by auto

definition sound_judgment :: "'typ judg ⇒ bool"
  where "sound_judgment t ≡ ∀ v ∈ fn_value_set. judges_type v t"

lemma min_common_type_is_common: assumes "v ∈ fn_value_set"
  shows "of_type v min_common_type"
  using of_type_Inf assms by fastforce (* huh wow *)

theorem fn_type_judgment_sound: "sound_judgment fn_type_judgment"
  unfolding sound_judgment_def
  using fn_type_judgment_Of min_common_type_is_common by fastforce

lemma sound_nonempty: assumes "fn_value_set ≠ {}" and "sound_judgment t"
  obtains t0 where "t = Of t0"
  using assms by (cases t) (simp_all add: sound_judgment_def)

theorem fn_type_judgment_minimal: "sound_judgment t ⟹ fn_type_judgment ≤ t"
proof (cases "fn_value_set = {}")
  case True
  then show ?thesis using fn_type_judgment_Empty less_eq_empty by auto
next
  case nonempty: False
  with nonempty have judged: "fn_type_judgment = Of min_common_type"
    using fn_type_judgment_Of by fast
  assume sound: "sound_judgment t"
  with nonempty obtain t0 where t0_def: "t = Of t0" using sound_nonempty by blast
  from sound t0_def have "∀ v ∈ fn_value_set. of_type v t0"
    using sound_judgment_def judges_type_def by simp
  then have "t0 ∈ {t. ∀ v ∈ fn_value_set. of_type v t}" by blast
  then have "min_common_type ≤ t0" by (simp add: Inf_lower)
  with judged t0_def show ?thesis using less_eq_judg_def by fastforce
qed

abbreviation leqAll where "leqAll xs ys ≡ list_all2 (≤) xs ys"

lemma of_types_mono: ‹of_typeAll vs ts ⟹ leqAll ts ts' ⟹ of_typeAll vs ts'›
  by (rule list_all2_trans[where ?bs = ts and ?P1.0=of_type and ?P2.0=less_eq])
    (auto intro: of_type_mono)

end

context
  fixes ts :: "'typ list" and ts'
  assumes ts_leq: "leqAll ts ts'"
begin

lemma fn_value_set_mono: "fn_value_set ts ⊆ fn_value_set ts'"
  unfolding fn_value_set_def using ts_leq of_types_mono by auto

lemma sound_judgment_mono1: "sound_judgment ts' t ⟹ sound_judgment ts t"
  unfolding sound_judgment_def using fn_value_set_mono by fast

theorem type_judgment_mono: "(fn_type_judgment ts) ≤ (fn_type_judgment ts')"
  using fn_type_judgment_sound fn_type_judgment_minimal sound_judgment_mono1
  by auto

end

definition arity where "arity n ≡ ∃vs. length vs = n ∧ mf vs ≠ None"

definition max_arity where "max_arity ≡ Max {n. arity n}"

definition signature_ith where
  ‹signature_ith i ≡ ⨅{t. ∀vs. length vs ≥ Suc i ∧ mf vs ≠ None ⟶ of_type (vs ! i) t}›

definition signature where
  ‹signature = map signature_ith [0..<max_arity]›
