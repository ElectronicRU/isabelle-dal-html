Theory Aggregate

theory Aggregate
  imports Content_Law

section ‹Aggregation›

text ‹An aggregate folds the entries of an inner expression over each fiber of the frame's LOD
  with a commutative semigroup operation.  The fold adjoins a unit for the empty fiber, which
  gives no entry; it is not ‹⊥›: a monotone operation with unit ‹⊥› would be inflationary
  (‹x = x ⊕ ⊥ ⊑ x ⊕ y›), which rules out sums on flat values.›

definition opt_op :: "('v::pcpo → 'v → 'v) ⇒ 'v option ⇒ 'v option ⇒ 'v option" where
  "opt_op f a b = (case a of None ⇒ b | Some x ⇒ Some (case b of None ⇒ x | Some y ⇒ f⋅x⋅y))"

lemma comm_monoid_opt_op:
  assumes "abel_semigroup (λa b. f⋅a⋅b)"
  shows "comm_monoid (opt_op f) None"
proof -
  interpret abel_semigroup "λa b. f⋅a⋅b" by (fact assms)
  show ?thesis
    by standard (auto simp: opt_op_def assoc commute left_commute split: option.splits)
qed

definition agg_fold :: "('v::pcpo → 'v → 'v) ⇒ ('a ⇀ 'v) ⇒ 'a set ⇒ 'v option" where
  "agg_fold f v A = comm_monoid_set.F (opt_op f) None v A"

context
  fixes f :: "'v::pcpo → 'v → 'v"
  assumes f: "abel_semigroup (λa b. f⋅a⋅b)"
begin

interpretation A: comm_monoid_set "opt_op f" None
  unfolding comm_monoid_set_def by (rule comm_monoid_opt_op[OF f])

lemma agg_fold_empty: "agg_fold f v {} = None"
  by (simp add: agg_fold_def)

lemma agg_fold_singleton: "agg_fold f v {a} = v a"
  by (simp add: agg_fold_def opt_op_def split: option.splits)

lemma agg_fold_nonempty: "agg_fold f v A ≠ None ⟹ A ≠ {}"
  by (auto simp: agg_fold_empty)

lemma agg_fold_present: "finite A ⟹ a ∈ A ⟹ v a ≠ None ⟹ agg_fold f v A ≠ None"
  unfolding agg_fold_def by (subst A.remove) (auto simp: opt_op_def)

lemma agg_fold_absent: "(⋀a. a ∈ A ⟹ v a = None) ⟹ agg_fold f v A = None"
  unfolding agg_fold_def by (rule A.neutral) simp

lemma agg_fold_cong: "A = B ⟹ (⋀a. a ∈ B ⟹ v a = w a) ⟹ agg_fold f v A = agg_fold f w B"
  unfolding agg_fold_def by (simp cong: A.cong)

text ‹Folding over two images of one set of rows, identified alike, with the same entries.›

lemma agg_fold_image_cong:
  assumes "⋀y z. y ∈ Z ⟹ z ∈ Z ⟹ g y = g z ⟷ g' y = g' z"
    and "⋀y. y ∈ Z ⟹ v (g y) = w (g' y)"
  shows "agg_fold f v (g ` Z) = agg_fold f w (g' ` Z)"
  unfolding agg_fold_def
proof (rule A.reindex_bij_witness[where i = "λb. g (SOME y. y ∈ Z ∧ g' y = b)"
      and j = "λa. g' (SOME y. y ∈ Z ∧ g y = a)"])
  fix a assume "a ∈ g ` Z"
  then obtain y where y: "y ∈ Z" "g y = a" by blast
  let ?y = "SOME y. y ∈ Z ∧ g y = a"
  have y': "?y ∈ Z" "g ?y = a" using someI[of "λy. y ∈ Z ∧ g y = a", OF conjI[OF y]] by simp_all
  let ?z = "SOME z. z ∈ Z ∧ g' z = g' ?y"
  have z: "?z ∈ Z" "g' ?z = g' ?y" using someI[of "λz. z ∈ Z ∧ g' z = g' ?y" ?y] y'(1) by simp_all
  show "g ?z = a" using assms(1)[OF z(1) y'(1)] z(2) y'(2) by simp
  show "g' ?y ∈ g' ` Z" using y'(1) by blast
  show "w (g' ?y) = v a" using assms(2)[OF y'(1)] y'(2) by simp
next
  fix b assume "b ∈ g' ` Z"
  then obtain y where y: "y ∈ Z" "g' y = b" by blast
  let ?y = "SOME y. y ∈ Z ∧ g' y = b"
  have y': "?y ∈ Z" "g' ?y = b" using someI[of "λy. y ∈ Z ∧ g' y = b", OF conjI[OF y]] by simp_all
  let ?z = "SOME z. z ∈ Z ∧ g z = g ?y"
  have z: "?z ∈ Z" "g ?z = g ?y" using someI[of "λz. z ∈ Z ∧ g z = g ?y" ?y] y'(1) by simp_all
  show "g' ?z = b" using assms(1)[OF z(1) y'(1)] z(2) y'(2) by simp
  show "g ?y ∈ g ` Z" using y'(1) by blast
qed

end

lemma agg_fold_bij:
  assumes f: "abel_semigroup (λa b. f⋅a⋅b)" and h: "bij_betw h A B"
  shows "agg_fold f (v ∘ h) A = agg_fold f v B"
proof -
  interpret A: comm_monoid_set "opt_op f" None
    unfolding comm_monoid_set_def by (rule comm_monoid_opt_op[OF f])
  show ?thesis unfolding agg_fold_def comp_def by (rule A.reindex_bij_betw[OF h])
qed

text ‹The rows of a frame in the fiber of a key at an LOD.›

definition afiber :: "'r lower_pd ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k semi_factor ⇒ 'k ⇒ 'r set" where
  "afiber data F c k = {y ∈ lower_maximal data. frows F y ∧ rproj c y = k}"

text ‹‹aggregate R f E› folds the inner values over the inner keys below each key of the frame,
  the inner expression being evaluated under the frame refined by the extra requested dimensions
  ‹R› (a grouping inside the aggregate).  The inner keys are those where the inner expression is
  present: its own domain, so for a lawful inner expression they are data keys.  Its keys are the
  frame's, so it decides no detail: the lower parts of the inner entries move into the
  views' key parts, and the fold reads the fiber's classes at ‹emind E ⊔ R›.›

definition inner_keys :: "'k semi_factor ⇒ ('k, 'f, 'r, 'v) expr ⇒ 'r lower_pd ⇒ ('k semi_factor, 'f, 'r) frame ⇒ 'k ⇒ 'k set" where
  "inner_keys R E data F k = {κ. proj (kd E (finclude R F))⋅κ = κ ∧ proj (flod F)⋅κ = k ∧
     esem E data (finclude R F) κ ≠ None}"

text ‹For a lawful inner expression at a conforming dataset, its keys below ‹k› are the defined ones
  among the inner keys of the fiber of ‹k›.›

lemma inner_keys_fiber:
  assumes EL: "lawful E" and d: "conforms data"
  shows "inner_keys R E data F k = {κ ∈ rproj (kd E (finclude R F)) ` afiber data F (flod F) k.
      esem E data (finclude R F) κ ≠ None}"
proof -
  let ?X = "kd E (finclude R F)"
  have le: "flod F ⊑ ?X" by (simp add: flod_finclude below_trans[OF lattice_join_ge1 lattice_join_ge2])
  show ?thesis
  proof (rule set_eqI, rule iffI)
    fix κ assume "κ ∈ inner_keys R E data F k"
    then have kfix: "proj ?X⋅κ = κ" and below: "proj (flod F)⋅κ = k" and kdef: "esem E data (finclude R F) κ ≠ None"
      by (simp_all add: inner_keys_def)
    obtain x where x: "x ∈ lower_maximal data" "frows F x" and kx: "κ = rproj ?X x"
      using lawful_key_law[OF EL d kdef] by (auto simp: keyimg_def)
    have "rproj (flod F) x = k" using below unfolding kx by (simp only: proj_le[OF le])
    then show "κ ∈ {κ ∈ rproj ?X ` afiber data F (flod F) k. esem E data (finclude R F) κ ≠ None}"
      using x kx kdef by (auto simp: afiber_def)
  next
    fix κ assume "κ ∈ {κ ∈ rproj ?X ` afiber data F (flod F) k. esem E data (finclude R F) κ ≠ None}"
    then obtain x where x: "x ∈ afiber data F (flod F) k" and kx: "κ = rproj ?X x"
      and kdef: "esem E data (finclude R F) κ ≠ None" by blast
    have "rproj (flod F) x = k" using x by (simp add: afiber_def)
    then have "proj (flod F)⋅κ = k" unfolding kx by (simp only: proj_le[OF le])
    then show "κ ∈ inner_keys R E data F k" using kdef by (simp add: inner_keys_def kx)
  qed
qed

text ‹An inner entry, seen from outside the aggregate: its lower part and its requested part move
  into its view's key part, so the outer content is keyed by the inner keys.›

definition agg_entry :: "'k semi_factor ⇒ ('k, 'r) entry ⇒ ('k, 'r) entry" where
  "agg_entry R e = e⦇esel := sel ⊥ (eupper e), eview := (ecols e, ekeys e ⊔ (elower e ⊔ (eupper e ⊓ R)))⦈"

lemma agg_entry_simps [simp]:
  "elower (agg_entry R e) = ⊥" "eupper (agg_entry R e) = eupper e"
  "eguard (agg_entry R e) = eguard e" "eview (agg_entry R e) = (ecols e, ekeys e ⊔ (elower e ⊔ (eupper e ⊓ R)))"
  by (simp_all add: agg_entry_def)

lemma keyed_view_raise: "keyed_view (c, a) ⟹ keyed_view (c, a ⊔ W)"
proof -
  assume "keyed_view (c, a)"
  then have "keyed_view ((c, a) ⊔ (⊥, W))" by (rule keyed_view_sup[OF _ keyed_view_key])
  then show ?thesis by (simp only: lattice_join_prod fst_conv snd_conv lattice_join_bot_right)
qed

lemma entry_wf_agg_entry: "entry_wf e ⟹ entry_wf (agg_entry R e)"
  unfolding entry_wf_def agg_entry_simps using keyed_view_raise[of "ecols e" "ekeys e"] by simp

text ‹The content semantics folds the inner content semantics over the fold entry's keys, reading
  each inner entry's content at a key off the outer content: the pairs whose key has the inner key's
  part at the entry's lower and requested parts, cut back to the inner key part.›

definition agg_read :: "'k semi_factor ⇒ ('k, 'r) entry ⇒ 'k ⇒ ('k × 'r) set ⇒ ('k × 'r) set" where
  "agg_read R e p X = (λx. (proj (ekeys e)⋅(fst x), snd x)) `
     {x ∈ X. proj (elower e ⊔ (eupper e ⊓ R))⋅(fst x) = proj (elower e ⊔ (eupper e ⊓ R))⋅p}"

text ‹On maps the read picks, at each inner key, the one outer key of the slice that projects to
  it, and drops the inner key when there is none or more than one.  Which keys collide is decided by
  the domain alone, so the read is monotone on all maps; on contents nothing collides.›

definition agg_src :: "'k semi_factor ⇒ ('k, 'r) entry ⇒ 'k ⇒ ('k ⇒ 'r option) ⇒ 'k ⇒ 'k ⇒ bool" where
  "agg_src R e p X q q0 ⟷ q0 ∈ dom X ∧
     proj (elower e ⊔ (eupper e ⊓ R))⋅q0 = proj (elower e ⊔ (eupper e ⊓ R))⋅p ∧ proj (ekeys e)⋅q0 = q"

definition agg_pick :: "'k semi_factor ⇒ ('k, 'r) entry ⇒ 'k ⇒ ('k ⇒ 'r option) ⇒ 'k ⇒ 'r option" where
  "agg_pick R e p X q = (if ∃!q0. agg_src R e p X q q0 then X (THE q0. agg_src R e p X q q0) else None)"

definition agg_inner :: "'k semi_factor ⇒ ('k, 'r) entry list ⇒ ('k ⇒ 'r option) list ⇒ 'k ⇒ ('k ⇒ 'r option) list" where
  "agg_inner R es Xs p = map2 (λe X. agg_pick R e p X) es Xs"

text ‹A graph is ∗‹shaped› for an entry when it is single-valued with keys at the entry's key part.
  Outer keys at the outer key part are determined by their parts at the inner key part and at the
  lower and requested parts, so on shaped outer graphs the pick agrees with the graph read.›

definition gshaped :: "('k, 'r) entry ⇒ ('k × 'r) set ⇒ bool" where
  "gshaped e X ⟷ single_valued X ∧ (∀x∈X. proj (ekeys e)⋅(fst x) = fst x)"

lemma cgraph_gshaped: "entry_wf e ⟹ conforms data ⟹ gshaped e (cgraph data e F k)"
  unfolding gshaped_def using cgraph_single_valued[of e data F k] by (auto simp: cgraph_def cpair_def)

lemma agg_read_key_eq:
  assumes X: "gshaped (agg_entry R e) X" and Y: "gshaped (agg_entry R e) Y"
    and q: "(q, v) ∈ X" and q': "(q', w) ∈ Y"
    and A: "proj (elower e ⊔ (eupper e ⊓ R))⋅q = proj (elower e ⊔ (eupper e ⊓ R))⋅q'"
    and s: "proj (ekeys e)⋅q = proj (ekeys e)⋅q'"
  shows "q = q'"
proof -
  let ?S = "ekeys e ⊔ (elower e ⊔ (eupper e ⊓ R))"
  have "q = proj ?S⋅q" "q' = proj ?S⋅q'" using X Y q q' by (force simp: gshaped_def)+
  moreover have "proj ?S⋅q = proj ?S⋅q'" using A s by (simp only: proj_sup_eq)
  ultimately show ?thesis by simp
qed

lemma agg_read_single_valued:
  assumes X: "gshaped (agg_entry R e) X"
  shows "single_valued (agg_read R e p X)"
proof (rule single_valuedI)
  fix q u u' assume u: "(q, u) ∈ agg_read R e p X" and u': "(q, u') ∈ agg_read R e p X"
  obtain q1 where 1: "(q1, u) ∈ X" "q = proj (ekeys e)⋅q1"
    "proj (elower e ⊔ (eupper e ⊓ R))⋅q1 = proj (elower e ⊔ (eupper e ⊓ R))⋅p"
    using u unfolding agg_read_def by force
  obtain q2 where 2: "(q2, u') ∈ X" "q = proj (ekeys e)⋅q2"
    "proj (elower e ⊔ (eupper e ⊓ R))⋅q2 = proj (elower e ⊔ (eupper e ⊓ R))⋅p"
    using u' unfolding agg_read_def by force
  have "q1 = q2" by (rule agg_read_key_eq[OF X X 1(1) 2(1)]) (use 1(2,3) 2(2,3) in simp_all)
  then show "u = u'" using single_valuedD[of X] X 1(1) 2(1) by (simp add: gshaped_def)
qed

lemma agg_pick_graph:
  assumes X: "gshaped (agg_entry R e) (graph X)"
  shows "agg_pick R e p X q = gmap (agg_read R e p (graph X)) q"
proof -
  let ?A = "elower e ⊔ (eupper e ⊓ R)" and ?P = "agg_src R e p X q"
  have uniq: "q1 = q2" if 1: "?P q1" and 2: "?P q2" for q1 q2
  proof -
    obtain v w where v: "X q1 = Some v" and w: "X q2 = Some w" using 1 2 by (auto simp: agg_src_def)
    show ?thesis
      by (rule agg_read_key_eq[OF X X, of q1 v q2 w]) (use 1 2 v w in ‹simp_all add: graph_def agg_src_def›)
  qed
  show ?thesis
  proof (cases "∃q0. ?P q0")
    case True
    then obtain q0 where q0: "?P q0" by blast
    then obtain v where v: "X q0 = Some v" by (auto simp: agg_src_def)
    have ex1: "∃!q0. ?P q0" using q0 uniq by blast
    have th: "(THE q0. ?P q0) = q0" using q0 uniq by blast
    have "(q, v) ∈ agg_read R e p (graph X)"
      unfolding agg_read_def using q0 v by (force simp: graph_def agg_src_def intro: rev_image_eqI)
    then have "gmap (agg_read R e p (graph X)) q = Some v" by (rule gmap_apply[OF agg_read_single_valued[OF X]])
    then show ?thesis unfolding agg_pick_def using ex1 th v by simp
  next
    case False
    then have "q ∉ fst ` agg_read R e p (graph X)" unfolding agg_read_def by (force simp: graph_def agg_src_def)
    then show ?thesis unfolding agg_pick_def gmap_def using False by auto
  qed
qed

definition agg_csem :: "'k semi_factor ⇒ ('v::pcpo → 'v → 'v) ⇒ ('k, 'f, 'r, 'v) expr ⇒ ('k ⇒ 'r option) list ⇒ 'v option" where
  "agg_csem R f E Xs = agg_fold f (λp. csem E (agg_inner R (edims E) (tl Xs) p))
     {p ∈ dom (hd Xs). csem E (agg_inner R (edims E) (tl Xs) p) ≠ None}"

lemma map2_map: "map2 g xs (map h xs) = map (λx. g x (h x)) xs"
  by (induct xs) simp_all

definition aggregate :: "'k semi_factor ⇒ ('v::pcpo → 'v → 'v) ⇒ ('k, 'f, 'r, 'v) expr ⇒ ('k, 'f, 'r, 'v) expr" where
  "aggregate R f E = ⦇emind = ⊥,
     edims = mk_entry (sel ⊥ lattice_top) gtrue (⊥, emind E ⊔ R) # map (agg_entry R) (edims E),
     esem = λdata F k. agg_fold f (esem E data (finclude R F)) (inner_keys R E data F k),
     csem = agg_csem R f E⦈"

lemma aggregate_simps [simp]:
  "emind (aggregate R f E) = ⊥"
  "edims (aggregate R f E) = mk_entry (sel ⊥ lattice_top) gtrue (⊥, emind E ⊔ R) # map (agg_entry R) (edims E)"
  "esem (aggregate R f E) data F k = agg_fold f (esem E data (finclude R F)) (inner_keys R E data F k)"
  "csem (aggregate R f E) = agg_csem R f E"
  by (simp_all add: aggregate_def)

text ‹The fold entry reads the fiber.›

lemma agg_fiber: "proj (flod F)⋅k = k ⟹ group data ⊥ lattice_top F k = afiber data F (flod F) k"
  by (simp add: group_top[of ⊥, simplified lattice_join_bot_left] afiber_def lattice_join_bot_left)

text ‹An inner group is the outer group through the same upper part, cut down by the inner key's
  lower part and requested part.›

lemma agg_group:
  assumes y: "rproj (flod F) y = k" and am: "a ⊑ m"
  shows "group data a ρ (finclude R F) (rproj (m ⊔ flod (finclude R F)) y)
      = {y' ∈ group data ⊥ ρ F k. rproj (a ⊔ (ρ ⊓ R)) y' = rproj (a ⊔ (ρ ⊓ R)) y}"
proof -
  let ?X = "m ⊔ flod (finclude R F)" and ?A = "a ⊔ (ρ ⊓ R)" and ?L = "(ρ ⊓ flod F)"
  have lod: "a ⊔ (ρ ⊓ flod (finclude R F)) = ?A ⊔ ?L" by (simp add: flod_finclude truncate_sup lattice_join_ac)
  have mR: "m ⊔ R ⊑ ?X" by (simp add: flod_finclude lattice_join_mono lattice_join_ge2)
  have AX: "?A ⊑ ?X" by (rule below_trans[OF lattice_join_mono[OF am truncate_below] mR])
  have LX: "?L ⊑ ?X"
    by (rule below_trans[OF truncate_below]) (simp add: flod_finclude below_trans[OF lattice_join_ge1 lattice_join_ge2])
  have kA: "proj ?A⋅(rproj ?X y) = rproj ?A y" by (rule proj_le[OF AX])
  have kL: "proj ?L⋅(rproj ?X y) = proj ?L⋅k"
  proof -
    have "proj ?L⋅(rproj ?X y) = rproj ?L y" by (rule proj_le[OF LX])
    also have "… = proj ?L⋅(rproj (flod F) y)" by (rule proj_le[OF truncate_below, symmetric])
    finally show ?thesis by (simp only: y)
  qed
  show ?thesis unfolding group_def lod proj_sup_eq kA kL by (auto simp: lattice_join_bot_left proj_bot)
qed

text ‹Inner keys of a fiber correspond to their parts at ‹m ⊔ R›.›

lemma agg_keys_bij:
  "bij_betw (Rep_cfun (proj (m ⊔ R))) (rproj (m ⊔ flod (finclude R F)) ` afiber data F (flod F) k)
      (rproj (m ⊔ R) ` afiber data F (flod F) k)"
proof -
  let ?X = "m ⊔ flod (finclude R F)" and ?Z = "afiber data F (flod F) k"
  have le: "m ⊔ R ⊑ ?X" by (simp add: flod_finclude lattice_join_mono lattice_join_ge2)
  have pr: "proj (m ⊔ R)⋅(rproj ?X y) = rproj (m ⊔ R) y" for y by (rule proj_le[OF le])
  show ?thesis
  proof (rule bij_betw_imageI)
    show "inj_on (Rep_cfun (proj (m ⊔ R))) (rproj ?X ` ?Z)"
    proof (rule inj_onI)
      fix κ1 κ2 assume k1: "κ1 ∈ rproj ?X ` ?Z" and k2: "κ2 ∈ rproj ?X ` ?Z"
        and e: "proj (m ⊔ R)⋅κ1 = proj (m ⊔ R)⋅κ2"
      obtain y1 where y1: "y1 ∈ ?Z" "κ1 = rproj ?X y1" using k1 by blast
      obtain y2 where y2: "y2 ∈ ?Z" "κ2 = rproj ?X y2" using k2 by blast
      have "rproj (m ⊔ R) y1 = rproj (m ⊔ R) y2" using e unfolding y1(2) y2(2) pr .
      moreover have "rproj (flod F) y1 = rproj (flod F) y2" using y1(1) y2(1) by (simp add: afiber_def)
      ultimately have "rproj ((m ⊔ R) ⊔ flod F) y1 = rproj ((m ⊔ R) ⊔ flod F) y2"
        by (simp only: rproj_sup_eq)
      moreover have "(m ⊔ R) ⊔ flod F = ?X" by (simp add: flod_finclude lattice_join_ac)
      ultimately show "κ1 = κ2" by (simp add: y1(2) y2(2))
    qed
    show "Rep_cfun (proj (m ⊔ R)) ` rproj ?X ` ?Z = rproj (m ⊔ R) ` ?Z"
      by (simp only: image_image pr)
  qed
qed

text ‹The content graph of an inner entry at an inner key is read off the outer entry's: the
  pairs whose key has the inner key's part at the entry's lower and requested parts.›

lemma agg_content_inner:
  assumes ES: "semilawful E" and e: "e ∈ set (edims E)" and yZ: "y ∈ afiber data G (flod G) kG"
    and κ: "κ = rproj (emind E ⊔ flod (finclude R G)) y" and py: "proj (emind E ⊔ R)⋅κ = p"
  shows "cgraph data e (finclude R G) κ = agg_read R e p (cgraph data (agg_entry R e) G kG)"
proof -
  let ?m = "emind E" and ?a = "elower e" and ?ρ = "eupper e" and ?s = "ekeys e"
  have am: "?a ⊑ ?m" by (rule lower_below_mind[OF ES e])
  let ?A = "?a ⊔ (?ρ ⊓ R)"
  have AmR: "?A ⊑ ?m ⊔ R" by (rule lattice_join_mono[OF am truncate_below])
  have mR: "?m ⊔ R ⊑ ?m ⊔ flod (finclude R G)" by (simp add: flod_finclude lattice_join_mono lattice_join_ge2)
  have yA: "rproj ?A y = proj ?A⋅p"
  proof -
    have "proj ?A⋅p = proj ?A⋅κ" unfolding py[symmetric] by (rule proj_le[OF AmR])
    also have "… = rproj ?A y" unfolding κ by (rule proj_le[OF below_trans[OF AmR mR]])
    finally show ?thesis by simp
  qed
  have gy: "rproj (flod G) y = kG" using yZ by (simp add: afiber_def)
  have grp: "group data ?a ?ρ (finclude R G) κ = {y' ∈ group data ⊥ ?ρ G kG. rproj ?A y' = proj ?A⋅p}"
    unfolding κ agg_group[OF gy am] yA ..
  have pA: "proj ?A⋅(rproj (?s ⊔ ?A) y') = rproj ?A y'" for y' by (rule proj_le[OF lattice_join_ge2])
  have ps: "proj ?s⋅(rproj (?s ⊔ ?A) y') = rproj ?s y'" for y' by (rule proj_le[OF lattice_join_ge1])
  let ?h = "λy'. (rproj (?s ⊔ ?A) y', y' ⇂ ecols e)"
  let ?Y = "efiber data (agg_entry R e) G kG"
  have "{x ∈ ?h ` ?Y. proj ?A⋅(fst x) = proj ?A⋅p} = ?h ` {y' ∈ ?Y. rproj ?A y' = proj ?A⋅p}"
    by (auto simp: pA)
  then have "agg_read R e p (cgraph data (agg_entry R e) G kG)
      = cpair e ` {y' ∈ ?Y. rproj ?A y' = proj ?A⋅p}"
    unfolding agg_read_def cgraph_def cpair_def agg_entry_simps fst_conv snd_conv by (simp add: image_image ps)
  moreover have "{y' ∈ ?Y. rproj ?A y' = proj ?A⋅p} = efiber data e (finclude R G) κ"
    unfolding efiber_def grp by auto
  ultimately show ?thesis unfolding cgraph_def by simp
qed

theorem lawful_aggregate:
  assumes EL: "lawful E" and f: "abel_semigroup (λa b. f⋅a⋅b)"
  shows "lawful (aggregate R f E)"
proof -
  have ES: "semilawful E" by (rule lawful_semilawful[OF EL])
  let ?m = "emind E"
  let ?h = "mk_entry (sel ⊥ lattice_top) gtrue (⊥, ?m ⊔ R) :: ('k, 'r) entry"
  have bj: "⊥ ⊔ X = X" for X :: "'k semi_factor" by (rule lattice_join_bot_left)
  note fiber = agg_fiber
  note bij = agg_keys_bij[of ?m R]
  show ?thesis
  proof (rule lawfulI, simp_all only: aggregate_simps)
    show "entry_wf e" if "e ∈ set (?h # map (agg_entry R) (edims E))" for e
      using that by (auto intro: entry_wf_view keyed_view_key supported_gtrue entry_wf_agg_entry semilawful_wf[OF ES])
    show "(eupper e ⊓ ⊥) ⊑ elower e" if "e ∈ set (?h # map (agg_entry R) (edims E))" for e
      using that by (auto simp: lattice_meet_bot_right)
    show "(⊥::'k semi_factor) = fold (⊔) (map elower (?h # map (agg_entry R) (edims E))) ⊥"
    proof -
      have l: "map elower (?h # map (agg_entry R) (edims E)) = map (λ_. ⊥) (?h # map (agg_entry R) (edims E))"
        by simp
      show ?thesis by (simp only: l fold_join_bots)
    qed
    show "keyimg data F (⊥ ⊔ flod F) k"
      if d: "conforms data" and "agg_fold f (esem E data (finclude R F)) (inner_keys R E data F k) ≠ None"
      for data F k
    proof -
      have "inner_keys R E data F k ≠ {}" using agg_fold_nonempty[OF f that(2)] by auto
      then have "afiber data F (flod F) k ≠ {}" by (auto simp: inner_keys_fiber[OF EL d])
      then show ?thesis by (auto simp: afiber_def keyimg_def bj)
    qed
    show "agg_fold f (esem E data (finclude R F)) (inner_keys R E data F k)
        = agg_csem R f E (contents data (?h # map (agg_entry R) (edims E)) F k)"
      if d: "conforms data" and k: "proj (⊥ ⊔ flod F)⋅k = k" for data F k
    proof -
      have k1: "proj (flod F)⋅k = k" using k by (simp only: bj)
      let ?Z = "afiber data F (flod F) k"
      let ?K = "rproj (?m ⊔ flod (finclude R F)) ` ?Z"
      let ?P = "rproj (?m ⊔ R) ` ?Z" and ?π = "Rep_cfun (proj (?m ⊔ R))"
      let ?v = "esem E data (finclude R F)"
      let ?X = "contents data (map (agg_entry R) (edims E)) F k"
      let ?Φ = "λp. csem E (agg_inner R (edims E) ?X p)"
      have hd: "dom (hd (contents data (?h # map (agg_entry R) (edims E)) F k)) = ?P"
        by (simp add: content_def cgraph_def cpair_def efiber_def fiber[OF k1] image_image)
      have tl: "tl (contents data (?h # map (agg_entry R) (edims E)) F k) = ?X"
        by (simp only: list.map(2) list.sel(3))
      define ψ where "ψ = inv_into ?K ?π"
      have b: "bij_betw ψ ?P ?K" unfolding ψ_def by (rule bij_betw_inv_into[OF bij])
      have sP: "?π ` ?K = ?P" by (rule bij_betw_imp_surj_on[OF bij])
      have pt: "?Φ p = ?v (ψ p)" if p: "p ∈ ?P" for p
      proof -
        have pK: "ψ p ∈ ?K" unfolding ψ_def by (rule inv_into_into) (simp only: sP p)
        obtain y where y: "y ∈ ?Z" "ψ p = rproj (?m ⊔ flod (finclude R F)) y" using pK by blast
        have py: "proj (?m ⊔ R)⋅(ψ p) = p" unfolding ψ_def by (rule f_inv_into_f) (simp only: sP p)
        have "agg_inner R (edims E) ?X p = map (λe. agg_pick R e p (content data (agg_entry R e) F k)) (edims E)"
          unfolding agg_inner_def map_map map2_map by (simp add: comp_def)
        also have "… = contents data (edims E) (finclude R F) (ψ p)"
        proof (rule map_cong[OF refl])
          fix e assume e: "e ∈ set (edims E)"
          have wf: "entry_wf (agg_entry R e)" by (rule entry_wf_agg_entry[OF semilawful_wf[OF ES e]])
          have g: "graph (content data (agg_entry R e) F k) = cgraph data (agg_entry R e) F k"
            by (rule graph_content[OF wf d])
          have sh: "gshaped (agg_entry R e) (graph (content data (agg_entry R e) F k))"
            unfolding g by (rule cgraph_gshaped[OF wf d])
          show "agg_pick R e p (content data (agg_entry R e) F k) = content data e (finclude R F) (ψ p)"
          proof
            fix q
            have "agg_pick R e p (content data (agg_entry R e) F k) q = gmap (agg_read R e p (cgraph data (agg_entry R e) F k)) q"
              by (simp only: agg_pick_graph[OF sh] g)
            then show "agg_pick R e p (content data (agg_entry R e) F k) q = content data e (finclude R F) (ψ p) q"
              by (simp only: content_def agg_content_inner[OF ES e y(1) y(2) py])
          qed
        qed
        finally have c: "agg_inner R (edims E) ?X p = contents data (edims E) (finclude R F) (ψ p)" .
        have "proj (kd E (finclude R F))⋅(ψ p) = ψ p" by (simp only: y(2) proj_idem)
        then show ?thesis by (simp only: c semilawful_csem[OF ES d])
      qed
      let ?Q = "{p ∈ ?P. ?v (ψ p) ≠ None}"
      have D: "inner_keys R E data F k = ψ ` ?Q"
        unfolding inner_keys_fiber[OF EL d] bij_betw_imp_surj_on[OF b, symmetric] by auto
      have T: "{p ∈ ?P. ?Φ p ≠ None} = ?Q" using pt by auto
      have bQ: "bij_betw ψ ?Q (ψ ` ?Q)"
        by (rule inj_on_imp_bij_betw[OF inj_on_subset[OF bij_betw_imp_inj_on[OF b]]]) blast
      have "agg_fold f ?v (ψ ` ?Q) = agg_fold f (?v ∘ ψ) ?Q"
        by (rule agg_fold_bij[OF f bQ, symmetric])
      also have "… = agg_fold f ?Φ ?Q"
      proof (rule agg_fold_cong[OF f refl])
        fix p assume "p ∈ ?Q"
        then show "(?v ∘ ψ) p = ?Φ p" by (simp only: comp_apply pt mem_Collect_eq)
      qed
      finally show ?thesis unfolding agg_csem_def hd tl T D .
    qed
  qed
qed
