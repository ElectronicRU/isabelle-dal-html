Theory Ast

theory Ast
  imports Expr_Combinators Value
begin

section ‹Expression syntax›

text ‹The abstract syntax of the expression language, after name resolution: references to
  dimensions and to table columns, literals, calls of function symbols, aggregations, requested
  grouping (‹GROUP BY›), explicit cuts (‹FOR›) and filters (‹WHERE›).  Values are concrete:
  literals and expressions are data (@{typ "scalar datum"}), dimensions are keys
  (@{typ "scalar with_na"}).  The conditions of cuts are boolean expressions on dimensions, which
  serve as slices; filters are row-level boolean expressions, which may read columns.
  Not yet covered: ‹DISTINCT›, ‹ISOLATE›, ‹TOTAL›, and the dimension kind (which needs
  staging).›

text ‹Function symbols: the connectives, equality and the ‹NA› test are known to the language,
  for normal forms, coarsening and pins; other symbols are interpreted by a table.›

datatype 'fn fsym = FAnd | FOr | FNot | FEq | FIsNA | FOther 'fn

datatype ('d, 'c, 'fn, 'ag) ast =
  Dim 'd
| Col 'c
| Lit "scalar datum"
| Call "'fn fsym" "('d, 'c, 'fn, 'ag) ast list"
| Aggr 'ag "('d, 'c, 'fn, 'ag) ast"
| GroupBy "'d list" "('d, 'c, 'fn, 'ag) ast"
| For "'d list" "('d, 'c, 'fn, 'ag) ast" "('d, 'c, 'fn, 'ag) ast"
| Where "('d, 'c, 'fn, 'ag) ast" "('d, 'c, 'fn, 'ag) ast"

section ‹The known function symbols›

text ‹Booleans as data, and the n-ary connectives: strict, so every missing argument is
  infectious.  The empty conjunction is true, the empty disjunction false.›

definition d_bool :: "tr → scalar datum" where
  "d_bool = na_emb oo Known oo SBool"

fixrec d_ands :: "scalar datum ltuple → scalar datum" where
  "d_ands⋅LNil = d_bool⋅TT"
| "xs ≠ ⊥ ⟹ d_ands⋅(LCons⋅x⋅xs) = dlift2⋅sand⋅x⋅(d_ands⋅xs)"

fixrec d_ors :: "scalar datum ltuple → scalar datum" where
  "d_ors⋅LNil = d_bool⋅FF"
| "xs ≠ ⊥ ⟹ d_ors⋅(LCons⋅x⋅xs) = dlift2⋅sor⋅x⋅(d_ors⋅xs)"

fixrec arg1 :: "('a::domain → 'b::domain) → 'a ltuple → 'b" where
  "arg1⋅f⋅(LCons⋅x⋅LNil) = f⋅x"

fixrec arg2 :: "('a::domain → 'a → 'b::domain) → 'a ltuple → 'b" where
  "arg2⋅f⋅(LCons⋅x⋅(LCons⋅y⋅LNil)) = f⋅x⋅y"

definition d_eq :: "scalar datum → scalar datum → scalar datum" where
  "d_eq = dlift2⋅(Λ x y. SBool⋅(scalar_eq⋅x⋅y))"

definition d_is_na :: "scalar datum → scalar datum" where
  "d_is_na = (Λ x. d_bool⋅(is_na⋅x))"

primrec fsym_sem :: "('fn ⇒ scalar datum ltuple → scalar datum) ⇒ 'fn fsym ⇒ scalar datum ltuple → scalar datum" where
  "fsym_sem other FAnd = d_ands"
| "fsym_sem other FOr = d_ors"
| "fsym_sem other FNot = arg1⋅(dlift1⋅snot)"
| "fsym_sem other FEq = arg2⋅d_eq"
| "fsym_sem other FIsNA = arg1⋅d_is_na"
| "fsym_sem other (FOther fn) = other fn"

primrec fsym_arity :: "('fn ⇒ nat ⇒ bool) ⇒ 'fn fsym ⇒ nat ⇒ bool" where
  "fsym_arity other FAnd n ⟷ True"
| "fsym_arity other FOr n ⟷ True"
| "fsym_arity other FNot n ⟷ n = 1"
| "fsym_arity other FEq n ⟷ n = 2"
| "fsym_arity other FIsNA n ⟷ n = 1"
| "fsym_arity other (FOther fn) n ⟷ other fn n"

lemma truth_d_bool [simp]: "truth_d⋅(d_bool⋅t) = t"
  by (cases t rule: trE) (simp_all add: d_bool_def truth_d_def)

text ‹Truth of the connectives: a conjunction is true where all conjuncts are; a disjunction
  where all disjuncts are defined and one is true.  Both are defined where all arguments are.›

lemma tr_and_TT_iff: "tr_and⋅x⋅y = TT ⟷ x = TT ∧ y = TT"
  by (cases x rule: trE; cases y rule: trE) (simp_all add: TT_def FF_def)

lemma tr_and_bot_iff: "tr_and⋅x⋅y = ⊥ ⟷ x = ⊥ ∨ y = ⊥"
  by (cases x rule: trE; cases y rule: trE) (simp_all add: TT_def FF_def)

lemma tr_or_TT_iff: "tr_or⋅x⋅y = TT ⟷ x ≠ ⊥ ∧ y ≠ ⊥ ∧ (x = TT ∨ y = TT)"
  by (cases x rule: trE; cases y rule: trE) (simp_all add: TT_def FF_def)

lemma tr_or_bot_iff: "tr_or⋅x⋅y = ⊥ ⟷ x = ⊥ ∨ y = ⊥"
  by (cases x rule: trE; cases y rule: trE) (simp_all add: TT_def FF_def)

lemma truth_d_ands:
  "truth_d⋅(d_ands⋅(ltuple_of xs)) = TT ⟷ (∀x∈set xs. truth_d⋅x = TT)"
  "truth_d⋅(d_ands⋅(ltuple_of xs)) ≠ ⊥ ⟷ (∀x∈set xs. truth_d⋅x ≠ ⊥)"
  by (induct xs) (simp_all add: ltuple_of_not_bottom truth_d_and[simplified] tr_and_TT_iff tr_and_bot_iff)

lemma truth_d_ors:
  "truth_d⋅(d_ors⋅(ltuple_of xs)) = TT ⟷ (∀x∈set xs. truth_d⋅x ≠ ⊥) ∧ (∃x∈set xs. truth_d⋅x = TT)"
  "truth_d⋅(d_ors⋅(ltuple_of xs)) ≠ ⊥ ⟷ (∀x∈set xs. truth_d⋅x ≠ ⊥)"
  by (induct xs) (auto simp: ltuple_of_not_bottom truth_d_or[simplified] tr_or_TT_iff tr_or_bot_iff)

section ‹Dimensions and the key reading›

text ‹Dimensions name LODs of the keys, each with a value read off the key there.  A dimension's
  value reads only its LOD, and at its LOD it is the whole information: the value orders the
  key's part at the LOD.  Nothing ties dimensions to each other: hierarchies are inclusions of
  their LODs.  Every compact key lists dimensions it lives at, possibly more than it needs (a
  year key may list the day).  Expressions on dimensions alone, with the known and the tabled
  function symbols, read keys directly.›

primrec tuple_cfun :: "('r::cpo → 'v::domain) list ⇒ 'r → 'v ltuple" where
  "tuple_cfun [] = (Λ y. LNil)"
| "tuple_cfun (g # gs) = (Λ y. LCons⋅(g⋅y)⋅(tuple_cfun gs⋅y))"

lemma tuple_cfun_apply: "tuple_cfun gs⋅y = ltuple_of (map (λg. g⋅y) gs)"
  by (induct gs) simp_all

definition key_datum :: "scalar with_na → scalar datum" where
  "key_datum = na_bind⋅(Val oo Known)"

lemma key_datum_simps [simp]:
  "key_datum⋅⊥ = ⊥" "key_datum⋅NA = NA" "key_datum⋅(Val⋅x) = Val⋅(Known⋅x)"
  by (simp_all add: key_datum_def)

locale key_language =
  fixes dim_lod :: "'d ⇒ 'k::lod_domain semi_factor"
    and dim_val :: "'d ⇒ 'k → scalar with_na"
    and key_dims :: "'k ⇒ 'd list"
    and fn_other :: "'fn ⇒ scalar datum ltuple → scalar datum"
  assumes dim_compact: "compact (dim_lod d)"
    and dim_reads: "dim_val d⋅(proj (dim_lod d)⋅k) = dim_val d⋅k"
    and dim_reflects: "dim_val d⋅k ⊑ dim_val d⋅k' ⟹ proj (dim_lod d)⋅k ⊑ proj (dim_lod d)⋅k'"
    and key_dims_lives: "compact K ⟹ proj (fold (⊔) (map dim_lod (key_dims K)) ⊥)⋅K = K"
begin

definition lod :: "'d list ⇒ 'k semi_factor" where
  "lod ds = fold (⊔) (map dim_lod ds) ⊥"

lemma dim_below_iff: "dim_val d⋅k ⊑ dim_val d⋅k' ⟷ proj (dim_lod d)⋅k ⊑ proj (dim_lod d)⋅k'"
  by (metis dim_reflects dim_reads monofun_cfun_arg)

lemma dim_eq_iff: "dim_val d⋅k = dim_val d⋅k' ⟷ proj (dim_lod d)⋅k = proj (dim_lod d)⋅k'"
  by (simp add: po_eq_conv dim_below_iff)

lemma fold_join_least:
  "b ⊑ c ⟹ (⋀x. x ∈ set xs ⟹ x ⊑ c) ⟹ fold (⊔) xs b ⊑ (c :: 'a::lattice_complete)"
  by (induct xs arbitrary: b) (simp_all add: join_belowI)

lemma lod_below_iff: "lod ds ⊑ c ⟷ (∀d∈set ds. dim_lod d ⊑ c)"
proof
  assume "lod ds ⊑ c"
  moreover have "dim_lod d ⊑ lod ds" if "d ∈ set ds" for d
    unfolding lod_def by (rule fold_join_upper) (simp add: that)
  ultimately show "∀d∈set ds. dim_lod d ⊑ c" by (blast intro: below_trans)
qed (unfold lod_def, rule fold_join_least, auto)

lemma compact_lod: "compact (lod ds)"
  unfolding lod_def by (rule compact_fold_sup) (auto simp: dim_compact)

text ‹A key is below another as soon as each of its listed dimensions' values is.›

lemma proj_fold_below:
  assumes "proj b⋅K ⊑ k" and "⋀d. d ∈ set ds ⟹ proj (dim_lod d)⋅K ⊑ k"
  shows "proj (fold (⊔) (map dim_lod ds) b)⋅K ⊑ k"
  using assms by (induct ds arbitrary: b) (simp_all add: proj_sup join_belowI)

lemma key_dims_below:
  assumes K: "compact K" and le: "⋀d. d ∈ set (key_dims K) ⟹ dim_val d⋅K ⊑ dim_val d⋅k"
  shows "K ⊑ k"
proof -
  have "proj (fold (⊔) (map dim_lod (key_dims K)) ⊥)⋅K ⊑ k"
  proof (rule proj_fold_below)
    show "proj ⊥⋅K ⊑ k" by (simp add: proj_bot)
    show "proj (dim_lod d)⋅K ⊑ k" if "d ∈ set (key_dims K)" for d
      using dim_reflects[OF le[OF that]] proj_below by (rule below_trans)
  qed
  then show ?thesis by (simp only: key_dims_lives[OF K])
qed

subsection ‹The key reading›

fun kdims :: "('d, 'c, 'fn, 'ag) ast ⇒ 'd list" where
  "kdims (Dim d) = [d]"
| "kdims (Call fn es) = concat (map kdims es)"
| "kdims _ = []"

fun kval :: "('d, 'c, 'fn, 'ag) ast ⇒ 'k → scalar datum" where
  "kval (Dim d) = key_datum oo dim_val d"
| "kval (Lit v) = (Λ _. v)"
| "kval (Call fn es) = fsym_sem fn_other fn oo tuple_cfun (map kval es)"
| "kval _ = ⊥"

definition reads_in :: "'k semi_factor ⇒ ('d, 'c, 'fn, 'ag) ast ⇒ bool" where
  "reads_in A e ⟷ (∀d∈set (kdims e). dim_lod d ⊑ A)"

lemma kval_reads: "reads_in A e ⟹ kval e⋅(proj A⋅k) = kval e⋅k"
proof (induct e rule: kval.induct)
  case (1 d)
  have "dim_val d⋅(proj A⋅k) = dim_val d⋅(proj (dim_lod d)⋅(proj A⋅k))" by (simp only: dim_reads)
  also have "… = dim_val d⋅k" using 1 by (simp add: reads_in_def proj_le dim_reads)
  finally show ?case by simp
next
  case (3 fn es)
  then have "map (λe. kval e⋅(proj A⋅k)) es = map (λe. kval e⋅k) es" by (auto simp: reads_in_def)
  then show ?case by (simp only: kval.simps cfcomp2 tuple_cfun_apply map_map comp_def)
qed simp_all

definition kholds :: "('d, 'c, 'fn, 'ag) ast ⇒ 'k ⇒ bool" where
  "kholds e k ⟷ truth_d⋅(kval e⋅k) = TT"

lemma kholds_up: "kholds e k ⟹ k ⊑ k' ⟹ kholds e k'"
  unfolding kholds_def by (metis TT_below_iff monofun_cfun_arg)

lemma kholds_open: "adm (λk. ¬ kholds e k)"
  unfolding kholds_def by (simp add: compact_TT eq_commute[of _ TT])

lemma reads_in_meet: "reads_in (A ⊓ B) e ⟷ reads_in A e ∧ reads_in B e"
  unfolding reads_in_def
  by (blast intro: lattice_meet_greatest below_trans[OF _ lattice_meet_le1] below_trans[OF _ lattice_meet_le2])

subsection ‹Coarsening conditions›

text ‹Coarsening a condition to an LOD drops what reads outside it, as in the source:
  conjunctions keep their surviving conjuncts (and vanish when none survives), disjunctions
  vanish as soon as one disjunct does, anything else survives whole or not at all.  Dropping
  only ever replaces a part in positive position by truth, so it weakens: what was true stays
  true, what was defined stays defined.  Negation is not looked into; normal forms push it to
  the literals first.›

fun crs :: "'k semi_factor ⇒ ('d, 'c, 'fn, 'ag) ast ⇒ ('d, 'c, 'fn, 'ag) ast option" where
  "crs A (Call FAnd es) = (if es ≠ [] ∧ filter (λr. r ≠ None) (map (crs A) es) = [] then None
     else Some (Call FAnd (map the (filter (λr. r ≠ None) (map (crs A) es)))))"
| "crs A (Call FOr es) = (if None ∈ set (map (crs A) es) then None
     else Some (Call FOr (map the (map (crs A) es))))"
| "crs A e = (if reads_in A e then Some e else None)"

lemma set_kept: "set (map the (filter (λr. r ≠ None) (map f xs))) = {y. ∃x∈set xs. f x = Some y}"
  by (induct xs) auto

lemma set_all_some:
  "None ∉ set (map f xs) ⟹ set (map the (map f xs)) = {y. ∃x∈set xs. f x = Some y}"
  by (induct xs) auto

lemma crs_inside: "reads_in A e ⟹ crs A e = Some e"
proof (induct A e rule: crs.induct)
  case (1 A es)
  have "map (crs A) es = map Some es" using 1(1) 1(3) by (simp add: reads_in_def)
  then show ?case by (simp add: comp_def filter_map cong: map_cong)
next
  case (2 A es)
  have "map (crs A) es = map Some es" using 2(1) 2(3) by (simp add: reads_in_def)
  then show ?case by (simp add: comp_def cong: map_cong)
qed simp_all

lemma crs_reads: "crs A e = Some e' ⟹ reads_in A e' ∧ set (kdims e') ⊆ set (kdims e)"
proof (induct A e arbitrary: e' rule: crs.induct)
  case (1 A es)
  have IH: "reads_in A c' ∧ set (kdims c') ⊆ set (kdims c)" if "c ∈ set es" "crs A c = Some c'" for c c'
    using 1(1)[OF that] .
  obtain es' where e': "e' = Call FAnd es'" and es': "set es' = {c'. ∃c∈set es. crs A c = Some c'}"
    using 1(3) set_kept[of "crs A" es] by (auto split: if_splits)
  show ?case unfolding e' using es' IH by (fastforce simp: reads_in_def)
next
  case (2 A es)
  have IH: "reads_in A c' ∧ set (kdims c') ⊆ set (kdims c)" if "c ∈ set es" "crs A c = Some c'" for c c'
    using 2(1)[OF that] .
  obtain es' where e': "e' = Call FOr es'" and es': "set es' = {c'. ∃c∈set es. crs A c = Some c'}"
    using 2(3) set_all_some[of "crs A" es] by (auto split: if_splits)
  show ?case unfolding e' using es' IH by (fastforce simp: reads_in_def)
qed (auto split: if_splits)

lemma kept_kept:
  "map the (filter (λr. r ≠ None) (map g (map the (filter (λr. r ≠ None) (map f xs)))))
    = map the (filter (λr. r ≠ None) (map (λx. Option.bind (f x) g) xs))"
  by (induct xs) (auto split: option.split)

lemma all_some_bind:
  "None ∉ set (map f xs) ⟹ map g (map the (map f xs)) = map (λx. Option.bind (f x) g) xs"
  by (induct xs) (auto, metis option.exhaust_sel bind.simps(2))

lemma crs_bind: "Option.bind (crs B e) (crs A) = crs (A ⊓ B) e"
proof (induct B e arbitrary: A rule: crs.induct)
  case (1 B es)
  have IH: "map (crs (A ⊓ B)) es = map (λc. Option.bind (crs B c) (crs A)) es"
    using 1(1) by simp
  show ?case
  proof (cases "es ≠ [] ∧ filter (λr. r ≠ None) (map (crs B) es) = []")
    case True
    then have "filter (λr. r ≠ None) (map (crs (A ⊓ B)) es) = []"
      unfolding IH by (auto simp: filter_empty_conv)
    then show ?thesis using True by simp
  next
    case False
    have kk: "map the (filter (λr. r ≠ None) (map (crs A) (map the (filter (λr. r ≠ None) (map (crs B) es)))))
        = map the (filter (λr. r ≠ None) (map (crs (A ⊓ B)) es))" unfolding kept_kept IH ..
    have nil: "map the xs = [] ⟷ xs = []" for xs :: "'x option list" by simp
    show ?thesis
      unfolding crs.simps(1)[of B] if_not_P[OF False] bind.simps(2) crs.simps(1)[of A] crs.simps(1)[of "A ⊓ B"] kk
      apply (rule if_cong[OF _ refl refl])
      using False arg_cong[OF kk, of "λxs. xs = []"] unfolding nil by (cases es) auto
  qed
next
  case (2 B es)
  have IH: "map (crs (A ⊓ B)) es = map (λc. Option.bind (crs B c) (crs A)) es"
    using 2(1) by simp
  show ?case
  proof (cases "None ∈ set (map (crs B) es)")
    case True
    then have "None ∈ set (map (crs (A ⊓ B)) es)" unfolding IH by force
    then show ?thesis using True by simp
  next
    case False
    have "map (crs A) (map the (map (crs B) es)) = map (crs (A ⊓ B)) es"
      unfolding IH by (rule all_some_bind[OF False])
    then show ?thesis
      unfolding crs.simps(2)[of B] if_not_P[OF False] bind.simps(2) crs.simps(2)[of A] crs.simps(2)[of "A ⊓ B"]
      by simp
  qed
qed (auto simp: reads_in_meet)

lemma kval_call [simp]: "kval (Call fn es)⋅k = fsym_sem fn_other fn⋅(ltuple_of (map (λe. kval e⋅k) es))"
  by (simp add: tuple_cfun_apply comp_def)

text ‹Coarsening weakens: true stays true, defined stays defined.›

lemma crs_weakens:
  "crs A e = Some e' ⟹ (truth_d⋅(kval e⋅k) = TT ⟶ truth_d⋅(kval e'⋅k) = TT) ∧
     (truth_d⋅(kval e⋅k) ≠ ⊥ ⟶ truth_d⋅(kval e'⋅k) ≠ ⊥)"
proof (induct A e arbitrary: e' rule: crs.induct)
  case (1 A es)
  have IH: "(truth_d⋅(kval c⋅k) = TT ⟶ truth_d⋅(kval c'⋅k) = TT) ∧
      (truth_d⋅(kval c⋅k) ≠ ⊥ ⟶ truth_d⋅(kval c'⋅k) ≠ ⊥)" if "c ∈ set es" "crs A c = Some c'" for c c'
    using 1(1)[OF that] .
  obtain es' where e': "e' = Call FAnd es'" and es': "set es' = {c'. ∃c∈set es. crs A c = Some c'}"
    using 1(3) set_kept[of "crs A" es] by (auto split: if_splits)
  show ?case unfolding e' kval_call fsym_sem.simps using es' IH by (auto simp: truth_d_ands)
next
  case (2 A es)
  have IH: "(truth_d⋅(kval c⋅k) = TT ⟶ truth_d⋅(kval c'⋅k) = TT) ∧
      (truth_d⋅(kval c⋅k) ≠ ⊥ ⟶ truth_d⋅(kval c'⋅k) ≠ ⊥)" if "c ∈ set es" "crs A c = Some c'" for c c'
    using 2(1)[OF that] .
  have all: "∃c'. crs A c = Some c'" if "c ∈ set es" for c
    using 2(3) that by (auto split: if_splits) (metis image_eqI option.exhaust)
  obtain es' where e': "e' = Call FOr es'" and es': "set es' = {c'. ∃c∈set es. crs A c = Some c'}"
    using 2(3) set_all_some[of "crs A" es] by (auto split: if_splits)
  have defd: "∀c'∈set es'. truth_d⋅(kval c'⋅k) ≠ ⊥" if "∀c∈set es. truth_d⋅(kval c⋅k) ≠ ⊥"
    using that IH unfolding es' by blast
  have T: "truth_d⋅(kval (Call FOr xs)⋅k) = TT ⟷
      (∀c∈set xs. truth_d⋅(kval c⋅k) ≠ ⊥) ∧ (∃c∈set xs. truth_d⋅(kval c⋅k) = TT)"
    and D: "truth_d⋅(kval (Call FOr xs)⋅k) ≠ ⊥ ⟷ (∀c∈set xs. truth_d⋅(kval c⋅k) ≠ ⊥)" for xs
    unfolding kval_call fsym_sem.simps by (simp_all add: truth_d_ors)
  show ?case unfolding e' T D
  proof (rule conjI; rule impI)
    assume "(∀c∈set es. truth_d⋅(kval c⋅k) ≠ ⊥) ∧ (∃c∈set es. truth_d⋅(kval c⋅k) = TT)"
    then have d: "∀c∈set es. truth_d⋅(kval c⋅k) ≠ ⊥" and "∃c∈set es. truth_d⋅(kval c⋅k) = TT" by blast+
    then obtain c c' where c: "c ∈ set es" "truth_d⋅(kval c⋅k) = TT" and c': "crs A c = Some c'" using all by blast
    have "c' ∈ set es'" using c(1) c' unfolding es' by blast
    then show "(∀c∈set es'. truth_d⋅(kval c⋅k) ≠ ⊥) ∧ (∃c∈set es'. truth_d⋅(kval c⋅k) = TT)"
      using defd[OF d] IH[OF c(1) c'] c(2) by blast
  next
    show "∀c∈set es'. truth_d⋅(kval c⋅k) ≠ ⊥" if "∀c∈set es. truth_d⋅(kval c⋅k) ≠ ⊥"
      by (rule defd[OF that])
  qed
qed (auto split: if_splits)

lemma reads_in_mono: "reads_in A e ⟹ A ⊑ B ⟹ reads_in B e"
  unfolding reads_in_def by (blast intro: below_trans)

lemma reads_in_lod: "reads_in (lod (kdims e)) e"
  unfolding reads_in_def using lod_below_iff[of "kdims e" "lod (kdims e)"] by simp

end

section ‹Conditions as slices›

text ‹A slice is a finite set of conditions, read as their conjunction; it is written at the
  join of their dimensions' LODs.  Pins conjoin a test per dimension the key lists: ‹IS NA›,
  ‹NOT IS NA›, or equality with the key's value, as the value is ‹NA›, the not-null root, or a
  scalar.›

type_synonym ('d, 'c, 'fn, 'ag) slice = "('d, 'c, 'fn, 'ag) ast fset"

lemma lattice_lub_empty: "lub {} = (⊥ :: 'a::lattice_complete)"
  by (rule bottomI) (rule lattice_lub_least, simp)

lemma lattice_lub_Un: "lub (A ∪ B) = lub A ⊔ (lub B :: 'a::lattice_complete)"
proof (rule below_antisym)
  show "lub (A ∪ B) ⊑ lub A ⊔ lub B"
    by (rule lattice_lub_least, rule is_ubI)
      (auto intro: below_trans[OF lattice_lub_upper lattice_join_ge1] below_trans[OF lattice_lub_upper lattice_join_ge2])
  show "lub A ⊔ lub B ⊑ lub (A ∪ B)"
    by (rule join_belowI; rule lattice_lub_least, rule is_ubI, rule lattice_lub_upper) simp_all
qed

lemma lattice_lub_compact:
  "finite S ⟹ (⋀x. x ∈ S ⟹ compact x) ⟹ compact (lub S :: 'k::{bifinite, distributive_cpo} semi_factor)"
proof (induct rule: finite_induct)
  case empty
  then show ?case by (simp add: lattice_lub_empty)
next
  case (insert x S)
  have "lub (insert x S) = lub {x} ⊔ lub S" using lattice_lub_Un[of "{x}" S] by simp
  moreover have "lub {x} = x" by (rule lub_singleton)
  ultimately show ?case using insert by (simp add: compact_semi_factor_sup)
qed

lemma adm_not_ball_finite:
  "finite S ⟹ (⋀e. e ∈ S ⟹ adm (λk. ¬ P e k)) ⟹ adm (λk. ¬ (∀e∈S. P e k))"
  by (induct rule: finite_induct) (simp_all add: adm_disj)

lemma selects_slice_open [simp]: "selects (slice_open S) k ⟷ (∀e∈fset S. kholds e k)"

  have up: "∀x y. (∀e∈fset S. kholds e x) ⟶ x ⊑ y ⟶ (∀e∈fset S. kholds e y)"

  have "(λk. ∀e∈fset S. kholds e k) ∈ {P. (∀x y. P x ⟶ x ⊑ y ⟶ P y) ∧ adm (λx. ¬ P x)}"
    using up adm by (simp only: mem_Collect_eq)

subsection ‹The slicer laws›

lemma slice_lod_union: "slice_lod (S |∪| T) = slice_lod S ⊔ slice_lod T"
  by (simp add: slice_lod_def image_Un lattice_lub_Un)

lemma slice_lod_empty: "slice_lod {||} = ⊥"
  by (simp add: slice_lod_def lattice_lub_empty)

lemma slice_open_union: "slice_open (S |∪| T) = inf (slice_open S) (slice_open T)"
  by (rule sel_eqI) auto

lemma slice_open_empty: "slice_open {||} = top"
  by (rule sel_eqI) simp

lemma lod_below_slice_lod: "e |∈| S ⟹ lod (kdims e) ⊑ slice_lod S"
  unfolding slice_lod_def by (rule lattice_lub_upper) simp

lemma reads_in_slice_lod: "e |∈| S ⟹ reads_in (slice_lod S) e"
  by (rule reads_in_mono[OF reads_in_lod lod_below_slice_lod])

lemma slice_open_lod: "selects (slice_open S) (proj (slice_lod S)⋅k) ⟷ selects (slice_open S) k"
  by (simp add: kholds_def kval_reads[OF reads_in_slice_lod])

lemma compact_slice_lod: "compact (slice_lod S)"
  unfolding slice_lod_def by (rule lattice_lub_compact) (auto simp: compact_lod)

lemma slice_coarsen_union: "slice_coarsen A (S |∪| T) = slice_coarsen A S |∪| slice_coarsen A T"
  by (subst fset_inject[symmetric]) (auto simp: fset_slice_coarsen)

lemma slice_coarsen_inside: "slice_lod S ⊑ A ⟹ slice_coarsen A S = S"
proof (subst fset_inject[symmetric])
  assume A: "slice_lod S ⊑ A"
  have "crs A e = Some e" if "e |∈| S" for e
    by (rule crs_inside[OF reads_in_mono[OF reads_in_slice_lod[OF that] A]])
  then show "fset (slice_coarsen A S) = fset S" by (auto simp: fset_slice_coarsen)
qed

lemma slice_lod_coarsen: "slice_lod (slice_coarsen A S) ⊑ slice_lod S ⊓ A"
  unfolding slice_lod_def
proof (rule lattice_lub_least, rule is_ubI)
  fix x assume "x ∈ (λe. lod (kdims e)) ` fset (slice_coarsen A S)"
  then obtain e e' where e: "e |∈| S" and c: "crs A e = Some e'" and x: "x = lod (kdims e')"
    by (auto simp: fset_slice_coarsen)
  have r: "reads_in A e'" and sub: "set (kdims e') ⊆ set (kdims e)" using crs_reads[OF c] by simp_all
  have "x ⊑ lod (kdims e)" unfolding x lod_below_iff using sub lod_below_iff[of "kdims e" "lod (kdims e)"] by auto
  then have "x ⊑ lub ((λe. lod (kdims e)) ` fset S)"
    using lod_below_slice_lod[OF e] unfolding slice_lod_def by (rule below_trans)
  moreover have "x ⊑ A" unfolding x lod_below_iff using r by (simp add: reads_in_def)
  ultimately show "x ⊑ lub ((λe. lod (kdims e)) ` fset S) ⊓ A" by (rule lattice_meet_greatest)
qed

lemma slice_coarsen_cut:
  "selects (slice_open S) k ⟹ selects (slice_open (slice_coarsen A S)) (proj A⋅k)"
proof (simp only: selects_slice_open fset_slice_coarsen, intro ballI)
  fix e' assume S: "∀e∈fset S. kholds e k" and "e' ∈ {e'. ∃e∈fset S. crs A e = Some e'}"
  then obtain e where e: "e |∈| S" and c: "crs A e = Some e'" by auto
  have "kholds e' k" using S e crs_weakens[OF c, of k] by (simp add: kholds_def)
  then show "kholds e' (proj A⋅k)" using crs_reads[OF c] by (simp add: kholds_def kval_reads)
qed

lemma slice_coarsen_slod: "slice_coarsen A S = slice_coarsen (A ⊓ slice_lod S) S"
proof (subst fset_inject[symmetric])
  have "crs (A ⊓ slice_lod S) e = crs A e" if "e |∈| S" for e
    using crs_bind[of "slice_lod S" e A] crs_inside[OF reads_in_slice_lod[OF that]] by simp
  then show "fset (slice_coarsen A S) = fset (slice_coarsen (A ⊓ slice_lod S) S)"
    by (auto simp: fset_slice_coarsen)
qed

lemma slice_coarsen_coarsen: "slice_coarsen A (slice_coarsen B S) = slice_coarsen (A ⊓ B) S"
proof (subst fset_inject[symmetric])
  have "crs (A ⊓ B) e = Some e'' ⟷ (∃e'. crs B e = Some e' ∧ crs A e' = Some e'')" for e e''
    by (simp only: crs_bind[symmetric] bind_eq_Some_conv)
  then show "fset (slice_coarsen A (slice_coarsen B S)) = fset (slice_coarsen (A ⊓ B) S)"
    unfolding fset_slice_coarsen by blast
qed

text ‹Scalars are flat, so a defined scalar is maximal; so are ‹NA› and a key value
  ‹Val⋅s› for a defined scalar ‹s›.  The upsets of key values are therefore exactly what the
  three tests select.›

lemma scalar_maximal: "s ≠ ⊥ ⟹ s ⊑ x ⟹ x = (s :: scalar)"
  by (cases s; cases x) (auto simp: flat_below_iff)

lemma truth_d_is_na: "truth_d⋅(d_is_na⋅(key_datum⋅w)) = is_na⋅w"
  by (cases w) (simp_all add: d_is_na_def)

lemma truth_d_eq_key:
  assumes "s ≠ ⊥"
  shows "truth_d⋅(d_eq⋅(key_datum⋅w)⋅(key_datum⋅(Val⋅s))) = TT ⟷ w = Val⋅s"
proof -
  have strict: "(Λ x y. SBool⋅(scalar_eq⋅x⋅y))⋅⊥⋅y = ⊥" "(Λ x y. SBool⋅(scalar_eq⋅x⋅y))⋅x⋅⊥ = ⊥" for x y
    by simp_all
  have "truth_d⋅(d_eq⋅a⋅b) = scalar_eq⋅(plain⋅a)⋅(plain⋅b)" for a b
    unfolding d_eq_def truth_d_def by (simp add: plain_dlift2[OF strict])
  moreover have "scalar_eq⋅(plain⋅(key_datum⋅w))⋅s = TT ⟷ w = Val⋅s"
  proof (cases w)
    case (Val x)
    then show ?thesis using assms by (cases "x = ⊥") (simp_all add: scalar_eq_defined TT_def)
  qed simp_all
  ultimately show ?thesis by simp
qed

lemma kholds_pin_test:
  "(∀e∈set (pin_test K d). kholds e k) ⟷ dim_val d⋅K ⊑ dim_val d⋅k"
proof -
  let ?v = "dim_val d⋅K" and ?w = "dim_val d⋅k"
  show ?thesis
  proof (cases ?v)
    case bottom
    then show ?thesis by (simp add: pin_test_def)
  next
    case NA
    have "is_na⋅?w = TT ⟷ ?w = NA" by (cases ?w) simp_all
    moreover have "NA ⊑ ?w ⟷ ?w = NA" by (cases ?w) simp_all
    ultimately show ?thesis using NA by (simp add: pin_test_def kholds_def truth_d_is_na)
  next
    case (Val s)
    show ?thesis
    proof (cases "s = ⊥")
      case True
      have "neg⋅(is_na⋅?w) = TT ⟷ Val⋅⊥ ⊑ ?w" by (cases ?w) simp_all
      then show ?thesis using Val True
        by (simp add: pin_test_def kholds_def truth_d_is_na truth_d_not[simplified])
    next
      case False
      have "Val⋅s ⊑ ?w ⟷ ?w = Val⋅s"
        by (cases ?w) (auto dest: scalar_maximal[OF False])
      then show ?thesis using Val False
        by (simp add: pin_test_def kholds_def truth_d_eq_key[OF False, simplified])
    qed
  qed
qed

    by (auto simp: slice_pin_def fset_of_list.rep_eq)
qed
