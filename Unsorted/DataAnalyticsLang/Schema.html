Theory Schema

theory Schema
  imports LOD "HOLCF-extra.LowerPD_Representation" "HOLCF-extra.Lattice_Complete"
begin

default_sort "{bifinite, distributive_cpo}"

section ‹Schemas: presence rules and chase rules›

text ‹A schema on a row domain consists of presence rules and chase rules.

  A presence rule is triggered by a ∗‹decisive› antecedent: a predicate of one part of the row,
  false at bottom and constant above every nonbottom value of that part (so it is decided as soon
  as the part is present, as for presence itself or for ‹is_na› on a separated sum).  Its consequent
  is any up-closed predicate (presence, maximality, a decided value test).

  A chase rule acts on pairs of rows: when its guard holds of ‹(x, y)› it raises ‹x› by a
  contribution computed from ‹x› and ‹y›, in the rows with a top adjoined (an incompatible
  contribution is a conflict).  Guards are up-closed and Scott-open.  Functional dependencies
  are chase rules (guard: equal maximal determinants, contribution: the partner's dependent part),
  and so are join columns, which stay uncollapsed: under joint presence each key side is raised by
  the other side transported into its coordinates (a unary rule, ignoring the partner).

  Each presence rule is either ∗‹upheld› by every chase rule (one step between conforming rows
  produces a row satisfying it) or ∗‹repaired› by a set of functional dependencies among the
  chase rules: their determinants lie inside the trigger part and are forced maximal by the
  antecedent, and the consequent is carried along their determinants and dependents.  A triggered
  row then got its trigger part from a conforming source row, and the repairing dependencies copy the
  source's evidence across.›

datatype 'r presence_rule =
  Presence_Rule (trigger: "'r semi_factor") (antecedent: "'r ⇒ bool") (consequent: "'r ⇒ bool")

datatype 'r chase_rule =
  Chase_Rule (guard: "'r ⇒ 'r ⇒ bool") (contribution: "'r → 'r → 'r top")

subsection ‹Antecedents and consequents›

definition decisive :: "'r semi_factor ⇒ ('r ⇒ bool) ⇒ bool" where
  "decisive p P ⟷ (∀x. P x = P (x ⇂ p)) ∧ ¬ P ⊥ ∧
    (∀x y. x ⇂ p ≠ ⊥ ⟶ x ⊑ y ⟶ P y = P x)"

definition up_closed :: "('r::po ⇒ bool) ⇒ bool" where
  "up_closed C ⟷ (∀x y. C x ⟶ x ⊑ y ⟶ C y)"

lemma decisive_part: "decisive p P ⟹ P x = P (x ⇂ p)"
  unfolding decisive_def by blast

lemma decisive_bot: "decisive p P ⟹ ¬ P ⊥"
  unfolding decisive_def by blast

lemma decisive_above: "decisive p P ⟹ x ⇂ p ≠ ⊥ ⟹ x ⊑ y ⟹ P y = P x"
  unfolding decisive_def by blast

lemma decisive_present: "decisive p P ⟹ P x ⟹ x ⇂ p ≠ ⊥"
  unfolding decisive_def by metis

lemma up_closedD: "up_closed C ⟹ C x ⟹ x ⊑ y ⟹ C y"
  unfolding up_closed_def by blast

text ‹Decisive antecedents are Scott-open, so their negations are admissible; up-closed
  consequents are admissible outright.›

lemma adm_not_decisive:
  assumes P: "decisive p P"
  shows "adm (λx. ¬ P x)"
proof (rule admI)
  fix Y assume Y: "chain Y" and nP: "∀i. ¬ P (Y i)"
  show "¬ P (⨆i. Y i)"
  proof
    assume lub: "P (⨆i. Y i)"
    have "(⨆i. Y i) ⇂ p ≠ ⊥" using decisive_present[OF P lub] .
    then have "(⨆i. Y i ⇂ p) ≠ ⊥" using Y by (simp add: contlub_cfun_arg)
    then obtain i where i: "Y i ⇂ p ≠ ⊥" by fastforce
    have "P (⨆i. Y i) = P (Y i)" by (rule decisive_above[OF P i is_ub_thelub[OF Y]])
    with lub nP show False by simp
  qed
qed

lemma adm_up_closed: "up_closed C ⟹ adm C"
  by (rule admI) (blast intro: up_closedD is_ub_thelub)

subsection ‹Guards›

definition open_guard :: "('r ⇒ 'r ⇒ bool) ⇒ bool" where
  "open_guard g ⟷ (∀x y x' y'. g x y ⟶ x ⊑ x' ⟶ y ⊑ y' ⟶ g x' y') ∧
    (∀X Y. chain X ⟶ chain Y ⟶ g (⨆i. X i) (⨆i. Y i) ⟶ (∃i. g (X i) (Y i)))"

lemma open_guard_up: "open_guard g ⟹ g x y ⟹ x ⊑ x' ⟹ y ⊑ y' ⟹ g x' y'"
  unfolding open_guard_def by blast

lemma open_guard_lub:
  "open_guard g ⟹ chain X ⟹ chain Y ⟹ g (⨆i. X i) (⨆i. Y i) ⟹ ∃i. g (X i) (Y i)"
  unfolding open_guard_def by blast

lemma open_guard_lub1: "open_guard g ⟹ chain X ⟹ g (⨆i. X i) y ⟹ ∃i. g (X i) y"
  using open_guard_lub[of g X "λ_. y"] by simp

lemma open_guard_lub2: "open_guard g ⟹ chain Y ⟹ g x (⨆i. Y i) ⟹ ∃i. g x (Y i)"
  using open_guard_lub[of g "λ_. x" Y] by simp

text ‹A maximal part does not move under refinement, and with finitary determinants it is
  compact, so it is reached after finitely many approximants.›

lemma chain_part_reaches:
  assumes C: "chain C" and k: "(⨆i. C i) ⇂ p = k" and ck: "compact k"
  obtains n where "⋀j. n ≤ j ⟹ C j ⇂ p = k"
proof -
  have lub: "k = (⨆i. C i ⇂ p)" using k C by (simp add: contlub_cfun_arg)
  have Cp: "chain (λi. C i ⇂ p)" using C by simp
  obtain n where n: "k ⊑ C n ⇂ p"
    using compact_below_lub_iff[OF ck Cp] lub by auto
  have "C j ⇂ p = k" if j: "n ≤ j" for j
  proof (rule below_antisym)
    show "C j ⇂ p ⊑ k" unfolding lub by (rule is_ub_thelub[OF Cp])
    show "k ⊑ C j ⇂ p" using n monofun_cfun_arg[OF chain_mono[OF C j]] by (rule below_trans)
  qed
  then show ?thesis by (rule that)
qed

subsection ‹Functional dependencies and key joins as chase rules›

text ‹A set of rows satisfies a strong functional dependency ‹a → b› when rows whose
  ‹a›-parts are equal and maximal have equal ‹b›-parts.  A maximal ‹a›-part does not move under
  refinement.›

definition fd :: "'r semi_factor ⇒ 'r semi_factor ⇒ 'r set ⇒ bool" where
  "fd a b R ⟷
    (∀x∈R. ∀y∈R. maximal_at a (x ⇂ a) ⟶ x ⇂ a = y ⇂ a ⟶ x ⇂ b = y ⇂ b)"

lemma fdD:
  "fd a b R ⟹ x ∈ R ⟹ y ∈ R ⟹ maximal_at a (x ⇂ a) ⟹ x ⇂ a = y ⇂ a ⟹
    x ⇂ b = y ⇂ b"
  unfolding fd_def by blast

lemma maximal_at_proj_above:
  assumes m: "maximal_at a (x ⇂ a)" and le: "x ⊑ y"
  shows "y ⇂ a = x ⇂ a"
proof (rule maximal_at_eq[OF m])
  show "proj a⋅(y ⇂ a) = y ⇂ a" by simp
  show "x ⇂ a ⊑ y ⇂ a" using le by (rule monofun_cfun_arg)
qed

definition fd_guard :: "'r semi_factor ⇒ 'r ⇒ 'r ⇒ bool" where
  "fd_guard a x y ⟷ maximal_at a (x ⇂ a) ∧ x ⇂ a = y ⇂ a"

definition fd_rule :: "'r semi_factor ⇒ 'r semi_factor ⇒ 'r chase_rule" where
  "fd_rule a b = Chase_Rule (fd_guard a) (Λ x y. down⋅(y ⇂ b))"

lemma fd_rule_simps [simp]:
  "guard (fd_rule a b) = fd_guard a" "contribution (fd_rule a b)⋅x⋅y = down⋅(y ⇂ b)"
  by (simp_all add: fd_rule_def)

lemma open_fd_guard:
  assumes a: "finitary_semi_factor a"
  shows "open_guard (fd_guard a)"
  unfolding open_guard_def
proof (intro conjI allI impI)
  fix x y x' y' assume g: "fd_guard a x y" and x: "x ⊑ x'" and y: "y ⊑ y'"
  have m: "maximal_at a (x ⇂ a)" and eq: "x ⇂ a = y ⇂ a" using g unfolding fd_guard_def by auto
  have my: "maximal_at a (y ⇂ a)" using m eq by simp
  show "fd_guard a x' y'"
    using m maximal_at_proj_above[OF m x] maximal_at_proj_above[OF my y] eq
    by (simp add: fd_guard_def)
next
  fix X Y :: "nat ⇒ _" assume X: "chain X" and Y: "chain Y" and g: "fd_guard a (⨆i. X i) (⨆i. Y i)"
  define k where "k = (⨆i. X i) ⇂ a"
  have m: "maximal_at a k" and yk: "(⨆i. Y i) ⇂ a = k" using g by (auto simp add: fd_guard_def k_def)
  have ck: "compact k" using a by (simp add: k_def finitary_semi_factor_def finitary_def)
  obtain n where n: "⋀j. n ≤ j ⟹ X j ⇂ a = k" using chain_part_reaches[OF X k_def[symmetric] ck] by blast
  obtain n' where n': "⋀j. n' ≤ j ⟹ Y j ⇂ a = k" using chain_part_reaches[OF Y yk ck] by blast
  have "fd_guard a (X (max n n')) (Y (max n n'))" using n n' m by (simp add: fd_guard_def)
  then show "∃i. fd_guard a (X i) (Y i)" by blast
qed

text ‹A key join between parts ‹S› and ‹T› (the two key sides), with transports ‹φ› from
  ‹S›-coordinates to ‹T›-coordinates and ‹ψ› back: under joint presence each side is raised
  by the other.›

definition both_present :: "'r semi_factor ⇒ 'r semi_factor ⇒ 'r ⇒ 'r ⇒ bool" where
  "both_present S T x y ⟷ x ⇂ S ≠ ⊥ ∧ x ⇂ T ≠ ⊥"

definition key_join_rules :: "'r semi_factor ⇒ 'r semi_factor ⇒ ('r → 'r) ⇒ ('r → 'r) ⇒ 'r chase_rule set" where
  "key_join_rules S T φ ψ =
    {Chase_Rule (both_present S T) (Λ x y. down⋅(φ⋅(x ⇂ S))),
     Chase_Rule (both_present S T) (Λ x y. down⋅(ψ⋅(x ⇂ T)))}"

lemma present_lub: "chain X ⟹ (⨆i. X i) ⇂ S ≠ ⊥ ⟹ ∃n. ∀j≥n. X j ⇂ S ≠ ⊥"
proof -
  assume X: "chain X" and p: "(⨆i. X i) ⇂ S ≠ ⊥"
  then have "(⨆i. X i ⇂ S) ≠ ⊥" by (simp add: contlub_cfun_arg)
  then obtain n where n: "X n ⇂ S ≠ ⊥" by fastforce
  have "X j ⇂ S ≠ ⊥" if "n ≤ j" for j
    using n monofun_cfun_arg[OF chain_mono[OF X that], of "proj S"] by (auto simp add: bottomI)
  then show ?thesis by blast
qed

lemma open_both_present: "open_guard (both_present S T)"
  unfolding open_guard_def
proof (intro conjI allI impI)
  fix x y x' y' assume g: "both_present S T x y" and x: "x ⊑ x'"
  have "x ⇂ S ⊑ x' ⇂ S" "x ⇂ T ⊑ x' ⇂ T" using x by (simp_all add: monofun_cfun_arg)
  then show "both_present S T x' y'" using g by (auto simp add: both_present_def bottomI)
next
  fix X Y :: "nat ⇒ _" assume X: "chain X" and g: "both_present S T (⨆i. X i) (⨆i. Y i)"
  obtain n where n: "∀j≥n. X j ⇂ S ≠ ⊥" using present_lub[OF X] g by (auto simp add: both_present_def)
  obtain n' where n': "∀j≥n'. X j ⇂ T ≠ ⊥" using present_lub[OF X] g by (auto simp add: both_present_def)
  have "both_present S T (X (max n n')) (Y (max n n'))" using n n' by (simp add: both_present_def)
  then show "∃i. both_present S T (X i) (Y i)" by blast
qed

lemma open_key_join_rules: "c ∈ key_join_rules S T φ ψ ⟹ open_guard (guard c)"
  by (auto simp add: key_join_rules_def open_both_present)

subsection ‹The combined chase step›

text ‹One step applies every rule to a pair of rows and takes the join of the row and all firing
  contributions, in the complete lattice of rows with a top.›

definition guarded :: "'r chase_rule ⇒ 'r ⇒ 'r ⇒ 'r top" where
  "guarded c x y = (if guard c x y then contribution c⋅x⋅y else ⊥)"

definition rules_step :: "'r chase_rule set ⇒ 'r ⇒ 'r ⇒ 'r top" where
  "rules_step C x y = down⋅x ⊔ lub ((λc. guarded c x y) ` C)"

lemma rules_step_above: "down⋅x ⊑ rules_step C x y"
  unfolding rules_step_def by (rule joinableD(1)[OF joinable_lattice])

lemma rules_step_guarded: "c ∈ C ⟹ guarded c x y ⊑ rules_step C x y"
  unfolding rules_step_def
  by (rule below_trans[OF lattice_lub_upper joinableD(2)[OF joinable_lattice]]) simp

lemma cont_guarded_if:
  assumes up: "⋀x x'. g x ⟹ x ⊑ x' ⟹ g x'" and op: "⋀Y. chain Y ⟹ g (⨆i. Y i) ⟹ ∃i. g (Y i)"
    and f: "cont f"
  shows "cont (λx. if g x then f x else ⊥)"
proof (rule contI2)
  show "monofun (λx. if g x then f x else ⊥)"
    by (rule monofunI) (auto dest: up intro: cont2monofunE[OF f])
  fix Y assume Y: "chain Y" and ch: "chain (λi. if g (Y i) then f (Y i) else ⊥)"
  show "(if g (⨆i. Y i) then f (⨆i. Y i) else ⊥) ⊑ (⨆i. if g (Y i) then f (Y i) else ⊥)"
  proof (cases "g (⨆i. Y i)")
    case True
    obtain n where n: "g (Y n)" using op[OF Y True] by blast
    have gn: "g (Y (i + n))" for i by (rule up[OF n chain_mono[OF Y]]) simp
    have "f (⨆i. Y i) = (⨆i. f (Y (i + n)))"
      by (simp add: cont2contlubE[OF f Y] lub_range_shift[OF ch2ch_cont[OF f Y]])
    also have "… = (⨆i. if g (Y (i + n)) then f (Y (i + n)) else ⊥)" by (simp add: gn)
    also have "… = (⨆i. if g (Y i) then f (Y i) else ⊥)" by (rule lub_range_shift[OF ch])
    finally show ?thesis using True by simp
  qed simp
qed

lemma cont_lattice_lub:
  fixes f :: "'i::type ⇒ 'a::cpo ⇒ 'b::lattice_complete"
  assumes f: "⋀i. i ∈ I ⟹ cont (f i)"
  shows "cont (λx. lub ((λi. f i x) ` I))"
proof (rule contI2)
  show "monofun (λx. lub ((λi. f i x) ` I))"
  proof (rule monofunI, rule lattice_lub_least, rule is_ubI, elim imageE)
    fix x y z i assume xy: "x ⊑ y" and z: "z = f i x" and i: "i ∈ I"
    have "f i x ⊑ f i y" by (rule cont2monofunE[OF f[OF i] xy])
    also have "f i y ⊑ lub ((λi. f i y) ` I)" using i by (simp add: lattice_lub_upper)
    finally show "z ⊑ lub ((λi. f i y) ` I)" by (simp add: z)
  qed
  fix Y assume Y: "chain Y" and ch: "chain (λn. lub ((λi. f i (Y n)) ` I))"
  show "lub ((λi. f i (⨆n. Y n)) ` I) ⊑ (⨆n. lub ((λi. f i (Y n)) ` I))"
  proof (rule lattice_lub_least, rule is_ubI, elim imageE)
    fix z i assume z: "z = f i (⨆n. Y n)" and i: "i ∈ I"
    have "f i (⨆n. Y n) = (⨆n. f i (Y n))" by (rule cont2contlubE[OF f[OF i] Y])
    also have "… ⊑ (⨆n. lub ((λi. f i (Y n)) ` I))"
      by (rule lub_mono[OF ch2ch_cont[OF f[OF i] Y] ch]) (simp add: lattice_lub_upper i)
    finally show "z ⊑ (⨆n. lub ((λi. f i (Y n)) ` I))" by (simp add: z)
  qed
qed

lemma cont_lattice_join:
  fixes f g :: "'a::cpo ⇒ 'b::lattice_complete"
  assumes f: "cont f" and g: "cont g"
  shows "cont (λx. f x ⊔ g x)"
proof -
  have img: "(λb. (if b then f else g) x) ` UNIV = {f x, g x}" for x by (auto simp add: UNIV_bool)
  have "(λx. f x ⊔ g x) = (λx. lub ((λb. (if b then f else g) x) ` UNIV))"
    by (simp add: img lub_eqI[OF joinable_lub[OF joinable_lattice]])
  moreover have "cont (if b then f else g)" for b by (cases b) (simp_all add: f g)
  ultimately show ?thesis using cont_lattice_lub[of UNIV "λb. if b then f else g"] by simp
qed

lemma cont_guarded1: "open_guard (guard c) ⟹ cont (λx. guarded c x y)"
  unfolding guarded_def
  by (rule cont_guarded_if) (auto dest: open_guard_lub1 intro: open_guard_up)

lemma cont_guarded2: "open_guard (guard c) ⟹ cont (λy. guarded c x y)"
  unfolding guarded_def
  by (rule cont_guarded_if) (auto dest: open_guard_lub2 intro: open_guard_up)

lemma cont_rules_step1: "∀c∈C. open_guard (guard c) ⟹ cont (λx. rules_step C x y)"
  unfolding rules_step_def
  by (intro cont_lattice_join cont_lattice_lub cont_guarded1) simp_all

lemma cont_rules_step2: "∀c∈C. open_guard (guard c) ⟹ cont (λy. rules_step C x y)"
  unfolding rules_step_def
  by (intro cont_lattice_join cont_lattice_lub cont_guarded2) simp_all

definition step_cfun :: "'r chase_rule set ⇒ 'r → 'r → 'r top" where
  "step_cfun C = (Λ x y. rules_step C x y)"

lemma step_cfun_beta: "∀c∈C. open_guard (guard c) ⟹ step_cfun C⋅x⋅y = rules_step C x y"
  unfolding step_cfun_def
  by (simp add: cont2cont_LAM cont_rules_step1 cont_rules_step2)

text ‹Lifted to rows with a top, which is absorbing.›

definition step_top :: "'r chase_rule set ⇒ 'r top → 'r top → 'r top" where
  "step_top C = (Λ U V. Iftop Top (Λ u. Iftop Top (step_cfun C⋅u) V) U)"

lemma cont_step_top_inner: "cont (λu. Iftop Top (step_cfun C⋅u) V)"
  by (cases V) (simp_all add: Iftop_down Iftop_top)

lemma step_top_down: "step_top C⋅(down⋅u)⋅(down⋅v) = step_cfun C⋅u⋅v"
proof -
  have inner: "(Λ u. Iftop Top (step_cfun C⋅u) V)⋅w = Iftop Top (step_cfun C⋅w) V" for V w
    by (simp add: cont_step_top_inner)
  have c1: "cont (λV. Iftop Top (Λ u. Iftop Top (step_cfun C⋅u) V) U)" for U
    by (cases U) (simp_all add: Iftop_down Iftop_top inner cont_Iftop)
  have c2: "cont (λU. Iftop Top (Λ u. Iftop Top (step_cfun C⋅u) V) U)" for V
    by (rule cont_Iftop) simp
  show ?thesis
    unfolding step_top_def by (simp add: cont2cont_LAM c1 c2 Iftop_down inner cont_Iftop)
qed

subsection ‹The chase on the Hoare powerdomain›

text ‹One round of steps applies the step to every pair of members, as a double bind with no
  explicit sup over the rules.  The chase of ‹X› is the least ‹Y› above ‹X› closed under steps.›

definition chase_steps :: "'r chase_rule set ⇒ 'r top lower_pd → 'r top lower_pd" where
  "chase_steps C = (Λ Y. lower_bind⋅Y⋅(Λ U. lower_bind⋅Y⋅(Λ V. {step_top C⋅U⋅V}♭)))"

lemma chase_steps_beta: "chase_steps C⋅Y = lower_bind⋅Y⋅(Λ U. lower_bind⋅Y⋅(Λ V. {step_top C⋅U⋅V}♭))"
  by (simp add: chase_steps_def)

definition pd_chase :: "'r chase_rule set ⇒ 'r top lower_pd → 'r top lower_pd" where
  "pd_chase C = (Λ X. fix⋅(Λ Y. X ∪♭ Y ∪♭ chase_steps C⋅Y))"

lemma pd_chase_beta: "pd_chase C⋅X = fix⋅(Λ Y. X ∪♭ Y ∪♭ chase_steps C⋅Y)"
  by (simp add: pd_chase_def)

lemma chase_steps_mem:
  assumes U: "U ∈♭ Y" and V: "V ∈♭ Y"
  shows "step_top C⋅U⋅V ∈♭ chase_steps C⋅Y"
proof -
  have "{step_top C⋅U⋅V}♭ ⊑ lower_bind⋅Y⋅(Λ V. {step_top C⋅U⋅V}♭)"
    using lower_bind_mem[OF V, of "Λ V. {step_top C⋅U⋅V}♭"] by simp
  also have "… ⊑ chase_steps C⋅Y"
    using lower_bind_mem[OF U, of "Λ U. lower_bind⋅Y⋅(Λ V. {step_top C⋅U⋅V}♭)"]
    by (simp add: chase_steps_beta)
  finally show ?thesis by (simp add: lower_mem_equiv)
qed

lemma chase_steps_below:
  assumes st: "⋀U V. U ∈♭ Y ⟹ V ∈♭ Y ⟹ step_top C⋅U⋅V ∈♭ Z"
  shows "chase_steps C⋅Y ⊑ Z"
  unfolding chase_steps_beta
proof (rule lower_bind_below, simp)
  fix U assume U: "U ∈♭ Y"
  show "lower_bind⋅Y⋅(Λ V. {step_top C⋅U⋅V}♭) ⊑ Z"
    by (rule lower_bind_below) (simp add: lower_mem_equiv[symmetric] st[OF U])
qed

lemma pd_chase_unfold: "pd_chase C⋅X = X ∪♭ pd_chase C⋅X ∪♭ chase_steps C⋅(pd_chase C⋅X)"
  unfolding pd_chase_beta by (subst fix_eq) simp

lemma pd_chase_above: "X ⊑ pd_chase C⋅X"
proof -
  have "X ⊑ X ∪♭ pd_chase C⋅X ∪♭ chase_steps C⋅(pd_chase C⋅X)"
    by (rule below_trans[OF lower_plus_below1 lower_plus_below1])
  also have "… = pd_chase C⋅X" by (rule pd_chase_unfold[symmetric])
  finally show ?thesis .
qed

lemma pd_chase_closed: "chase_steps C⋅(pd_chase C⋅X) ⊑ pd_chase C⋅X"
proof -
  have "chase_steps C⋅(pd_chase C⋅X) ⊑ X ∪♭ pd_chase C⋅X ∪♭ chase_steps C⋅(pd_chase C⋅X)"
    by (rule lower_plus_below2)
  also have "… = pd_chase C⋅X" by (rule pd_chase_unfold[symmetric])
  finally show ?thesis .
qed

lemma pd_chase_least: "X ⊑ Y ⟹ chase_steps C⋅Y ⊑ Y ⟹ pd_chase C⋅X ⊑ Y"
  unfolding pd_chase_beta by (rule fix_least_below) (simp add: lower_plus_least)

subsection ‹Lawful datasets›

definition obeys :: "'r presence_rule ⇒ 'r ⇒ bool" where
  "obeys r x ⟷ (antecedent r x ⟶ consequent r x)"

definition row_obeys :: "'r presence_rule set ⇒ 'r ⇒ bool" where
  "row_obeys R x ⟷ (∀r∈R. obeys r x)"

text ‹Both conditions are stated on compact members, which makes them admissible.  Presence:
  every compact member lies below a conforming member.  Chase: steps between compact members do not
  conflict and land in the set.›

definition obeys_presence_pd :: "'r presence_rule set ⇒ 'r lower_pd ⇒ bool" where
  "obeys_presence_pd R X ⟷
    (∀x. compact x ⟶ x ∈♭ X ⟶ (∃l. row_obeys R l ∧ x ⊑ l ∧ l ∈♭ X))"

definition chase_closed_pd :: "'r chase_rule set ⇒ 'r lower_pd ⇒ bool" where
  "chase_closed_pd C X ⟷ (∀x y. compact x ⟶ compact y ⟶ x ∈♭ X ⟶ y ∈♭ X ⟶
    (∃w. w ∈♭ X ∧ rules_step C x y = down⋅w))"

lemma adm_obeys_presence_pd: "adm (obeys_presence_pd (R :: 'r presence_rule set))"
proof (rule admI)
  fix Y :: "nat ⇒ 'r lower_pd"
  assume Y: "chain Y" and all: "∀i. obeys_presence_pd R (Y i)"
  show "obeys_presence_pd R (⨆i. Y i)"
    unfolding obeys_presence_pd_def
  proof (intro allI impI)
    fix x assume x: "compact x" and mem: "x ∈♭ (⨆i. Y i)"
    obtain i where "x ∈♭ Y i" by (rule lower_mem_compact_lub[OF Y x mem])
    then obtain l where "row_obeys R l" "x ⊑ l" "l ∈♭ Y i"
      using all x unfolding obeys_presence_pd_def by blast
    then show "∃l. row_obeys R l ∧ x ⊑ l ∧ l ∈♭ (⨆i. Y i)" using lower_mem_lub[OF Y] by blast
  qed
qed

lemma adm_chase_closed_pd: "adm (chase_closed_pd (C :: 'r chase_rule set))"
proof (rule admI)
  fix Y :: "nat ⇒ 'r lower_pd"
  assume Y: "chain Y" and all: "∀i. chase_closed_pd C (Y i)"
  show "chase_closed_pd C (⨆i. Y i)"
    unfolding chase_closed_pd_def
  proof (intro allI impI)
    fix x y assume x: "compact x" and y: "compact y"
      and mx: "x ∈♭ (⨆i. Y i)" and my: "y ∈♭ (⨆i. Y i)"
    obtain i where i: "x ∈♭ Y i" by (rule lower_mem_compact_lub[OF Y x mx])
    obtain j where j: "y ∈♭ Y j" by (rule lower_mem_compact_lub[OF Y y my])
    have "x ∈♭ Y (max i j)" "y ∈♭ Y (max i j)" using lower_mem_chain[OF Y] i j by simp_all
    then obtain w where "w ∈♭ Y (max i j)" "rules_step C x y = down⋅w"
      using all x y unfolding chase_closed_pd_def by blast
    then show "∃w. w ∈♭ (⨆i. Y i) ∧ rules_step C x y = down⋅w" using lower_mem_lub[OF Y] by blast
  qed
qed

text ‹With open guards, closure under steps between compact members extends to all members,
  limits included: the steps between the approximants converge to the step between the members.›

theorem chase_closed_pd_mem:
  fixes x y :: 'r
  assumes og: "∀c∈C. open_guard (guard c)" and cl: "chase_closed_pd C X"
    and x: "x ∈♭ X" and y: "y ∈♭ X"
  shows "∃w. w ∈♭ X ∧ rules_step C x y = down⋅w"
proof -
  obtain ap :: "nat ⇒ 'r → 'r" where ap: "approx_chain ap" using bifinite by blast
  define xs where "xs = (λi. ap i⋅x)"
  define ys where "ys = (λi. ap i⋅y)"
  have xs: "chain xs" and ys: "chain ys"
    unfolding xs_def ys_def using approx_chain.chain_approx[OF ap] by (simp_all add: ch2ch_Rep_cfunL)
  have lub_x: "(⨆i. xs i) = x" and lub_y: "(⨆i. ys i) = y"
    using approx_chain.chain_approx[OF ap] approx_chain.lub_approx[OF ap]
    by (simp_all add: xs_def ys_def flip: contlub_cfun_fun)
  have "∃w. w ∈♭ X ∧ rules_step C (xs i) (ys i) = down⋅w" for i
  proof -
    have "xs i ∈♭ X" "ys i ∈♭ X"
      using x y approx_chain.approx_below[OF ap] by (simp_all add: xs_def ys_def lower_mem_below)
    then show ?thesis using cl approx_chain.compact_approx[OF ap]
      unfolding chase_closed_pd_def xs_def ys_def by blast
  qed
  then obtain ws where ws: "⋀i. ws i ∈♭ X" and eq: "⋀i. rules_step C (xs i) (ys i) = down⋅(ws i)"
    by metis
  define s where "s = step_cfun C"
  have s: "s⋅u⋅v = rules_step C u v" for u v by (simp add: s_def step_cfun_beta og)
  have ws_chain: "chain ws"
  proof (rule chainI)
    fix i
    have "s⋅(xs i)⋅(ys i) ⊑ s⋅(xs (Suc i))⋅(ys (Suc i))"
      by (intro monofun_cfun_arg monofun_cfun chainE xs ys)
    then show "ws i ⊑ ws (Suc i)" by (simp add: s eq)
  qed
  have "rules_step C x y = s⋅(⨆i. xs i)⋅(⨆i. ys i)" by (simp add: s lub_x lub_y)
  also have "… = (⨆i. s⋅(xs i)⋅(ys i))"
    by (simp add: contlub_cfun_arg[OF xs] lub_APP[OF ch2ch_Rep_cfunR[OF xs] ys])
  also have "… = (⨆i. down⋅(ws i))" by (simp add: s eq)
  also have "… = down⋅(⨆i. ws i)" by (simp add: contlub_cfun_arg[OF ws_chain])
  finally show ?thesis using admD[OF adm_lower_mem ws_chain] ws by blast
qed

subsection ‹The chase as the least conforming dataset above›

text ‹Through ‹down›, a dataset is closed exactly when one round of steps stays inside it.›

lemma chase_closed_pd_steps:
  assumes og: "∀c∈C. open_guard (guard c)"
  shows "chase_closed_pd C X ⟷ chase_steps C⋅(lower_map⋅down⋅X) ⊑ lower_map⋅down⋅X"
proof
  assume cl: "chase_closed_pd C X"
  show "chase_steps C⋅(lower_map⋅down⋅X) ⊑ lower_map⋅down⋅X"
  proof (rule chase_steps_below)
    fix U V assume "U ∈♭ lower_map⋅down⋅X" "V ∈♭ lower_map⋅down⋅X"
    then obtain u v where u: "u ∈♭ X" "U = down⋅u" and v: "v ∈♭ X" "V = down⋅v"
      by (auto simp add: lower_mem_map_down)
    obtain w where "w ∈♭ X" "rules_step C u v = down⋅w"
      using chase_closed_pd_mem[OF og cl u(1) v(1)] by blast
    then show "step_top C⋅U⋅V ∈♭ lower_map⋅down⋅X"
      by (auto simp add: u v step_top_down step_cfun_beta og lower_mem_map_down)
  qed
next
  assume st: "chase_steps C⋅(lower_map⋅down⋅X) ⊑ lower_map⋅down⋅X"
  show "chase_closed_pd C X"
    unfolding chase_closed_pd_def
  proof (intro allI impI)
    fix x y assume x: "x ∈♭ X" and y: "y ∈♭ X"
    have "step_top C⋅(down⋅x)⋅(down⋅y) ∈♭ chase_steps C⋅(lower_map⋅down⋅X)"
      by (intro chase_steps_mem lower_map_mem x y)
    then have "rules_step C x y ∈♭ lower_map⋅down⋅X"
      using st by (auto simp add: lower_mem_equiv step_top_down step_cfun_beta og intro: below_trans)
    then show "∃w. w ∈♭ X ∧ rules_step C x y = down⋅w" by (simp add: lower_mem_map_down)
  qed
qed

theorem pd_chase_fixed_iff:
  assumes og: "∀c∈C. open_guard (guard c)"
  shows "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X ⟷ chase_closed_pd C X"
proof
  assume fx: "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X"
  show "chase_closed_pd C X"
    using pd_chase_closed[of C "lower_map⋅down⋅X"] by (simp add: fx chase_closed_pd_steps[OF og])
next
  assume cl: "chase_closed_pd C X"
  show "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X"
  proof (rule below_antisym)
    show "pd_chase C⋅(lower_map⋅down⋅X) ⊑ lower_map⋅down⋅X"
      by (rule pd_chase_least) (simp_all add: chase_closed_pd_steps[OF og, symmetric] cl)
  qed (rule pd_chase_above)
qed

corollary pd_chase_fixed:
  "∀c∈C. open_guard (guard c) ⟹ chase_closed_pd C X ⟹ pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X"
  by (simp add: pd_chase_fixed_iff)

text ‹A chase without conflict is again a dataset, and it is closed.›

lemma pd_chase_Top_free:
  assumes og: "∀c∈C. open_guard (guard c)" and nT: "¬ Top ∈♭ pd_chase C⋅(lower_map⋅down⋅X)"
  obtains X' where "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X'" "X ⊑ X'" "chase_closed_pd C X'"
proof -
  obtain X' where eq: "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X'"
    using nT by (rule lower_map_down_onto)
  have "X ⊑ X'" using pd_chase_above[of "lower_map⋅down⋅X" C] by (simp add: eq lower_map_down_below_iff)
  moreover have "chase_closed_pd C X'"
    using pd_chase_closed[of C "lower_map⋅down⋅X"] by (simp add: eq chase_closed_pd_steps[OF og])
  ultimately show ?thesis using eq that by blast
qed

text ‹The chase is the least closed dataset above ‹X›, and it conflicts exactly when there is
  no closed dataset above ‹X›.›

theorem pd_chase_least_closed:
  assumes og: "∀c∈C. open_guard (guard c)" and le: "X ⊑ Z" and cl: "chase_closed_pd C Z"
  obtains X' where "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X'"
    "X ⊑ X'" "chase_closed_pd C X'" "X' ⊑ Z"
proof -
  have below: "pd_chase C⋅(lower_map⋅down⋅X) ⊑ lower_map⋅down⋅Z"
    by (rule pd_chase_least)
      (simp_all add: lower_map_down_below_iff le chase_closed_pd_steps[OF og, symmetric] cl)
  have "¬ Top ∈♭ pd_chase C⋅(lower_map⋅down⋅X)"
    using below Top_not_mem_map_down[of Z] by (auto simp add: lower_mem_equiv intro: below_trans)
  then obtain X' where eq: "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X'"
    and "X ⊑ X'" "chase_closed_pd C X'"
    by (rule pd_chase_Top_free[OF og])
  moreover have "X' ⊑ Z" using below by (simp add: eq lower_map_down_below_iff)
  ultimately show ?thesis by (rule that)
qed

theorem pd_chase_conflict_iff:
  assumes og: "∀c∈C. open_guard (guard c)"
  shows "Top ∈♭ pd_chase C⋅(lower_map⋅down⋅X) ⟷ ¬ (∃Z. X ⊑ Z ∧ chase_closed_pd C Z)"
proof
  assume T: "Top ∈♭ pd_chase C⋅(lower_map⋅down⋅X)"
  show "¬ (∃Z. X ⊑ Z ∧ chase_closed_pd C Z)"
  proof
    assume "∃Z. X ⊑ Z ∧ chase_closed_pd C Z"
    then obtain Z where "X ⊑ Z" "chase_closed_pd C Z" by blast
    then obtain X' where "pd_chase C⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X'"
      by (rule pd_chase_least_closed[OF og])
    with T show False by (simp add: Top_not_mem_map_down)
  qed
next
  assume nc: "¬ (∃Z. X ⊑ Z ∧ chase_closed_pd C Z)"
  show "Top ∈♭ pd_chase C⋅(lower_map⋅down⋅X)"
  proof (rule ccontr)
    assume "¬ Top ∈♭ pd_chase C⋅(lower_map⋅down⋅X)"
    then obtain X' where "X ⊑ X'" "chase_closed_pd C X'" by (rule pd_chase_Top_free[OF og])
    with nc show False by blast
  qed
qed

text ‹Monotonicity is free: the chase is continuous, so conflicts propagate upwards.›

corollary pd_chase_conflict_up:
  "X ⊑ Y ⟹ Top ∈♭ pd_chase C⋅X ⟹ Top ∈♭ pd_chase C⋅Y"
  by (rule lower_mem_below[OF _ below_refl], unfold lower_mem_equiv)
    (erule below_trans, erule monofun_cfun_arg)

subsection ‹The laws on maximal members›

text ‹A maximal member absorbs a step exactly when every firing rule contributes below it.  Since
  every member lies below a maximal one and the step is monotone, closure under steps is the same
  as the rules holding at the maximal members.›

lemma rules_step_absorbed:
  "rules_step C m y = down⋅m ⟷ (∀c∈C. guard c m y ⟶ contribution c⋅m⋅y ⊑ down⋅m)"
proof (intro iffI ballI impI)
  fix c assume eq: "rules_step C m y = down⋅m" and c: "c ∈ C" and g: "guard c m y"
  show "contribution c⋅m⋅y ⊑ down⋅m"
    using rules_step_guarded[OF c, of m y] g by (simp add: eq guarded_def)
next
  assume all: "∀c∈C. guard c m y ⟶ contribution c⋅m⋅y ⊑ down⋅m"
  have "lub ((λc. guarded c m y) ` C) ⊑ down⋅m"
    by (rule lattice_lub_least, rule is_ubI) (auto simp add: guarded_def all)
  then have "rules_step C m y ⊑ down⋅m"
    unfolding rules_step_def by (rule join_belowI[OF below_refl])
  then show "rules_step C m y = down⋅m" using rules_step_above by (rule below_antisym)
qed

definition rule_holds :: "'r chase_rule ⇒ 'r lower_pd ⇒ bool" where
  "rule_holds c X ⟷
    (∀m∈lower_maximal X. ∀y. y ∈♭ X ⟶ guard c m y ⟶ contribution c⋅m⋅y ⊑ down⋅m)"

theorem chase_closed_pd_iff_rule_holds:
  assumes og: "∀c∈C. open_guard (guard c)"
  shows "chase_closed_pd C X ⟷ (∀c∈C. rule_holds c X)"
proof
  assume cl: "chase_closed_pd C X"
  show "∀c∈C. rule_holds c X"
    unfolding rule_holds_def
  proof (intro ballI allI impI)
    fix c m y assume c: "c ∈ C" and m: "m ∈ lower_maximal X" and y: "y ∈♭ X" and g: "guard c m y"
    have mx: "m ∈♭ X" using m by (simp add: lower_maximal_def)
    obtain w where w: "w ∈♭ X" and eq: "rules_step C m y = down⋅w"
      using chase_closed_pd_mem[OF og cl mx y] by blast
    have "m ⊑ w" using rules_step_above[of m C y] by (simp add: eq)
    then have "w = m" using m w by (simp add: lower_maximal_def)
    then have "rules_step C m y = down⋅m" by (simp add: eq)
    then show "contribution c⋅m⋅y ⊑ down⋅m" using c g by (simp add: rules_step_absorbed)
  qed
next
  assume all: "∀c∈C. rule_holds c X"
  show "chase_closed_pd C X"
    unfolding chase_closed_pd_def
  proof (intro allI impI)
    fix x y assume x: "x ∈♭ X" and y: "y ∈♭ X"
    obtain m where m: "m ∈ lower_maximal X" "x ⊑ m" using x by (rule lower_mem_maximal_above)
    have "rules_step C m y = down⋅m"
      using all m(1) y unfolding rules_step_absorbed rule_holds_def by blast
    moreover have "rules_step C x y ⊑ rules_step C m y"
      by (rule cont2monofunE[OF cont_rules_step1[OF og] m(2)])
    ultimately have le: "rules_step C x y ⊑ down⋅m" by simp
    then obtain w where w: "rules_step C x y = down⋅w" by (cases "rules_step C x y") auto
    have "w ∈♭ X" using le m(1) by (auto simp add: w lower_maximal_def intro: lower_mem_below)
    then show "∃w. w ∈♭ X ∧ rules_step C x y = down⋅w" using w by blast
  qed
qed

text ‹For a functional dependency the law is the dependency on the maximal members.›

theorem rule_holds_fd: "rule_holds (fd_rule a b) X ⟷ fd a b (lower_maximal X)"
proof
  assume h: "rule_holds (fd_rule a b) X"
  show "fd a b (lower_maximal X)"
    unfolding fd_def
  proof (intro ballI impI)
    fix x y assume x: "x ∈ lower_maximal X" and y: "y ∈ lower_maximal X"
      and m: "maximal_at a (x ⇂ a)" and eq: "x ⇂ a = y ⇂ a"
    have absorbed: "v ⇂ b ⊑ u ⇂ b"
      if u: "u ∈ lower_maximal X" and v: "v ∈ lower_maximal X" and g: "fd_guard a u v" for u v
    proof -
      have "v ∈♭ X" using v by (simp add: lower_maximal_def)
      then have "v ⇂ b ⊑ u" using h u g by (simp add: rule_holds_def)
      then have "proj b⋅(v ⇂ b) ⊑ u ⇂ b" by (rule monofun_cfun_arg)
      then show ?thesis by simp
    qed
    have "fd_guard a x y" "fd_guard a y x" using m eq by (auto simp add: fd_guard_def)
    then show "x ⇂ b = y ⇂ b"
      using absorbed[OF x y] absorbed[OF y x] by (blast intro: below_antisym)
  qed
next
  assume f: "fd a b (lower_maximal X)"
  show "rule_holds (fd_rule a b) X"
    unfolding rule_holds_def
  proof (intro ballI allI impI)
    fix m y assume m: "m ∈ lower_maximal X" and y: "y ∈♭ X" and g: "guard (fd_rule a b) m y"
    obtain m' where m': "m' ∈ lower_maximal X" "y ⊑ m'" using y by (rule lower_mem_maximal_above)
    have ma: "maximal_at a (m ⇂ a)" and eq: "m ⇂ a = y ⇂ a"
      using g unfolding fd_rule_simps fd_guard_def by auto
    have my: "maximal_at a (y ⇂ a)" using ma unfolding eq .
    have "m' ⇂ a = y ⇂ a" by (rule maximal_at_proj_above[OF my m'(2)])
    then have "m ⇂ a = m' ⇂ a" using eq by simp
    then have "m ⇂ b = m' ⇂ b" by (rule fdD[OF f m(1) m'(1) ma])
    moreover have "y ⇂ b ⊑ m' ⇂ b" using m'(2) by (rule monofun_cfun_arg)
    ultimately have "y ⇂ b ⊑ m ⇂ b" by simp
    also have "m ⇂ b ⊑ m" by (rule proj_below)
    finally show "contribution (fd_rule a b)⋅m⋅y ⊑ down⋅m" by simp
  qed
qed

corollary fd_maximal_members:
  "∀c∈C. open_guard (guard c) ⟹ chase_closed_pd C X ⟹ fd_rule a b ∈ C ⟹
    fd a b (lower_maximal X)"
  using chase_closed_pd_iff_rule_holds rule_holds_fd by blast

section ‹The schema locale›

definition upholds :: "'r presence_rule set ⇒ 'r chase_rule ⇒ 'r presence_rule ⇒ bool" where
  "upholds R c r ⟷ (∀x y z. row_obeys R x ⟶ row_obeys R y ⟶ guard c x y ⟶
    down⋅x ⊔ contribution c⋅x⋅y = down⋅z ⟶ obeys r z)"

definition repairs :: "('r semi_factor × 'r semi_factor) set ⇒ 'r presence_rule ⇒ bool" where
  "repairs D r ⟷
    (∀(a, b)∈D. a ⊑ trigger r ∧ (∀x. antecedent r x ⟶ maximal_at a (x ⇂ a))) ∧
    (∀x y. consequent r y ⟶ (∀(a, b)∈D. y ⇂ a ⊑ x ⇂ a ∧ y ⇂ b ⊑ x ⇂ b) ⟶
      consequent r x)"

record 'r schema_rules =
  presence :: "'r presence_rule set"
  chase :: "'r chase_rule set"

hide_const (open) schema_rules.truncate

locale schema =
  fixes rules :: "'r schema_rules"
  assumes antecedent_decisive: "r ∈ presence rules ⟹ decisive (trigger r) (antecedent r)"
    and consequent_up_closed: "r ∈ presence rules ⟹ up_closed (consequent r)"
    and guard_open: "∀c∈chase rules. open_guard (guard c)"
    and rule_maintained: "r ∈ presence rules ⟹
      (∀c∈chase rules. upholds (presence rules) c r) ∨
      (∃D. (∀(a, b)∈D. fd_rule a b ∈ chase rules) ∧ repairs D r)"
begin

abbreviation presence_rules :: "'r presence_rule set" where "presence_rules ≡ presence rules"
abbreviation chase_rules :: "'r chase_rule set" where "chase_rules ≡ chase rules"

abbreviation conforming_row :: "'r ⇒ bool" where
  "conforming_row ≡ row_obeys presence_rules"

definition conforms :: "'r lower_pd ⇒ bool" where
  "conforms X ⟷ obeys_presence_pd presence_rules X ∧ chase_closed_pd chase_rules X"

lemma conforming_row_bot: "conforming_row ⊥"
  unfolding row_obeys_def obeys_def by (blast dest: decisive_bot[OF antecedent_decisive])

lemma adm_conforming_row: "adm conforming_row"
  unfolding row_obeys_def obeys_def imp_conv_disj
  by (intro adm_ball adm_disj adm_not_decisive[OF antecedent_decisive]
      adm_up_closed[OF consequent_up_closed])

text ‹Lawful rows are closed under compatible joins: a triggered antecedent of the join is
  decided by the joinand whose trigger part is present, and that joinand's consequent goes up.›

lemma conforming_row_join:
  assumes x: "conforming_row x" and y: "conforming_row y" and j: "x ⇑ y"
  shows "conforming_row (x ⊔ y)"
  unfolding row_obeys_def obeys_def
proof (intro ballI impI)
  fix r assume r: "r ∈ presence_rules" and P: "antecedent r (x ⊔ y)"
  note D = antecedent_decisive[OF r] and C = consequent_up_closed[OF r]
  have "(x ⊔ y) ⇂ trigger r ≠ ⊥" by (rule decisive_present[OF D P])
  then have "x ⇂ trigger r ≠ ⊥ ∨ y ⇂ trigger r ≠ ⊥" by (auto simp add: proj_join[OF j])
  then show "consequent r (x ⊔ y)"
  proof
    assume "x ⇂ trigger r ≠ ⊥"
    then have "antecedent r x" using P decisive_above[OF D _ joinableD(1)[OF j]] by blast
    then have "consequent r x" using x r by (simp add: row_obeys_def obeys_def)
    then show ?thesis by (rule up_closedD[OF C _ joinableD(1)[OF j]])
  next
    assume "y ⇂ trigger r ≠ ⊥"
    then have "antecedent r y" using P decisive_above[OF D _ joinableD(2)[OF j]] by blast
    then have "consequent r y" using y r by (simp add: row_obeys_def obeys_def)
    then show ?thesis by (rule up_closedD[OF C _ joinableD(2)[OF j]])
  qed
qed

lemma adm_conforms: "adm conforms"
  unfolding conforms_def by (intro adm_conj adm_obeys_presence_pd adm_chase_closed_pd)

theorem conforms_chase_fixed: "conforms X ⟹ pd_chase chase_rules⋅(lower_map⋅down⋅X) = lower_map⋅down⋅X"
  unfolding conforms_def using pd_chase_fixed[OF guard_open] by blast

corollary conforms_fd:
  "conforms X ⟹ fd_rule a b ∈ chase_rules ⟹ fd a b (lower_maximal X)"
  unfolding conforms_def using fd_maximal_members[OF guard_open] by blast

lemma conforms_iff_rule_holds:
  "conforms X ⟷ obeys_presence_pd presence_rules X ∧ (∀c∈chase_rules. rule_holds c X)"
  unfolding conforms_def by (simp add: chase_closed_pd_iff_rule_holds[OF guard_open])
