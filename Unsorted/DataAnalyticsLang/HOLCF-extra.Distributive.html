Theory HOLCF-extra.Distributive

theory Distributive imports Bounded_Complete begin

section ‹Distributive bounded-complete cpos›

text ‹A bounded-complete cpo is ∗‹distributive› when binary meet distributes over joins of
  joinable pairs.›

class distributive_cpo = bounded_complete +
  assumes meet_join_distrib: "a ⇑ b ⟹ c ⊓ (a ⊔ b) = (c ⊓ a) ⊔ (c ⊓ b)"
begin

lemma meet_join_distrib2: "a ⇑ b ⟹ (a ⊔ b) ⊓ c = (a ⊓ c) ⊔ (b ⊓ c)"
  by (simp add: meet_join_distrib meet_sym)

lemma join_meet_distrib: "a ⊔ (b ⊓ c) = (a ⊔ b) ⊓ (a ⊔ c)" if "a ⇑ b" "a ⇑ c"
proof -
  have d1: "(a ⊔ b) ⊓ c = (a ⊓ c) ⊔ (b ⊓ c)"
    by (auto intro: meet_join_distrib2 ‹a ⇑ b›)
  have "a ⊔ (b ⊓ c) = (a ⊔ (a ⊓ c)) ⊔ (b ⊓ c)"
    by (simp add: join_absorb_meet2)
  also have "... = a ⊔ ((a ⊓ c) ⊔ (b ⊓ c))"
    by (rule join_assoc, simp_all add: that d1[symmetric])
  also have "... = a ⊔ ((a ⊔ b)⊓ c)"
    by (simp add: d1)
  also have "... = ((a ⊔ b) ⊓ a) ⊔ ((a ⊔ b) ⊓ c)"
    by (simp add: that)
  also have "... = (a ⊔ b) ⊓ (a ⊔ c)"
    by (simp add: that meet_join_distrib meet_sym)
  finally show ?thesis .
qed

end

class scott_domain = "domain" + distributive_cpo

section ‹Instances›

text ‹Two joinable elements of a flat domain are comparable, so distributivity is a case split.›

instance flat ⊆ distributive_cpo
proof
  fix a b c :: 'a assume ab: "a ⇑ b"
  have mb: "x ⊓ ⊥ = ⊥" for x :: 'a by (simp add: meet_sym meet_absorbI)
  from joinableD[OF ab] have "a = ⊥ ∨ b = ⊥ ∨ a = b"
    by (metis ax_flat)
  then show "c ⊓ (a ⊔ b) = c ⊓ a ⊔ (c ⊓ b)"
    by (elim disjE) (simp_all add: mb join_bottom)
qed

instance "fun" :: (type, distributive_cpo) distributive_cpo
proof
  fix a b c :: "'a ⇒ 'b"
  assume ab: "a ⇑ b"
  have ptw: "a x ⇑ b x" for x
  proof -
    from ab obtain z where "a ⊑ z" "b ⊑ z" unfolding joinable_def by blast
    thus "a x ⇑ b x" by (meson fun_belowD joinableI)
  qed
  have cab: "(c ⊓ a) ⇑ (c ⊓ b)" by (rule joinableI[OF meet_below1 meet_below1])
  have "c ⊓ (a ⊔ b) = (λx. c x ⊓ (a x ⊔ b x))"
    using ab by (simp add: fun_meet fun_join)
  also have "... = (λx. (c x ⊓ a x) ⊔ (c x ⊓ b x))"
    using ptw by (simp add: meet_join_distrib)
  also have "... = (c ⊓ a) ⊔ (c ⊓ b)"
    using cab by (simp add: fun_meet fun_join)
  finally show "c ⊓ (a ⊔ b) = (c ⊓ a) ⊔ (c ⊓ b)" .
qed

instance u :: (distributive_cpo) distributive_cpo
proof
  fix a b c :: "'a u" assume ab: "a ⇑ b"
  show "c ⊓ (a ⊔ b) = c ⊓ a ⊔ (c ⊓ b)"
  proof (cases a; cases b; cases c)
    fix a' b' c' assume *: "a = up⋅a'" "b = up⋅b'" "c = up⋅c'"
    with ab have a'b': "a' ⇑ b'" by simp
    have "c' ⊓ a' ⇑ c' ⊓ b'" by (rule joinableI[OF meet_below1 meet_below1])
    with * a'b' show ?thesis by (simp add: up_join meet_join_distrib)
  qed simp_all
qed

instance ssum :: (distributive_cpo, distributive_cpo) distributive_cpo
proof
  fix a b c :: "'a ⊕ 'b" assume ab: "a ⇑ b"
  have ml: "x ⊓ x' ⇑ x ⊓ x''" for x x' x'' :: 'a by (rule joinableI[OF meet_below1 meet_below1])
  have mr: "y ⊓ y' ⇑ y ⊓ y''" for y y' y'' :: 'b by (rule joinableI[OF meet_below1 meet_below1])
  show "c ⊓ (a ⊔ b) = c ⊓ a ⊔ (c ⊓ b)"
    using ab by (cases a; cases b; cases c)
      (simp_all add: sinl_join sinr_join meet_join_distrib ml mr)
qed

instance prod :: (distributive_cpo, distributive_cpo) distributive_cpo
proof
  fix a b c :: "'a × 'b" assume ab: "a ⇑ b"
  then have c: "fst a ⇑ fst b" "snd a ⇑ snd b" by (simp_all add: prod_joinable_iff)
  have "c ⊓ a ⇑ c ⊓ b" by (rule joinableI[OF meet_below1 meet_below1])
  with ab c show "c ⊓ (a ⊔ b) = c ⊓ a ⊔ (c ⊓ b)"
    by (simp add: prod_meet prod_join meet_join_distrib)
qed

section ‹Continuity of meet, and distributivity from compacts›

text ‹In a bifinite bounded-complete domain binary meet is continuous: this is the
  ‹approximants› observation — a compact below @{term ‹(⨆i. X i) ⊓ y›} already lies
  below some @{term ‹X i ⊓ y›}.›

lemma approximants_meet:
  "approximants ((x::'a::{bifinite, bounded_complete}) ⊓ y) = approximants x ∩ approximants y"
  unfolding approximants_def by (auto simp add: meet_below_iff)

lemma cont_meet_arg: "cont (λx::'a::{bifinite, bounded_complete}. x ⊓ y)"
proof -
  have mono: "monofun (λx::'a. x ⊓ y)"
  proof (rule monofunI)
    fix a b :: 'a assume "a ⊑ b"
    have "a ⊓ y ⊑ b" using meet_below1[of a y] ‹a ⊑ b› by (rule below_trans)
    moreover have "a ⊓ y ⊑ y" by (rule meet_below2)
    ultimately show "a ⊓ y ⊑ b ⊓ y" by (rule meet_greatest)
  qed
  show ?thesis
  proof (rule contI2[OF mono])
    fix X :: "nat ⇒ 'a" assume X: "chain X"
    assume chXy: "chain (λi. X i ⊓ y)"
    show "(⨆i. X i) ⊓ y ⊑ (⨆i. X i ⊓ y)"
      by (rule compact_basis.belowI) (simp add: compact_basis.rep_lub X chXy approximants_meet)
  qed
qed

lemma cont_meet_arg2: "cont (λy::'a::{bifinite, bounded_complete}. x ⊓ y)"
  using cont_meet_arg[of x] by (simp add: meet_sym)

lemma cont2cont_meet [simp, cont2cont]:
  fixes f g :: "'b::cpo ⇒ 'a::{bifinite, bounded_complete}"
  assumes contf: "cont f" and contg: "cont g" shows "cont (λx. f x ⊓ g x)"
  by (rule cont_apply [OF contg cont_meet_arg2 cont_compose [OF cont_meet_arg contf]])

lemma meet_lub_chains:
  fixes X Y :: "nat ⇒ 'a::{bifinite, bounded_complete}"
  assumes "chain X" "chain Y" shows "(⨆i. X i) ⊓ (⨆i. Y i) = (⨆i. X i ⊓ Y i)"
proof -
  have c: "cont (λp::'a × 'a. fst p ⊓ snd p)" by simp
  have ch: "chain (λi. (X i, Y i))" using assms by (simp add: chain_def)
  have "(⨆i. (X i, Y i)) = ((⨆i. X i), (⨆i. Y i))" using assms by (rule lub_Pair)
  with cont2contlubE[OF c ch] show ?thesis by simp
qed

text ‹Joins of pointwise joinable chains: the lubs are joinable, and joining commutes with lubs.›

lemma lub_join_chains:
  fixes X Y :: "nat ⇒ 'a::bounded_complete"
  assumes X: "chain X" and Y: "chain Y" and j: "⋀i. X i ⇑ Y i"
  shows "chain (λi. X i ⊔ Y i)" "(⨆i. X i) ⇑ (⨆i. Y i)"
    "(⨆i. X i ⊔ Y i) = (⨆i. X i) ⊔ (⨆i. Y i)"
proof -
  have XY: "X i ⊑ (⨆i. X i)" "Y i ⊑ (⨆i. Y i)" for i using X Y by (simp_all add: is_ub_thelub)
  show ch: "chain (λi. X i ⊔ Y i)"
  proof (rule chainI)
    fix i
    have "X i ⊑ X (Suc i) ⊔ Y (Suc i)" "Y i ⊑ X (Suc i) ⊔ Y (Suc i)"
      using chainE[OF X, of i] chainE[OF Y, of i] joinableD[OF j[of "Suc i"]]
      by (blast intro: below_trans)+
    then show "X i ⊔ Y i ⊑ X (Suc i) ⊔ Y (Suc i)" by (rule join_belowI)
  qed
  have "(⨆i. X i) ⊑ (⨆i. X i ⊔ Y i)" "(⨆i. Y i) ⊑ (⨆i. X i ⊔ Y i)"
    using joinableD[OF j] by (blast intro: lub_mono X Y ch)+
  then show jl: "(⨆i. X i) ⇑ (⨆i. Y i)" by (rule joinableI)
  show "(⨆i. X i ⊔ Y i) = (⨆i. X i) ⊔ (⨆i. Y i)"
  proof (rule below_antisym)
    show "(⨆i. X i ⊔ Y i) ⊑ (⨆i. X i) ⊔ (⨆i. Y i)"
    proof (rule lub_below[OF ch])
      fix i
      show "X i ⊔ Y i ⊑ (⨆i. X i) ⊔ (⨆i. Y i)"
        using XY joinableD[OF jl] by (blast intro: join_belowI below_trans)
    qed
    show "(⨆i. X i) ⊔ (⨆i. Y i) ⊑ (⨆i. X i ⊔ Y i)"
      using ‹(⨆i. X i) ⊑ _› ‹(⨆i. Y i) ⊑ _› by (rule join_belowI)
  qed
qed

text ‹Distributivity through any chain of maps with lub ‹ID› (e.g. the ‹take› functions of a
  domain, or an approximation chain): it suffices to meet with approximants ‹t n⋅c›, against
  arbitrary joinable ‹a›, ‹b›; the limit is taken in ‹c› only, by continuity of meet.›

lemma chain_distrib:
  fixes t :: "nat ⇒ 'a::{bifinite, bounded_complete} → 'a" and a b c :: 'a
  assumes t: "chain t" "(⨆i. t i) = ID"
    and law: "⋀n a b. a ⇑ b ⟹ t n⋅c ⊓ (a ⊔ b) = t n⋅c ⊓ a ⊔ t n⋅c ⊓ b"
    and "a ⇑ b"
  shows "c ⊓ (a ⊔ b) = c ⊓ a ⊔ c ⊓ b"
proof -
  have ch: "chain (λn. t n⋅c)" by (rule ch2ch_Rep_cfunL[OF t(1)])
  have reach: "(⨆n. t n⋅c) = c" by (simp add: contlub_cfun_fun[OF t(1), symmetric] t(2))
  have lim: "c ⊓ y = (⨆n. t n⋅c ⊓ y)" and chy: "chain (λn. t n⋅c ⊓ y)" for y
    using cont2contlubE[OF cont_meet_arg ch, of y] ch2ch_cont[OF cont_meet_arg ch, of y]
    by (simp_all add: reach)
  have j: "t n⋅c ⊓ a ⇑ t n⋅c ⊓ b" for n by (rule joinableI[OF meet_below1 meet_below1])
  have "c ⊓ (a ⊔ b) = (⨆n. t n⋅c ⊓ (a ⊔ b))" by (rule lim)
  also have "… = (⨆n. t n⋅c ⊓ a ⊔ t n⋅c ⊓ b)" by (simp only: law[OF ‹a ⇑ b›])
  also have "… = (⨆n. t n⋅c ⊓ a) ⊔ (⨆n. t n⋅c ⊓ b)" by (rule lub_join_chains(3)[OF chy chy j])
  also have "… = c ⊓ a ⊔ c ⊓ b" by (simp only: lim[symmetric])
  finally show ?thesis .
qed

lemma distributive_cpo_chainI:
  fixes t :: "nat ⇒ 'a::{bifinite, bounded_complete} → 'a"
  assumes t: "chain t" "(⨆i. t i) = ID"
    and law: "⋀n a b c. a ⇑ b ⟹ t n⋅c ⊓ (a ⊔ b) = t n⋅c ⊓ a ⊔ t n⋅c ⊓ b"
  shows "OFCLASS('a, distributive_cpo_class)"
  by intro_classes (rule chain_distrib[OF t law])

text ‹For a fixed ‹c›, the law against compact joinable pairs extends to all joinable pairs
  (approximating ‹a› and ‹b›; meet is continuous in its second argument).›

lemma compact_meet_distrib:
  fixes a b c :: "'a::{bifinite, bounded_complete}"
  assumes law: "⋀a b. compact a ⟹ compact b ⟹ a ⇑ b ⟹ c ⊓ (a ⊔ b) = c ⊓ a ⊔ c ⊓ b"
    and "a ⇑ b"
  shows "c ⊓ (a ⊔ b) = c ⊓ a ⊔ c ⊓ b"
proof -
  from bifinite obtain approx :: "nat ⇒ 'a → 'a" where "approx_chain approx" ..
  then interpret approx_chain approx .
  have reach: "(⨆i. approx i⋅x) = x" for x
    by (simp add: contlub_cfun_fun[OF chain_approx, symmetric])
  have ch: "chain (λi. approx i⋅x)" for x by (rule ch2ch_Rep_cfunL[OF chain_approx])
  have lim: "c ⊓ (⨆i. X i) = (⨆i. c ⊓ X i)" and chX: "chain (λi. c ⊓ X i)" if "chain X" for X
    using cont2contlubE[OF cont_meet_arg2 that] ch2ch_cont[OF cont_meet_arg2 that] by simp_all
  define A where "A = (λi. approx i⋅a)"
  define B where "B = (λi. approx i⋅b)"
  have chs: "chain A" "chain B" unfolding A_def B_def by (rule ch)+
  have AB: "A i ⇑ B i" for i
    unfolding A_def B_def using ‹a ⇑ b› by (rule cfun_arg_joinable)
  have j: "c ⊓ A i ⇑ c ⊓ B i" for i by (rule joinableI[OF meet_below1 meet_below1])
  have "c ⊓ (a ⊔ b) = c ⊓ ((⨆i. A i) ⊔ (⨆i. B i))" by (simp only: A_def B_def reach)
  also have "… = c ⊓ (⨆i. A i ⊔ B i)" by (simp only: lub_join_chains(3)[OF chs AB])
  also have "… = (⨆i. c ⊓ (A i ⊔ B i))" by (rule lim[OF lub_join_chains(1)[OF chs AB]])
  also have "… = (⨆i. c ⊓ A i ⊔ c ⊓ B i)"
    using AB by (simp add: law A_def B_def)
  also have "… = (⨆i. c ⊓ A i) ⊔ (⨆i. c ⊓ B i)"
    by (rule lub_join_chains(3)[OF chX[OF chs(1)] chX[OF chs(2)] j])
  also have "… = c ⊓ a ⊔ c ⊓ b"
    by (simp only: lim[OF chs(1), symmetric] lim[OF chs(2), symmetric]) (simp only: A_def B_def reach)
  finally show ?thesis .
qed

text ‹So distributivity of a bifinite bounded-complete domain need only be checked on compacts:
  the approximation-chain instance of @{text chain_distrib}.›

lemma distributive_cpo_compactI:
  assumes "⋀a b c::'a::{bifinite, bounded_complete}. compact a ⟹ compact b ⟹ compact c ⟹
    a ⇑ b ⟹ c ⊓ (a ⊔ b) = c ⊓ a ⊔ (c ⊓ b)"
  shows "OFCLASS('a, distributive_cpo_class)"
proof intro_classes
  fix a b c :: 'a assume "a ⇑ b"
  from bifinite obtain approx :: "nat ⇒ 'a → 'a" where "approx_chain approx" ..
  then interpret approx_chain approx .
  show "c ⊓ (a ⊔ b) = c ⊓ a ⊔ c ⊓ b"
    by (rule chain_distrib[OF chain_approx lub_approx _ ‹a ⇑ b›], rule compact_meet_distrib,
      rule assms) simp_all
qed
