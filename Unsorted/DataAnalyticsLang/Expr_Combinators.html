Theory Expr_Combinators

theory Expr_Combinators
  imports Content_Law
begin

section ‹Combinators›

text ‹Each combinator transforms the entries of its inner expressions' dims along with their
  semantics, and is lawful on lawful inner expressions.  Collecting a list of expressions into a
  tuple, followed by a map, expresses every operator of any arity.›

subsection ‹Tuples›

text ‹A collected entry is present where every component is; its value is a tuple that is lazy in
  the components, so an undefined component does not make the tuple undefined.  No order classes
  are needed for values; they will be once tuples are key values.›

domain ('a::"domain") ltuple = LNil | LCons (lazy "'a") "'a ltuple"

fun ltuple_of :: "'a::domain list ⇒ 'a ltuple" where
  "ltuple_of [] = LNil"
| "ltuple_of (x # xs) = LCons⋅x⋅(ltuple_of xs)"

lemma ltuple_of_not_bottom: "ltuple_of xs ≠ ⊥"
  by (induct xs) simp_all

lemma those_not_None_iff: "those xs ≠ None ⟷ (∀x∈set xs. x ≠ None)"
  by (induct xs) (auto split: option.splits)

context dataspace
begin

subsection ‹Keys›

text ‹The key at an LOD ‹c›.  Its only entry is the group of rows with that key part, the frame's
  filters coarsened to ‹c›, keyed there and reading no columns, so its content is that key part;
  it is present where that group is nonempty, so it is gated at its own row LOD, not at the frame's
  keys.›

lemma join_meet_self: "c ⊔ (c ⊓ L) = c" for c :: "'k semi_factor"
  by (rule lattice_join_absorb1[OF lattice_meet_le1])

definition key_at :: "'k semi_factor ⇒ ('k, 'f, 'r, 'k) expr" where
  "key_at c = ⦇emind = c, edims = [mk_entry (sel c c) gtrue (⊥, c)],
     esem = λdata F k. if group data c c F k = {} then None else Some (proj c⋅k),
     csem = λXs. if hd Xs = Map.empty then None else Some (the_elem (dom (hd Xs)))⦈"

lemma key_at_simps [simp]:
  "emind (key_at c) = c" "edims (key_at c) = [mk_entry (sel c c) gtrue (⊥, c)]"
  "esem (key_at c) data F k = (if group data c c F k = {} then None else Some (proj c⋅k))"
  "csem (key_at c) Xs = (if hd Xs = Map.empty then None else Some (the_elem (dom (hd Xs))))"
  by (simp_all add: key_at_def)

lemma group_own: "y ∈ group data c c F k ⟹ rproj c y = proj c⋅k"
  by (simp add: group_def join_meet_self)

lemma content_key_at:
  "content data (mk_entry (sel c c) gtrue (⊥, c)) F k = (if group data c c F k = {} then Map.empty else [proj c⋅k ↦ ⊥])"
proof -
  have "cgraph data (mk_entry (sel c c) gtrue (⊥, c)) F k = (λ_. (proj c⋅k, ⊥)) ` group data c c F k"
    unfolding cgraph_def efiber_def cpair_def by (rule image_cong) (simp_all add: group_own semi_factor_p_bot)
  then show ?thesis unfolding content_def by (simp add: image_constant_conv)
qed

theorem semilawful_key_at: "semilawful (key_at c)"
proof (rule semilawfulI, simp_all only: key_at_simps)
  show "entry_wf e" if "e ∈ set [mk_entry (sel c c) gtrue (⊥, c)]" for e
    using that by (simp add: entry_wf_view keyed_view_key supported_gtrue)
  show "c = fold (⊔) (map elower [mk_entry (sel c c) gtrue (⊥, c)]) ⊥" by (simp add: lattice_join_bot_right)
  show "(if group data c c F k = {} then None else Some (proj c⋅k))
      = (if hd (contents data [mk_entry (sel c c) gtrue (⊥, c)] F k) = Map.empty then None
         else Some (the_elem (dom (hd (contents data [mk_entry (sel c c) gtrue (⊥, c)] F k)))))" for data F k
    by (simp only: list.map list.sel content_key_at) simp
qed

subsection ‹Table columns›

text ‹A table column reads a section of the row.  An external oracle names the key part that
  determines it in every conforming dataset; then the rows of a key's group carry one section,
  and the column's value is the column function of it.  Its view is that section, keyed at the
  oracle's key part.›

definition column :: "'r semi_factor ⇒ 'k semi_factor ⇒ ('r → 'w::cpo) ⇒ ('k, 'f, 'r, 'w) expr" where
  "column c a v = ⦇emind = a, edims = [mk_entry (sel a a) gtrue (c, a)],
     esem = λdata F k. if group data a a F k = {} then None
       else Some (THE w. w ∈ (λy. v⋅(y ⇂ c)) ` group data a a F k),
     csem = λXs. if hd Xs = Map.empty then None else Some (v⋅(the (hd Xs (the_elem (dom (hd Xs))))))⦈"

lemma column_simps [simp]:
  "emind (column c a v) = a" "edims (column c a v) = [mk_entry (sel a a) gtrue (c, a)]"
  "esem (column c a v) data F k = (if group data a a F k = {} then None
       else Some (THE w. w ∈ (λy. v⋅(y ⇂ c)) ` group data a a F k))"
  "csem (column c a v) Xs = (if hd Xs = Map.empty then None else Some (v⋅(the (hd Xs (the_elem (dom (hd Xs)))))))"
  by (simp_all add: column_def)

text ‹The rows of a column's group carry one section.›

lemma column_group_section:
  assumes fd: "universally (λP. vfd P (⊥, a) (c, ⊥))"
    and d: "conforms data" and x: "x ∈ group data a a F k" and y: "y ∈ group data a a F k"
  shows "y ⇂ c = x ⇂ c"
proof -
  have "y ⇂ fst (c, a) = x ⇂ fst (c, a)"
  proof (rule view_agree[OF keyed_view_columns[OF fd] d])
    show "x ∈ lower_maximal data" "y ∈ lower_maximal data" using x y by (simp_all add: group_def)
    show "rproj a y = rproj a x" using x y by (simp add: group_own)
  qed simp
  then show ?thesis by simp
qed

theorem semilawful_column:
  assumes fd: "universally (λP. vfd P (⊥, a) (c, ⊥))"
  shows "semilawful (column c a v)"
proof (rule semilawfulI, simp_all only: column_simps)
  show "entry_wf e" if "e ∈ set [mk_entry (sel a a) gtrue (c, a)]" for e
    using that by (simp add: entry_wf_view keyed_view_columns[OF fd] supported_gtrue)
  show "a = fold (⊔) (map elower [mk_entry (sel a a) gtrue (c, a)]) ⊥"
    by (simp add: lattice_join_bot_right)
  show "(if group data a a F k = {} then None else Some (THE w. w ∈ (λy. v⋅(y ⇂ c)) ` group data a a F k))
      = (if hd (contents data [mk_entry (sel a a) gtrue (c, a)] F k) = Map.empty then None
         else Some (v⋅(the (hd (contents data [mk_entry (sel a a) gtrue (c, a)] F k)
           (the_elem (dom (hd (contents data [mk_entry (sel a a) gtrue (c, a)] F k))))))))"
    if d: "conforms data" for data F k
  proof (cases "group data a a F k = {}")
    case False
    then obtain x where x: "x ∈ group data a a F k" by blast
    have sec: "(λy. y ⇂ c) ` group data a a F k = {x ⇂ c}"
      using column_group_section[OF fd d x] x by blast
    have "cgraph data (mk_entry (sel a a) gtrue (c, a)) F k = (λ_. (proj a⋅k, x ⇂ c)) ` group data a a F k"
      unfolding cgraph_def efiber_def cpair_def
      by (rule image_cong) (simp_all add: group_own column_group_section[OF fd d x])
    then have C: "content data (mk_entry (sel a a) gtrue (c, a)) F k = [proj a⋅k ↦ x ⇂ c]"
      unfolding content_def using False by (simp add: image_constant_conv)
    have V: "(λy. v⋅(y ⇂ c)) ` group data a a F k = {v⋅(x ⇂ c)}"
      using sec by (simp only: image_image[of "Rep_cfun v" "λy. y ⇂ c", symmetric, unfolded comp_def]) simp
    show ?thesis using False by (simp only: list.map list.sel C V) simp
  next
    case True
    have "cgraph data (mk_entry (sel a a) gtrue (c, a)) F k = {}" using True by (simp add: cgraph_def efiber_def)
    then show ?thesis using True by (simp add: content_def)
  qed
qed

subsection ‹Maps›

definition smap :: "('v::cpo → 'w::cpo) ⇒ ('k, 'f, 'r, 'v) expr ⇒ ('k, 'f, 'r, 'w) expr" where
  "smap g E = ⦇emind = emind E, edims = edims E, esem = λdata F k. map_option (Rep_cfun g) (esem E data F k),
     csem = λXs. map_option (Rep_cfun g) (csem E Xs)⦈"

lemma smap_simps [simp]:
  "emind (smap g E) = emind E" "edims (smap g E) = edims E"
  "esem (smap g E) data F k = map_option (Rep_cfun g) (esem E data F k)"
  "csem (smap g E) Xs = map_option (Rep_cfun g) (csem E Xs)"
  by (simp_all add: smap_def)

theorem semilawful_smap: assumes "semilawful E" shows "semilawful (smap g E)"
proof -
  have ES: "semilawful E" by (fact assms)
  show ?thesis
  proof (rule semilawfulI, simp_all only: smap_simps)
    show "entry_wf e" if "e ∈ set (edims E)" for e by (rule semilawful_wf[OF ES that])
    show "emind E = fold (⊔) (map elower (edims E)) ⊥" by (rule semilawful_mind_lub[OF ES])
    show "map_option (Rep_cfun g) (esem E data F k) = map_option (Rep_cfun g) (csem E (contents data (edims E) F k))"
      if "conforms data" "proj (emind E ⊔ flod F)⋅k = k" for data F k
      by (simp only: semilawful_csem[OF ES that])
  qed
qed

theorem lawful_smap: assumes "lawful E" shows "lawful (smap g E)"
proof -
  have EL: "lawful E" by (fact assms)
  show ?thesis
  proof (rule lawful_semiI[OF semilawful_smap[OF lawful_semilawful[OF assms]]], simp_all only: smap_simps)
    show "(eupper e ⊓ emind E) ⊑ elower e" if "e ∈ set (edims E)" for e
      by (rule lawful_saturated[OF EL that])
    show "keyimg data F (emind E ⊔ flod F) k" if d: "conforms data" and "map_option (Rep_cfun g) (esem E data F k) ≠ None" for data F k
      using that(2) by (intro lawful_key_law[OF EL d]) simp
  qed
qed

text ‹A function of the key at an LOD.›

definition value_at :: "'k semi_factor ⇒ ('k → 'v::cpo) ⇒ ('k, 'f, 'r, 'v) expr" where
  "value_at c g = smap g (key_at c)"

lemma value_at_simps [simp]:
  "emind (value_at c g) = c" "edims (value_at c g) = [mk_entry (sel c c) gtrue (⊥, c)]"
  "esem (value_at c g) data F k = (if group data c c F k = {} then None else Some (g⋅(proj c⋅k)))"
  "csem (value_at c g) Xs = (if hd Xs = Map.empty then None else Some (g⋅(the_elem (dom (hd Xs)))))"
  by (simp_all add: value_at_def)

theorem semilawful_value_at: "semilawful (value_at c g)"
  unfolding value_at_def by (rule semilawful_smap[OF semilawful_key_at])

subsection ‹‹RESTRICT››

text ‹Restricting conjoins the rows a test passes to the ambient filter: inner fibers are the
  outer fibers' rows passing it, so the test is conjoined to each entry's guard.›

definition restrict_entry :: "('r, 'k) guard ⇒ ('k, 'r) entry ⇒ ('k, 'r) entry" where
  "restrict_entry γ e = e⦇eguard := gand γ (eguard e)⦈"

lemma restrict_entry_simps [simp]:
  "esel (restrict_entry γ e) = esel e"
  "eguard (restrict_entry γ e) = gand γ (eguard e)" "eview (restrict_entry γ e) = eview e"
  by (simp_all add: restrict_entry_def)

definition restrict :: "('r, 'k) guard ⇒ ('k, 'f, 'r, 'v) expr ⇒ ('k, 'f, 'r, 'v) expr" where
  "restrict γ E = ⦇emind = emind E, edims = map (restrict_entry γ) (edims E),
     esem = λdata F. esem E data (frestrict (gsel γ) F), csem = csem E⦈"

lemma restrict_simps [simp]:
  "emind (restrict γ E) = emind E"
  "edims (restrict γ E) = map (restrict_entry γ) (edims E)"
  "esem (restrict γ E) data F = esem E data (frestrict (gsel γ) F)"
  "csem (restrict γ E) = csem E"
  by (simp_all add: restrict_def)

lemma group_frestrict:
  "group data a ρ (frestrict φ F) k = {y ∈ group data a ρ F k. selects φ y}"
  by (auto simp: group_def)

lemma content_frestrict:
  "content data e (frestrict (gsel γ) F) k = content data (restrict_entry γ e) F k"
  unfolding content_def cgraph_def efiber_def cpair_def by (auto simp: group_frestrict)

theorem semilawful_restrict:
  assumes g: "supported γ" and ES: "semilawful E"
  shows "semilawful (restrict γ E)"
proof (rule semilawfulI, simp_all only: restrict_simps)
  show "entry_wf e" if "e ∈ set (map (restrict_entry γ) (edims E))" for e
    using that semilawful_wf[OF ES] entry_wf_guard[OF _ g] by (auto simp: restrict_entry_def)
  show "emind E = fold (⊔) (map elower (map (restrict_entry γ) (edims E))) ⊥"
    by (simp add: comp_def semilawful_mind_lub[OF ES])
  show "esem E data (frestrict (gsel γ) F) k = csem E (contents data (map (restrict_entry γ) (edims E)) F k)"
    if d: "conforms data" and k: "proj (emind E ⊔ flod F)⋅k = k" for data F k
  proof -
    have "proj (emind E ⊔ flod (frestrict (gsel γ) F))⋅k = k" using k by simp
    then show ?thesis by (simp add: semilawful_csem[OF ES d] content_frestrict comp_def)
  qed
qed

theorem lawful_restrict:
  assumes g: "supported γ" and EL: "lawful E"
  shows "lawful (restrict γ E)"
proof (rule lawful_semiI[OF semilawful_restrict[OF g lawful_semilawful[OF EL]]], simp_all only: restrict_simps)
  show "(eupper e ⊓ emind E) ⊑ elower e" if "e ∈ set (map (restrict_entry γ) (edims E))" for e
    using that by (auto intro: lawful_saturated[OF EL])
  show "keyimg data F (emind E ⊔ flod F) k" if d: "conforms data" and "esem E data (frestrict (gsel γ) F) k ≠ None" for data F k
  proof -
    have "keyimg data (frestrict (gsel γ) F) (emind E ⊔ flod F) k"
      using lawful_key_law[OF EL that] by simp
    then show ?thesis by (rule keyimg_mono[rotated]) (simp add: frows_frestrict)
  qed
qed

subsection ‹‹FOR› and ‹FIXED››

text ‹‹E FOR τ G›: the inner expression under the frame with LOD and slice truncated to ‹τ›
  and ‹G› conjoined, at the truncated key.  Its upper parts meet ‹τ›; each entry's guard is
  conjoined with ‹G› coarsened to the entry's upper part.  It is not gated: the result is
  broadcast over the dropped detail.  ‹FIXED D› is ‹FOR D› without a filter.›

definition for_entry :: "'k semi_factor ⇒ 'f ⇒ ('k, 'r) entry ⇒ ('k, 'r) entry" where
  "for_entry τ G e = e⦇esel := sel (elower e) (eupper e ⊓ τ), eguard := gand (guard_holds (coarsen (eupper e) G)) (eguard e)⦈"

lemma for_entry_simps [simp]:
  "elower (for_entry τ G e) = elower e" "eupper (for_entry τ G e) = eupper e ⊓ τ"
  "eguard (for_entry τ G e) = gand (guard_holds (coarsen (eupper e) G)) (eguard e)"
  "eview (for_entry τ G e) = eview e"
  by (simp_all add: for_entry_def)

definition filtfor :: "'k semi_factor ⇒ 'f ⇒ ('k, 'f, 'r, 'v) expr ⇒ ('k, 'f, 'r, 'v) expr" where
  "filtfor τ G E = ⦇emind = emind E, edims = map (for_entry τ G) (edims E),
     esem = λdata F k. esem E data (ffor τ G F) (proj (emind E ⊔ flod (ffor τ G F))⋅k), csem = csem E⦈"

abbreviation fixed :: "'k semi_factor ⇒ ('k, 'f, 'r, 'v) expr ⇒ ('k, 'f, 'r, 'v) expr" where
  "fixed D ≡ filtfor D ❙1"

lemma filtfor_simps [simp]:
  "emind (filtfor τ G E) = emind E"
  "edims (filtfor τ G E) = map (for_entry τ G) (edims E)"
  "esem (filtfor τ G E) data F k = esem E data (ffor τ G F) (proj (emind E ⊔ flod (ffor τ G F))⋅k)"
  "csem (filtfor τ G E) = csem E"
  by (simp_all add: filtfor_def)

text ‹Inside the truncated frame, at the truncated key, a group is the outer group through the
  composite mask, cut down by the filter coarsened to the inner upper part.›

lemma group_ffor:
  assumes am: "a ⊑ m"
  shows "group data a ρ (ffor τ G F) (proj (m ⊔ flod (ffor τ G F))⋅k)
       = {y ∈ group data a (ρ ⊓ τ) F k. holds (coarsen ρ G) (rkey⋅y)}"
proof -
  let ?X = "a ⊔ ((ρ ⊓ τ) ⊓ flod F)"
  have lod: "a ⊔ (ρ ⊓ flod (ffor τ G F)) = ?X" by (simp add: truncate_inf_comp)
  have le: "?X ⊑ m ⊔ flod (ffor τ G F)"
    unfolding flod_ffor truncate_inf_comp by (rule lattice_join_mono[OF am truncate_below])
  have key: "proj ?X⋅(proj (m ⊔ flod (ffor τ G F))⋅k) = proj ?X⋅k" by (rule proj_le[OF le])
  have sl: "holds (coarsen ρ (slice (ffor τ G F))) k' ⟷
      holds (coarsen (ρ ⊓ τ) (slice F)) k' ∧ holds (coarsen ρ G) k'" for k'
    by (simp add: coarsen_conj coarsen_coarsen)
  show ?thesis unfolding group_def lod key sl by auto
qed

lemma content_ffor:
  assumes "elower e ⊑ m"
  shows "content data e (ffor τ G F) (proj (m ⊔ flod (ffor τ G F))⋅k) = content data (for_entry τ G e) F k"
  unfolding content_def cgraph_def efiber_def cpair_def group_ffor[OF assms] by auto

lemma content_const_top:
  assumes "proj (c ⊔ flod F)⋅k = k"
  shows "content data (mk_entry (sel c lattice_top) gtrue ⊥) F k
       = (if keyimg data F (c ⊔ flod F) k then [⊥ ↦ ⊥] else Map.empty)"
proof -
  have "cgraph data (mk_entry (sel c lattice_top) gtrue ⊥) F k = (λ_. (⊥, ⊥)) ` group data c lattice_top F k"
    unfolding cgraph_def efiber_def cpair_def by (simp add: proj_bot semi_factor_p_bot)
  moreover have "gmap {⊥ :: 'k × 'r} = [⊥ ↦ ⊥]" using gmap_singleton[of "⊥::'k" "⊥::'r"] by simp
  ultimately show ?thesis unfolding content_def by (simp add: image_constant_conv keyimg_group_top[OF assms])
qed

theorem semilawful_filtfor: assumes "semilawful E" shows "semilawful (filtfor τ G E)"
proof -
  have ES: "semilawful E" by (fact assms)
  let ?m = "emind E"
  have inner: "contents data (edims E) (ffor τ G H) (proj (?m ⊔ flod (ffor τ G H))⋅κ)
      = contents data (map (for_entry τ G) (edims E)) H κ" for data H κ
    unfolding map_map comp_def by (rule map_cong[OF refl], rule content_ffor[OF lower_below_mind[OF ES]])
  show ?thesis
  proof (rule semilawfulI, simp_all only: filtfor_simps)
    show "entry_wf e" if "e ∈ set (map (for_entry τ G) (edims E))" for e
      using that semilawful_wf[OF ES]
      by (auto simp: entry_wf_def supported_gand supported_guard_holds)
    show "?m = fold (⊔) (map elower (map (for_entry τ G) (edims E))) ⊥"
      by (simp add: comp_def semilawful_mind_lub[OF ES])
    show "esem E data (ffor τ G F) (proj (?m ⊔ flod (ffor τ G F))⋅k)
        = csem E (contents data (map (for_entry τ G) (edims E)) F k)"
      if d: "conforms data" for data F k
      by (simp only: semilawful_csem[OF ES d proj_idem] inner)
  qed
qed

subsection ‹Including dimensions›

text ‹Evaluating under the frame raised by ‹M› is the same as raising each entry's lower part by
  its masked part of ‹M›: the entry-level form of ‹finclude›.›

definition include_entry :: "'k semi_factor ⇒ ('k, 'r) entry ⇒ ('k, 'r) entry" where
  "include_entry M e = e⦇esel := sel (elower e ⊔ (eupper e ⊓ M)) (eupper e)⦈"

lemma include_entry_simps [simp]:
  "elower (include_entry M e) = elower e ⊔ (eupper e ⊓ M)" "eupper (include_entry M e) = eupper e"
  "eguard (include_entry M e) = eguard e" "eview (include_entry M e) = eview e"
  by (simp_all add: include_entry_def)

lemma content_include_entry: "content data (include_entry M e) F k = content data e (finclude M F) k"
proof -
  have lod: "(elower e ⊔ (eupper e ⊓ M)) ⊔ (eupper e ⊓ flod F) = elower e ⊔ (eupper e ⊓ flod (finclude M F))"
    by (simp add: flod_finclude truncate_sup lattice_join_ac)
  have "group data (elower e ⊔ (eupper e ⊓ M)) (eupper e) F k = group data (elower e) (eupper e) (finclude M F) k"
    by (rule group_cong) (simp_all only: amb_finclude slice_finclude lod)
  then show ?thesis unfolding content_def cgraph_def efiber_def cpair_def by simp
qed

lemma entry_wf_include_entry: "entry_wf e ⟹ entry_wf (include_entry M e)"
  unfolding entry_wf_def include_entry_simps .

subsection ‹The gate›

text ‹The gate turns a semilawful expression into a lawful one: it evaluates under the frame
  including the minimal LOD, which saturates the entries (their lower parts take in the masked
  minimal LOD), and it is present exactly at the frame's data keys, which a presence entry at the
  full key records.›

definition gate :: "('k, 'f, 'r, 'v) expr ⇒ ('k, 'f, 'r, 'v) expr" where
  "gate E = ⦇emind = emind E,
     edims = mk_entry (sel (emind E) lattice_top) gtrue ⊥ # map (include_entry (emind E)) (edims E),
     esem = λdata F k. if keyimg data F (emind E ⊔ flod F) k
       then esem E data (finclude (emind E) F) k else None,
     csem = λXs. if hd Xs = Map.empty then None else csem E (tl Xs)⦈"

lemma gate_simps [simp]:
  "emind (gate E) = emind E"
  "edims (gate E) = mk_entry (sel (emind E) lattice_top) gtrue ⊥ # map (include_entry (emind E)) (edims E)"
  "esem (gate E) data F k = (if keyimg data F (emind E ⊔ flod F) k
       then esem E data (finclude (emind E) F) k else None)"
  "csem (gate E) Xs = (if hd Xs = Map.empty then None else csem E (tl Xs))"
  by (simp_all add: gate_def)

theorem lawful_gate: assumes "semilawful E" shows "lawful (gate E)"
proof -
  have ES: "semilawful E" by (fact assms)
  let ?m = "emind E"
  let ?h = "mk_entry (sel ?m lattice_top) gtrue ⊥ :: ('k, 'r) entry"
  have kd: "?m ⊔ flod (finclude ?m F) = ?m ⊔ flod F" for F
    by (simp add: flod_finclude lattice_join_ac)
  show ?thesis
  proof (rule lawfulI, simp_all only: gate_simps)
    show "entry_wf e" if "e ∈ set (?h # map (include_entry ?m) (edims E))" for e
      using that by (auto intro: entry_wf_view keyed_view_bot_view supported_gtrue entry_wf_include_entry semilawful_wf[OF ES])
    show "(eupper e ⊓ ?m) ⊑ elower e" if "e ∈ set (?h # map (include_entry ?m) (edims E))" for e
      using that by (auto simp: truncate_top intro: lattice_join_ge2)
    show "?m = fold (⊔) (map elower (?h # map (include_entry ?m) (edims E))) ⊥"
    proof -
      have "fold (⊔) (map elower (map (include_entry ?m) (edims E))) ?m = ?m"
        by (rule fold_join_absorb)
          (auto intro: lattice_join_least[OF lower_below_mind[OF ES] truncate_below])
      then show ?thesis by (simp add: lattice_join_bot_right)
    qed
    show "keyimg data F (?m ⊔ flod F) k"
      if "conforms data" and "(if keyimg data F (?m ⊔ flod F) k then esem E data (finclude ?m F) k else None) ≠ None"
      for data F k
      using that by (simp split: if_splits)
    show "(if keyimg data F (?m ⊔ flod F) k then esem E data (finclude ?m F) k else None)
        = (if hd (contents data (?h # map (include_entry ?m) (edims E)) F k) = Map.empty then None
           else csem E (tl (contents data (?h # map (include_entry ?m) (edims E)) F k)))"
      if d: "conforms data" and k: "proj (?m ⊔ flod F)⋅k = k" for data F k
    proof -
      have h: "hd (contents data (?h # map (include_entry ?m) (edims E)) F k) = Map.empty
          ⟷ ¬ keyimg data F (?m ⊔ flod F) k"
        by (simp only: list.map list.sel content_const_top[OF k]) simp
      have t: "tl (contents data (?h # map (include_entry ?m) (edims E)) F k)
          = contents data (edims E) (finclude ?m F) k"
        by (simp add: comp_def content_include_entry)
      have "proj (?m ⊔ flod (finclude ?m F))⋅k = k" by (simp only: kd k)
      then show ?thesis by (simp only: h t semilawful_csem[OF ES d]) simp
    qed
  qed
qed

subsection ‹Collect›

text ‹The components are evaluated under the frame itself, each at its own key; their entries are
  the collected expression's entries, unchanged.  Collecting is semilawful: it is not gated, and a
  component sees nothing of the others' minimal LODs.  The gate widens it back to the full key.›

text ‹Cutting a list into consecutive blocks of given lengths.›

fun blocks :: "nat list ⇒ 'a list ⇒ 'a list list" where
  "blocks [] xs = []"
| "blocks (n # ns) xs = take n xs # blocks ns (drop n xs)"

lemma blocks_concat: "blocks (map (λx. length (f x)) xs) (concat (map f xs)) = map f xs"
  by (induct xs) simp_all

definition collect :: "('k, 'f, 'r, 'a::domain) expr list ⇒ ('k, 'f, 'r, 'a ltuple) expr" where
  "collect Es = ⦇emind = fold (⊔) (map emind Es) ⊥, edims = concat (map edims Es),
     esem = λdata F k. map_option ltuple_of (those (map (λE. esem E data F (proj (kd E F)⋅k)) Es)),
     csem = λXs. map_option ltuple_of (those (map (λ(E, Ys). csem E Ys) (zip Es (blocks (map (λE. length (edims E)) Es) Xs))))⦈"

lemma collect_simps [simp]:
  "emind (collect Es) = fold (⊔) (map emind Es) ⊥"
  "edims (collect Es) = concat (map edims Es)"
  "esem (collect Es) data F k = map_option ltuple_of (those (map (λE. esem E data F (proj (kd E F)⋅k)) Es))"
  "csem (collect Es) Xs = map_option ltuple_of (those (map (λ(E, Ys). csem E Ys) (zip Es (blocks (map (λE. length (edims E)) Es) Xs))))"
  by (simp_all add: collect_def)

text ‹The contents of a semilawful expression only see the key at its own key LOD.›

lemma contents_own_key:
  assumes E: "semilawful E"
  shows "contents data (edims E) F (proj (kd E F)⋅k) = contents data (edims E) F k"
proof (rule map_cong[OF refl])
  fix e assume e: "e ∈ set (edims E)"
  have "proj (elower e ⊔ (eupper e ⊓ flod F))⋅(proj (kd E F)⋅k) = proj (elower e ⊔ (eupper e ⊓ flod F))⋅k"
    by (rule proj_le[OF group_below_kd[OF E e]])
  then show "content data e F (proj (kd E F)⋅k) = content data e F k"
    by (intro content_group_cong) (simp add: group_def)
qed

theorem semilawful_collect:
  assumes law: "⋀E. E ∈ set Es ⟹ semilawful E"
  shows "semilawful (collect Es)"
proof (rule semilawfulI, simp_all only: collect_simps)
  show "entry_wf e" if "e ∈ set (concat (map edims Es))" for e
    using that by (auto intro: semilawful_wf[OF law])
  show "fold (⊔) (map emind Es) ⊥ = fold (⊔) (map elower (concat (map edims Es))) ⊥"
  proof (rule below_antisym)
    let ?R = "fold (⊔) (map elower (concat (map edims Es))) ⊥"
    show "fold (⊔) (map emind Es) ⊥ ⊑ ?R"
    proof (rule fold_join_least)
      fix x assume "x ∈ set (map emind Es)"
      then obtain E where E: "E ∈ set Es" and x: "x = emind E" by auto
      show "x ⊑ ?R" unfolding x semilawful_mind_lub[OF law[OF E]]
      proof (rule fold_join_least)
        fix a assume "a ∈ set (map elower (edims E))"
        then show "a ⊑ ?R" by (intro fold_join_upper) (use E in force)
      qed simp
    qed simp
    show "?R ⊑ fold (⊔) (map emind Es) ⊥"
    proof (rule fold_join_least)
      fix a assume "a ∈ set (map elower (concat (map edims Es)))"
      then obtain E e where E: "E ∈ set Es" and e: "e ∈ set (edims E)" and a: "a = elower e" by auto
      show "a ⊑ fold (⊔) (map emind Es) ⊥"
        unfolding a by (rule below_trans[OF lower_below_mind[OF law[OF E] e] fold_join_upper]) (simp add: E)
    qed simp
  qed
  show "map_option ltuple_of (those (map (λE. esem E data F (proj (kd E F)⋅k)) Es))
      = map_option ltuple_of (those (map (λ(E, Ys). csem E Ys)
          (zip Es (blocks (map (λE. length (edims E)) Es) (contents data (concat (map edims Es)) F k)))))"
    if d: "conforms data" for data F k
  proof -
    have c: "contents data (concat (map edims Es)) F k = concat (map (λE. contents data (edims E) F k) Es)"
      by (simp add: map_concat comp_def)
    have b: "blocks (map (λE. length (edims E)) Es) (contents data (concat (map edims Es)) F k)
        = map (λE. contents data (edims E) F k) Es"
      using blocks_concat[of "λE. contents data (edims E) F k" Es] by (simp only: c length_map)
    have z: "map (λ(E, Ys). csem E Ys) (zip Es (map (λE. contents data (edims E) F k) Es))
        = map (λE. csem E (contents data (edims E) F k)) Es"
      by (induct Es) simp_all
    have "esem E data F (proj (kd E F)⋅k) = csem E (contents data (edims E) F k)" if E: "E ∈ set Es" for E
      by (simp only: semilawful_csem[OF law[OF E] d proj_idem] contents_own_key[OF law[OF E]])
    then show ?thesis by (simp only: b z cong: map_cong)
  qed
qed
