Theory Congruence

theory Congruence
  imports HOLCF
begin

default_sort pcpo

section ‹Continuous congruences (functional-dependence relations)›

text ‹
  A ∗‹continuous congruence› on a cpo \<^typ>‹'a› is a relation \<^term>‹R› that is transitive,
  refines the cpo order \<^term>‹(⊑)›, and is admissible (Scott-closed) in each argument: a
  coarser ``below''.  The motivating instance is the kernel order \<^term>‹λx y. f⋅x ⊑ f⋅y› of a
  continuous map; a bigger relation identifies more points, i.e.\ is ∗‹coarser›.

  These are used only as ∗‹predicates›, to state functional dependence between levels of
  detail; levels of detail themselves are syntactic dimension sets (see the frame
  development).  We deliberately do not build a type of congruences: the only reason to do so
  would be meets, and finitely flat-generated congruences are not closed under them.
›

subsection ‹Continuous congruences (admissible refining preorders)›

locale is_cong =
  fixes R :: "'a ⇒ 'a ⇒ bool"
  assumes trans: "R x y ⟹ R y z ⟹ R x z"
    and belowD: "x ⊑ y ⟹ R x y"
    and adm1: "adm (λx. R x y)"
    and adm2: "adm (λy. R x y)"
begin
lemmas axioms = trans belowD adm1 adm2
end

lemma (in is_cong) refl: "R x x"
  by (auto simp: trans belowD)

lemma is_cong_top: "is_cong top"
  by (simp add: is_cong_def)

lemma is_cong_below: "is_cong (λx y. x ⊑ y)"
  by (auto simp: is_cong_def intro: below_trans adm_below cont_id cont_const)

lemma is_cong_Inter:
  assumes "⋀i. i ∈ I ⟹ is_cong (R i)"
  shows "is_cong (λx y. ∀i∈I. R i x y)"
proof
  show "∀i ∈ I. R i x y ⟹ ∀i ∈ I. R i y z ⟹ ∀i ∈ I. R i x z" for x y z
    using assms[THEN is_cong.trans] by blast
  show "x ⊑ y ⟹ ∀i ∈ I. R i x y" for x y
    using assms[THEN is_cong.belowD] by blast
  show "adm (λx. ∀i ∈ I. R i x y)" for y
    using assms[THEN is_cong.adm1] adm_ball[of I] by force 
  show "adm (λy. ∀i ∈ I. R i x y)" for x
    using assms[THEN is_cong.adm2] adm_ball[of I] by force
qed

lemma Inf_iff: "Inf ℛ = (λx y. ∀R ∈ ℛ. R x y)"
  by fast

corollary is_cong_Inf: "(⋀R. R ∈ ℛ ⟹ is_cong R) ⟹ is_cong (Inf ℛ)"
  unfolding Inf_iff
  using is_cong_Inter[of "ℛ" "(λx. x)"] .

lemma inf2_def: "inf R S = (λx y. R x y ∧ S x y)"
  by auto

lemma is_cong_inf2:
  assumes "is_cong R" "is_cong S" shows "is_cong (inf R S)"
  unfolding inf2_def
proof
  from assms show "R x y ∧ S x y ⟹ R y z ∧ S y z ⟹ R x z ∧ S x z" for x y z
    by (blast dest: is_cong.trans)
  from assms show "x ⊑ y ⟹ R x y ∧ S x y" for x y
    by (blast dest: is_cong.belowD)
  from assms show "adm (λx. R x y ∧ S x y)" for y
    by (intro adm_conj) (blast intro: assms dest: is_cong.adm1)+
  from assms show "adm (λy. R x y ∧ S x y)" for x
    by (intro adm_conj) (blast intro: assms dest: is_cong.adm2)+
qed

subsection ‹The generated congruence (least cong above a relation)›

definition gen :: "('a ⇒ 'a ⇒ bool) ⇒ ('a ⇒ 'a ⇒ bool)" where
  "gen B = (λx y. ∀R. is_cong R ⟶ (∀u v. B u v ⟶ R u v) ⟶ R x y)"

lemma is_cong_gen: "is_cong (gen B)"
proof -
  have *: "gen B = (λx y. ∀R∈{R. is_cong R ∧ (∀u v. B u v ⟶ R u v)}. R x y)"
    by (auto simp: gen_def fun_eq_iff)
  show ?thesis unfolding * by (rule is_cong_Inter) auto
qed

lemma gen_upper: "B x y ⟹ gen B x y"
  by (auto simp: gen_def)

lemma gen_least: "is_cong R ⟹ (⋀u v. B u v ⟹ R u v) ⟹ gen B x y ⟹ R x y"
  by (auto simp: gen_def)


lemma map_is_cong: assumes "is_cong R" "cont f" shows "is_cong (inv_imagep R f)"
  unfolding inv_imagep_def
proof (standard)
  from assms(1) show "R (f x) (f y) ⟹ R (f y) (f z) ⟹ R (f x) (f z)" for x y z
    by (rule is_cong.trans)
  from assms(1)[THEN is_cong.belowD] assms(2)[THEN cont2monofunE]
  show "x ⊑ y ⟹ R (f x) (f y)" for x y
    by presburger
  from assms(2) assms(1)[THEN is_cong.adm1]
  show "adm (λx. R (f x) (f y))" for y by (rule adm_subst)
  from assms(2) assms(1)[THEN is_cong.adm2]
  show "adm (λy. R (f x) (f y))" for x by (rule adm_subst)
qed


subsection ‹Flat LODs (the generators)›

text ‹A flat congruence is one whose quotient has height at most 1: a bottom class of
  ``undefined'' points (each ‹R›-below everything) and an antichain of value classes above.
  These arise exactly from continuous maps into flat domains, and generate the LOD frame.›

locale is_flat_cong = is_cong +
  assumes height1: "R x y ⟹ R y x ∨ (∀z. R x z)"

lemma is_flat_cong_eq_on:
  fixes f :: "'a → 'b::flat"
  shows "is_flat_cong (λx y. f⋅x ⊑ f⋅y)"
proof unfold_locales
  show "f⋅x ⊑ f⋅y ⟹ f⋅y ⊑ f⋅z ⟹ f⋅x ⊑ f⋅z" for x y z by (rule below_trans)
  show "x ⊑ y ⟹ f⋅x ⊑ f⋅y" for x y by (rule monofun_cfun_arg)
  show "adm (λx. f⋅x ⊑ f⋅y)" for y by simp
  show "adm (λy. f⋅x ⊑ f⋅y)" for x by simp
  show "f⋅x ⊑ f⋅y ⟹ f⋅y ⊑ f⋅x ∨ (∀z. f⋅x ⊑ f⋅z)" for x y
    by (metis ax_flat minimal)
qed

text ‹The family generated by the flats under arbitrary cong-joins, i.e.\ boolean-level
  intersections (\<^term>‹Inf› at the predicate level = \<^const>‹Sup› at the cong level).›

definition is_flat_gen :: "('a ⇒ 'a ⇒ bool) ⇒ bool" where
  "is_flat_gen R ⟷ (∃ℱ. (∀f∈ℱ. is_flat_cong f) ∧ R = Inf ℱ)"

lemma is_flat_gen_flat: "is_flat_cong f ⟹ is_flat_gen f"
  unfolding is_flat_gen_def by (rule exI[of _ "{f}"]) simp

text ‹Meet of two flats is flat: the generated congruence stays height at most 1.  No induction,
  no distributivity of @{typ 'a} — the symmetric companion @{term U} is squeezed in between.›

lemma is_flat_cong_meet:
  fixes f g :: "'a ⇒ 'a ⇒ bool"
  assumes f: "is_flat_cong f" and g: "is_flat_cong g"
  shows "is_flat_cong (gen (sup f g))"
proof -
  let ?R = "gen (sup f g)"
  have R: "is_cong ?R" by (rule is_cong_gen)
  have cf: "is_cong f" using f by (simp add: is_flat_cong_def)
  note fh = is_flat_cong.height1[OF f]
  note gh = is_flat_cong.height1[OF g]
  have sup1: "?R a b" if "f a b" for a b
    using that by (intro gen_upper) simp
  have sup2: "?R a b" if "g a b" for a b
    using that by (intro gen_upper) simp
  define U :: "'a ⇒ 'a ⇒ bool"
    where "U = (λx y. (∀z. ?R x z) ∨ (?R x y ∧ ?R y x))"
  have Ucong: "is_cong U"
  proof unfold_locales
    fix x y z assume xy: "U x y" and yz: "U y z"
    show "U x z" unfolding U_def
      using xy[unfolded U_def] yz[unfolded U_def] is_cong.trans[OF R] by metis
  next
    fix x y :: "'a" assume "x ⊑ y"
    hence fxy: "f x y" by (rule is_cong.belowD[OF cf])
    hence Rxy: "?R x y" by (rule sup1)
    from fh[OF fxy] show "U x y" unfolding U_def
    proof (elim disjE)
      assume "f y x" hence "?R y x" by (rule sup1)
      with Rxy show "(∀z. ?R x z) ∨ (?R x y ∧ ?R y x)" by blast
    next
      assume "∀z. f x z" hence "∀z. ?R x z" using sup1 by blast
      thus "(∀z. ?R x z) ∨ (?R x y ∧ ?R y x)" by blast
    qed
  next
    fix y show "adm (λx. U x y)" unfolding U_def
      by (intro adm_disj adm_all adm_conj is_cong.adm1[OF R] is_cong.adm2[OF R])
  next
    fix x show "adm (λy. U x y)" unfolding U_def
      by (intro adm_disj adm_const adm_conj is_cong.adm2[OF R] is_cong.adm1[OF R])
  qed
  have supU: "U u v" if huv: "sup f g u v" for u v
    unfolding U_def using sup1 sup2 fh gh
    by (metis sup1 fh sup2 gh that sup_apply sup_bool_def)
  have RU: "?R x y ⟹ U x y" for x y
    by (rule gen_least[OF Ucong supU])
  show "is_flat_cong ?R"
  proof unfold_locales
    show "?R x y ⟹ ?R y z ⟹ ?R x z" for x y z by (rule is_cong.trans[OF R])
    show "x ⊑ y ⟹ ?R x y" for x y by (rule is_cong.belowD[OF R])
    show "adm (λx. ?R x y)" for y by (rule is_cong.adm1[OF R])
    show "adm (λy. ?R x y)" for x by (rule is_cong.adm2[OF R])
    show "?R x y ⟹ ?R y x ∨ (∀z. ?R x z)" for x y
      using RU unfolding U_def by blast
  qed
qed

lemma gen_inf: "gen R = Inf {z. z ≥ R ∧ is_cong z}"
  unfolding gen_def
  by blast
